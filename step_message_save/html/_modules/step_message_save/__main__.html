  1"""It is the entry point of the program.
  2
  3The program receives the arguments from the command line and based on the arguments it
  4prints the configuration JSON schema, the configuration template or it loads the
  5configuration from a file. If no argument is given, it prints the help of the program.
  6The program waits the MQTT ITS ETSI messages in format ASN.1 and publishes them to a
  7Redis stream.
  8
  9Complexity:
 10    - Time: O(n), where n is the number of ITS/Redis messages processed over the program's
 11        lifetime; each message is handled in O(1) time, and each connect/reconnect cycle
 12        performs O(1) setup work.
 13    - Space: O(r), where r is the configured number of concurrent STEP receiver tasks
 14        (``stepReceiverNumber``); handling an individual message uses O(1) additional space.

165async def _step_receive_run(
166    instance_name: str, step_broker: StepBroker, queue: Queue[str], logger: Logger
167) -> None:
168    """Receives messages from the STEP gateway and publishes them to Redis streams.
169
170    This function listens for ITS messages from the STEP broker, logs details about each message,
171    and publishes CAM or DENM messages to the appropriate Redis streams using the adapter or producer.
172    Messages without payloads are skipped with a warning.
173
174    Args:
175        instance_name (str): The name of the STEP client instance.
176        queue (Queue[str]): Queue to publish received messages to.
177        step_broker (StepBroker): STEP broker instance to receive messages from.
178        logger (Logger): Logger instance for logging.
179
180    Returns:
181        None
182
183    Complexity:
184        - Time: O(n), where n is the number of messages received from the STEP broker; each
185          message is dispatched and processed in O(1) time via a type->label dict lookup
186          instead of a sequential isinstance chain.
187        - Space: O(1), aside from the constant-size resources held by the adapter and producer
188          Redis stream clients.
189
190    Note:
191        This function is intended to run within an asyncio event loop and will process messages
192        as they arrive from the STEP broker.
193    """
194    if logger.isEnabledFor(level=logging.INFO):
195        logger.info(
196            msg=f"Instance {instance_name} listening messages from STEP gateway"
197        )
198    async for message in step_broker.its_messages:
199        payload = message.its_payload
200        if logger.isEnabledFor(level=logging.INFO):
201            logger.info(
202                msg=f"Received {message.its_payload.message_type}message from STEP gateway"
203            )
204        await queue.put(item=payload.to_json(indent=None))

207async def _step_message_write_run(
208    output_file: Path, queue: Queue[str], logger: Logger
209) -> None:
210    """Writes JSON messages received from concurrent STEP receiver tasks to a single
211    output file.
212
213    This function runs as a single dedicated task that owns the output file handle
214    exclusively, consuming JSON-serialized messages from a queue fed by multiple STEP
215    receiver tasks running concurrently. Because only this task ever touches the file,
216    writes are naturally serialized without needing a lock. Writes are buffered and go
217    directly to the file without an explicit flush; they only reach disk once the
218    buffer fills or the file is closed.
219
220    Args:
221        output_file (Path): Path of the file where JSON messages are appended.
222        queue (asyncio.Queue[str]): Queue of JSON-serialized messages produced by the
223            STEP receiver tasks.
224        logger (Logger): Logger instance for logging.
225
226    Returns:
227        None
228
229    Complexity:
230        - Time: O(n), where n is the number of messages written over the task's
231          lifetime; each message is dequeued and written to the file in O(1) time.
232        - Space: O(1), aside from the queue itself, which is owned by the caller.
233
234    Note:
235        This function is intended to run within an asyncio event loop until cancelled
236        (e.g. when the enclosing TaskGroup is cancelled on shutdown), at which point the
237        output file is closed as the cancellation propagates out of the ``with`` block.
238    """
239    if logger.isEnabledFor(level=logging.INFO):
240        logger.info(msg=f"Writing STEP messages to {output_file}")
241    output_file.parent.mkdir(parents=True, exist_ok=True)
242    with output_file.open(mode="w", encoding="utf-8", newline="") as file:
243        while True:
244            line = await queue.get()
245            try:
246                file.write(line + "\n")
247            finally:
248                queue.task_done()

251async def _main_async(
252    instance_name: str, config: StepMessageSaveConfig, logger: Logger
253) -> None:
254    """Main asynchronous function that manages the STEP client's core operations.
255
256    This function sets up exception handling, initializes connections to Redis and the STEP gateway,
257    and manages message receiving and sending tasks in an infinite loop with reconnection
258    capability.
259
260    Args:
261        instance_name (str): The name of the STEP client instance.
262        config (StepMessageSaveConfig): Configuration object containing all necessary settings
263        logger (Logger): Logger instance for recording program events
264
265    Returns:
266        None
267
268    Complexity:
269        - Time: O(1) per reconnect iteration to set up the Redis pool, broker, and receiver
270          tasks, excluding the unbounded time spent awaiting messages within those tasks.
271        - Space: O(r), where r is ``config.stepReceiverNumber``, for the concurrently running
272          STEP receiver tasks created on each connection cycle.
273
274    Note:
275        The function runs continuously until external termination, handling connection
276        drops by attempting to reconnect after the interval specified in config.
277    """
278    get_running_loop().set_exception_handler(
279        handler=lambda _, context: _exception_handler(context=context, logger=logger)
280    )
281    if logger.isEnabledFor(level=logging.INFO):
282        logger.info(msg="Program started")
283    while True:
284        try:
285            async with StepBroker(
286                instance_name=instance_name, config=config.stepBroker, logger=logger
287            ) as step_broker:
288                await step_broker.subscribe_execution()
289                queue = Queue[str](maxsize=config.maxFileWriterQueueRooms)
290                async with TaskGroup() as group:
291                    # Create STEP receive thread. First to get a permanent connection
292                    for count in range(config.stepReceiverNumber):
293                        group.create_task(
294                            coro=_step_receive_run(
295                                instance_name=f"{instance_name}-{count}",
296                                step_broker=step_broker,
297                                queue=queue,
298                                logger=logger,
299                            )
300                        )
301                    # Create STEP message write thread. Only one to write in the file
302                    group.create_task(
303                        coro=_step_message_write_run(
304                            output_file=Path(config.outputFile),
305                            queue=queue,
306                            logger=logger,
307                        )
308                    )
309        except Exception:
310            if logger.isEnabledFor(level=logging.ERROR):
311                logger.exception(msg="Exception unmanaged")
312            await asyncio_sleep(delay=config.reconnectInterval)
313            if logger.isEnabledFor(level=logging.INFO):
314                logger.info(msg=f"Connection retry after {config.reconnectInterval}")

379def main() -> int:  # noqa: C901, PLR0912
380    """Main entry point for the application.
381
382    This function initializes the application, processes command-line arguments,
383    loads configuration, sets up logging, and executes the main asynchronous event loop.
384
385    It handles the following operations:
386
387        - Printing help when no arguments are provided
388        - Generating JSON schema file when requested
389        - Creating configuration template file when requested
390        - Loading configuration from specified file
391        - Setting up Windows event loop policy if running on Windows
392        - Running the main asynchronous function
393
394    Exception handling:
395
396        - Catches interruptions (Ctrl+C) gracefully
397        - Logs unhandled exceptions
398
399    Returns:
400        int: Exit code indicating program result status:
401
402            - ExitCode.OK (0): Success
403            - ExitCode.USAGE (1): Improper usage (missing/invalid command-line arguments)
404            - ExitCode.ERROR (2): An unhandled exception occurred
405            - ExitCode.ABORTED (3): Program was interrupted (e.g. Ctrl+C or SIGTERM)
406
407    Complexity:
408        - Time: O(1) for argument parsing and setup, excluding the unbounded time spent
409          running the main asyncio event loop via ``_main_async``.
410        - Space: O(1), aside from the configuration object and logger held for the
411          duration of program execution.
412    """
413    if IS_WINDOWS:
414        # Windows: Handle CTRL_BREAK_EVENT
415        sigbreak = getattr(signal, "SIGBREAK", None)
416        if sigbreak is not None:
417            signal.signal(signalnum=sigbreak, handler=_signal_handler)
418    else:
419        # Linux/Unix: Handle SIGINT (in case default behavior is overridden)
420        signal.signal(signal.SIGINT, _signal_handler)
421        # Linux/Unix: Handle SIGTERM (sent by container runtimes on stop)
422        signal.signal(signal.SIGTERM, _signal_handler)
423    result = ExitCode.OK
424    logger = StepLogger.get_logger(name=None, config=None, instance_name=None)
425    try:
426        parser = _args_parse()
