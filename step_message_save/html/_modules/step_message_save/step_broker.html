  1"""STEP Gateway Module for MQTT Communication.
  2
  3This module provides the necessary components for a secure, configurable MQTT-based gateway
  4within the STEP ecosystem. It connects to the STEP MQTT broker and subscribes to ITS
  5(Intelligent Transport Systems) message topics on its behalf.
  6
  7The module implements a gateway that supports:
  8
  9    - Secure TLS/SSL MQTT communications with configurable parameters
 10    - Topic subscription with configurable Quality of Service (QoS) controls
 11    - Asynchronous operation with context management for clean resource handling
 12    - Support for various ITS message types, with focus on CAM (Cooperative Awareness Messages)
 13    - Clean error handling and comprehensive logging
 14
 15The STEP Broker class leverages async/await patterns throughout to ensure non-blocking
 16operations, making it suitable for high-throughput messaging environments in
 17ITS applications.
 18
 19Complexity:
 20    - Time: O(t), where t is the number of configured MQTT topics; subscribing to and
 21        converting the publish-topic configuration scales with t, while publishing or
 22        receiving an individual ITS message is O(1).
 23    - Space: O(t), for the subscription and publish-topic configuration retained by the
 24        gateway.
 25"""

 43class StepBroker(AsyncComManager):
 44    """STEP gateway implementing AsyncComManager.
 45
 46    This class provides a high-level client for connecting to a STEP service
 47    using MQTT protocol. It handles secure connections, topic subscriptions
 48    and message management according to the provided configuration.
 49
 50    The class manages:
 51
 52        - TLS/SSL connections with certificate validation
 53        - MQTT connection parameters (host, port, auth, timeouts, etc.)
 54        - Topic subscriptions with configurable QoS levels
 55        - Message queuing and delivery
 56
 57    It implements the AsyncComManager interface and supports asynchronous
 58    context manager protocol for safe resource management.
 59
 60    Attributes:
 61        instance_name (str): Unique identifier for the STEP client instance.
 62        config (StepBrokerConfig): Configuration object containing STEP broker settings.
 63        logger (Logger): Logger instance for logging messages and errors.
 64
 65    Complexity:
 66        - Time: O(t), where t is the number of configured subscription/publish topics;
 67            construction and subscription scale with t, while publishing a single ITS message
 68            is O(1).
 69        - Space: O(t), for the topic subscription and publish-topic configuration held by
 70            the instance.
 71    """
 72
 73    __slots__ = ("_config", "_logger")
 74
 75    def __init__(
 76        self, instance_name: str, config: StepBrokerConfig, logger: Logger
 77    ) -> None:
 78        """Initialize the StepBroker instance.
 79
 80        Args:
 81            instance_name (str): The name identifier for this broker instance.
 82            config (StepBrokerConfig): Configuration object containing STEP broker settings
 83                including MQTT client parameters, protocol settings, and SSL configuration.
 84            logger (Logger): Logger instance for recording broker operations and events.
 85
 86        Returns:
 87            None
 88
 89        Complexity:
 90            - Time: O(t), where t is the number of configured publish topics; each topic is
 91              converted into a ``PublishTopicParams`` instance in O(1) time.
 92            - Space: O(t), for the list of converted publish topic parameters.
 93
 94        Note:
 95            This initializer sets up the STEP broker with MQTT communication capabilities,
 96            configures SSL/TLS parameters if enabled, and initializes the async communication
 97            manager with all necessary parameters for STEP protocol communication.
 98        """
 99        self._config = config
100        self._logger = logger
101        tls_context: ssl.SSLContext | None = None
102        if self._config.mqttClient.sslEnable:
103            tls_context = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
104            tls_context.minimum_version = ssl.TLSVersion.TLSv1_2
105            if self._config.mqttClient.sslCaCertsFile:
106                tls_context.load_verify_locations(
107                    cafile=self._config.mqttClient.sslCaCertsFile
108                )
109            if not self._config.mqttClient.sslVerifyServerCert:
110                tls_context.check_hostname = False
111                tls_context.verify_mode = ssl.CERT_NONE
112                if self._logger.isEnabledFor(level=logging.WARNING):
113                    self._logger.warning(
114                        msg="TLS certificate verification is DISABLED. MITM attacks possible."
115                    )
116            if self._config.mqttClient.sslClientCertFile:
117                tls_context.load_cert_chain(
118                    certfile=self._config.mqttClient.sslClientCertFile,
119                    keyfile=self._config.mqttClient.sslClientKeyFile,
120                    password=self._config.mqttClient.sslClientKeyFilePassword,
121                )
122        async_com_manager_params = AsyncComManagerParams(
123            instance_name=instance_name,
124            host=self._config.mqttClient.host,
125            port=self._config.mqttClient.port,
126            username=self._config.mqttClient.username or "",
127            password=self._config.mqttClient.password or "",
128            timeout=self._config.mqttProtocol.timeout,
129            station_id=None,
130            keepalive=self._config.mqttProtocol.keepalive,
131            bind_address="",
132            bind_port=0,
133            clean_start=self._config.mqttProtocol.cleanStart,
134            queue_type=None,
135            max_queued_incoming_messages=self._config.mqttProtocol.maxQueuedIncomingMessages,
136            max_queued_outgoing_messages=self._config.mqttProtocol.maxQueuedOutgoingMessages,
137            max_inflight_messages=self._config.mqttProtocol.maxInflightMessages,
138            max_concurrent_outgoing_calls=self._config.mqttProtocol.maxConcurrentOutgoingCalls,
139            tls_context=tls_context,
140            tls_params=None,
141            proxy=None,
142            socket_options=None,
143            session_expiry=self._config.mqttProtocol.sessionExpiry,
144            receive_maximum=self._config.mqttProtocol.receiveMaximum,
145            maximum_packet_size=self._config.mqttProtocol.maximumPacketSize,
146            topic_alias_maximum=0,
147            publish_topics_params=None,
148            cache_dir=self._config.cacheDir,
149            logger=self._logger,
150        )
151        super().__init__(params=async_com_manager_params)

171    async def subscribe_execution(self) -> None:
172        """Subscribe to a list of topics with specified MQTT options.
173
174        This method subscribes to a list of topics specified in the configuration
175        with the following MQTT options:
176
177            - QoS level from config
178            - No local messages (noLocal=True)
179            - Do not retain as published (retainAsPublished=False)
180            - Do not send retained messages (retainHandling=RETAIN_DO_NOT_SEND)
181
182        Raises:
183            StepMessageSaveError: If subscription fails or the granted QoS level
184                exceeds the configured maximum. A broker is allowed to grant a lower
185                QoS than requested, so a downgrade is accepted rather than treated as
186                a failure.
187
188        Returns:
189            None
190
191        Complexity:
192            - Time: O(t), where t is the number of configured subscription topics; each topic
193                is subscribed and validated in O(1) time.
194            - Space: O(t), for the list of topic/options tuples and the returned subscription
195                results.
196        """
197        if self._config.mqttProtocol.subscriptionTopics is not None:
198            sub_results = await self.subscribe(
199                topic=[
200                    (
201                        topic_params.topic,
202                        SubscribeOptions(
203                            qos=self._config.mqttProtocol.maxSubscriptionQos,
204                            noLocal=False,
205                            retainAsPublished=False,
206                            retainHandling=SubscribeOptions.RETAIN_DO_NOT_SEND,
207                        ),
208                    )
209                    for topic_params in self._config.mqttProtocol.subscriptionTopics
210                ]
211            )
212            if len(sub_results) != len(self._config.mqttProtocol.subscriptionTopics):
213                raise StepMessageSaveError(
214                    f"Subscription results {len(sub_results)} do not match the number "
215                    f"of topics {len(self._config.mqttProtocol.subscriptionTopics)}."
216                )
217            for topic, result in zip(
218                self._config.mqttProtocol.subscriptionTopics, sub_results, strict=True
219            ):
220                result_value = (
221                    result.value
222                    if isinstance(result, ReasonCode)
223                    else cast("int", result)
224                )
225                if result_value > self._config.mqttProtocol.maxSubscriptionQos:
226                    raise StepMessageSaveError(
227                        f"Failed to subscribe to {topic}: {result}"
228                    )
229                if self._logger.isEnabledFor(level=logging.INFO):
230                    self._logger.info(msg=f"Subscribed to {topic}: {result}")
