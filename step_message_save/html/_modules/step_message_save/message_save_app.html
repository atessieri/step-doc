  1"""Workload of the STEP Message Save program.
  2
  3Defines the ``AbstractApp`` subclass handed to ``AppRunner`` by
  4``step_message_save.__main__``: it reads the MQTT credential from the environment,
  5connects to the STEP MQTT broker, receives the ITS ETSI messages with concurrent
  6receiver tasks and appends them as JSON lines to an output file through a single
  7writer task, reconnecting after any failure. On a shutdown signal the messages
  8still queued are drained to the output file.
  9
 10Complexity:
 11    - Time: O(n), where n is the number of ITS messages processed over the
 12      program's lifetime; each message is handled in O(1) time, and each
 13      connect/reconnect cycle performs O(1) setup work.
 14    - Space: O(r + q), where r is the configured number of concurrent STEP receiver
 15      tasks (``stepReceiverNumber``) and q the queue capacity
 16      (``maxFileWriterQueueRooms``); handling an individual message uses O(1)
 17      additional space.
 18
 19Classes:
 20    - ExitCode: Enumeration of the program exit codes. Complexity: time O(1),
 21      space O(1).
 22    - MessageSaveApp: Workload run by ``AppRunner``: the STEP broker manager, and
 23      the drain of the queue on shutdown. Complexity: time O(n) over the program's
 24      lifetime plus O(q) at shutdown, space O(r + q).
 25"""

 74class MessageSaveApp(AbstractApp[ExitCode]):
 75    """Workload of the program, run by ``AppRunner``: the STEP broker manager, and the
 76    drain of the queue on shutdown.
 77
 78    Holds the state shared by ``work`` and ``on_shutdown``: the MQTT credential
 79    read from the environment, the output file and the queue between the receiver
 80    tasks and the writer task. The queue is created once, so messages still queued
 81    when a connection drops are written by the writer task of the next reconnect
 82    cycle. The output file is opened in append mode, so messages already on disk
 83    are never erased, whether by a reconnect or by a restart of the process itself.
 84
 85    Attributes:
 86        instance_name (str): The name of the STEP client instance.
 87        config (StepMessageSaveConfig): Configuration object containing all
 88            necessary settings.
 89        logger (Logger): Logger instance for recording program events.
 90
 91    Complexity:
 92        - Time: O(n), where n is the number of ITS messages processed by
 93          ``work`` over the program's lifetime, plus O(q) at shutdown to drain
 94          the q messages still queued.
 95        - Space: O(r + q), where r is ``config.stepReceiverNumber`` and q is
 96          ``config.maxFileWriterQueueRooms``.
 97    """
 98
 99    _MQTT_URL_VAR_NAME: ClassVar[str] = "MQTT_URL"
100    _MQTT_PORT_VAR_NAME: ClassVar[str] = "MQTT_PORT"
101    _MQTT_USERNAME_VAR_NAME: ClassVar[str] = "MQTT_USERNAME"
102    _MQTT_PASSWORD_VAR_NAME: ClassVar[str] = "MQTT_PASSWORD"  # noqa: S105
103
104    __slots__ = (
105        "_config",
106        "_instance_name",
107        "_logger",
108        "_mqtt_credential",
109        "_output_file",
110        "_queue",
111    )
112
113    def __init__(
114        self, instance_name: str, config: StepMessageSaveConfig, logger: Logger
115    ) -> None:
116        """Store the settings, read the MQTT credential from the environment and
117        create the queue shared by every reconnect cycle.
118
119        Args:
120            instance_name (str): The name of the STEP client instance.
121            config (StepMessageSaveConfig): Configuration object containing all
122                necessary settings.
123            logger (Logger): Logger instance for recording program events.
124
125        Raises:
126            ValueError: If the MQTT credential cannot be read from the
127                environment (see ``_get_mqtt_credential``).
128
129        Complexity:
130            - Time: O(1), a few attribute assignments and a fixed number of
131              environment variable lookups.
132            - Space: O(1), the queue is created empty.
133        """
134        self._instance_name = instance_name
135        self._config = config
136        self._logger = logger
137        # Read here and not in work(): _step_broker_run retries on any exception,
138        # so a missing variable would never end the program
139        self._mqtt_credential = self._get_mqtt_credential()
140        self._output_file = Path(config.outputFile)
141        # Not bound to a loop at creation (Python >= 3.10): created before the run
142        self._queue = Queue[str](maxsize=config.maxFileWriterQueueRooms)
143
144    def _get_mqtt_credential(self) -> MqttCredential:
145        """Load MQTT broker connection details from environment variables.
146
147        Reads the broker URL, port, username, and password from the environment
148        (loading a ``.env`` file first via ``load_dotenv``), using the
149        ``_MQTT_URL_VAR_NAME``, ``_MQTT_PORT_VAR_NAME``, ``_MQTT_USERNAME_VAR_NAME``,
150        and ``_MQTT_PASSWORD_VAR_NAME`` variable names, and logs host, port and
151        username at INFO level (the password is never logged). The username and
152        password are optional; the URL and port are required.
153
154        Returns:
155            MqttCredential: The MQTT connection details assembled from the
156                environment variables.
157
158        Raises:
159            ValueError: If the ``MQTT_URL`` or ``MQTT_PORT`` environment variable
160                is not set, if ``MQTT_PORT`` is not an integer in the 1-65535
161                range, or if ``MQTT_URL`` is empty/whitespace-only.
162
163        Complexity:
164            - Time: O(1), a fixed number of environment variables is read.
165            - Space: O(1), a single fixed-size frozen dataclass instance is allocated.
166        """
167        load_dotenv()
168        url = os.getenv(self._MQTT_URL_VAR_NAME)
169        if url is None:
170            raise ValueError(
171                f"Environment variable {self._MQTT_URL_VAR_NAME} is not set."
172            )
173        port = os.getenv(self._MQTT_PORT_VAR_NAME)
174        if port is None:
175            raise ValueError(
176                f"Environment variable {self._MQTT_PORT_VAR_NAME} is not set."

178        username = os.getenv(self._MQTT_USERNAME_VAR_NAME)
179        password = os.getenv(self._MQTT_PASSWORD_VAR_NAME)

181            self._logger.info(
182                msg="Creating MqttCredential with the following settings: "

186            mqtt_url=url,
187            mqtt_port=int(port),

201            level (int): Logging level the logger must be enabled for (e.g.,

204        Returns:

207        Complexity:

218                            f"from {message.station_id}: {message.message}"

222    async def _step_receive_run(
223        self, instance_name: str, step_broker: StepBroker
224    ) -> None:
225        """Receive messages from the STEP gateway and queue them for the file writer.
226
227        Listens for ITS messages from the STEP broker, logs details about each
228        message, and enqueues each message's JSON representation for the dedicated
229        file writer task to persist.
230
231        Args:
232            instance_name (str): The name of the receiver task: the instance name
233                followed by the task index.
234            step_broker (StepBroker): STEP broker instance to receive messages from.
235
236        Returns:
237            None: Messages are enqueued for the writer task; nothing is returned.
238
239        Note:
240            This method is intended to run within an asyncio event loop and will
241            process messages as they arrive from the STEP broker.
242
243        Complexity:
244            - Time: O(n), where n is the number of messages received from the STEP
245              broker; each message is JSON-serialized and enqueued in O(1) time.
246            - Space: O(1), aside from the queue shared with the file writer task.
247        """
248        if self._logger.isEnabledFor(level=logging.INFO):
249            self._logger.info(
250                msg=f"Instance {instance_name} listening messages from STEP gateway"
251            )
252        async for message in step_broker.its_messages:
253            payload = message.its_payload
254            self._log_received_message(message=payload, level=logging.INFO)
255            await self._queue.put(item=payload.to_json(indent=None))

257    async def _step_message_write_run(self) -> None:
258        """Write the JSON messages received from the concurrent STEP receiver tasks
259        to the output file.
260
261        Runs as a single dedicated task that owns the output file exclusively,
262        consuming JSON-serialized messages from the queue fed by the STEP receiver
263        tasks running concurrently. Because only this task ever touches the file,
264        writes are naturally serialized without needing a lock. The file is opened
265        in append mode, so messages written in a previous run, or in a previous
266        reconnect cycle of this run, are never erased. Writes are buffered and go
267        directly to the file without an explicit flush; they only reach disk once
268        the buffer fills or the file is closed.
269
270        Returns:
271            None: Runs until cancelled; messages are written to the file as a side
272                effect and nothing is returned.
273
274        Note:
275            This method is intended to run within an asyncio event loop until
276            cancelled (e.g. when the enclosing TaskGroup is cancelled on shutdown
277            or reconnect). The file is closed here when the task is cancelled.
278
279        Complexity:
280            - Time: O(n), where n is the number of messages written over the task's
281              lifetime; each message is dequeued and written to the file in O(1)
282              time.
283            - Space: O(1), aside from the queue itself.
284        """
285        if self._logger.isEnabledFor(level=logging.INFO):
286            self._logger.info(msg=f"Writing STEP messages to {self._output_file}")
287        with self._output_file.open(mode="a", encoding="utf-8", newline="") as file:
288            while True:
289                line = await self._queue.get()
290                try:
291                    file.write(line + "\n")
292                finally:
293                    self._queue.task_done()
294
295    async def _step_broker_run(self) -> None:
296        """Manage the STEP broker connection: connect, subscribe and run the receiver
297        and writer tasks in a dedicated TaskGroup, reconnecting after any unhandled
298        exception.
299
300        Every connection cycle opens a ``StepBroker``, subscribes, and runs
301        ``config.stepReceiverNumber`` receiver tasks plus a single writer task in a
302        ``TaskGroup`` nested in the broker context, so the broker only disconnects
303        once the group has unwound. When the cycle fails, the exception is logged
304        and a new cycle starts after ``config.reconnectInterval`` seconds. The same
305        queue and output file are used on every cycle, so messages still queued
306        when a connection drops are written by the writer of the next cycle.
307
308        Returns:
309            None: Never returns normally; runs until cancelled.
310
311        Note:
312            Runs until cancelled by ``AppRunner``'s TaskGroup on shutdown;
313            the cancellation propagates from wherever this is waiting, including the
314            connection attempt and the reconnect sleep. Non-transient errors (e.g. an
315            unwritable output file) are retried every ``reconnectInterval`` as well,
316            which is at least one second, so the retry loop is rate-limited.
317
318        Complexity:
319            - Time: O(1) per reconnect iteration to set up the broker and the tasks,
320              excluding the unbounded time spent awaiting messages within those
321              tasks.
322            - Space: O(r), where r is ``config.stepReceiverNumber``, for the
323              concurrently running STEP receiver tasks created on each connection
324              cycle.
325        """
326        while True:
327            try:
328                async with StepBroker(
329                    instance_name=self._instance_name,
330                    config=self._config.stepBroker,
331                    mqtt_credential=self._mqtt_credential,
332                    logger=self._logger,
333                ) as step_broker:
334                    await step_broker.subscribe_execution()
335                    async with TaskGroup() as group:
336                        # Create STEP receive tasks
337                        for count in range(self._config.stepReceiverNumber):
338                            group.create_task(
339                                coro=self._step_receive_run(
340                                    instance_name=f"{self._instance_name}-{count}",
341                                    step_broker=step_broker,
342                                )
343                            )
344                        # Create STEP message write task. Only one to write in the file
345                        group.create_task(coro=self._step_message_write_run())
346            except* Exception:
347                if self._logger.isEnabledFor(level=logging.ERROR):
348                    self._logger.exception(msg="Exception unmanaged")
349                await asyncio_sleep(delay=self._config.reconnectInterval)
350                if self._logger.isEnabledFor(level=logging.INFO):
351                    self._logger.info(
352                        msg=f"Connection retry after {self._config.reconnectInterval}"
353                    )
354
355    def _drain_queue(self) -> int:
356        """Append every message still queued to the output file, without awaiting.
357
358        Called by ``on_shutdown`` once ``AppRunner``'s TaskGroup has unwound, so
359        the writer task has closed its file and this is the only handle open on
360        the output.
361
362        Returns:
363            int: Number of messages written.
364
365        Complexity:
366            - Time: O(q), where q is the number of messages still queued.
367            - Space: O(1), messages are written one at a time.
368        """
369        count = 0
370        with self._output_file.open(mode="a", encoding="utf-8", newline="") as file:
371            while not self._queue.empty():
372                file.write(self._queue.get_nowait() + "\n")
373                self._queue.task_done()
374                count += 1
375        return count

398    @override
399    def on_shutdown(self, stop_signal: Signals) -> ExitCode:
400        """Drain the queue to the output file once a shutdown signal stopped ``work``.
401
402        Called by ``AppRunner`` after every task has unwound, so the writer task
403        has closed its file and this is the only handle open on the output. A
404        message is lost only if it fails to write to the file, or if a receiver
405        task was cancelled while blocked handing it off to a full queue.
406
407        Args:
408            stop_signal (Signals): The signal that stopped the run.
409
410        Returns:
411            ExitCode: ExitCode.ABORTED.
412
413        Complexity:
414            - Time: O(q), where q is the number of messages still queued.
415            - Space: O(1), messages are written one at a time.
416        """
417        # The writer is gone: persist what is still queued
418        written = self._drain_queue()
419        if self._logger.isEnabledFor(level=logging.INFO):
420            self._logger.info(
421                msg=(
422                    f"Program interrupted by {stop_signal.name}, "
423                    f"{written} queued messages saved"
424                )
425            )
426        return ExitCode.ABORTED
