150    @pytest.mark.parametrize(argnames="queue_rooms", argvalues=[1, 1000, 10000])
151    def test___init__(self, mocker: MockerFixture, *, queue_rooms: int) -> None:
152        """Test that ``__init__`` stores the settings, passes the reconnect interval
153        to ``ReconnectingApp``, reads the MQTT credential
154        through ``MqttCredential.from_env``, derives the output file path from the
155        configuration and creates an empty queue bounded by
156        ``maxFileWriterQueueRooms``.
157
158        Args:
159            mocker (MockerFixture): pytest-mock fixture for creating mock objects.
160            queue_rooms (int): The configured ``maxFileWriterQueueRooms``.
161
162        Returns:
163            None
164        """
165        config = StepMessageSaveConfig.get_default().model_copy(
166            update={"maxFileWriterQueueRooms": queue_rooms}
167        )
168        mock_logger = _mock_logger(mocker, logger_enabled=True)
169        mock_from_env = mocker.patch(
170            "step_message_save.message_save_app.MqttCredential.from_env",
171            return_value=_MQTT_CREDENTIAL,
172        )
173        app = MessageSaveApp(
174            instance_name=_INSTANCE_NAME, config=config, logger=mock_logger
175        )
176        assert app._instance_name == _INSTANCE_NAME
177        assert app._config is config
178        assert app._reconnect_interval == config.reconnectInterval
179        mock_from_env.assert_called_once_with(logger=mock_logger)
180        assert app._mqtt_credential is _MQTT_CREDENTIAL
181        assert app._output_file == Path(config.outputFile)
182        assert app._queue.maxsize == queue_rooms
183        assert app._queue.empty()
184        mock_logger.info.assert_not_called()
185        # Last: the identity check narrows the mock to Logger for the type checker
186        assert app._logger is mock_logger

436            nested (bool): Whether the output directory has missing parents.
437
438        Returns:
439            None
440        """

442        output_dir = tmp_path / "a" / "b" if nested else tmp_path
443        output_file = output_dir / "output.jsonl"
444        config = StepMessageSaveConfig.get_default().model_copy(
445            update={"outputFile": str(output_file)}

448        app = _create_app(mocker, config=config, logger=mock_logger)
449        mock_step_broker = mocker.MagicMock()
450        mock_step_broker.__aenter__ = mocker.AsyncMock(return_value=mock_step_broker)
451        mock_step_broker.__aexit__ = mocker.AsyncMock(return_value=None)
452        mock_step_broker.subscribe_execution = mocker.AsyncMock()
453        mock_step_broker_class = mocker.patch(

466        mock_order.attach_mock(mock_step_broker.subscribe_execution, "subscribe")

506        assert mock_task_group.create_task.call_args_list[-1] == mocker.call(
507            coro=mock_step_message_write_run.return_value
508        )
509        # The broker wraps the task group: it connects and subscribes before the
510        # tasks start, and only disconnects once the group has unwound
511        assert [name for name, _, _ in mock_order.mock_calls] == [
512            "broker_enter",
