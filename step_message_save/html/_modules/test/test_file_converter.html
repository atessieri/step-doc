 41    ) -> None:
 42        """Initialize the stub.
 43
 44        Args:
 45            name (str): The name recorded in ``calls`` when called.
 46            extension (str): The extension returned by ``get_extension``.
 47            calls (list[str]): The shared list the stub appends its name to.
 48            accepts (bool): Whether the stub accepts the file.

 50        Returns:
 51            None

 54        self._extension = extension
 55        self._calls = calls

 63            str: The extension.

 67    def __call__(self, file_path: Path) -> list[FacilityMessage]:  # noqa: ARG002
 68        """Record the call, then accept or reject the file.
 69
 70        Args:
 71            file_path (Path): Path to the file to parse.
 72
 73        Returns:
 74            list[FacilityMessage]: An empty list when the stub accepts the file.

 76        Raises:
 77            LoaderError: The stub rejects the file.

 79        self._calls.append(self.name)
 80        if not self._accepts:
