 40def _validate_with(model: type[BaseModel], field: str, value: Any) -> BaseModel:
 41    """Validate ``model``'s default values with ``field`` replaced by ``value``.
 42
 43    Args:
 44        model (type[BaseModel]): The configuration model class exposing a
 45            ``get_default()`` factory.
 46        field (str): The name of the field to override.
 47        value (Any): The value to assign to ``field``.
 48
 49    Returns:
 50        BaseModel: The validated model instance.
 51    """
 52    data = model.get_default().model_dump()  # type: ignore[attr-defined]
 53    data[field] = value
 54    return model.model_validate(data)

 57def _assert_model_config(model: type[BaseModel]) -> None:
 58    """Assert that ``model`` is frozen, strict, and forbids extra fields, and that
 59    mutating its default instance raises ``ValidationError``.
 60
 61    Args:
 62        model (type[BaseModel]): The configuration model class exposing a
 63            ``get_default()`` factory.
 64
 65    Returns:
 66        None
 67    """
 68    assert model.model_config["frozen"] is True

 70    assert model.model_config["extra"] == "forbid"
 71    instance = model.get_default()  # type: ignore[attr-defined]
