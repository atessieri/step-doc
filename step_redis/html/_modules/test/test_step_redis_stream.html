667        Returns:
668            None
669        """

675            mock_redis_object.xreadgroup = mocker.AsyncMock(
676                side_effect=[exception, *responses]
677            )
678        else:
679            mock_redis_object.xreadgroup = mocker.AsyncMock(side_effect=responses)

688        )
689        # Test

694                consumer_name=consumer_name,
695                block_timeout=block_timeout,

697            )
