142async def mock_receive_generator(message: str) -> AsyncGenerator[str]:
143    """Asynchronous generator that yields a mock JSON message and then raises a
144    ValueError to simulate the end of iteration.
145
146    Args:
147        message (str): The mock JSON message to yield.
148
149    Yields:
150        str: The mock JSON message.
151
152    Raises:
153        ValueError: Always raised after yielding the mock message
154            to indicate the generator should stop.
155
156    Returns:
157        None
158    """

160    raise ValueError("Stop iteration")

163async def mock_receive_generator_sequence(messages: list[str]) -> AsyncGenerator[str]:
164    """Asynchronous generator that yields each mock JSON message in ``messages``, in
165    order, and then raises a ValueError to simulate the end of iteration.
166
167    Args:
168        messages (list[str]): The mock JSON messages to yield, in order.
169
170    Yields:
171        str: Each mock JSON message, in order.
172
173    Raises:
174        ValueError: Always raised after yielding every message
175            to indicate the generator should stop.
176
177    Returns:
178        None
179    """
180    for message in messages:
181        yield message
182    raise ValueError("Stop iteration")
