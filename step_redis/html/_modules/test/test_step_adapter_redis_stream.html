  1"""Unit tests for the StepAdapterRedisStream class.
  2
  3This module provides comprehensive test coverage for the ``StepAdapterRedisStream``
  4class, which extends ``StepRedisStream`` to provide specialized functionality for
  5publishing and receiving adapter messages in CloudEvents format via Redis Streams.
  6
  7The test suite validates:
  8
  9    - Class initialization and configuration with Redis pool and stream parameters
 10    - Asynchronous context manager protocol (``__aenter__``)
 11    - Publishing CAM (Cooperative Awareness Message) data wrapped in CloudEvents format
 12    - Receiving and deserializing adapter messages from Redis Streams
 13    - Proper interaction with parent class methods and external dependencies
 14
 15All tests utilize mocking to isolate the unit under test from external dependencies
 16such as Redis connections, parent class implementations, and CloudEvents schemas.
 17
 18Test Class:
 19    TestStepAdapterRedisStream: Contains all unit tests for StepAdapterRedisStream.
 20
 21Dependencies:
 22    - pytest: Testing framework
 23    - pytest-mock: Mocking utilities for pytest

 25    - step_redis: Redis Stream components and StepAdapterRedisStream implementation
 26
 27See Also:
 28    - ``step_redis.StepAdapterRedisStream``: The class being tested

 69    async def test_publish_cam_message(
 70        self, mocker: MockerFixture, log_enabled: bool
 71    ) -> None:
 72        """Test the publish_cam_message method.
 73
 74        Verifies that publish_cam_message correctly wraps a CamDataV1 message in a CloudEvents format
 75        and calls the parent's publish_message method with the serialized JSON.
 76
 77        Args:
 78            mocker (MockerFixture): The pytest-mock fixture used to create mock objects.
 79            log_enabled (bool): Flag to enable or disable logging.
 80
 81        Returns:
 82            None
 83        """
 84        # Arrange
 85        mock_pool = mocker.MagicMock(spec=StepRedisPool)
 86        config = StepRedisStreamParams(
 87            stream_id="test_stream",
 88            group_id="test_group",
 89            wait_timeout_ms=1000,
 90            published_message_max_retained=100,
 91            consumer_name="test_consumer",
 92            ttl_seconds=10,
 93        )
 94        mock_logger = mocker.MagicMock(spec=Logger)
 95        mock_logger.isEnabledFor = mocker.MagicMock(return_value=log_enabled)
 96        # Patch the parent's __init__ method to avoid initialization issues
 97        mocker.patch.object(
 98            target=StepRedisStream,
 99            attribute="__init__",
100            autospec=True,
101            side_effect=fake_redis_stream_init,
102        )
103        mock_publish_message = mocker.patch.object(
104            target=StepRedisStream,
105            attribute="publish_message",
106            new_callable=mocker.AsyncMock,
107        )
108        # Mock the CloudEvents wrapper classes
109        mock_adapter_cloud_events_cam_data = mocker.MagicMock()
110        mocker.patch(
111            "step_redis.step_adapter_redis_stream.AdapterCloudEventsCamDataV1",
112            return_value=mock_adapter_cloud_events_cam_data,
113        )
114        mock_adapter_cloud_events_json_root_object = mocker.MagicMock(
115            spec=AdapterCloudEventsJsonRoot
116        )
117        mock_model_dump_json = "serialized_json_message"
118        mock_adapter_cloud_events_json_root_object.model_dump_json.return_value = (
119            mock_model_dump_json
120        )
121        mocker.patch(
122            "step_redis.step_adapter_redis_stream.AdapterCloudEventsJsonRoot",
123            return_value=mock_adapter_cloud_events_json_root_object,
124        )
125        adapter_stream = StepAdapterRedisStream(
126            pool=mock_pool, config=config, logger=mock_logger
127        )
128        # Manually set the required attributes since __init__ is patched
129        adapter_stream._config = config

131        # Create a mock CamDataV1 message with required attributes

147            mock_logger.info.assert_not_called()
