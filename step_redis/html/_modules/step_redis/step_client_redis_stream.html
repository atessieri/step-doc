  1"""STEP Client Redis Integration.
  2
  3This module provides the StepClientRedisStream class for Redis-based communication in the STEP client,
  4enabling asynchronous handling of ITS messages (DENM) using Redis streams.
  5
  6Key features:
  7    - Publishes CloudEvents-compliant DENM messages to Redis streams.
  8    - Receives and parses messages from configured streams.
  9    - Manages stream TTL and uses configuration for stream/group/consumer settings.
 10
 11The StepClientRedisStream class extends StepRedisStream, providing methods for publishing and receiving
 12ITS messages in CloudEvents format, tailored for Intelligent Transport Systems (ITS) applications.
 13
 14Complexity:
 15    - Time: Setup operations (``__init__``, ``__aenter__``) run in O(1). Publishing
 16      (``publish_denm_message``) runs in O(n), where n is the serialized DENM message
 17      size, dominated by CloudEvents JSON serialization and the Redis XADD round-trip.
 18      Receiving (``receive_client_message``) runs in O(m) per yielded message, where m
 19      is the received JSON payload size, dominated by JSON parsing and Pydantic
 20      validation.
 21    - Space: O(1) for setup operations. O(n) per publish call and O(m) per received
 22      message, bounded by the size of the single message being serialized or parsed at a
 23      time; no message history is retained in memory.
 24"""

 60class StepClientRedisStream(StepRedisStream):
 61    """Redis stream client for STEP ITS message exchange.
 62
 63    Extends StepRedisStream to provide asynchronous handling of ITS messages (DENM) using Redis streams.
 64    Supports publishing CloudEvents-compliant DENM messages and receiving messages from configured streams.
 65    Automatically manages stream TTL and uses configuration for stream/group/consumer settings.
 66
 67    Attributes:
 68        config (StepRedisStreamParams): Configuration object
 69            containing Redis stream settings.
 70        health_check_interval (int): Health-check interval in seconds, copied from
 71            the pool's ``health_check_interval`` property at construction time.
 72        logger (Logger): Logger instance for logging operations.
 73        redis_pool (Redis): Async Redis client constructed from the given connection
 74            pool, used for every Redis command issued by this stream.
 75        logging_timer (LoggingTimer): Timer used to log the duration of publish
 76            operations, constructed with prefix ``"step_redis"``.
 77
 78    Complexity:
 79        - Time: O(n), the worst case across ``publish_denm_message`` and
 80          ``receive_client_message``, where n is the serialized/received message
 81          payload size; dominated by CloudEvents JSON serialization or parsing plus
 82          the underlying Redis round-trip.
 83        - Space: O(n), for the single message payload held in memory during a
 84          publish or receive call; no message history is retained.
 85    """
 86
 87    __slots__ = ("_logging_timer",)
 88
 89    def __init__(
 90        self, pool: StepRedisPool, config: StepRedisStreamParams, logger: Logger
 91    ) -> None:
 92        """Initialize the Redis-based step adapter.
 93
 94        Args:
 95            pool (StepRedisPool): Redis connection pool for managing connections.
 96            config (StepRedisStreamParams): Configuration object containing Redis stream settings.
 97            logger (Logger): Logger instance for logging operations.
 98
 99        Returns:
100            None
101
102        Note:
103            ``ttl_seconds`` is not derived automatically; it must be set explicitly in
104            ``config`` and must exceed ``wait_timeout_ms`` converted to seconds (this is
105            enforced by ``StepRedisStreamParams.__post_init__``).
106
107        Complexity:
108            - Time: O(1), delegates directly to ``StepRedisStream.__init__``.
109            - Space: O(1), stores references only, no copies of ``pool``, ``config``, or
110              ``logger``.
111        """
112        super().__init__(pool=pool, config=config, logger=logger)
113        self._logging_timer = LoggingTimer(
114            logger=self._logger, prefix="step_redis", log_level=LogMode.INFO
115        )

136    async def publish_denm_message(self, message: DenmDataV1) -> None:
137        """Publish a DENM (Decentralized Environmental Notification Message) to the STEP
138        Client stream.
139
140        Wraps the provided DENM message in a CloudEvents-compliant JSON structure,
141        assigns a unique ID and timestamp, and publishes it to the configured
142        Redis producer stream.
143
144        Args:
145            message (DenmDataV1): The DENM message data to be published.
146
147        Returns:
148            None
149
150        Note:
151            Publishes the message and logs the reference time and serialized message.
152
153        Complexity:
154            - Time: O(n), where n is the size (field count/byte length) of the DENM
155              message; dominated by CloudEvents JSON serialization plus the network
156              round-trip of the underlying Redis XADD call.
157            - Space: O(n), for the serialized JSON payload held in memory before being
158              sent to Redis.
159        """
160        with self._logging_timer(func_name="client-publish_denm_message"):
161            message_client = ClientCloudEventsJsonRoot(
162                root=ClientCloudEventsDenmDataV1(
163                    id=str(object=uuid4()),
164                    source=self._config.consumer_name,
165                    specversion=SpecVersion.V1_0,
166                    type=ClientMessageType.DENM,
167                    dataschema=ClientDataSchemaType.DENM_DATA_V1_SCHEMA,
168                    datacontenttype=DataContentType.APPLICATION_JSON,
169                    time=str(object=ItsTime()),
170                    data=message,
171                )
172            ).model_dump_json()
173        if self._logger.isEnabledFor(level=logging.INFO):
174            self._logger.info(
175                msg=(
176                    f"Publish DENM to Client: time {message.managementContainer.referenceTime}, "
177                    f"{message_client}"
178                )
179            )
180        await self.publish_message(message=message_client)

182    async def receive_client_message(self) -> AsyncGenerator[ClientCloudEventsJsonRoot]:
183        """Asynchronously receive and yield CloudEvents messages from the Redis stream.
184
185        Iterates over messages received from the configured Redis stream, logging each message
186        and yielding a validated ClientCloudEventsJsonRoot object.
187
188        A message that fails Pydantic validation is logged and discarded; it is not
189        yielded and does not stop iteration.
190
191        Yields:
192            ClientCloudEventsJsonRoot: Parsed and validated CloudEvents message object.
193
194        Returns:
195            None
196
197        Complexity:
198            - Time: O(m) per yielded message, where m is the byte length of the received
199              JSON payload; dominated by JSON parsing and Pydantic validation (the
200              generator itself runs indefinitely).
201            - Space: O(m) per yielded message, for the parsed
202              ``ClientCloudEventsJsonRoot`` object held in memory.
203        """
204        async for message_json in self.receive_message():
205            if self._logger.isEnabledFor(level=logging.INFO):
206                self._logger.info(msg=f"Received message: {message_json}")
207            try:
208                yield ClientCloudEventsJsonRoot.model_validate_json(
209                    json_data=message_json, strict=True
210                )
211            except ValidationError:
212                if self._logger.isEnabledFor(level=logging.ERROR):
213                    self._logger.exception(
214                        msg=f"Discarding invalid message: {message_json}"
215                    )
