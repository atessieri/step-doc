  1"""Redis connection settings shared by the STEP applications.
  2
  3This module holds the two halves of a Redis connection description, kept apart so
  4that secrets are never written to a configuration file:
  5
  6    #. RedisConnection: A Pydantic model with the connection settings that belong
  7       to an application configuration file (database, socket timeouts, TCP
  8       keepalive, SSL/TLS, health check, pool timeout, protocol, retry count).
  9    #. RedisCredential: A frozen dataclass with the Redis host, port, username and
 10       password, read from the environment (optionally via a ``.env`` file) by
 11       ``RedisCredential.from_env``.
 12
 13``StepRedisPoolParams.from_connection`` combines both into the parameters of a
 14``StepRedisPool``.
 15
 16Example:
 17    >>> from step_redis.step_redis_config import RedisConnection, RedisCredential
 18    >>> RedisConnection.get_default().socketTimeout
 19    40
 20    >>> RedisCredential(
 21    ...     redis_url="localhost",
 22    ...     redis_port=6379,
 23    ...     redis_username=None,
 24    ...     redis_password=None,
 25    ... ).redis_port
 26    6379
 27
 28Complexity:
 29    - Time: O(1), every operation validates or copies a fixed number of fields
 30      (``RedisCredential.from_env`` also reads the ``.env`` file, if any).
 31    - Space: O(1), every instance holds a fixed number of fields.
 32"""

 46class RedisConnection(BaseModel):
 47    """Redis connection configuration model based on Pydantic's BaseModel.
 48
 49    This class provides a strongly typed configuration for Redis connections
 50    with comprehensive validation rules. It covers the database number, socket
 51    configurations, SSL settings, and connection pool properties; the Redis host,
 52    port and credentials come from the environment (see ``RedisCredential``).
 53    The model enforces validation rules on all fields to
 54    ensure proper configuration. It also includes a utility method to create a default
 55    configuration instance.
 56
 57    Attributes:
 58        db (int): Redis database number to use
 59        socketTimeout (int): Socket timeout in seconds for receive operations (at least 1);
 60            it must exceed the input stream ``waitTimeout`` by at least 5 seconds
 61            (checked by the application configuration that embeds this model)
 62        socketConnectTimeout (int): Socket timeout in seconds for connection operations
 63            (at least 1)
 64        socketKeepIdle (int): Time in seconds before sending first keepalive probe
 65            (1-32767, the Linux ``TCP_KEEPIDLE`` limit)
 66        socketKeepAlive (int): Interval in seconds between keepalive probes (1-32767, the
 67            Linux ``TCP_KEEPINTVL`` limit)
 68        socketKeepCnt (int): Number of keepalive probes before dropping connection (1-127,
 69            the Linux ``TCP_KEEPCNT`` limit)
 70        sslEnable (bool): Whether to use SSL/TLS for the connection
 71        sslClientKeyFile (str | None): Path to the SSL client key file
 72        sslClientKeyFilePassword (str | None): Password for the SSL client key file
 73        sslClientCertFile (str | None): Path to the SSL client certificate file
 74        sslCertRequested (bool): Whether to verify the Redis server's TLS certificate (CA
 75            chain and hostname)

372    @classmethod
373    def from_env(cls, logger: Logger) -> "RedisCredential":
374        """Read the Redis credential from the environment.
375
376        Loads the ``.env`` file found in the current working directory or its
377        nearest parent, if any, then reads ``REDIS_URL``, ``REDIS_PORT``,
378        ``REDIS_USERNAME`` and ``REDIS_PASSWORD``. The username and password are
379        optional and an empty value counts as unset; the URL and port are
380        required. The settings are logged at INFO level, username and password
381        excluded (only whether a username is set). A commented template of the
382        ``.env`` file is ``doc/source/env/.env.redis.examples`` in the repository.
383
384        Args:
385            logger (Logger): Logger used for the INFO line with the settings.
386
387        Returns:
388            RedisCredential: The credential assembled from the environment.
389
390        Raises:
391            ValueError: If ``REDIS_URL`` or ``REDIS_PORT`` is not set, if the port
392                is not an integer, or if the URL is empty or the port out of range.
393
394        Complexity:
395            - Time: O(1), a fixed number of environment variables is read.
396            - Space: O(1), a single fixed-size instance is allocated.
397        """
398        # Without usecwd, find_dotenv() starts from this module's directory (inside
399        # site-packages once installed), not from where the program is launched
400        load_dotenv(dotenv_path=find_dotenv(usecwd=True))
401        redis_url = os.getenv(key=cls._URL_VAR_NAME)
402        if redis_url is None:
403            raise ValueError(f"Environment variable {cls._URL_VAR_NAME} is not set.")
404        redis_port = os.getenv(key=cls._PORT_VAR_NAME)
405        if redis_port is None:
406            raise ValueError(f"Environment variable {cls._PORT_VAR_NAME} is not set.")
407        try:
408            port = int(redis_port)
409        except ValueError as exc:
410            raise ValueError(
411                f"Environment variable {cls._PORT_VAR_NAME} must be an integer, "
412                f"got {redis_port!r}."
413            ) from exc
414        # An empty value counts as unset
415        redis_username = os.getenv(key=cls._USERNAME_VAR_NAME) or None
416        redis_password = os.getenv(key=cls._PASSWORD_VAR_NAME) or None
417        if logger.isEnabledFor(level=logging.INFO):
418            logger.info(
419                msg=(
420                    "Creating RedisCredential with the following settings: "
421                    f"redis_host={redis_url}, redis_port={port}, "
422                    f"redis_auth={redis_username is not None}"
423                )
424            )
425        return cls(
426            redis_url=redis_url,
427            redis_port=port,
428            redis_username=redis_username,
429            redis_password=redis_password,
430        )
