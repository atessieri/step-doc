  1"""Asynchronous Redis Stream interface for message processing with consumer groups.
  2
  3This module provides the StepRedisStream class and StepRedisStreamParams dataclass
  4for working with Redis Streams in an asynchronous context. It supports:
  5
  6    - Creating and managing Redis streams and consumer groups
  7    - Publishing messages with configurable maximum stream length
  8    - Consuming messages with automatic acknowledgment
  9    - Managing stream TTL (Time-To-Live)
 10    - Connection health monitoring via ping
 11
 12The StepRedisStream class is designed to be used as an async context manager.
 13It requires a StepRedisPool, a StepRedisStreamParams, and a Logger.
 14
 15Example:
 16    >>> import asyncio
 17    ... import logging
 18    ... from step_redis.step_redis_pool import StepRedisPool, StepRedisPoolParams
 19    ... from step_redis.step_redis_stream import (
 20    ...     StepRedisStream,
 21    ...     StepRedisStreamParams,
 22    ... )
 23    ...
 24    ...
 25    ... async def main():
 26    ...     logger = logging.getLogger()
 27    ...     pool_params = StepRedisPoolParams(host="localhost", port=6379, db=0, ...)
 28    ...     pool = StepRedisPool(params=pool_params, logger=logger)
 29    ...     config = StepRedisStreamParams(
 30    ...         stream_id="my_stream",
 31    ...         group_id="my_group",
 32    ...         wait_timeout_ms=5000,
 33    ...         max_length=1000,
 34    ...         consumer_name="consumer1",
 35    ...         ttl_seconds=3600,
 36    ...     )
 37    ...     async with StepRedisStream(pool, config, logger) as stream:
 38    ...         await stream.create_input_stream()
 39    ...         await stream.publish_message("Hello, Redis!")
 40    ...         async for message in stream.receive_message():
 41    ...             print(f"Received: {message}")
 42    ...             break
 43    ...
 44    ...
 45    ... asyncio.run(main())
 46
 47Complexity:
 48    - Time: Setup operations (``__init__``, ``__aenter__``) run in O(1). Single-command
 49      stream management (``create_input_stream``, ``update_stream_ttl``,
 50      ``get_stream_ttl``, ``ping``) runs in O(1), each performing one Redis round-trip.
 51      Publishing (``publish_message``) runs in O(k), where k is the length of the
 52      message string, dominated by the Redis XADD round-trip. Receiving
 53      (``receive_message``) runs in O(m) per yielded message, where m is the received
 54      payload size, dominated by the XREADGROUP call and decoding.
 55    - Space: O(1) for setup and single-command operations. O(k) per publish call and
 56      O(m) per received message, bounded by the size of the single message being sent or
 57      parsed at a time; no message history is retained in memory.
 58"""

 78@dataclass(frozen=True, kw_only=True, slots=True)
 79class StepRedisStreamParams:
 80    """Configuration parameters for StepRedisStream.
 81
 82    This dataclass defines the required parameters for configuring a Redis stream
 83    with consumer groups. All fields are validated for correct usage.
 84
 85    Attributes:
 86        stream_id (str): Identifier for the Redis stream (at least 1 character).
 87        group_id (str | None): Identifier for the Redis consumer group (at least 1 character, optional).
 88        wait_timeout_ms (int | None): Timeout in milliseconds for waiting to receive a message (non-negative, optional).
 89        max_length (int | None): Maximum length of the stream (positive integer, defaults to 1). Note:
 90            leaving this at its default of 1 trims the stream to (approximately) a single entry on every
 91            publish, which discards backlog for any multi-consumer or replay scenario; set it explicitly
 92            to a larger value unless single-entry retention is actually the desired behavior.
 93        consumer_name (str): Name of the consumer (at least 1 character).
 94        ttl_seconds (int | None): Time-to-live in seconds for the stream (positive integer if provided, optional).
 95            When both ``ttl_seconds`` and ``wait_timeout_ms`` are provided, ``ttl_seconds`` must be strictly
 96            greater than ``wait_timeout_ms / 1000``; otherwise Redis can expire and delete the stream while
 97            ``receive_message`` is still blocked waiting on it.
 98
 99    Note:
100        The dataclass is immutable (frozen) and uses slots for memory efficiency.
101    """
102
103    stream_id: str
104    group_id: str | None = None
105    wait_timeout_ms: int | None = None
106    max_length: int | None = 1
107    consumer_name: str
108    ttl_seconds: int | None = None

273        if self._logger.isEnabledFor(level=logging.DEBUG):
274            self._logger.debug(msg="Redis stream context manager exited")
275        await self._redis_pool.aclose()

277    async def create_input_stream(self) -> None:
278        """Create a Redis stream and consumer group if they do not exist.
279
280        This method attempts to create a Redis consumer group for the configured stream.
281        If the group already exists, it logs an informational message and continues.
282        If a TTL is set in the configuration, it updates the stream's expiration.
283
284        Raises:
285            ValueError: If ``group_id`` is not provided in the configuration.
286
287        Returns:
288            None
289
290        Complexity:
291            - Time: O(1), a single Redis XGROUP CREATE call plus, when a TTL is
292              configured, one additional O(1) EXPIRE call.
293            - Space: O(1), no additional data structures are allocated.
294        """
295        if self._config.group_id is None:
296            raise ValueError("group_id must be provided to create a consumer group")
297        try:
298            response = await self._redis_pool.xgroup_create(
299                name=self._config.stream_id,
300                groupname=self._config.group_id,
301                mkstream=True,
302            )
303            if self._logger.isEnabledFor(level=logging.DEBUG):
304                self._logger.debug(
305                    msg=f"Created input stream {response}: {self._config.stream_id} with group {self._config.group_id}"
306                )
307        except RedisResponseError as e:
308            # redis-py does not expose a structured error code for XGROUP CREATE,
309            # so "group already exists" can only be distinguished by matching the
310            # server's error-message prefix. This is brittle across Redis
311            # versions/forks (e.g. KeyDB, Valkey) but is the only signal available.
312            if str(object=e).startswith("BUSYGROUP"):
313                if self._logger.isEnabledFor(level=logging.INFO):
314                    self._logger.info(
315                        msg=f"Consumer group '{self._config.group_id}' already exists"
316                    )
317            else:
318                raise
319        if self._config.ttl_seconds is not None:
320            await self.update_stream_ttl(ttl_seconds=self._config.ttl_seconds)

322    async def update_stream_ttl(self, ttl_seconds: int) -> bool:
323        """Update the Time-To-Live (TTL) expiration for the configured Redis stream.
324
325        This method updates the TTL expiration for the Redis stream specified in the configuration.
326        It uses the Redis EXPIRE command to set or replace the TTL for the stream. If the TTL is
327        successfully updated, it returns True. If the stream does not exist or the TTL could not
328        be set, it returns False.
329
330        Args:
331            ttl_seconds (int): The number of seconds until the stream should expire.
332                Must be a positive integer.
333
334        Returns:
335            bool: True if the TTL was successfully updated, False otherwise.
336
337        Complexity:
338            - Time: O(1), a single Redis EXPIRE call.
339            - Space: O(1), no additional data structures are allocated.
340        """
341        if ttl_seconds <= 0:
342            raise ValueError(f"TTL seconds must be a positive integer: {ttl_seconds}")
343        if self._logger.isEnabledFor(level=logging.DEBUG):
344            self._logger.debug(
345                msg=f"Updating TTL to {ttl_seconds} seconds for stream {self._config.stream_id}"
346            )
347        # The Redis EXPIRE command returns 1 if the timeout was set, 0 if the key doesn't exist
348        result = int(
349            await self._redis_pool.expire(name=self._config.stream_id, time=ttl_seconds)
350        )
351        if result != 1:
352            if self._logger.isEnabledFor(level=logging.WARNING):
353                self._logger.warning(
354                    msg=f"Failed to update TTL for stream {self._config.stream_id}, unexpected error"
355                )
356            return False
357        if self._logger.isEnabledFor(level=logging.DEBUG):
358            self._logger.debug(
359                msg=f"Successfully updated TTL for stream {self._config.stream_id}"
360            )
361        return True

363    async def get_stream_ttl(self) -> int:
364        """Get the remaining Time-To-Live (TTL) in seconds for the configured Redis
365        stream.
366
367        This method retrieves the TTL for the stream specified in the configuration.
368        It uses the Redis TTL command, which returns the remaining time to live for a key in seconds.
369
370        Returns:
371            int: The number of seconds until the stream expires. Returns -2 if the stream does not exist.
372            Returns -1 if the stream exists but has no expiration set.
373
374        Note:
375            A return value of -2 indicates the stream does not exist.
376            A return value of -1 indicates the stream exists but has no associated expiration.
377
378        Complexity:
379            - Time: O(1), a single Redis TTL call.
380            - Space: O(1), no additional data structures are allocated.
381        """
382        if self._logger.isEnabledFor(level=logging.DEBUG):
383            self._logger.debug(msg=f"Getting TTL for stream {self._config.stream_id}")
384        ttl = int(await self._redis_pool.ttl(name=self._config.stream_id))
385        if self._logger.isEnabledFor(level=logging.DEBUG):
386            if ttl == -2:  # noqa: PLR2004
387                self._logger.debug(
388                    msg=f"Stream {self._config.stream_id} does not exist"
389                )
390            elif ttl == -1:
391                self._logger.debug(
392                    msg=f"Stream {self._config.stream_id} exists but has no expiration set"
393                )
394            else:
395                self._logger.debug(
396                    msg=f"Stream {self._config.stream_id} will expire in {ttl} seconds"
397                )

422    async def publish_message(self, message: str) -> None:
423        """Publish a message to the configured Redis stream.
424
425        This method publishes a message to the Redis stream specified in the configuration.
426        The message is stored as a field named 'payload' in the stream entry. The stream
427        is trimmed to the maximum length specified in the configuration using approximate
428        trimming for performance.
429
430        Args:
431            message (str): The message to publish.
432
433        Raises:
434            ValueError: If ``max_length`` is not provided in the configuration.
435
436        Returns:
437            None
438
439        Complexity:
440            - Time: O(k), where k is the length of ``message``; dominated by the network
441              round-trip of the Redis XADD call (approximate trimming keeps the trim

443            - Space: O(k), for the message payload held in memory before being sent.

462    async def receive_message(self) -> AsyncGenerator[str]:  # noqa: C901, PLR0912
463        """Asynchronously yield messages from the configured Redis stream using consumer
464        groups.
465
466        This async generator continuously reads messages from the Redis stream using the
467        configured consumer group and consumer name. It blocks for the specified timeout
468        (in milliseconds) on each attempt. If a message is received, its payload is yielded.
469        If a TTL is configured, the stream's TTL is updated after each attempt.
470
471        Yields:
472            str: The payload of each received message.
473
474        Returns:
475            None
476
477        Note:
478            This method uses ``noack=True``, so messages are automatically acknowledged
479            upon delivery and not added to the pending entries list.
480
481        Complexity:
482            - Time: O(m) per yielded message, where m is the byte length of the received
483              message payload; dominated by the Redis XREADGROUP call and string
484              decoding (the generator itself runs indefinitely).
485            - Space: O(m) per yielded message, for the single message payload held in
486              memory; no message history is retained.
487        """
488        if self._config.group_id is None:
489            raise ValueError(
490                "group_id must be provided in the configuration to receive messages"
491            )
492        if self._config.wait_timeout_ms is None:
493            raise ValueError(
494                "wait_timeout_ms must be provided in the configuration to receive messages"
495            )
496        while True:
497            try:
498                response = await self._redis_pool.xreadgroup(
499                    groupname=self._config.group_id,
500                    consumername=self._config.consumer_name,
501                    streams={self._config.stream_id: ">"},
502                    block=self._config.wait_timeout_ms,
503                    count=1,
504                    noack=True,
505                )
506                if self._logger.isEnabledFor(level=logging.DEBUG):
507                    self._logger.debug(msg=f"Received response: {response}")
508                stream_entries = self._extract_stream_entries(response=response)
509                if stream_entries:
510                    entry = stream_entries[0]
511                    message_id = str(object=entry[0])
512                    message = str(object=entry[1]["payload"])
513                    if self._logger.isEnabledFor(level=logging.DEBUG):
514                        self._logger.debug(
515                            msg=f"Received message {message_id} from Redis"
516                        )
517                    yield message
518                elif self._logger.isEnabledFor(level=logging.DEBUG):
519                    self._logger.debug(
520                        msg=f"No stream {self._config.stream_id} in the response"
521                    )
522            except RedisTimeoutError as e:
523                if self._logger.isEnabledFor(level=logging.DEBUG):
524                    self._logger.debug(msg=f"RedisTimeoutError: {e}")
525            except RedisResponseError as e:
526                # As with the BUSYGROUP check in create_input_stream, redis-py
527                # exposes no structured error code, so NOGROUP (the one case
528                # recoverable by recreating the group) can only be
529                # distinguished by the server's error-message prefix.
530                if not str(object=e).startswith("NOGROUP"):
531                    raise
532                if self._logger.isEnabledFor(level=logging.ERROR):
533                    self._logger.exception(
534                        msg=(
535                            f"NOGROUP error while reading stream "
536                            f"{self._config.stream_id}. Recreating the consumer "
537                            "group and retrying."
538                        )
539                    )
540                await self.create_input_stream()
541            except KeyError as e:
542                if self._logger.isEnabledFor(level=logging.WARNING):
543                    self._logger.warning(msg=f"KeyError: {e}")
544            finally:
545                if self._config.ttl_seconds is not None:
546                    await self.update_stream_ttl(ttl_seconds=self._config.ttl_seconds)

548    def _extract_stream_entries(
549        self, response: XReadGroupResponse | None
550    ) -> list[Any] | None:
551        """Normalize the ``XREADGROUP`` response into a list of stream entries.
552
553        The wire/Python shape of the response from ``redis-py`` depends on the
554        configured RESP ``protocol``: with the default ``legacy_responses=True``,
555        ``protocol=2`` yields ``list[[stream_name, entries]]`` where ``entries`` is
556        already the flat list of ``(id, fields)`` tuples, while ``protocol=3``
557        yields ``dict[stream_name, [entries]]`` where the entries list is wrapped
558        in one extra single-element list (redis-py's own RESP3 parsing adds this
559        extra layer that RESP2 does not have). A blocking read that times out with
560        no data typically returns ``None``. This method normalizes all of these
561        shapes and returns only the flat entries list for the configured stream,
562        or ``None`` if there are none.
563
564        Args:
565            response (XReadGroupResponse | None): The raw response returned by
566                ``redis-py``'s ``xreadgroup`` call.
567
568        Returns:
569            list[Any] | None: The entries for the configured stream, or ``None`` if
570                the response contains no entries for it.
571
572        Complexity:
573            - Time: O(s), where s is the number of streams in the response
574              (typically 1, since only one stream is requested).
575            - Space: O(1), no additional data structures are allocated beyond the
576              returned reference.
577        """
578        if response is None:
579            return None
580        if isinstance(response, dict):
581            entries_wrapper = response.get(self._config.stream_id)
582            return entries_wrapper[0] if entries_wrapper else None  # type: ignore[return-value]
583        for stream_name, entries in response:
584            if stream_name == self._config.stream_id:
585                return entries or None
586        return None
