  1"""Redis connection pool management for asyncio applications.
  2
  3This module provides a specialized Redis connection pool implementation that extends the
  4BlockingConnectionPool from redis-py with enhanced features for asyncio applications. It
  5offers both SSL and non-SSL connections to Redis servers with configurable TCP keepalive
  6settings to maintain persistent connections, particularly useful for long-running services.
  7
  8The module consists of two main components:
  9
 10    #. StepRedisPoolParams: A dataclass that encapsulates all configuration parameters for
 11    the Redis connection pool, including network settings, authentication credentials,
 12    SSL/TLS configuration, and pool behavior settings.
 13
 14    #. StepRedisPool: A connection pool implementation that leverages the asyncio capabilities
 15    of redis-py and provides context management for proper resource cleanup through the
 16    async context manager protocol.
 17
 18Key Features:
 19
 20    - Comprehensive SSL/TLS support for secure Redis connections
 21    - TCP keepalive configuration for connection stability
 22    - Configurable connection limits and timeouts
 23    - Proper resource management through async context managers
 24    - Automatic response decoding
 25    - Integrated logging support
 26
 27Example:
 28    >>> import asyncio
 29    ... from logging import getLogger
 30    ... from step_redis.step_redis_pool import StepRedisPoolParams, StepRedisPool
 31    ...
 32    ...
 33    ... async def main():
 34    ...     # Configure Redis connection parameters
 35    ...     params = StepRedisPoolParams(
 36    ...         host="redis.example.com",
 37    ...         port=6379,
 38    ...         username="user",  # Optional
 39    ...         password="password",
 40    ...         db=0,
 41    ...         ssl_enable=True,
 42    ...         # ... other configuration parameters
 43    ...     )
 44    ...
 45    ...     logger = getLogger()
 46    ...
 47    ...     # Use the connection pool with async context manager
 48    ...     async with StepRedisPool(params=params, logger=logger) as pool:
 49    ...         # Create a Redis client using the pool
 50    ...         redis = Redis.from_pool(pool)
 51    ...         # Perform Redis operations
 52    ...         await redis.set("key", "value")
 53    ...         value = await redis.get("key")
 54    ...
 55    ...
 56    ... asyncio.run(main())
 57
 58Complexity:
 59    - Time: Pool construction (``StepRedisPool.__init__``) and context-manager entry
 60      (``__aenter__``) run in O(1), building fixed-size keepalive/SSL option mappings
 61      with no network I/O. Context-manager exit (``__aexit__``) runs in O(c), where c is
 62      the number of connections currently held by the pool, since ``aclose`` disconnects
 63      each pooled connection.
 64    - Space: O(1) for pool construction and entry, storing only ``params`` and
 65      ``logger`` references. O(c) transiently during teardown, bounded by
 66      ``max_connections``; no per-message data is retained.
 67"""

 85@dataclass(kw_only=True, frozen=True, slots=True)
 86class StepRedisPoolParams:
 87    """Parameters for configuring a Redis connection pool.
 88
 89    Attributes:
 90        host (str): The Redis server host.
 91        port (int): The Redis server port.
 92        username (str | None): The Redis username for authentication, if any.
 93        password (str | None): The Redis password for authentication.
 94        db (int): The Redis database index.
 95        socket_timeout (int): Socket timeout in seconds.
 96        socket_connect_timeout (int): Socket connection timeout in seconds.
 97        socket_keep_idle (int): TCP keepalive idle time (``TCP_KEEPIDLE``/``TCP_KEEPALIVE``),
 98            in seconds, before the first keepalive probe is sent.
 99        socket_keep_alive (int): TCP keepalive probe interval in seconds (``TCP_KEEPINTVL``),
100            i.e. the time between successive keepalive probes. Despite the name, this does
101            not toggle keepalive on/off: TCP keepalive is always enabled by this pool
102            (``socket_keepalive=True``) regardless of this value.
103        socket_keep_cnt (int): Maximum number of TCP keepalive probes.
104        ssl_enable (bool): Whether to enable SSL/TLS connection.
105        ssl_client_key_file (str | None): Path to client's private key file.
106        ssl_client_key_file_password (str | None): Password for the client's private key file.
107        ssl_client_cert_file (str | None): Path to client's certificate file.
108        ssl_cert_requested (bool): Whether the client verifies the server's certificate.
109            When ``True``, the connection requires and validates the server certificate
110            (``ssl_cert_reqs="required"``) and checks the server hostname
111            (``ssl_check_hostname=True``). When ``False``, both server-certificate
112            verification and hostname checking are disabled entirely (not just
113            client-certificate/mTLS behavior) - the connection remains encrypted but is
114            not authenticated and is vulnerable to man-in-the-middle attacks. Only set
115            this to ``False`` in trusted/test environments.
116        ssl_ca_certs_file (str | None): Path to CA certificates file.
117        health_check_interval (int): Interval in seconds for health checks.
118        max_connections (int): Maximum number of connections in the pool.
119        connection_available_timeout (int | None): Maximum time to wait for a connection to be
120            available.
121        protocol (int): Redis protocol version to use.
122        retry_count (int): Maximum number of retries for a Redis command (``PING``,
123            ``XADD``, ``XREADGROUP``, etc.) after a ``ConnectionError`` - notably
124            the Redis server closing a connection that has been idle longer than
125            its configured ``timeout`` (see ``redis.conf``). Each retry
126            transparently reconnects and waits using an equal-jitter exponential
127            backoff (redis-py's ``default_backoff()``) before the error is
128            finally raised to the caller. Deliberately scoped to
129            ``ConnectionError`` only (not ``TimeoutError``): retrying a
130            ``TimeoutError`` on a long-blocking command such as
131            ``XREADGROUP ... BLOCK <ms>`` would silently re-issue the full block
132            duration on every attempt, which can starve callers that rely on a
133            timely ``TimeoutError`` (e.g. to refresh a stream's TTL between
134            polls) and is not needed anyway, since a plain read timeout does not
135            imply the connection itself is broken. Set to 0 to disable retries
136            and preserve the previous fail-fast behavior.
137
138    Complexity:
139        - Time: O(1), the dataclass has no methods beyond the auto-generated
140          ``__init__`` (which assigns each field directly, since it is frozen
141          and has no defaults); no ``__post_init__`` is defined on this class.
142        - Space: O(1), stores only the fixed set of scalar/optional configuration
143          fields.
144    """

169class StepRedisPool(BlockingConnectionPool):
170    """A custom Redis connection pool class that extends BlockingConnectionPool and
171    provides context management capabilities with async support. This class provides a
172    blocking connection pool for Redis connections with support for both SSL and non-SSL
173    connections. It configures connection parameters based on the provided
174    StepRedisPoolParams and includes TCP keepalive settings for maintaining persistent
175    connections. The class implements the asynchronous context manager protocol
176    (`__aenter__` and `__aexit__`), allowing it to be used with the async with statement
177    for proper resource management.
178
179    Attributes:
180        params (StepRedisPoolParams): Configuration parameters for the Redis pool including:
181
182            - Connection details (host, port, db, username, password)
183            - SSL settings (ssl_enable, ssl_keyfile, ssl_certfile, ssl_ca_certs_file, etc.)
184            - Socket settings (timeouts, keepalive options)
185            - Pool settings (max_connections, health_check_interval)
186            - Protocol and response decoding options
187
188        logger (Logger): Logger instance for logging connection pool events and errors.
189
190    Example:
191        >>> redis_params = StepRedisPoolParams(
192        ...    host="redis.example.com",
193        ...    port=6379,
194        ...    ssl_enable=True,
195        ...    # ... other parameters
196        ... logger = Logger("redis_connection")
197        ... async with StepRedisPool(params=redis_params, logger=logger) as redis_pool:
198        ...    # Use the redis pool here
199        ...    # Connections are automatically managed and cleaned up
200
201    Note:
202        - When SSL is enabled, the pool uses SSLConnection and configures all required SSL
203          parameters
204        - When SSL is disabled, the pool uses standard Connection
205        - TCP keepalive settings are configured for both connection types
206        - All responses are automatically decoded (decode_responses=True)
207        - ``health_check_interval`` (the interval in seconds between active
208          health-check pings, copied from ``params.health_check_interval`` at
209          construction time) is exposed read-only via the
210          ``health_check_interval`` property below, documented there rather
211          than in this class's ``Attributes:`` to avoid a duplicate Sphinx
212          object description. The remaining fields of ``params`` (host, port,
213          credentials, SSL settings, timeouts, etc.) are passed through to
214          ``BlockingConnectionPool.__init__`` and are not retained on this
215          instance.
216
217    Complexity:
218        - Time: O(c), the worst case, driven by ``__aexit__``, where c is the number
219          of connections currently held by the pool (``aclose`` disconnects each
220          one). Every other method (``__init__``, ``health_check_interval``,
221          ``__aenter__``) runs in O(1).
222        - Space: O(1) across all methods; state is limited to ``params``, the
223          logger, and a small fixed-size keepalive options mapping.
224    """
225
226    __slots__ = ("_health_check_interval", "_logger")
227
228    def __init__(self, params: StepRedisPoolParams, logger: Logger) -> None:
229        """Initialize the Redis connection pool with the provided parameters.
230
231        This constructor creates either an SSL-enabled or standard Redis connection pool
232        based on the configuration parameters. It configures connection timeouts, keepalive
233        settings, and other connection-specific options.
234
235        Args:
236            params (StepRedisPoolParams): Configuration parameters for the Redis pool including:
237
238                - Connection details (host, port, db, username, password)
239                - SSL settings (ssl_enable, ssl_keyfile, ssl_certfile, ssl_ca_certs_file, etc.)
240                - Socket settings (timeouts, keepalive options)
241                - Pool settings (max_connections, health_check_interval)
242                - Protocol and response decoding options
243
244            logger (Logger): Logger instance for logging connection pool events and errors.
245
246        Returns:
247            None
248
249        Note:
250            - When SSL is enabled, uses SSLConnection with TLSv1.2 minimum version
251            - Socket keepalive is always enabled with configurable TCP parameters
252            - Response decoding is enabled by default
253            - Health checks are performed at the specified interval
254            - Failed commands (PING, XADD, XREADGROUP) are retried up to
255              ``params.retry_count`` times with jittered exponential backoff on
256              ``ConnectionError`` only, transparently reconnecting - this covers
257              the Redis server's idle-connection timeout scenario. ``TimeoutError``
258              is deliberately excluded so a long-blocking ``XREADGROUP ... BLOCK``
259              read timeout still surfaces immediately to the caller instead of
260              silently re-blocking for the full duration on every retry
261
262        Complexity:
263            - Time: O(1), builds a fixed number of keepalive options and delegates to
264              ``BlockingConnectionPool.__init__``; no connections are opened yet.
265            - Space: O(1), stores only the logger reference, the health-check interval,
266              and a small fixed-size keepalive options mapping.
267        """
268        self._health_check_interval = params.health_check_interval
269        self._logger = logger
270        if sys.platform == "darwin":
271            # macOS uses TCP_KEEPALIVE (not TCP_KEEPIDLE) for the idle-time option.
272            socket_keepalive_options = {
273                socket.TCP_KEEPALIVE: params.socket_keep_idle,  # type: ignore[attr-defined]  # pylint: disable=no-member
274                socket.TCP_KEEPINTVL: params.socket_keep_alive,
275                socket.TCP_KEEPCNT: params.socket_keep_cnt,
276            }
277        else:
278            # TCP_KEEPIDLE is the POSIX/Linux/Windows option name and is correct
279            # for every non-macOS platform (win32, linux, and other
280            # TCP_KEEPIDLE-supporting platforms), rather than assuming only
281            # win32/linux behave this way and defaulting everything else to the
282            # macOS-only TCP_KEEPALIVE option.
283            socket_keepalive_options = {
284                socket.TCP_KEEPIDLE: params.socket_keep_idle,
285                socket.TCP_KEEPINTVL: params.socket_keep_alive,
286                socket.TCP_KEEPCNT: params.socket_keep_cnt,
287            }
288        retry = Retry(
289            backoff=default_backoff(),  # type: ignore[no-untyped-call]
290            retries=params.retry_count,
291            supported_errors=(RedisConnectionError,),
292        )
293        if params.ssl_enable:
294            super().__init__(
295                connection_class=SSLConnection,
296                host=params.host,
297                port=params.port,
298                db=params.db,
299                username=params.username,
300                password=params.password,
301                socket_timeout=params.socket_timeout,
302                socket_connect_timeout=params.socket_connect_timeout,
303                socket_keepalive=True,
304                socket_keepalive_options=socket_keepalive_options,
305                retry=retry,
306                ssl_keyfile=params.ssl_client_key_file,
307                ssl_password=params.ssl_client_key_file_password,
308                ssl_certfile=params.ssl_client_cert_file,
309                ssl_cert_reqs=("required" if params.ssl_cert_requested else None),
310                ssl_ca_certs=params.ssl_ca_certs_file,
311                ssl_check_hostname=params.ssl_cert_requested,
312                ssl_min_version=ssl.TLSVersion.TLSv1_2,
313                max_connections=params.max_connections,
314                timeout=params.connection_available_timeout,
315                health_check_interval=params.health_check_interval,
316                protocol=params.protocol,
317                decode_responses=True,
318            )
319        else:
320            super().__init__(
321                connection_class=Connection,
322                host=params.host,
323                port=params.port,
324                db=params.db,
325                username=params.username,
326                password=params.password,
327                socket_timeout=params.socket_timeout,
328                socket_connect_timeout=params.socket_connect_timeout,
329                socket_keepalive=True,
330                socket_keepalive_options=socket_keepalive_options,
331                retry=retry,
332                max_connections=params.max_connections,
333                protocol=params.protocol,
334                health_check_interval=params.health_check_interval,
335                timeout=params.connection_available_timeout,
336                decode_responses=True,
