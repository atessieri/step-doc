 143async def _async_iter(items: list[Any], exc: Exception | None) -> AsyncGenerator[Any]:
 144    """Yield each item of ``items``, then raise ``exc`` if given.
 145
 146    Args:
 147        items (list[Any]): The items to yield.
 148        exc (Exception | None): The exception raised once the items are exhausted,
 149            or ``None`` to end normally.
 150
 151    Yields:
 152        Any: Each item of ``items``.
 153
 154    Raises:
 155        Exception: ``exc``, if it is not ``None``.
 156    """
 157    for item in items:
 158        yield item
 159    if exc is not None:
 160        raise exc
