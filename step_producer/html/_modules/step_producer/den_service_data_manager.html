  1"""DenServiceDataManager Module.
  2
  3This module provides the ``DenServiceDataManager`` class for managing
  4Decentralized Environmental Notification Message (DENM) data related
  5to collision risk events. It is designed for use in Intelligent
  6Transport Systems (ITS) and handles the creation, storage, updating, and
  7expiration of DENM records. The manager maintains efficient indexes for
  8quick retrieval and manipulation of DENM records, and integrates with
  9logging for debugging and monitoring.
 10
 11Thread Safety:
 12    No locking is used around the internal indexes. All methods are
 13    synchronous (they never await), so one instance can be shared by several
 14    asyncio tasks of the same event loop, as StepProducerApp does through its
 15    DenServiceManager; it must not be shared between threads.
 16
 17Key Features:
 18    - Creation and update of DENM records for collision scenarios
 19    - Indexing for fast lookup by record key (reference/involved actor pair)
 20      and by ActionId, plus a reference-time ordered SortedList for expiry
 21    - Expiration and cleanup of outdated DENM records
 22    - Integration with ITS data models and logging
 23
 24Complexity:
 25    - Time: O(k log k + k log N), dominated by
 26      DenServiceDataManager.insert_station_collision_array (station sort
 27      plus per-station SortedList insert/remove), k = number of stations
 28      in the collision array processed, N = total tracked DENM records.
 29    - Space: O(k), dominated by the same method's returned DENM message
 30      list, aside from the O(N) growth of the internal indexes across all
 31      currently tracked records.
 32
 33Classes:
 34    - DenmRecordInfo: Frozen dataclass pairing a DenmDataV1 record with its
 35      reference time and creation-time monotonic timestamp. Complexity:
 36      time O(1), space O(1).
 37    - DenServiceDataManager: Creates, indexes, updates, and expires DENM
 38      records for a single station. Complexity: time O(k log k + k
 39      log N), space O(k), k = number of stations in a processed collision
 40      array, N = total tracked DENM records (worst case, in
 41      insert_station_collision_array).
 42"""

 79@dataclass(frozen=True, kw_only=True, slots=True)
 80class DenmRecordInfo:
 81    """DenmRecordInfo holds DENM data and its associated reference time.
 82
 83    The reference time and the monotonic creation time are derived in
 84    ``__post_init__`` and cannot be passed to the constructor: the reference time
 85    is parsed from ``denm_data.managementContainer.referenceTime``, the monotonic
 86    time is read from ``time.monotonic()``.
 87
 88    Attributes:
 89        denm_data (DenmDataV1): The DENM data object containing event information.
 90        record_key (RecordKey): The unique key identifying this record's
 91            reference/involved actor pair, computed once when the record is
 92            created and reused for removal, avoiding any need to
 93            reconstruct it later from (schema-Optional) fields on denm_data.
 94
 95    Complexity:
 96        - Time: O(1), dominated by __post_init__ (one ITS time parse and a
 97          monotonic() call).
 98        - Space: O(1), the four fields are fixed size (aside from the
 99          referenced DenmDataV1).
100    """
101
102    denm_data: DenmDataV1
103    record_key: RecordKey
104    reference_time: ItsTime = field(init=False)
105    monotonic_time_sec: int = field(init=False)

107    def __post_init__(self) -> None:
108        """Derive the reference time and the monotonic creation time.
109
110        Returns:
111            None: The ``reference_time`` and ``monotonic_time_sec`` fields are
112                set in place on the frozen instance.
113
114        Complexity:
115            - Time: O(1), one ITS time parse and a monotonic() call.
116            - Space: O(1), a single new ItsTime.
117        """
118        object.__setattr__(
119            self,
120            "reference_time",
121            ItsTime.create_from_iso_time(
122                time_str=self.denm_data.managementContainer.referenceTime
123            ),
124        )
125        object.__setattr__(self, "monotonic_time_sec", int(monotonic()))

128class DenServiceDataManager:
129    """Manages the DENM records of the collision risk events of one station.
130
131    This class is responsible for creating, storing, updating and
132    managing DENM records related to potential collisions. It maintains various indexes
133    for efficient retrieval and processing of DENM records. DenServiceDataManager
134    organizes collision data into structured records, handles the lifecycle of these
135    records including creation, update and expiration, and provides methods to access
136    and manipulate the stored data.
137
138    Attributes:
139        station_id (int): The unique identifier for the station generating these DENM messages.
140        time_after_collision_sec (int): Time to continue broadcasting a DENM after a collision, in seconds.
141        denm_last_update_max_delay_ms (int): Maximum allowed delay, in milliseconds, for a DENM record's reference time before it is treated as expired.
142        logger (Logger): Logger instance for recording operations and debugging information.
143

262    def _create_denm_record_info(
263        self,
264        *,
265        action_id: ActionId,
266        record_key: RecordKey,
267        detection_time: str,
268        station_collision_array: StationCollisionArrayV1,
269        position_reference: PositionInfo,
270        position_reference_time: ItsTime,
271        station: PositionInfo,
272    ) -> DenmRecordInfo:
273        """Creates the DENM record of a collision risk between two stations.
274
275        This method constructs a DENM record containing information about a potential collision
276        between vehicles. The DENM includes management data, situation information, location
277        details, and pre-crash information. The situation's event type is the collision
278        risk cause code, with the sub cause code selected by the type of the involved
279        station (see ``_COLLISION_RISK_SUB_CAUSE_BY_STATION_TYPE``; UNAVAILABLE for a
280        type it does not map).
281
282        Args:
283            action_id (ActionId): Action identifier for the DENM
284            record_key (RecordKey): The unique key identifying this record's
285                reference/involved actor pair, stored on the returned
286                DenmRecordInfo for later removal.
287            detection_time (str): The ITS time of the detection event
288            station_collision_array (StationCollisionArrayV1): Information about the
289                collision scenario
290            position_reference (PositionInfo): Position information of the reference station
291            position_reference_time (ItsTime): The ITS time of the position reference station
292            station (PositionInfo): Position information of the other involved station
293
294        Returns:
295            DenmRecordInfo: A complete DENM record containing all necessary information about the
296                potential collision event
297
298        Complexity:
299            - Time: O(1), constructs a fixed number of nested Pydantic
300              models from the given arguments.
301            - Space: O(1), aside from the newly constructed record.
302        """
303        estimated_collision_time = ItsTime.create_from_iso_time(
304            time_str=station_collision_array.estimatedCollisionTime
305        )
306        time_to_collision_sec = (
307            estimated_collision_time.time_difference(other_time=position_reference_time)
308            / ItsTime.SEC_TO_MS_FACTOR
309        )
310        sub_cause_code = self._COLLISION_RISK_SUB_CAUSE_BY_STATION_TYPE.get(
311            station.stationInfo.stationType, CollisionRiskSubCauseCode.UNAVAILABLE
312        )
313        return DenmRecordInfo(
314            record_key=record_key,
315            denm_data=DenmDataV1(
316                type=MessageId.DENM_PDU,
317                stationId=self._station_id,
318                managementContainer=ManagementContainerDenm(
319                    actionId=action_id,
320                    detectionTime=detection_time,
321                    referenceTime=position_reference.time,
322                    termination=None,
323                    eventPosition=position_reference.position,
324                    awarenessDistance=None,
325                    trafficDirection=None,
326                    validityDuration=max(0, ceil(time_to_collision_sec))
327                    + self._time_after_collision_sec,
328                    transmissionInterval=None,
329                    stationType=position_reference.stationInfo.stationType,
330                    referenceStationId=position_reference.stationInfo.stationId,
331                    referenceObjectId=position_reference.objectId,
332                ),
333                situationContainer=SituationContainer(
334                    informationQuality=station_collision_array.informationQuality,
335                    eventType=CollisionRiskCause(
336                        causeCode=CauseCode.COLLISION_RISK, subCauseCode=sub_cause_code
337                    ),
338                ),
339                locationContainer=LocationContainer(
340                    eventSpeed=position_reference.speed,
341                    eventPositionHeading=position_reference.heading,
342                    roadType=station_collision_array.roadType,
343                ),
344                preCrashContainer=PreCrashContainer(
345                    perceivedObject=PerceivedObject(
346                        objectId=station.objectId,
347                        objectType=station.stationInfo.stationType,
348                        # Station type confidence allows 0, object type confidence
349                        # starts at 1: map 0 to None (unavailable)
350                        objectTypeConfidence=station.stationInfo.stationTypeConfidence
351                        or None,
352                        measurementTime=station.time,
353                        position=station.position,
354                        speed=station.speed,
355                        heading=station.heading,
356                        objectDimensionZ=None,
357                        objectDimensionY=None,
358                        objectDimensionX=None,
359                        objectPerceptionQuality=None,
360                        positionPredicted=None,
361                    ),
362                    stationIdInvolved=station.stationInfo.stationId,
363                    timeToCollision=time_to_collision_sec,
364                    impactSection=None,
365                    estimatedBrakingDistance=station_collision_array.estimatedBrakingDistance,
366                    estimatedCollisionLatitude=station_collision_array.estimatedCollisionLatitude,
367                    estimatedCollisionLongitude=station_collision_array.estimatedCollisionLongitude,
368                ),
369            ),
370        )

372    def _create_action_id(self, record_key: RecordKey) -> ActionId:
373        """Generates or retrieves an ActionId for the given record key.
374
375        If the record key already exists in the internal index, returns the
376        associated ActionId. Otherwise, creates a new ActionId from the station ID
377        and the next free sequence number: the sequence number wraps around after
378        ``ActionId.SEQUENCE_NUMBER_MAX`` and skips the values still held by a
379        tracked record, so a live ActionId is never reissued.
380
381        Args:
382            record_key (RecordKey): The key identifying the record.
383
384        Returns:
385            ActionId: The action identifier associated with the record key.
386
387        Raises:
388            OverflowError: If every sequence number is held by a tracked record.
389
390        Complexity:
391            - Time: O(1) average, a dict lookup plus, for a new key, one candidate
392              per sequence number already in use (O(S) worst case, S =
393              ``ActionId.SEQUENCE_NUMBER_MAX + 1``, when the index is full).
394            - Space: O(1), aside from the newly created ActionId.
395        """
396        if record_key in self._record_key_index:
397            action_id = self._record_key_index[
398                record_key
399            ].denm_data.managementContainer.actionId
400            if self._logger.isEnabledFor(level=logging.DEBUG):
401                self._logger.debug(
402                    msg=f"Found existing record {record_key} action id: {action_id}"
403                )
404            return action_id
405        # Skip the sequence numbers still held by a tracked record, so a wrap-around
406        # never reissues a live ActionId
407        for _ in range(ActionId.SEQUENCE_NUMBER_MAX + 1):
408            action_id = ActionId(
409                originatingStationId=self._station_id,
410                sequenceNumber=self._sequence_number,
411            )
412            # Wrap around after the schema's maximum sequence number
413            self._sequence_number = (self._sequence_number + 1) % (
414                ActionId.SEQUENCE_NUMBER_MAX + 1
415            )
416            if action_id not in self._action_id_index:
417                if self._logger.isEnabledFor(level=logging.DEBUG):
418                    self._logger.debug(
419                        msg=f"New record {record_key} action id: {action_id}"
420                    )
421                return action_id
422        raise OverflowError(
423            f"All {ActionId.SEQUENCE_NUMBER_MAX + 1} action ids are in use"
424        )

457    def insert_station_collision_array(
458        self, station_collision_array: StationCollisionArrayV1
459    ) -> list[DenmDataV1]:
460        """Creates or updates the DENM records of a station collision array.
461
462        The stations are ordered and the first one becomes the position reference;
463        a DENM record is built for the reference paired with each other station.
464        The records are first built without touching the indexes and committed
465        only when all of them were built: if any construction raises, no record of
466        the array is created or updated.
467
468        Args:
469            station_collision_array (StationCollisionArrayV1): The collision array containing
470                stations information and detection time.
471
472        Returns:
473            list[DenmDataV1]: The new or updated DENM records, one per distinct
474                station pair.
475
476        Note:
477            - Each station pair gets a unique record key based on station IDs and
478              object IDs; a pair repeated in the same array is built once, the last
479              occurrence winning.
480            - If a record with the same key already exists, its action ID is reused,
481              otherwise a new one is created.
482            - The reference of the DENM is always the first station in the ordered list.
483
484        Complexity:
485            - Time: O(k log k + k log N), k = number of stations in the
486              array, N = total tracked DENM records; the O(k log k) term is
487              _station_order's sort, the O(k log N) term is one SortedList
488              insert (and, for updates, one removal) per station pair.
489            - Space: O(k), for the ordered stations, the pending records and the
490              returned DENM message list.
491        """
492        ordered_stations = self._station_order(
493            stations=station_collision_array.stations
494        )
495        position_reference: PositionInfo | None = None
496        position_reference_time: ItsTime | None = None
497        # Build phase: no index is touched, so a failure leaves them unchanged
498        pending: dict[RecordKey, DenmRecordInfo] = {}
499        for station in ordered_stations:
500            if self._logger.isEnabledFor(level=logging.DEBUG):
501                self._logger.debug(
502                    msg=f"Processing station: {station.stationInfo.stationId}, objectId: {station.objectId}"
503                )
504            if (position_reference is None) or (position_reference_time is None):
505                position_reference = station
506                position_reference_time = ItsTime.create_from_iso_time(
507                    time_str=position_reference.time
508                )
509                if self._logger.isEnabledFor(level=logging.DEBUG):
510                    self._logger.debug(msg="Set position reference station")
511                continue
512            # Create a unique record key based on the involved stations
513            record_key = self._create_record_key(
514                reference_station_id=position_reference.stationInfo.stationId,
515                reference_object_id=position_reference.objectId,
516                involved_station_id=station.stationInfo.stationId,
517                involved_object_id=station.objectId,
518            )
519            if self._logger.isEnabledFor(level=logging.DEBUG):
520                self._logger.debug(msg=f"record key: {record_key}")
521            # A pair repeated in this array keeps the action id given to it here
522            action_id = (
523                pending[record_key].denm_data.managementContainer.actionId
524                if record_key in pending
525                else self._create_action_id(record_key=record_key)
526            )
527            pending[record_key] = self._create_denm_record_info(
528                action_id=action_id,
529                record_key=record_key,
530                detection_time=station_collision_array.detectionTime,
531                station_collision_array=station_collision_array,
532                position_reference=position_reference,
533                position_reference_time=position_reference_time,
534                station=station,
535            )
536        # Commit phase: every record was built, apply them all
537        for record_key, record_info in pending.items():
538            previous = self._record_key_index.get(record_key)
539            if previous is not None:
540                # Existing record: remove it from the sorted list before the update
541                self._denm_sorted_records.remove(
542                    value=OrderedListRecord(
543                        reference_time_param=previous.reference_time,
544                        record_key_param=record_key,
545                    )
546                )
547            self._record_key_index[record_key] = record_info
548            self._action_id_index[
549                record_info.denm_data.managementContainer.actionId
550            ] = record_info
551            self._denm_sorted_records.add(
552                value=OrderedListRecord(
553                    reference_time_param=record_info.reference_time,
554                    record_key_param=record_key,
555                )
556            )
557        return [record_info.denm_data for record_info in pending.values()]

559    def _is_denm_valid(
560        self, now_monotonic_sec: int, record_info: DenmRecordInfo
561    ) -> bool:
562        """Tells whether a DENM record is still within its validity duration.
563
564        The age of the record is measured on the monotonic clock from its creation.
565
566        Args:
567            now_monotonic_sec (int): The current time in seconds from a monotonic clock.
568            record_info (DenmRecordInfo): The DENM record information, which includes
569                the record's monotonic timestamp and validity duration.
570
571        Returns:
572            bool: True if the DENM record is still valid (i.e., the elapsed time since the
573                record was created is less than its validity duration), or if the validity
574                duration is not set. False otherwise.
575
576        Complexity:
577            - Time: O(1), a fixed number of arithmetic comparisons.
578            - Space: O(1), no additional storage.
579        """
580        if self._logger.isEnabledFor(level=logging.DEBUG):
581            self._logger.debug(
582                msg=(
583                    f"Checking DENM validity: now monotonic {now_monotonic_sec} sec, "
584                    f"record monotonic time {record_info.monotonic_time_sec} sec, "
585                    f"validity duration "
586                    f"{record_info.denm_data.managementContainer.validityDuration} sec"
587                )
588            )
589        return (
590            (
591                now_monotonic_sec - record_info.monotonic_time_sec
592                < record_info.denm_data.managementContainer.validityDuration
593            )
594            if record_info.denm_data.managementContainer.validityDuration is not None
595            else True
596        )

598    def get_denm_records_clean_reference_expired(self) -> list[DenmDataV1]:
599        """Removes expired DENM records based on reference time and validity duration.
600
601        Iterates through DENM records in chronological order and removes those whose
602        reference time is older than the allowed delay. For each removed record, if it is
603        still within its validity duration, its data is collected for further processing.
604
605        Returns:
606            list[DenmDataV1]: List of DENM data for records that were removed and still valid.
607
608        Complexity:
609            - Time: O(log N + m log N), m = number of expired records found,
610              N = total tracked DENM records; the irange lookup is O(log N)
611              even when nothing expired, then each of the m records costs
612              O(1) average dict deletes plus one O(log N) pop(0) from the
613              SortedList.
614            - Space: O(m), for the returned list of still-valid removed
615              records.
616        """
617        reference_time_threshold = ItsTime()
618        reference_time_threshold -= self._denm_last_update_max_delay_ms
619        record_range = self._denm_sorted_records.irange(
620            minimum=None,
621            maximum=OrderedListRecord(
622                reference_time_param=reference_time_threshold,
623                record_key_param=RecordKey(
624                    reference_actor_id=ActorId(station_id=0, object_id=None),
625                    involved_actor_id=ActorId(station_id=0, object_id=None),
626                ),
627            ),
628            inclusive=(True, False),
629            reverse=False,
630        )
631        records_to_remove: list[DenmDataV1] = []
632        now_monotonic_sec = int(monotonic())
633        record_range_len = 0
634        for record in record_range:
635            record_tmp: OrderedListRecord = record
636            record_range_len += 1
637            record_info = self._record_key_index[record_tmp.record_key]
638            del self._record_key_index[record_tmp.record_key]
639            del self._action_id_index[
640                record_info.denm_data.managementContainer.actionId
641            ]
642            if self._logger.isEnabledFor(level=logging.DEBUG):
643                self._logger.debug(
644                    msg=(
645                        f"Removing DENM: key {record_tmp.record_key}, "
646                        f"action ID {record_info.denm_data.managementContainer.actionId}, "
647                        f"reference time {record_info.reference_time}"
648                    )
649                )
650            if self._is_denm_valid(
651                now_monotonic_sec=now_monotonic_sec, record_info=record_info
652            ):
653                records_to_remove.append(record_info.denm_data)
654        # Remove the record from the sorted list
655        for _ in range(record_range_len):
656            self._denm_sorted_records.pop(0)
657        return records_to_remove
