  1"""Command-line entry point of the STEP Producer program.
  2
  3The program receives the arguments from the command line and, based on them, writes
  4the configuration JSON schema and/or the configuration template to a directory, or
  5loads the configuration from a file and runs ``StepProducerApp`` through
  6``AppRunner``. If no argument is given, it prints the help of the program. The
  7running app provides the DEN service: it reads DENM records and station collision
  8arrays (the output of STEP Crash Prediction) from a Redis input stream, manages the
  9lifecycle of the DENM messages and publishes them to the client Redis stream (see
 10``step_producer.step_producer_app``).
 11
 12Usage:
 13    The module can be executed directly with various command-line options:
 14
 15        - -h, --help: Display the help (exit code OK)
 16        - -v, --version: Display version information (exit code OK)
 17        - -s, --conf_schema: Write the configuration JSON schema into a directory
 18        - -t, --conf_template: Write the configuration template into a directory
 19        - -c, --config: Load configuration from specified JSON file
 20        - -i, --instance: Specify the instance name
 21
 22Note:
 23    When the module uses the `-c` or `--config` option, also the -i or --instance option
 24    must be specified to set the instance name.
 25
 26Example:
 27    .. code-block:: console
 28
 29        python -m step_producer -c config.json -i producer_1
 30
 31Complexity:
 32    - Time: dominated by main, which runs ``StepProducerApp`` for its whole
 33      lifetime (see ``step_producer.step_producer_app``); the setup is O(1)
 34      relative to input size (fixed configuration field count).
 35    - Space: O(1) relative to input size for the setup; the running app needs
 36      O(k), k = stations in a collision array message.
 37
 38Functions:
 39    - _get_config_schema_file: Writes the configuration JSON schema file to a
 40      directory. Complexity: time O(1), space O(1) relative to input size.
 41    - _get_config_template_file: Writes the configuration template file to a
 42      directory. Complexity: time O(1), space O(1) relative to input size.
 43    - _args_parse: Builds the command-line argument parser. Complexity: time
 44      O(1), space O(1).
 45    - main: CLI entry point; parses arguments, loads configuration, and runs
 46      ``StepProducerApp`` through ``AppRunner``. Complexity: time O(1) for setup,
 47      dominated thereafter by ``StepProducerApp.work``; space O(1) for setup.
 48"""

 69def _get_config_schema_file(path: Path) -> None:
 70    """Writes the configuration JSON schema into a directory, created if missing.
 71
 72    Generates the configuration schema file for StepProducerConfig in that
 73    directory, then prints a confirmation message to standard output.
 74
 75    Args:
 76        path (Path): The directory path where the configuration schema file
 77            will be created.
 78
 79    Returns:
 80        None: The schema is written to a file; nothing is returned.
 81
 82    Complexity:
 83        - Time: O(1) relative to input size, dominated by
 84          ``StepProducerConfig.print_config_schema``.
 85        - Space: O(1) relative to input size, for the same reason.
 86    """
 87    path.mkdir(parents=True, exist_ok=True)
 88    StepProducerConfig.print_config_schema(path=path)
 89    print("Schemas are generated in the directory")  # noqa: T201

115def _args_parse() -> ArgumentParser:
116    """Build the command line argument parser for the step_producer application.
117
118    This function creates and configures, without parsing, an ArgumentParser with
119    the following options:
120
121        -h, --help: Show the help (added by argparse, ``add_help=True``)
122        -v, --version: Show program version
123        -s, --conf_schema: Write JSON schema into the specified directory
124        -t, --conf_template: Write configuration template into the specified
125            directory
126        -c, --config: Load configuration from specified file
127        -i, --instance: Specify the instance name
128
129    Note:
130        The -c/--config and -i/--instance options must be given together; ``main``
131        raises ``StepCliError`` when only one of them is present.
132
133    Returns:
134        ArgumentParser: Configured argument parser object; the caller parses
135            ``sys.argv`` with it.
136
137    Complexity:
138        - Time: O(1), a fixed number of arguments is registered on the parser.
139        - Space: O(1), the parser holds a constant number of argument definitions.
140    """
141    parser = ArgumentParser(
142        description="Program to generate DENM messages from STEP Crash Prediction output.",
143        epilog=(
144            "When the module uses the `-c` or `--config` option, also the -i or --instance "
145            "option must be specified to set the instance name"
146        ),
147        add_help=True,
148        allow_abbrev=False,
149        exit_on_error=True,
150    )
151    parser.add_argument(
152        "-v",
153        "--version",
154        help="Version of the program",
155        action="version",
156        version=f"%(prog)s version: {step_producer.__version__}",
157    )
158    parser.add_argument(
159        "-s",
160        "--conf_schema",
161        help="Write the configuration JSON schema into the directory (created if missing)",
162        metavar="<schema dir>",
163        dest="conf_schema_file",
164        type=str,
165    )
166    parser.add_argument(
167        "-t",
168        "--conf_template",
169        help="Write the configuration template into the directory (created if missing)",
170        metavar="<template dir>",

193def main() -> int:  # noqa: C901, PLR0912
194    """Main entry point for the application.
195
196    This function processes command-line arguments, loads the configuration, sets
197    up logging, and runs ``StepProducerApp`` through ``AppRunner``.
198
199    It handles the following operations:
200
201        - Printing help when no arguments are provided
202        - Mapping argparse's own exit (help, version, invalid or unknown option)
203          to an ``ExitCode`` instead of letting ``SystemExit`` escape
204        - Generating JSON schema file when requested
205        - Creating configuration template file when requested
206        - Rejecting -c/--config without -i/--instance and vice versa
207        - Loading configuration from specified file, reconfiguring the logger
208          from it and logging the program and library versions
209        - Using the selector event loop when running on Windows, the same loop
210          used by the other STEP programs run through ``AppRunner``
211        - Running ``StepProducerApp`` and returning its exit code
212
213    Signal handling (SIGINT/SIGTERM on POSIX, CTRL_C/CTRL_BREAK on Windows) is
214    owned by ``AppRunner``: a shutdown signal cancels ``StepProducerApp.work`` and
215    returns ``StepProducerApp.on_shutdown`` (``ExitCode.ABORTED``).
216
217    Exception handling:
218
219        - Catches interruptions (Ctrl+C) gracefully
220        - Logs CLI misuse (``StepCliError``) as a plain error line
221        - Logs unhandled exceptions with their traceback
222
223    Returns:
224        int: Exit code indicating program result status:
225
226            - ExitCode.OK (0): Success, including -h/--help and -v/--version
227            - ExitCode.USAGE (1): No arguments given, an invalid or unknown
228              option, or only one of -c/--config and -i/--instance
229            - ExitCode.ERROR (2): An unhandled exception occurred, or
230              ``StepProducerApp`` ended on an error a reconnect cannot fix
231              (``fatal_result``)
232            - ExitCode.ABORTED (3): Program was interrupted
233
234    Complexity:
235        - Time: O(1) relative to input size for argument parsing, configuration
236          loading and optional schema/template generation; the bulk of the
237          runtime is spent in ``StepProducerApp.work`` until interrupted.
238        - Space: O(1) relative to input size, for the configuration object and
239          any generated schema/template.
240    """
241    result = ExitCode.OK
242    logger = StepLogger.get_logger(name=None, config=None, instance_name=None)
243    try:
244        app_runner = AppRunner[ExitCode](
245            logger_param=logger,
246            loop_factory=asyncio.SelectorEventLoop if IS_WINDOWS else None,
247        )
248        parser = _args_parse()
249        if len(sys.argv) == 1:
250            parser.print_help()
251            return ExitCode.USAGE.value
252        try:
253            args = parser.parse_args()
254        except SystemExit as exc:
255            # argparse exits 0 for -h/-v and 2 for a usage error: map it to ExitCode
256            return (ExitCode.OK if exc.code in {0, None} else ExitCode.USAGE).value
257        if args.conf_schema_file is not None:
258            _get_config_schema_file(path=Path(args.conf_schema_file))
259        if args.conf_template_file is not None:
260            _get_config_template_file(path=Path(args.conf_template_file))
261        if (args.config_file is not None) and (args.instance_name is None):
262            raise StepCliError(  # noqa: TRY301
263                "The -i/--instance option must be specified when using -c/--config."
264            )
265        if (args.config_file is None) and (args.instance_name is not None):
266            raise StepCliError(  # noqa: TRY301
267                "The -c/--config option must be specified when using -i/--instance."
268            )
269        if (args.config_file is not None) and (args.instance_name is not None):
270            config = StepProducerConfig.model_validate_json(
271                json_data=Path(args.config_file).read_text(encoding="utf-8")
272            )
273            logger = StepLogger.get_logger(
274                name=None, config=config.log, instance_name=args.instance_name
275            )
276            StepLogger.logger_force_write(
277                logger=logger,
278                msg=f"STEP Producer: name {args.instance_name} version {step_producer.__version__}",
279            )
280            StepLogger.logger_force_write(logger=logger, msg="Library:")
281            StepLogger.logger_force_write(
282                logger=logger,
283                msg=f"    - step_cloud_schemas: {step_cloud_schemas.__version__}",
284            )
285            StepLogger.logger_force_write(
286                logger=logger, msg=f"    - step_redis: {step_redis.__version__}"
287            )
288            result = app_runner.run(
289                StepProducerApp(
290                    instance_name=args.instance_name, config=config, logger=logger
291                )
292            )
293    except KeyboardInterrupt:
294        if logger.isEnabledFor(level=logging.INFO):
295            logger.info(msg="Program interrupted")
296        result = ExitCode.ABORTED
297    except StepCliError as exc:
298        # Command line misuse: a plain error line, no traceback
299        if logger.isEnabledFor(level=logging.ERROR):
300            logger.error(msg=str(exc))  # noqa: TRY400
301        result = ExitCode.USAGE
302    except Exception:
303        if logger.isEnabledFor(level=logging.ERROR):
304            logger.exception(msg="Exception unmanaged")
305        result = ExitCode.ERROR
306    return result.value
