  1"""Workload of the STEP Producer program.
  2
  3Defines the ``ReconnectingApp`` subclass handed to ``AppRunner`` by
  4``step_producer.__main__``: it reads the Redis credential from the environment
  5(``RedisCredential.from_env``) and creates a single ``DenServiceManager`` kept for
  6the whole run, then each connection cycle (``cycle``) opens a Redis connection
  7pool and the client output stream and runs three concurrent tasks in a
  8``TaskGroup``: one listening to the input Redis
  9stream and dispatching DENM records and station collision arrays to the DEN
 10service (the DENM messages derived from a collision array are published to the
 11client Redis stream; a message that fails to process is dropped), one
 12periodically cleaning up the expired DENM records and publishing their
 13cancellations to the client Redis stream, and one running the Redis health
 14check. A failure tears the cycle down and the inherited ``work`` reconnects
 15after ``reconnectInterval`` seconds, keeping the tracked DENM records; an error a
 16reconnect cannot fix ends the run with ``ExitCode.ERROR``, and a shutdown
 17cancellation ends the loop.
 18
 19Message Types Processed:
 20
 21    - ProducerCloudEventsDenmDataV1: Individual DENM records
 22    - ProducerCloudEventsStationCollisionArrayV1: Arrays of station collision data
 23
 24Complexity:
 25    - Time: dominated by the three tasks run by ``StepProducerApp.cycle``
 26      (``_stream_receive_run``, ``_periodical_cleanup_run``,
 27      ``_redis_health_check_run``), which run indefinitely; the heaviest bounded
 28      per-call operation is ``_stream_receive_run``'s O(k log k + k log N)
 29      dispatch to ``DenServiceManager.process_station_collisions_received``, k =
 30      stations in a collision array message, N = total tracked DENM records.
 31    - Space: O(k), for the same call's returned message list; O(m) per cleanup
 32      iteration, m = expired records; O(1) elsewhere.
 33
 34Classes:
 35    - ExitCode: Enumeration of the program exit codes. Complexity: time O(1),
 36      space O(1).
 37    - StepProducerApp: Workload run by ``AppRunner``: the reconnecting DEN
 38      service on top of the Redis streams. Complexity: time O(k log k + k log N)
 39      per station collision message, space O(k).
 40"""

102class StepProducerApp(ReconnectingApp[ExitCode]):
103    """Workload run by ``AppRunner``: the reconnecting DEN service on Redis streams.
104
105    Holds the state shared by every connection cycle: the instance name, the
106    configuration, the Redis credential read once from the environment at
107    construction and the single ``DenServiceManager`` kept for the whole run, so
108    the tracked DENM records and the ActionId sequence survive a reconnect. Each
109    ``cycle`` opens a Redis connection pool and the shared client output stream
110    and runs the stream receive, periodical cleanup and Redis health check tasks
111    until one of them fails; the inherited ``work`` then reconnects after
112    ``reconnectInterval`` seconds, and an error that reconnecting cannot fix
113    (``_FATAL_ERRORS``) ends the run with ``fatal_result`` (``ExitCode.ERROR``).
114
115    Attributes:
116        instance_name (str): The name of the STEP Producer instance, also used as
117            the Redis stream consumer name.
118        config (StepProducerConfig): Configuration object containing all
119            necessary settings.
120        logger (Logger): Logger instance for recording program events.
121
122    Complexity:
123        - Time: O(k log k + k log N) per station collision message and O(log N)
124          per DENM record processed by ``cycle`` (through ``_stream_receive_run``),
125          plus O(log N + m log N) per cleanup iteration
126          (``_periodical_cleanup_run``); k = stations in a collision array, N =
127          total tracked DENM records, m = expired records.
128        - Space: O(k) per station collision message, O(m) per cleanup iteration.
129    """
130
131    # Maximum number of connections allowed in the Redis connection pool
132    _REDIS_MAX_CONNECTIONS: ClassVar[int] = 2
133
134    # Errors that a reconnect can never fix: the inherited work() returns
135    # fatal_result() instead of retrying. step_redis classifies rejected
136    # credentials, ACL denials, missing SSL files and untrusted certificates as
137    # StepRedisFatalError
138    _FATAL_ERRORS: ClassVar[tuple[type[Exception], ...]] = (StepRedisFatalError,)
139
140    __slots__ = ("_config", "_den_service", "_instance_name", "_redis_credential")
141
142    def __init__(
143        self, instance_name: str, config: StepProducerConfig, logger: Logger
144    ) -> None:
145        """Store the settings, read the Redis credential and create the DEN service.
146
147        The DEN service manager is created once and kept across reconnects. The
148        credential is read here rather than in ``cycle`` because the inherited
149        ``work`` retries on every error it does not classify as fatal, so a missing
150        variable would never end the program.
151
152        Args:
153            instance_name (str): The name of the STEP Producer instance, also used
154                as the Redis stream consumer name.
155            config (StepProducerConfig): Configuration object containing all
156                necessary settings.
157            logger (Logger): Logger instance for recording program events.
158
159        Raises:

231    async def _periodical_cleanup_run(
232        self, den_service: DenServiceManager, client_redis_stream: StepClientRedisStream
233    ) -> None:
234        """Periodically cleans up expired DENM records and publishes cancellation messages.
235
236        Loops forever: the DEN service manager's cleanup routine returns the
237        cancellation message of every expired DENM record, each one is published to
238        the client stream, and the loop sleeps ``denmCheckLastUpdateInterval``
239        seconds before the next run. The duration of each cleanup, publishing
240        included, is logged through a ``LoggingTimer``.
241
242        Args:
243            den_service (DenServiceManager): Service manager for DENM record handling.
244            client_redis_stream (StepClientRedisStream): Client output stream, shared
245                with ``_stream_receive_run``, the cancellations are published to.
246
247        Returns:
248            None: Never returns normally; runs until the enclosing task is cancelled
249                or an exception is raised.
250
251        Complexity:
252            - Time: O(log N + m log N) per iteration (the loop itself runs
253              indefinitely), m = expired records found in that iteration, N =
254              total tracked DENM records; dominated by
255              ``den_service.periodical_cleanup()`` plus one publish call per
256              expired message.
257            - Space: O(m) per iteration, for the list of expired messages.
258        """
259        logging_timer = LoggingTimer(
260            logger=self._logger, prefix="step_producer", log_level=LogMode.INFO
261        )
262        while True:
263            with logging_timer(self._periodical_cleanup_run):
264                if self._logger.isEnabledFor(level=logging.DEBUG):
265                    self._logger.debug(msg="Running DENM periodical cleanup")
266                for message in den_service.periodical_cleanup():
267                    await client_redis_stream.publish_denm_message(message=message)
268            await asyncio_sleep(delay=self._config.denmCheckLastUpdateInterval)

333    def _producer_message_dispatch(
334        self,
335        logging_timer: LoggingTimer,
336        den_service: DenServiceManager,
337        message: ProducerCloudEventsJsonRoot,
338    ) -> list[DenmDataV1]:
339        """Dispatch a received producer message to its handler by type.
340
341        Args:
342            logging_timer (LoggingTimer): Timer passed to the handlers.
343            den_service (DenServiceManager): Service manager that processes the
344                message.
345            message (ProducerCloudEventsJsonRoot): The validated CloudEvents message
346                received from the input stream.
347
348        Returns:
349            list[DenmDataV1]: The DENM messages to publish: those derived from a
350                station collision array, empty for a DENM record or an unknown type.
351
352        Complexity:
353            - Time: O(log N) for a DENM record, O(k log k + k log N) for a station
354              collision array (see the handlers), O(1) otherwise.
355            - Space: O(k) for a station collision array, O(1) otherwise.
356        """
357        match message.root:
358            case ProducerCloudEventsDenmDataV1():
359                self._denm_message_handler(
360                    logging_timer=logging_timer,
361                    den_service=den_service,
362                    message=message.root.data,
363                )
364            case ProducerCloudEventsStationCollisionArrayV1():
365                return self._station_collision_array_message_handler(
366                    logging_timer=logging_timer,
367                    den_service=den_service,
368                    message=message.root.data,
369                )
370            case _:
371                if self._logger.isEnabledFor(level=logging.WARNING):
372                    self._logger.warning(
373                        msg=f"Unknown message type: {type(message.root)}"
374                    )
375        return []

381        messages: list[DenmDataV1],

383        """Publish the DENM messages derived from a received message, timed as a batch.

406    async def _stream_receive_run(
407        self,
408        config: RedisStreams,
409        pool: StepRedisPool,
410        den_service: DenServiceManager,
411        client_redis_stream: StepClientRedisStream,
412    ) -> None:
413        """Listens to the input Redis stream and dispatches the received messages.
414
415        Opens a ``StepProducerRedisStream`` on the configured input stream and
416        consumer group (``_input_stream_params``), creates the input stream, then
417        iterates over the received producer messages, dispatched by
418        ``_producer_message_dispatch``:
419
420            - ``ProducerCloudEventsDenmDataV1``: handled by
421              ``_denm_message_handler``.
422            - ``ProducerCloudEventsStationCollisionArrayV1``: handled by
423              ``_station_collision_array_message_handler``; the derived DENM
424              messages are published by ``_denm_messages_publish``.
425            - Any other type: skipped with a warning.
426
427        A message whose processing raises ``ValueError`` (pydantic
428        ``ValidationError`` included, e.g. an invalid ITS time) is logged and
429        dropped, keeping the tracked DENM records; Redis errors are not caught, so
430        the inherited ``work`` reconnects.
431
432        Args:
433            config (RedisStreams): Redis streams configuration, used for the input
434                stream id, group id and wait timeout.
435            pool (StepRedisPool): Redis connection pool of the input stream client.
436            den_service (DenServiceManager): Service manager responsible for
437                processing DENM and station collision messages.
438            client_redis_stream (StepClientRedisStream): Client output stream, shared
439                with ``_periodical_cleanup_run``, the derived DENMs are published to.
440
441        Returns:
442            None: Never returns normally; runs until the enclosing task is cancelled
443                or an exception is raised.
444
445        Complexity:
446            - Time: per message, O(log N) for a DENM record or O(k log k + k log N)
447              for a station collision array, where N = total tracked DENM records
448              and k = stations in that message; the loop itself runs indefinitely.
449            - Space: O(k) per station collision message, for the returned list of
450              messages to publish; O(1) otherwise.
451        """
452        logging_timer = LoggingTimer(
453            logger=self._logger, prefix="step_producer", log_level=LogMode.INFO
454        )
455        async with StepProducerRedisStream(
456            pool=pool,
457            config=self._input_stream_params(config=config),
458            logger=self._logger,
459        ) as producer_redis_stream:
460            if self._logger.isEnabledFor(level=logging.INFO):
461                self._logger.info(
462                    msg=f"Instance {self._instance_name} listening for messages from Redis stream"
463                )
464            await producer_redis_stream.create_input_stream()
465            while True:
466                async for message in producer_redis_stream.receive_producer_message():
467                    if self._logger.isEnabledFor(level=logging.DEBUG):
468                        self._logger.debug(msg=f"Redis message: {message}")
469                    try:
470                        message_to_send_list = self._producer_message_dispatch(
471                            logging_timer=logging_timer,
472                            den_service=den_service,
473                            message=message,
474                        )
475                    except ValueError:
476                        # Message-level failure (pydantic ValidationError is a
477                        # ValueError, as is an invalid ITS time): drop this message,
478                        # keep the tracked DENM records. Redis errors are not
479                        # caught, so work() reconnects
480                        if self._logger.isEnabledFor(level=logging.ERROR):
481                            self._logger.exception(
482                                msg=f"Discarding {type(message.root).__name__} message"
483                            )
484                        continue
485                    if message_to_send_list:
486                        await self._denm_messages_publish(
487                            logging_timer=logging_timer,
488                            client_redis_stream=client_redis_stream,
489                            messages=message_to_send_list,
490                        )

529    @override
530    async def cycle(self) -> None:
531        """Open the Redis pool and the client output stream and run the tasks.
532
533        Opens a ``StepRedisPool`` (at most ``_REDIS_MAX_CONNECTIONS`` connections,
534        parameters from ``StepRedisPoolParams.from_connection``) and the client
535        output stream shared by the receive and cleanup tasks, and runs
536        ``_stream_receive_run``, ``_periodical_cleanup_run`` and
537        ``_redis_health_check_run`` in a ``TaskGroup`` on the ``DenServiceManager``
538        created at construction. The first failure propagates to the inherited
539        ``work``, which reconnects or ends the run.
540
541        Returns:
542            None: Returns only if every task ends normally, which they never do.
543
544        Complexity:
545            - Time: the indefinite work of the three tasks (see
546              ``_stream_receive_run`` and ``_periodical_cleanup_run``).
547            - Space: O(1) beyond the state of the three tasks.
548        """
549        async with (
550            StepRedisPool(
551                params=StepRedisPoolParams.from_connection(
552                    config=self._config.redis.connection,
553                    credential=self._redis_credential,
554                    max_connections=self._REDIS_MAX_CONNECTIONS,
555                    logger=self._logger,
556                ),
557                logger=self._logger,
558            ) as redis_pool,
559            StepClientRedisStream(
560                pool=redis_pool,
561                config=self._client_stream_params(config=self._config.redis.streams),
562                logger=self._logger,
563            ) as client_redis_stream,
564            TaskGroup() as group,
565        ):
566            # Create stream receive task. It dispatches the input stream messages
567            group.create_task(
568                coro=self._stream_receive_run(
569                    config=self._config.redis.streams,
570                    pool=redis_pool,
571                    den_service=self._den_service,
572                    client_redis_stream=client_redis_stream,
573                )
574            )
575            # Create DENM cleanup task. It cancels the expired DENM records
576            group.create_task(
577                coro=self._periodical_cleanup_run(
578                    den_service=self._den_service,
579                    client_redis_stream=client_redis_stream,
580                )
581            )
582            # Create health check task. It will check the health of the Redis connection
583            group.create_task(
584                coro=self._redis_health_check_run(
585                    config=self._config.redis.streams, pool=redis_pool
586                )
587            )
