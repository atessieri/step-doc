 70def _assert_model_config(model: type[BaseModel], title: str) -> None:
 71    """Assert that ``model`` is configured as a strict, frozen model forbidding
 72    extra fields, with the given schema title, and that it actually rejects an
 73    extra field and an assignment to a field of its default instance.
 74
 75    Args:
 76        model (type[BaseModel]): The Pydantic model class (with a static
 77            ``get_default`` factory).
 78        title (str): The expected ``model_config`` title.
 79
 80    Returns:
 81        None
 82    """
 83    config = model.model_config
 84    assert config["title"] == title
 85    assert config["extra"] == "forbid"
 86    assert config["frozen"] is True
 87    assert config["strict"] is True
 88    default = model.get_default()  # type: ignore[attr-defined]
 89    data = json.loads(default.model_dump_json())
 90    with pytest.raises(expected_exception=ValidationError, match="unexpectedField"):
 91        model.model_validate_json(json.dumps({**data, "unexpectedField": "x"}))
 92    first_field = next(iter(model.model_fields))
 93    with pytest.raises(expected_exception=ValidationError, match="frozen"):

117def _violating_value(constraint: Any, current: Any) -> Any:
118    """Return the closest value just outside ``constraint``.
119
120    Args:
121        constraint (Any): An ``annotated_types`` bound or length constraint.
122        current (Any): The JSON value of the field in the default instance, used
123            to shrink a list below its minimum length.
124
125    Returns:
126        Any: A JSON-serializable value violating ``constraint`` by the smallest
127            step (one below a lower bound, one above an upper bound, one item or
128            character short of or past a length limit).
129    """
130    # The bounds are typed as protocols (SupportsGe/SupportsLe); every bound in the
131    # configuration models is an int or a float
132    match constraint:
133        case Ge():
134            return cast("float", constraint.ge) - 1
135        case Gt(gt=bound) | Lt(lt=bound):
136            return bound
137        case Le():
138            return cast("float", constraint.le) + 1
139        case MinLen(min_length=length):
140            if isinstance(current, list):
141                return current[: length - 1]
142            return "x" * (length - 1)
143        case MaxLen(max_length=length):
144            return "x" * (length + 1)
145    raise AssertionError(f"Unexpected constraint {constraint!r}")  # pragma: no cover

148def _constraint_cases() -> list[Any]:
149    """Build one test case per field constraint of every configuration model.
150
151    The cases are read from the models' field metadata, so a constraint added
152    later is tested without touching this module.
153
154    Returns:
155        list[Any]: ``pytest.param(model, field, violating_value)`` entries.
156    """
157    cases: list[Any] = []
158    for model in _CONSTRAINED_MODELS:
159        defaults = json.loads(model.get_default().model_dump_json())  # type: ignore[attr-defined]
160        for name, field in model.model_fields.items():
161            for constraint in field.metadata:
162                if isinstance(constraint, _CONSTRAINT_TYPES):
163                    cases.append(
164                        pytest.param(
165                            model,
166                            name,
167                            _violating_value(constraint, defaults[name]),
168                            id=f"{model.__name__}.{name} {constraint!r}",
169                        )
170                    )
171    return cases

754        self, socket_timeout: int, wait_timeout: int, valid: bool
755    ) -> None:

757        the input stream wait timeout plus ``READ_TIMEOUT_MARGIN_MS`` and rejects a
758        shorter one, naming ``socketTimeout``.
759
760        Args:
761            socket_timeout (int): The ``connection.socketTimeout`` in seconds.

766            None
