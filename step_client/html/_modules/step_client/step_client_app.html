Source code for step_client.step_client_app

  1"""Workload of the STEP Client program.
  2
  3Defines the ``ReconnectingApp`` subclass handed to ``AppRunner`` by
  4``step_client.__main__``: it reads the MQTT credential (step_lib
  5``MqttCredential.from_env``) and the Redis credential (step_redis
  6``RedisCredential.from_env``) from the environment, then each connection cycle
  7(``cycle``) opens a Redis connection pool and connects to the STEP MQTT broker,
  8and runs three concurrent tasks in a ``TaskGroup``: one forwarding CAM/CPM
  9messages received from the STEP broker to the adapter Redis stream and DENM
 10messages to the producer Redis stream, one forwarding DENM messages read from the
 11input Redis stream to the STEP broker, and one running the Redis health check.
 12Any failure tears the cycle down and the inherited ``work`` reconnects after
 13``reconnectInterval`` seconds; an error a reconnect cannot fix ends the run with
 14``ExitCode.ERROR``, and a shutdown cancellation ends the loop instead. Delivery is
 15at-most-once: a message being forwarded when a transport fails is logged and
 16lost, because both the MQTT subscription and the Redis input stream (read
 17without acknowledgement) have already acknowledged it.
 18
 19Complexity:
 20    - Time: O(n * m), where n is the number of ITS/Redis messages processed over
 21      the program's lifetime and m the size (field count/byte length) of a
 22      message; each message is decoded, logged and re-serialized in O(m) time,
 23      and each connect/reconnect cycle performs O(t) setup work, t = number of
 24      configured MQTT subscription/publish topics.
 25    - Space: O(t + m) per reconnect cycle, for the StepBroker topic
 26      configuration and the payload of the message being handled; the three
 27      concurrent tasks are fixed in number.
 28
 29Classes:
 30    - ExitCode: Enumeration of the program exit codes. Complexity: time O(1),
 31      space O(1).
 32    - StepClientApp: Workload run by ``AppRunner``: the reconnecting STEP
 33      broker/Redis stream bridge. Complexity: time O(n * m) over the
 34      program's lifetime, space O(t + m).
 35"""

 38import logging
 39from asyncio import TaskGroup
 40from enum import IntEnum
 41from logging import Logger
 42from signal import Signals
 43from typing import ClassVar, override

105class StepClientApp(ReconnectingApp[ExitCode]):
106    """Workload of the program, run by ``AppRunner``: the reconnecting bridge
107    between the STEP MQTT broker and the Redis streams.
108
109    Holds the state shared by every connection cycle: the instance name, the
110    configuration and the MQTT and Redis credentials read once from the
111    environment at construction. Each ``cycle`` opens a fresh Redis connection
112    pool and ``StepBroker``, subscribes to the configured topics and runs the STEP
113    receive, Redis stream receive and Redis health check tasks until one of them
114    fails; the inherited ``work`` then reconnects after ``reconnectInterval``
115    seconds, and an error that reconnecting cannot fix (``_FATAL_ERRORS``) ends
116    the run with ``fatal_result`` (``ExitCode.ERROR``).
117
118    Attributes:
119        instance_name (str): The name of the STEP client instance, also used as
120            the MQTT client instance name and the Redis stream consumer name.
121        config (StepClientConfig): Configuration object containing all
122            necessary settings.
123        logger (Logger): Logger instance for recording program events.
124
125    Complexity:
126        - Time: O(n * m), where n is the number of ITS/Redis messages processed
127          by ``cycle`` (through ``_step_receive_run`` and ``_stream_receive_run``)
128          over the program's lifetime and m the size (field count/byte length)
129          of a message.
130        - Space: O(t + m), driven by ``cycle``: the StepBroker of each connection
131          cycle holds the t configured MQTT subscription/publish topics, and the
132          message being handled takes O(m).
133    """
134

136    # by the blocking XREADGROUP, the others for XADD/EXPIRE and the health check PING
137    _REDIS_MAX_CONNECTIONS: ClassVar[int] = 3
138
139    # Errors that a reconnect can never fix: the inherited work() returns
140    # fatal_result() instead of retrying. step_redis classifies rejected Redis
141    # credentials, ACL denials, missing SSL files and untrusted certificates as
142    # StepRedisFatalError; step_lib raises StepLibMqttFatalError when the broker
143    # CONNACK refuses the client's authentication or authorization
144    _FATAL_ERRORS: ClassVar[tuple[type[Exception], ...]] = (
145        StepRedisFatalError,
146        StepLibMqttFatalError,
147    )

298    async def _its_message_dispatch(
299        self,
300        adapter_stream: StepAdapterRedisStream,
301        producer_stream: StepProducerRedisStream,
302        message: ItsMessage,
303    ) -> None:
304        """Route an ITS message received from the STEP broker to its Redis stream.
305
306        CAM and CPM payloads go to the adapter stream, DENM payloads to the producer
307        stream; a message of any other type is skipped with a warning.
308
309        Args:
310            adapter_stream (StepAdapterRedisStream): Redis stream client for CAM and
311                CPM payloads.
312            producer_stream (StepProducerRedisStream): Redis stream client for DENM
313                payloads.
314            message (ItsMessage): The MQTT message received from the STEP broker,
315                with its decoded ITS payload.
316
317        Returns:
318            None: The payload is published as a side effect; nothing is returned.
319
320        Complexity:
321            - Time: O(m), m = size (field count/byte length) of the ITS payload; a
322              single type match and at most one O(m) log-and-publish call.
323            - Space: O(m), for the log line and the serialized payload.
324        """
325        match message.its_payload:
326            case CamMessageV1():
327                await self._cam_message_manage(
328                    adapter_stream=adapter_stream,
329                    message=message.its_payload,
330                    topic=message.topic,
331                    qos=message.qos,
332                )
333            case CpmMessageV1():
334                await self._cpm_message_manage(
335                    adapter_stream=adapter_stream,
336                    message=message.its_payload,
337                    topic=message.topic,
338                    qos=message.qos,
339                )
340            case DenmMessageV1():
341                await self._denm_message_manage(
342                    producer_stream=producer_stream,
343                    message=message.its_payload,
344                    topic=message.topic,
345                    qos=message.qos,
346                )
347            case _:
348                if self._logger.isEnabledFor(level=logging.WARNING):
349                    self._logger.warning(
350                        msg=f"Unknown message type: {message.its_payload}"
351                    )

353    async def _step_receive_run(
354        self, config: RedisStreams, pool: StepRedisPool, step_broker: StepBroker
355    ) -> None:
356        """Receives messages from the STEP broker and publishes them to Redis streams.
357
358        Opens the adapter and producer Redis stream clients, then iterates over the
359        ITS messages received from the STEP broker, routing each one through
360        ``_its_message_dispatch``: CAM and CPM payloads go to the adapter Redis
361        stream and DENM payloads to the producer Redis stream. Messages of unknown
362        type are skipped with a warning. A message whose CloudEvents envelope fails
363        validation (``ValidationError``) is logged and dropped without stopping the
364        loop; any other error (e.g. a Redis failure) gets a note on the exception
365        naming the message type - the message is lost, since MQTT already
366        acknowledged it - and propagates so that the inherited ``work`` logs it
367        once and reconnects. The processing time of each message is logged through a
368        ``LoggingTimer``.
369
370        Args:
371            config (RedisStreams): Redis streams configuration, used for the adapter
372                and producer output stream ids and maximum lengths.
373            pool (StepRedisPool): Redis connection pool shared by both stream clients.
374            step_broker (StepBroker): STEP broker instance to receive messages from.
375
376        Returns:
377            None: Returns only if the STEP broker message iterator ends; normally runs
378                until the enclosing task is cancelled or an exception is raised.
379
380        Complexity:
381            - Time: O(n * m), where n is the number of messages received from the
382              STEP broker and m the size (field count/byte length) of a message;
383              each message is dispatched and processed in O(m) time.
384            - Space: O(m), for the message being handled, aside from the
385              constant-size resources held by the adapter and producer Redis
386              stream clients.
387        """
388        logging_timer = LoggingTimer(
389            logger=self._logger, prefix="step_client", log_level=LogMode.INFO
390        )
391        async with (
392            StepAdapterRedisStream(
393                config=StepRedisStreamParams(
394                    stream_id=config.outputs.adapter.streamId,
395                    published_message_max_retained=config.outputs.adapter.maxLength,
396                    consumer_name=self._instance_name,
397                ),
398                logger=self._logger,
399                pool=pool,
400            ) as adapter_stream,
401            StepProducerRedisStream(
402                config=StepRedisStreamParams(
403                    stream_id=config.outputs.producer.streamId,
404                    published_message_max_retained=config.outputs.producer.maxLength,
405                    consumer_name=self._instance_name,
406                ),
407                logger=self._logger,
408                pool=pool,
409            ) as producer_stream,
410        ):
411            if self._logger.isEnabledFor(level=logging.INFO):
412                self._logger.info(
413                    msg=f"Instance {self._instance_name} listening messages from STEP broker"
414                )
415            async for message in step_broker.its_messages:
416                with logging_timer(func=self._step_receive_run):
417                    try:
418                        await self._its_message_dispatch(
419                            adapter_stream=adapter_stream,
420                            producer_stream=producer_stream,
421                            message=message,
422                        )
423                    except ValidationError:
424                        # Message-level failure (the CloudEvents envelope rejected
425                        # the payload): drop this message, keep the bridge running.
426                        # Redis errors are not caught, so work() reconnects
427                        if self._logger.isEnabledFor(level=logging.ERROR):
428                            self._logger.exception(
429                                msg=(
430                                    f"Discarding {message.its_payload.message_type} "
431                                    "message that could not be published to Redis"
432                                )
433                            )
434                    except Exception as exc:
435                        # Any other failure (e.g. Redis): the MQTT message is already
436                        # acknowledged, so it is lost; the note reaches the single log
437                        # written by work() before it reconnects
438                        exc.add_note(
439                            f"Discarding {message.its_payload.message_type} "
440                            "message, Redis publish failed: reconnecting"
441                        )
442                        raise

444    async def _stream_receive_run(
445        self, config: RedisStreams, pool: StepRedisPool, step_broker: StepBroker
446    ) -> None:
447        """Listens for messages on the input Redis stream and publishes DENMs to the
448        STEP broker.
449
450        Opens a ``StepClientRedisStream`` on the configured input stream and consumer
451        group (TTL from ``StepRedisStreamParams.ttl_seconds_from_wait_timeout``:
452        twice ``waitTimeout``, in seconds, at least 1), creates the
453        input stream, then iterates over the received client messages (the receive
454        generator never ends on its own): each CloudEvents DENM message is published
455        to the STEP broker through ``_denm_publish``; any other message type is
456        skipped with a warning. The processing time of each message is logged
457        through a ``LoggingTimer``.
458
459        Args:
460            config (RedisStreams): Redis streams configuration, used for the input
461                stream id, group id and wait timeout.
462            pool (StepRedisPool): Redis connection pool for stream operations.
463            step_broker (StepBroker): STEP broker client used to publish the DENMs.
464
465        Returns:
466            None: Never returns normally; runs until the enclosing task is cancelled
467                or an exception is raised.
468
469        Complexity:
470            - Time: O(n * m), where n is the number of messages received from the
471              Redis stream and m the size (field count/byte length) of a message;
472              each message is parsed, logged, matched and forwarded in O(m) time.
473            - Space: O(m), for the message being handled, aside from the
474              constant-size resources held by the Redis stream client.
475        """
476        logging_timer = LoggingTimer(
477            logger=self._logger, prefix="step_client", log_level=LogMode.INFO
478        )
479        async with StepClientRedisStream(
480            pool=pool,
481            config=StepRedisStreamParams(
482                stream_id=config.input.streamId,
483                group_id=config.input.groupId,
484                wait_timeout_ms=config.input.waitTimeout,
485                published_message_max_retained=None,
486                consumer_name=self._instance_name,
487                ttl_seconds=StepRedisStreamParams.ttl_seconds_from_wait_timeout(
488                    wait_timeout_ms=config.input.waitTimeout
489                ),
490            ),
491            logger=self._logger,
492        ) as redis_stream:
493            if self._logger.isEnabledFor(level=logging.INFO):
494                self._logger.info(
495                    msg=f"Instance {self._instance_name} listening for messages from Redis stream"
496                )
497            await redis_stream.create_input_stream()
498            # receive_client_message loops forever (blocking XREADGROUP with
499            # timeout), so this iteration only ends by raising
500            async for message in redis_stream.receive_client_message():
501                with logging_timer(self._stream_receive_run):
502                    if self._logger.isEnabledFor(level=logging.INFO):
503                        self._logger.info(
504                            msg=f"Instance {self._instance_name} redis message: {message}"
505                        )
506                    match message.root:
507                        case ClientCloudEventsDenmDataV1():
508                            await self._denm_publish(
509                                step_broker=step_broker, message=message.root
510                            )
511                        case _:
512                            if self._logger.isEnabledFor(level=logging.WARNING):
513                                self._logger.warning(
514                                    msg=f"Unknown message type: {message.root}"
515                                )

517    async def _denm_publish(
518        self, step_broker: StepBroker, message: ClientCloudEventsDenmDataV1
519    ) -> None:
520        """Publish a DENM read from the input Redis stream to the STEP broker.
521
522        A message-level failure (any ``StepLibError`` other than
523        ``StepLibMqttError``: encoding, unsupported type, backpressure) is logged
524        and the DENM is dropped, so the bridge keeps running. An MQTT transport
525        failure (``StepLibMqttError``) is annotated with the DENM, which is lost (the
526        input stream is read without acknowledgement), and propagates so that the
527        inherited ``work`` logs it once and reconnects.
528
529        Args:
530            step_broker (StepBroker): STEP broker client used to publish the DENM.
531            message (ClientCloudEventsDenmDataV1): The CloudEvents DENM message read
532                from the input Redis stream.
533
534        Returns:
535            None: The DENM is published as a side effect; nothing is returned.
536
537        Raises:
538            StepLibMqttError: On an MQTT transport failure.
539
540        Complexity:
541            - Time: O(m), m = size (field count/byte length) of the DENM, dominated
542              by ``StepBroker.client_publish_its_message``.
543            - Space: O(m), for the DENM message object and its serialized payload.
544        """
545        try:
546            await step_broker.client_publish_its_message(
547                message=DenmMessageV1(message_param=message.data)
548            )
549        except StepLibMqttError as exc:
550            # Transport failure: the DENM was already consumed from the input
551            # stream (no ack), so it is lost; the note reaches the single log
552            # written by work() before it reconnects
553            exc.add_note(f"Discarding DENM, MQTT transport failure: {message}")
554            raise
555        except StepLibError:
556            # Message-level failure: drop this DENM, keep the bridge running
557            if self._logger.isEnabledFor(level=logging.ERROR):
558                self._logger.exception(
559                    msg=f"Discarding DENM that could not be published: {message}"
560                )

607    @override
608    async def cycle(self) -> None:
609        """Open the Redis pool and the STEP broker and run the bridge tasks.
610
611        Opens a ``StepRedisPool`` (at most ``_REDIS_MAX_CONNECTIONS`` connections,
612        parameters from ``StepRedisPoolParams.from_connection``) and a
613        ``StepBroker``, subscribes to the configured topics and runs
614        ``_step_receive_run``, ``_stream_receive_run`` and
615        ``_redis_health_check_run`` in a ``TaskGroup``. The first failure
616        propagates to the inherited ``work``, which reconnects or ends the run.
617
618        Returns:
619            None: Returns only if every task ends normally, which they never do.
620
621        Complexity:
622            - Time: O(t) for the StepBroker setup and topic subscription, t =
623              number of configured subscription/publish topics, plus O(n * m) for
624              the n messages of size m processed by the receive tasks.
625            - Space: O(t + m), for the StepBroker topic configuration and the
626              message being handled by each task; the three tasks are fixed in
627              number.
628        """
629        async with (
630            StepRedisPool(
631                params=StepRedisPoolParams.from_connection(
632                    config=self._config.redis.connection,
633                    credential=self._redis_credential,
634                    max_connections=self._REDIS_MAX_CONNECTIONS,
635                    logger=self._logger,
636                ),
637                logger=self._logger,
638            ) as redis_pool,
639            StepBroker(
640                instance_name=self._instance_name,
641                sender_station_id=self._config.stationId,
642                sender_latitude=self._config.latitude,
643                sender_longitude=self._config.longitude,
644                config=self._config.stepBroker,
645                mqtt_credential=self._mqtt_credential,
646                logger=self._logger,
647            ) as step_broker,
648        ):
649            await step_broker.subscribe_execution()
650            async with TaskGroup() as group:
651                # Create STEP receive task. First to get a permanent connection
652                group.create_task(
653                    coro=self._step_receive_run(
654                        config=self._config.redis.streams,
655                        pool=redis_pool,
656                        step_broker=step_broker,
657                    )
658                )
659                # Create Redis receive task. It will receive messages from Redis stream
660                group.create_task(
661                    coro=self._stream_receive_run(
662                        config=self._config.redis.streams,
663                        pool=redis_pool,

687    @override
688    def on_shutdown(self, stop_signal: Signals) -> ExitCode:
689        """Log the shutdown signal once it stopped ``work``.
690
691        Called by ``AppRunner`` after every task has unwound; no cleanup is needed
692        here because the Redis pool, the STEP broker and the stream clients are
693        released by their async context managers in ``cycle``.
694
695        Args:
696            stop_signal (Signals): The signal that stopped the run, logged by name
697                at INFO level.
698
699        Returns:
700            ExitCode: Always ExitCode.ABORTED.
701
702        Complexity:
703            - Time: O(1), a single log call.
704            - Space: O(1), no additional data structures are allocated.
705        """
706        if self._logger.isEnabledFor(level=logging.INFO):
707            self._logger.info(msg=f"Program interrupted by {stop_signal.name}")
708        return ExitCode.ABORTED
