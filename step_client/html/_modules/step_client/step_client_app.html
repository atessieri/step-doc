  1"""Workload of the STEP Client program.
  2
  3Defines the ``AbstractApp`` subclass handed to ``AppRunner`` by
  4``step_client.__main__``: it reads the MQTT and Redis credentials from the
  5environment, opens a Redis connection pool and connects to the STEP MQTT broker,
  6then runs three concurrent tasks in a ``TaskGroup``: one forwarding CAM/CPM
  7messages received from the STEP broker to the adapter Redis stream and DENM
  8messages to the producer Redis stream, one forwarding DENM messages read from the
  9input Redis stream to the STEP broker, and one running the Redis health check.
 10Any failure tears the cycle down and, after ``reconnectInterval`` seconds,
 11reconnects; a shutdown cancellation ends the loop instead. Delivery is
 12at-most-once: a message being forwarded when a transport fails is logged and
 13lost, because both the MQTT subscription and the Redis input stream (read
 14without acknowledgement) have already acknowledged it.
 15
 16Complexity:
 17    - Time: O(n), where n is the number of ITS/Redis messages processed over the
 18      program's lifetime; each message is handled in O(1) time, and each
 19      connect/reconnect cycle performs O(t) setup work, t = number of
 20      configured MQTT subscription/publish topics.
 21    - Space: O(t) per reconnect cycle, for the StepBroker topic configuration;
 22      the three concurrent tasks are fixed in number and handling an individual
 23      message uses O(1) additional space.
 24
 25Classes:
 26    - ExitCode: Enumeration of the program exit codes. Complexity: time O(1),
 27      space O(1).
 28    - StepClientApp: Workload run by ``AppRunner``: the reconnecting STEP
 29      broker/Redis stream bridge. Complexity: time O(n) over the program's
 30      lifetime, space O(t).
 31"""

225    def _get_mqtt_redis_credential(self) -> MqttRedisCredential:
226        """Load MQTT broker and Redis connection details from environment variables.
227
228        Reads the MQTT and Redis URL, port, username, and password from the environment
229        (first loading the ``.env`` file found in the current working directory or
230        its nearest parent, if any): ``MQTT_URL``, ``MQTT_PORT``,
231        ``MQTT_USERNAME``, ``MQTT_PASSWORD``, ``REDIS_URL``,
232        ``REDIS_PORT``, ``REDIS_USERNAME`` and ``REDIS_PASSWORD`` (the names held by

235        The settings are logged at INFO level, usernames and passwords excluded (only
236        whether a username is set is logged).
237

240                environment variables.

244                environment variable is not set, if a port is not an integer (see
245                ``_port_parse``), or if ``MqttRedisCredential`` rejects an empty URL
246                or an out-of-range port.
247
248        Complexity:
249            - Time: O(1), a fixed number of environment variables is read.
250            - Space: O(1), a single fixed-size frozen dataclass is allocated.
251        """

253        # site-packages once installed), not from where the program is launched
254        load_dotenv(dotenv_path=find_dotenv(usecwd=True))

273        if redis_port is None:

276            )

278        redis_password = self._env_optional(var_name=self._REDIS_PASSWORD_VAR_NAME)

464    async def _step_receive_run(
465        self, config: RedisStreams, pool: StepRedisPool, step_broker: StepBroker
466    ) -> None:
467        """Receives messages from the STEP broker and publishes them to Redis streams.
468
469        Opens the adapter and producer Redis stream clients, then iterates over the
470        ITS messages received from the STEP broker, routing each one through
471        ``_its_message_dispatch``: CAM and CPM payloads go to the adapter Redis
472        stream and DENM payloads to the producer Redis stream. Messages of unknown
473        type are skipped with a warning. A message whose CloudEvents envelope fails
474        validation (``ValidationError``) is logged and dropped without stopping the
475        loop; any other error (e.g. a Redis failure) gets a note on the exception
476        naming the message type - the message is lost, since MQTT already
477        acknowledged it - and propagates so that ``work`` logs it once and
478        reconnects. The processing time of each message is logged through a
479        ``LoggingTimer``.
480
481        Args:
482            config (RedisStreams): Redis streams configuration, used for the adapter
483                and producer output stream ids and maximum lengths.
484            pool (StepRedisPool): Redis connection pool shared by both stream clients.
485            step_broker (StepBroker): STEP broker instance to receive messages from.
486
487        Returns:
488            None: Returns only if the STEP broker message iterator ends; normally runs
489                until the enclosing task is cancelled or an exception is raised.
490
491        Complexity:
492            - Time: O(n), where n is the number of messages received from the STEP
493              broker; each message is dispatched and processed in O(1) time.
494            - Space: O(1), aside from the constant-size resources held by the adapter
495              and producer Redis stream clients.
496        """
497        logging_timer = LoggingTimer(
498            logger=self._logger, prefix="step_client", log_level=LogMode.INFO
499        )
500        async with (
501            StepAdapterRedisStream(
502                config=StepRedisStreamParams(
503                    stream_id=config.outputs.adapter.streamId,
504                    published_message_max_retained=config.outputs.adapter.maxLength,
505                    consumer_name=self._instance_name,
506                ),
507                logger=self._logger,
508                pool=pool,
509            ) as adapter_stream,
510            StepProducerRedisStream(
511                config=StepRedisStreamParams(
512                    stream_id=config.outputs.producer.streamId,
513                    published_message_max_retained=config.outputs.producer.maxLength,
514                    consumer_name=self._instance_name,
515                ),
516                logger=self._logger,
517                pool=pool,
518            ) as producer_stream,
519        ):
520            if self._logger.isEnabledFor(level=logging.INFO):
521                self._logger.info(
522                    msg=f"Instance {self._instance_name} listening messages from STEP broker"
523                )
524            async for message in step_broker.its_messages:
525                with logging_timer(func=self._step_receive_run):
526                    try:
527                        await self._its_message_dispatch(
528                            adapter_stream=adapter_stream,
529                            producer_stream=producer_stream,
530                            message=message,
531                        )
532                    except ValidationError:
533                        # Message-level failure (the CloudEvents envelope rejected
534                        # the payload): drop this message, keep the bridge running.
535                        # Redis errors are not caught, so work() reconnects
536                        if self._logger.isEnabledFor(level=logging.ERROR):
537                            self._logger.exception(
538                                msg=(
539                                    f"Discarding {message.its_payload.message_type} "
540                                    "message that could not be published to Redis"
541                                )
542                            )
543                    except Exception as exc:
544                        # Any other failure (e.g. Redis): the MQTT message is already
545                        # acknowledged, so it is lost; the note reaches the single log
546                        # written by work() before it reconnects
547                        exc.add_note(
548                            f"Discarding {message.its_payload.message_type} "
549                            "message, Redis publish failed: reconnecting"
550                        )
551                        raise

575                or an exception is raised.

579              stream; each message is matched and forwarded in O(1) time.
580            - Space: O(1), aside from the constant-size resources held by the Redis
581              stream client.
582        """
583        logging_timer = LoggingTimer(
584            logger=self._logger, prefix="step_client", log_level=LogMode.INFO
585        )
586        async with StepClientRedisStream(
587            pool=pool,
588            config=StepRedisStreamParams(
589                stream_id=config.input.streamId,
590                group_id=config.input.groupId,
591                wait_timeout_ms=config.input.waitTimeout,
592                published_message_max_retained=None,
593                consumer_name=self._instance_name,
594                ttl_seconds=max(
595                    1, ceil(config.input.waitTimeout * 2 / ItsTime.SEC_TO_MS_FACTOR)
596                ),
597            ),
598            logger=self._logger,
599        ) as redis_stream:
600            if self._logger.isEnabledFor(level=logging.INFO):
601                self._logger.info(
602                    msg=f"Instance {self._instance_name} listening for messages from Redis stream"
603                )
604            await redis_stream.create_input_stream()
605            # receive_client_message loops forever (blocking XREADGROUP with
606            # timeout), so this iteration only ends by raising
607            async for message in redis_stream.receive_client_message():
608                with logging_timer(self._stream_receive_run):
609                    if self._logger.isEnabledFor(level=logging.INFO):
610                        self._logger.info(
611                            msg=f"Instance {self._instance_name} redis message: {message}"
612                        )
613                    match message.root:
614                        case ClientCloudEventsDenmDataV1():
615                            await self._denm_publish(
616                                step_broker=step_broker, message=message.root
617                            )
618                        case _:
619                            if self._logger.isEnabledFor(level=logging.WARNING):
620                                self._logger.warning(
621                                    msg=f"Unknown message type: {message.root}"
622                                )

624    async def _denm_publish(
625        self, step_broker: StepBroker, message: ClientCloudEventsDenmDataV1
626    ) -> None:
627        """Publish a DENM read from the input Redis stream to the STEP broker.
628
629        A message-level failure (any ``StepLibError`` other than
630        ``StepLibMqttError``: encoding, unsupported type, backpressure) is logged
631        and the DENM is dropped, so the bridge keeps running. An MQTT transport
632        failure (``StepLibMqttError``) is annotated with the DENM, which is lost (the
633        input stream is read without acknowledgement), and propagates so that
634        ``work`` logs it once and reconnects.
635
636        Args:
637            step_broker (StepBroker): STEP broker client used to publish the DENM.
638            message (ClientCloudEventsDenmDataV1): The CloudEvents DENM message read
639                from the input Redis stream.
640
641        Returns:
642            None: The DENM is published as a side effect; nothing is returned.
643
644        Raises:
645            StepLibMqttError: On an MQTT transport failure.
646
647        Complexity:
648            - Time: O(1), builds and publishes a single fixed-size message.
649            - Space: O(1), a single message object is allocated.
650        """
651        try:
652            await step_broker.client_publish_its_message(
653                message=DenmMessageV1(message_param=message.data)
654            )
655        except StepLibMqttError as exc:
656            # Transport failure: the DENM was already consumed from the input
657            # stream (no ack), so it is lost; the note reaches the single log
658            # written by work() before it reconnects
659            exc.add_note(f"Discarding DENM, MQTT transport failure: {message}")
660            raise
661        except StepLibError:
662            # Message-level failure: drop this DENM, keep the bridge running
663            if self._logger.isEnabledFor(level=logging.ERROR):
664                self._logger.exception(
665                    msg=f"Discarding DENM that could not be published: {message}"
666                )

669        self, config: RedisStreams, pool: StepRedisPool

672

675        ``healthCheckInterval`` + PING) and raises on a connection error, keeping the

678        Args:
679            config (RedisStreams): Redis streams configuration, used for the input stream
680                parameters (stream id, group id, wait timeout).
681            pool (StepRedisPool): Redis connection pool used by the health check stream.

683        Returns:

685                or an exception is raised.
686
687        Complexity:
688            - Time: O(1) per health check iteration, run indefinitely; each call to
689              ``run_health_check`` operates on a fixed amount of data.
690            - Space: O(1), aside from the constant-size resources held by the Redis
691              stream client.
692        """
693        async with StepRedisStream(
694            pool=pool,
695            config=StepRedisStreamParams(
696                stream_id=config.input.streamId,
697                group_id=config.input.groupId,
698                wait_timeout_ms=config.input.waitTimeout,
699                published_message_max_retained=None,
700                consumer_name=self._instance_name,
701                ttl_seconds=max(

704            ),
705            logger=self._logger,

774    @override
775    async def work(self) -> ExitCode:
776        """Run the STEP broker/Redis bridge, reconnecting after any failure.
777
778        Each cycle opens a ``StepRedisPool`` (at most ``_REDIS_MAX_CONNECTIONS``
779        connections) and a ``StepBroker``, subscribes to the configured topics and
780        runs ``_step_receive_run``, ``_stream_receive_run`` and
781        ``_redis_health_check_run`` in a ``TaskGroup``. Any exception is logged; if
782        the current task is being cancelled (shutdown in progress) a
783        ``CancelledError`` is raised instead of reconnecting, otherwise the next
784        cycle starts after ``config.reconnectInterval`` seconds. The loop never
785        exits normally: it runs until ``AppRunner`` cancels it on shutdown.
786
787        Returns:
788            ExitCode: Declared by ``AbstractApp``; never returned in practice, since
789                the loop only ends by raising.
790
791        Raises:
792            CancelledError: When an exception is caught while a shutdown
793                cancellation is in progress, to restore the cancellation.
794
795        Complexity:
796            - Time: O(t) per reconnect cycle for the StepBroker setup and topic
797              subscription, t = number of configured subscription/publish topics,
798              plus O(n) over the program's lifetime for the n messages processed
799              by the receive tasks.
800            - Space: O(t), for the StepBroker topic configuration of the current
801              cycle; the three tasks are fixed in number.
802        """
803        if self._logger.isEnabledFor(level=logging.INFO):
804            self._logger.info(msg="Program started")
805        while True:
806            try:
807                async with (
808                    StepRedisPool(
809                        params=self._redis_params_convert(
810                            config=self._config.redis.connection,
811                            redis_credential=self._mqtt_redis_credential,
812                            max_connections=self._REDIS_MAX_CONNECTIONS,
813                        ),
814                        logger=self._logger,
815                    ) as redis_pool,
816                    StepBroker(
817                        instance_name=self._instance_name,
818                        sender_station_id=self._config.stationId,
819                        sender_latitude=self._config.latitude,
820                        sender_longitude=self._config.longitude,
821                        config=self._config.stepBroker,
822                        mqtt_credential=self._mqtt_redis_credential,
823                        logger=self._logger,
824                    ) as step_broker,
825                ):
826                    await step_broker.subscribe_execution()
827                    async with TaskGroup() as group:
828                        # Create STEP receive task. First to get a permanent connection
829                        group.create_task(
830                            coro=self._step_receive_run(
831                                config=self._config.redis.streams,
832                                pool=redis_pool,
833                                step_broker=step_broker,
834                            )
835                        )
836                        # Create Redis receive task. It will receive messages from Redis stream
837                        group.create_task(
838                            coro=self._stream_receive_run(
839                                config=self._config.redis.streams,
840                                pool=redis_pool,
841                                step_broker=step_broker,
842                            )
843                        )
844                        # Create health check task. It will check the health of the Redis connection
845                        group.create_task(
846                            coro=self._redis_health_check_run(
847                                config=self._config.redis.streams, pool=redis_pool
848                            )
849                        )
850            except Exception as exc:
851                if self._logger.isEnabledFor(level=logging.ERROR):
852                    self._logger.exception(msg="Exception unmanaged")
853                task = current_task()
854                if task is not None and task.cancelling() > 0:
855                    # Shutdown in progress: an error raised while unwinding replaced
856                    # the CancelledError; restore it instead of reconnecting
857                    raise CancelledError from exc
858                await asyncio_sleep(delay=self._config.reconnectInterval)
859                if self._logger.isEnabledFor(level=logging.INFO):
860                    self._logger.info(
861                        msg=f"Connection retry after {self._config.reconnectInterval}"
862                    )
