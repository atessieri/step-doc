535    def test_eq_hash(
536        self,
537        cam1: Optional[CamDataV1 | str | int],
538        cam2: Optional[CamDataV1 | str | int],
539        expected_result: bool,
540        test_description: str,
541    ) -> None:
542        """Test equality and hash behavior for CamMessageV1.
543
544        This test checks that:
545
546            - Both None messages
547            - Equal messages (different instances)
548            - Different messages
549            - One None, one with message
550            - Comparison with different type
551            - Comparison with None type
552            - Comparison with integer
553
554        Args:
555            cam1: First CamDataV1, string, or int (or None).
556            cam2: Second CamDataV1, string, or int (or None).
557            expected_result: Expected result of equality and hash comparison.
558            test_description: Description of the test case.
559        """
560        # Arrange
561        if isinstance(cam1, (CamDataV1 | None)):
562            instance1 = CamMessageV1(message=cam1)
563        else:
564            instance1 = cam1  # type: ignore[assignment]
565        if isinstance(cam2, (CamDataV1 | None)):
566            instance2 = CamMessageV1(message=cam2)
567        else:
568            instance2 = cam2  # type: ignore[assignment]
569        hash1 = hash(instance1)
570        hash2 = hash(instance2)
571        # Act
572        result_equal = instance1 == instance2
573        result_hash = hash1 == hash2
574        # Assert
575        assert result_equal == expected_result, (
576            f"Test case '{test_description}' failed: "
577            f"expected {expected_result}, got {result_equal}"
578        )
579        assert result_hash == expected_result, (
580            f"Test case '{test_description}' failed: "
581            f"hash1: {hash1} hash2: {hash2}, expected {expected_result}, got {result_hash}"
582        )
