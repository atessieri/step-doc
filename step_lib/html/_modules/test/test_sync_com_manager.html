1493            """A BaseException that is not an Exception, like CancelledError."""

1515                - Time: O(1), builds a fixed, constant number of mock attributes.

1520            instance.__enter__.return_value = instance
1521            instance.__exit__.return_value = False
1522            if fails:
1523                instance.run.side_effect = _EscapesExceptionCatch("escaped")

1527            make_runner_instance(fails=True) for _ in range(failure_count)

1568            == failure_count + 1

1572                f"Connection status: {ConnectionStatus.RECONNECTING.name}"

2874    def test_stop_during_reconnect_does_not_hang_or_raise_unexpectedly(self) -> None:
2875        """Test that stop() completes cleanly when called while the manager is
2876        disconnected/reconnecting, instead of hanging or raising an unexpected
2877        CancelledError.
2878
2879        Unlike every other test in this file, this one deliberately runs a real
2880        background Thread with a real asyncio event loop (no mocking of
2881        asyncio_run_coroutine_threadsafe or the thread), because the bug this
2882        guards against only manifests from a genuine race between the calling
2883        thread and the event-loop thread: _message_publish_thread (the only
2884        code that ever calls _incoming_queue.task_done()) only exists while
2885        _async_run is inside a live connection, so calling stop() while
2886        disconnected previously left its _incoming_queue.join() pending forever
2887        or cancelled at loop teardown. The target port is bound and immediately
2888        released so the connection attempt is refused right away, reliably
2889        keeping the manager in the disconnected/reconnecting window for the
2890        whole test.
2891
2892        Returns:
2893            None
2894
2895        Complexity:
2896            - Time: O(1), a fixed, constant number of operations/assertions is performed
2897              with no loops over variable-length input.
2898            - Space: O(1), only a fixed, small number of local objects/references is
2899              created.
2900        """
2901        # Arrange
2902        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as probe:
2903            probe.bind(("127.0.0.1", 0))
2904            unused_port = probe.getsockname()[1]
2905        params = SyncComManagerParams(
2906            instance_name="test_stop_during_reconnect",
2907            host="127.0.0.1",
2908            port=unused_port,
2909            username="user",
2910            password="pass",
2911            reconnect_interval=1,
2912            timeout=5,
2913            max_incoming_queue_rooms=10,
2914            max_outgoing_queue_rooms=10,
2915            logger=logging.getLogger("test_stop_during_reconnect"),
2916        )
2917        manager = SyncComManager(params=params)
2918        manager.start()
2919        try:
2920            # No sleep needed: the connection attempt is refused essentially
2921            # instantly (nothing listens on unused_port), so the manager is
2922            # already disconnected/retrying by the time stop() below runs.
2923            start_time = time.monotonic()
2924            # Act
2925            manager.stop()
2926            elapsed = time.monotonic() - start_time
2927        finally:
2928            # Best-effort cleanup should stop() itself have failed/timed out,
2929            # so this test never leaves a background thread running.
2930            if manager._async_thread.is_alive():
2931                manager._async_thread_enable = False
2932                manager._async_thread.join(timeout=params.timeout)
2933        # Assert
2934        assert manager._async_loop is None
2935        assert not manager._async_thread.is_alive()
2936        # Comfortably under params.timeout: proves the primary fix (draining
2937        # _incoming_queue in _async_run before it returns) ran, rather than
2938        # stop() only succeeding via the defense-in-depth timeout fallback.
2939        assert elapsed < params.timeout
