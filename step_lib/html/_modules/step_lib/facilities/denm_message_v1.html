392    def get_message_changed_station_id(self, station_id: int) -> FacilityMessage:
393        """Return a new DENM message with the station ID changed.
394
395        This method builds a new DenmMessageV1 instance with the updated station ID
396        while preserving all other properties from the original message. The
397        station ID is updated both in the main message and in the originating
398        station ID of the action ID. The current instance is left untouched, so its
399        hash and equality remain unchanged.
400
401        Args:
402            station_id (int): The new station ID to set in the message.
403
404        Returns:
405            DenmMessageV1: A new message instance with the updated station ID, or
406            this same instance if the station ID is already up to date.
407
408        Complexity:
409            - Time: O(n), where n is the number of fields in the message, due to the
410              copy of the message and its nested managementContainer and actionId
411              containers.
412            - Space: O(n), where n is the number of fields in the message, for the
413              copied instance.
414        """
415        if self._message.stationId == station_id:
416            return self
417        return DenmMessageV1(
418            message_param=self._message.model_copy(
419                update={
420                    "stationId": station_id,
421                    "managementContainer": self._message.managementContainer.model_copy(
422                        update={
423                            "actionId": self._message.managementContainer.actionId.model_copy(
424                                update={"originatingStationId": station_id}
425                            )
426                        }
427                    ),
428                }
429            )
430        )
