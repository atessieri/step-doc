  1"""Module to serialize the ITS facility.
  2
  3This module provides functionality for encoding and decoding Intelligent Transport Systems (ITS)
  4messages using ASN.1 specifications. It supports various ITS message types including CAM
  5(Cooperative Awareness Messages) and DENM (Decentralized Environmental Notification Messages).
  6
  7The module includes:
  8
  9    - ASN.1 message compilation and caching
 10    - Message encoding/decoding between bytes and structured objects
 11    - Support for different ITS protocol versions
 12    - Conversion between ASN.1 and ORM object representations
 13
 14Main Components:
 15
 16    - FacilitySerializer: Main class handling message encoding/decoding operations
 17
 18Complexity:
 19    - Time: O(f) for construction (__init__ / _init_asn_context), where f is
 20      the total size of the bundled ASN.1 (.asn) grammar files, since the
 21      ASN.1 schemas are compiled once at initialization; O(n) for the

118    def _init_asn_context(self, cache_dir: str | None) -> Any:
119        """Retrieves and compiles ASN.1 context from .asn files.
120
121        This function finds all .asn files in the asn1_files resource directory,
122        compiles them using unaligned packed encoding rules (UPER), and returns
123        the compiled context object.
124
125        Args:
126            cache_dir (str | None): Directory to cache compiled ASN.1 files.
127                If None, no caching is performed.
128
129        Returns:
130            Any: Compiled ASN.1 context object that can be used for encoding/decoding.
131
132        Note:
133            The function uses the package resources system to access .asn files
134            bundled with the package.
135
136        Complexity:
137            - Time: O(f), where f is the total size of the .asn grammar files, to read
138              and compile them.
139            - Space: O(f), where f is the total size of the .asn grammar files, for the
140              resulting compiled ASN.1 context.
141        """
142        with resources.as_file(resources.files(asn1_files)) as path:
143            return compile_files(
144                filenames=[
145                    str(object=file.resolve())
146                    for file in Path(path).glob(pattern="*.asn")
147                    if file.is_file()
148                ],
149                cache_dir=cache_dir,
150                codec="uper",
151            )

153    def _to_facility_message(
154        self, raw_message: dict[Any, Any], message_type: MessageCustomType
155    ) -> FacilityMessage:
156        """Convert raw message data to a FacilityMessage based on the message type.
157
158        This method takes raw message data and a message type identifier, then converts it
159        into the appropriate FacilityMessage subclass. It also checks if the message came
160        from this station and ignores it if so.
161
162        Args:
163            raw_message (dict[Any, Any]): The raw message data to be converted.
164            message_type (MessageCustomType): Enum identifier specifying the type
165                of message and whether it is custom.
166
167        Returns:
168            FacilityMessage: The converted facility message.
169
170        Raises:
171            StepLibUnsupportedMessageError: If the message type is not supported.
172            StepLibEchoMessageError: If the message's station ID matches this station's
173                ID (to avoid processing our own messages).
174
175        Complexity:
176            - Time: O(n), where n is the number of fields in raw_message, to build the
177              corresponding FacilityMessage subclass.
178            - Space: O(n), where n is the number of fields in raw_message, for the
179              resulting FacilityMessage instance.
180        """
181        if self._logger.isEnabledFor(logging.DEBUG):
182            self._logger.debug("Decoded message %s: %s", message_type, raw_message)
183        facility_message: FacilityMessage
184        match message_type.message_type:
185            case MessageId.CAM_PDU:
186                facility_message = CamMessageV1.from_asn(
187                    data=raw_message, is_custom=message_type.is_custom
188                )
189            case MessageId.CPM_PDU:
190                facility_message = CpmMessageV1.from_asn(
191                    data=raw_message, is_custom=message_type.is_custom
192                )
193            case MessageId.DENM_PDU:
194                facility_message = DenmMessageV1.from_asn(
195                    data=raw_message, is_custom=message_type.is_custom
196                )
197            case _:
198                raise StepLibUnsupportedMessageError(
199                    f"Message {message_type} type not supported"
200                )
201        if (self._station_id is not None) and (
202            facility_message.station_id == self._station_id
203        ):
204            raise StepLibEchoMessageError(
205                f"Ignoring own message: {facility_message.station_id}"
206            )
207        return facility_message
