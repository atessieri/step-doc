409    def _convert_to_its_message(self, message: AiomqttMessage) -> ItsMessage | None:  # noqa: C901
410        """Deserialize a raw MQTT message into an ItsMessage.
411
412        Deserializes the binary payload, converts the result to a
413        ``FacilityMessage``, and wraps everything in an ``ItsMessage``
414        instance.  The internal transmit-error counter is reset on success
415        and incremented on failure; once ``MAX_RECEIVE_ERRORS`` consecutive
416        failures occur the exception is re-raised to the caller.
417
418        Args:
419            message (AiomqttMessage): The raw MQTT message whose ``payload``
420                attribute contains the encoded ITS message bytes.
421
422        Returns:
423            ItsMessage: The decoded message with both the original MQTT fields
424            and the deserialized ``FacilityMessage`` payload.
425
426        Raises:
427            StepLibFormatMessageError: If deserialization fails and the
428                consecutive error threshold has not yet been reached.
429            Exception: Re-raises the underlying exception once
430                ``MAX_RECEIVE_ERRORS`` consecutive failures have occurred.
431
432        Complexity:
433            - Time: O(n), where n is the length of message.payload in bytes, to
434              deserialize the GeoNet/BTP/facility layers and build the FacilityMessage.
435            - Space: O(n), where n is the length of message.payload in bytes, for the
436              resulting ItsMessage and FacilityMessage instances.
437        """
438        try:
439            its_message_stack = self._message_serializer.deserialize(
440                message=message.payload
441            )
442            its_message = ItsMessage(
443                topic=message.topic,
444                payload=message.payload,
445                qos=message.qos,
446                retain=message.retain,
447                mid=message.mid,
448                properties=message.properties,
449                its_payload_param=its_message_stack.facility_message,
450            )
451        except StepLibMessageNoDataUsefulError as exc:
452            if self._logger.isEnabledFor(level=logging.INFO):
453                self._logger.info(msg=str(object=exc))
454        except (
455            StepLibUnsupportedMessageError,
456            StepLibValueError,
457            StepLibKeyError,
458        ) as exc:
459            if self._logger.isEnabledFor(level=logging.WARNING):
460                self._logger.warning(msg=f"Error: {exc}")
461        except StepLibEchoMessageError as exc:
462            if self._logger.isEnabledFor(level=logging.INFO):
463                self._logger.info(msg=f"Echo message: {exc}")
464        except StepLibFormatMessageError:
465            self._log_deserialization_error(message=message)
466            if self._logger.isEnabledFor(level=logging.ERROR):
467                self._logger.exception(msg="Message format error")
468        except Exception:
469            self._receive_error_count += 1
470            if self._receive_error_count >= self.MAX_RECEIVE_ERRORS:
471                self._receive_error_count = 0
472                raise
473        else:
474            self._receive_error_count = 0
475            return its_message
476        return None

478    async def __anext__(self) -> ItsMessage:
479        """Retrieve and process the next message from the asynchronous message iterator.
480        This method fetches the next message from the underlying message iterator,
481        attempts to deserialize its payload, and constructs an ItsMessage object. If
482        deserialization is successful, the returned ItsMessage will contain the
483        deserialized payload in its_payload. If an error occurs during deserialization,
484        the error is logged and the message is returned with its original payload.
485
486        Returns:
487            ItsMessage: A message container with topic, payload, QoS, retain flag, message ID,
488            properties, and the deserialized ITS payload if successful.
489
490        Raises:
491            StopAsyncIteration: Implicitly when the underlying iterator is exhausted.
492
493        Note:
494            This method catches all exceptions during deserialization to ensure the iterator
495            continues functioning even if individual messages cannot be properly deserialized.
496
497        Complexity:
498            - Time: O(n), where n is the length of the raw message payload in bytes,
499              amortized across the (typically zero or one) messages that fail to
500              deserialize and are skipped before a valid one is returned.
501            - Space: O(n), where n is the length of the raw message payload in bytes,
502              for the resulting ItsMessage instance.
503        """
504        while True:
505            message = await anext(self._message_iterator)
506            if not isinstance(message.payload, bytes):
507                self._logger.warning(
508                    msg=f"Message payload is not bytes: {type(message.payload)}"
509                )
510            else:
511                its_message = self._convert_to_its_message(message=message)
512                if its_message is not None:
513                    return its_message
