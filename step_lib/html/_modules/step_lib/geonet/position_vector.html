  1"""GeoNetworking Position Vector Module.
  2
  3This module provides functionality for representing and handling position vectors used
  4in the GeoNetworking protocol for Intelligent Transport Systems (ITS) and Vehicular
  5Ad-Hoc Networks (VANETs).
  6
  7A position vector contains geographic location information and optional motion parameters
  8for a network node. It is a fundamental component in geographic addressing and routing
  9in vehicular communication systems.
 10
 11Key Components:
 12
 13    - PositionVectorType: Enumeration of position vector types (LONG or SHORT).
 14      Complexity: time O(1), space O(1).
 15    - PositionVector: Data class representing a node's position with attributes like
 16      geographic coordinates, speed, heading, and network address.
 17      Complexity: time O(1), space O(1).
 18
 19The module supports serialization and deserialization of position vectors to/from binary
 20format according to the ETSI TS 103 836-4-1 V2.1.1 GeoNetworking standard, allowing for
 21efficient transmission over the network.
 22
 23Complexity:
 24    - Time: O(1), a PositionVector is a fixed-size structure - 20 bytes for
 25      the SHORT form and 24 bytes for the LONG form (which adds accuracy,
 26      speed, and heading fields) - so every operation in this module -
 27      construction, validation, parsing from a byte stream, and
 28      serialization to bytes - touches a constant number of fields.
 29    - Space: O(1), only the fixed-size structure is held or produced,
 30      regardless of any surrounding payload.
 31"""

116    def __post_init__(self) -> None:
117        """Validates and adjusts position vector attributes after initialization.
118
119        This method performs the following checks and adjustments:
120
121            - If latitude is outside (-90.0, 90.0) degrees, clamps it to that range.
122            - If longitude is outside [-180.0, 180.0) degrees (upper bound
123              exclusive), wraps it back into that range (e.g. 190.0 becomes
124              -170.0, and 180.0 itself wraps to -180.0).
125            - If speed is provided and outside (0, 163.82) m/s, clamps it to that range.
126            - If heading is provided and outside (0, 360.0) degrees, clamps it to
127              that range.
128
129        Returns:
130            None: This method does not return a value; out-of-range fields are
131                rewritten in place via object.__setattr__ (the dataclass is frozen).
132
133        Note:
134            All out-of-range values are silently clamped or wrapped to the nearest
135            valid value rather than raising an exception.
136
137        Complexity:
138            - Time: O(1), a fixed, small number of bounds checks and clamps.
139            - Space: O(1), no additional storage is required.
140        """
141        if not self.LATITUDE_MIN <= self.latitude <= self.LATITUDE_MAX:
142            object.__setattr__(
143                self,
144                "latitude",
145                max(self.LATITUDE_MIN, min(self.latitude, self.LATITUDE_MAX)),
146            )
147        if not self.LONGITUDE_MIN <= self.longitude < self.LONGITUDE_MAX:
148            object.__setattr__(
149                self,
150                "longitude",
151                (self.longitude + self.LONGITUDE_HALF_RANGE) % (self.LONGITUDE_RANGE)
152                - self.LONGITUDE_HALF_RANGE,
153            )
154        if (
155            self.speed is not None
156        ) and not self.SPEED_MIN <= self.speed <= self.SPEED_MAX:
157            object.__setattr__(
158                self, "speed", max(self.SPEED_MIN, min(self.speed, self.SPEED_MAX))
159            )
160        if (
161            self.heading is not None
162        ) and not self.HEADING_MIN <= self.heading <= self.HEADING_MAX:
163            object.__setattr__(
164                self,
165                "heading",
166                max(self.HEADING_MIN, min(self.heading, self.HEADING_MAX)),

173        """Create a PositionVector object from a BytesIO stream.
174
175        This method reads the bytes from the given stream and constructs a PositionVector object
176        according to the specified packet type.
177        For all packet types, the following fields are included:
178
179            - GN_ADDR (8 bytes): The address.
180            - TST (4 bytes): The timestamp as delta time in milliseconds.
181            - LAT (4 bytes): The latitude scaled by LAT_LON_SCALE.
182            - LONG (4 bytes): The longitude scaled by LAT_LON_SCALE.
183
184        For the LONG packet type, additional fields are included:
185
186            - PAI (1 bit): Position accuracy indicator (is_accurate flag).
187            - S (15 bits): Speed value or SPEED_INVALID if None.
188            - H (2 bytes): Heading value or HEADING_INVALID if None.
189
190        Args:
191            stream (BytesIO): A bytes stream containing the position vector data.
192            packet_type (PositionVectorType): The type of position vector to be parsed
193                (LONG or SHORT).
194
195        Returns:
196            PositionVector: A new PositionVector object created from the stream data.
197
198        Note:
199            For LONG type vectors, the stream should contain additional information such as
200            position accuracy indicator, speed, and heading.
201            For SHORT type vectors, these values are set to defaults
202            (False for accuracy, None for speed and heading).
203
204        Complexity:
205            - Time: O(1), reads a fixed 20 bytes (SHORT) or 24 bytes (LONG) from the
206              stream.
207            - Space: O(1), no additional storage is required.
208        """
209        # GN_ADDR (8 bytes)
210        address = GeoNetworkAddress.from_bytes_stream(stream=stream)
211        # TST (4 bytes)
212        time = ItsTime.create_from_delta_time(
213            delta_time_ms=int.from_bytes(stream.read(4), byteorder="big", signed=False),
214            module=PositionVector.DELTA_TIME_MODULE,
215        )
216        # LAT (4 bytes)
217        latitude = (
218            int.from_bytes(bytes=stream.read(4), byteorder="big", signed=True)
219            / PositionVector.LAT_LON_SCALE
220        )
221        # LONG (4 bytes)
222        longitude = (
223            int.from_bytes(bytes=stream.read(4), byteorder="big", signed=True)
224            / PositionVector.LAT_LON_SCALE
225        )
226        if packet_type == PositionVectorType.LONG:
227            bits = ConstBitStream(bytes=stream.read(2))
228            # PAI (1 bit)
229            is_accurate = bool(bits.read(fmt="bool"))
230            # S (15 bits)
231            speed: int | None = int(bits.read(fmt="uint:15"))
232            if speed == PositionVector.SPEED_INVALID:
233                speed = None
234            # H (2 bytes)
235            heading: int | None = int.from_bytes(
236                bytes=stream.read(2), byteorder="big", signed=True
237            )
238            if heading == PositionVector.HEADING_INVALID:
239                heading = None
240        else:
241            is_accurate = False
242            speed = None
243            heading = None
244        return PositionVector(
245            address=address,
246            time=time,
247            latitude=latitude,
248            longitude=longitude,
249            is_accurate=is_accurate,
250            speed=speed / PositionVector.SPEED_SCALE if speed is not None else None,
251            heading=heading / PositionVector.HEADING_SCALE
252            if heading is not None
253            else None,
254        )

256    def to_bytes(self, packet_type: PositionVectorType) -> bytes:
257        """Converts the position vector to its binary representation.
258
259        This method serializes the position vector into a byte stream according to the
260        specified packet type.
261        For all packet types, the following fields are included:
262
263            - GN_ADDR (8 bytes): The address.
264            - TST (4 bytes): The timestamp as delta time in milliseconds.
265            - LAT (4 bytes): The latitude scaled by LAT_LON_SCALE.
266            - LONG (4 bytes): The longitude scaled by LAT_LON_SCALE.
267
268        For the LONG packet type, additional fields are included:
269
270            - PAI (1 bit): Position accuracy indicator (is_accurate flag).
271            - S (15 bits): Speed value or SPEED_INVALID if None.
272            - H (2 bytes): Heading value or HEADING_INVALID if None.
273
274        Args:
275            packet_type (PositionVectorType): The type of position vector packet to generate.
276                Determines which fields are included in the binary representation.
277
278        Returns:
279            bytes: Binary representation of the position vector.
280
281        Complexity:
282            - Time: O(1), produces a fixed 20-byte (SHORT) or 24-byte (LONG) output.
283            - Space: O(1), only the fixed-size result is allocated.
284        """
285        stream = BytesIO()
286        # GN_ADDR (8 bytes)
287        stream.write(self.address.to_bytes())
288        # TST (4 bytes)
289        stream.write(
290            self.time.get_delta_time_ms(module=self.DELTA_TIME_MODULE).to_bytes(
291                length=4, byteorder="big", signed=False
292            )
293        )
294        # LAT (4 bytes)
295        stream.write(
296            round(self.latitude * self.LAT_LON_SCALE).to_bytes(
297                length=4, byteorder="big", signed=True
298            )
299        )
300        # LONG (4 bytes)
301        stream.write(
302            round(self.longitude * self.LAT_LON_SCALE).to_bytes(
303                length=4, byteorder="big", signed=True
304            )
305        )
306        if packet_type == PositionVectorType.LONG:
307            bits = BitStream()
308            # PAI (1 bit)
309            bits.append(f"bool:1={self.is_accurate}")
310            # S (15 bits)
311            speed_int = (
312                round(self.speed * self.SPEED_SCALE)
313                if self.speed is not None
314                else self.SPEED_INVALID
315            )
316            bits.append(f"uint:15={speed_int}")
317            stream.write(bits.tobytes())
318            # H (2 bytes)
319            heading_int = (
320                round(self.heading * self.HEADING_SCALE)
321                if self.heading is not None
322                else self.HEADING_INVALID
323            )
324            stream.write((heading_int).to_bytes(length=2, byteorder="big", signed=True))
325        return stream.getvalue()
