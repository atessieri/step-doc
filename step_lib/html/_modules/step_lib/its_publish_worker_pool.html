   1"""ITS Publish Worker Pool Module.
   2
   3This module provides ``PublishWorkerPool``, the "ventilator" that lives in
   4the caller's process (process A) and owns the lifecycle of the publish
   5worker pool: binding the pool's single ZMQ endpoint, spawning workers that
   6connect to it, health-checking and restarting dead/blocked workers, and a
   7graceful-then-forceful shutdown. See
   8``doc/source/multiprocess_architecture.rst``,
   9``doc/source/ventilator_worker_protocol.rst``, and
  10``its_publish_worker.py`` (the worker side of this same pipeline).
  11
  12The pool is driven by an explicit state machine (``PoolState``) whose
  13transitions are serialized by one ``threading.Condition``, so ``start()``,
  14``close()``, ``send()`` and ``check_health()`` can be called concurrently
  15from different threads without racing each other.
  16
  17Complexity:
  18    - Time: O(1) for ``send``/``is_running``; O(w), where w is
  19      ``publish_worker_count``, for ``_dead_workers``/``_blocked_workers``
  20      (each scans every worker's metadata) and for
  21      ``start``/``close``/``check_health`` (each acts on every worker in
  22      turn).
  23    - Space: O(w) for the pool's own per-worker bookkeeping (processes,
  24      events, heartbeats), where w is ``publish_worker_count``.

 107        code (``GetLastError``). Always ``_WIN32_ERROR_SUCCESS`` on non-Windows

 123@contextmanager
 124def _interrupt_shielded_spawn(logger: Logger) -> Iterator[None]:
 125    """Make every process created inside the block start with Ctrl+C ignored.
 126
 127    Ctrl+C reaches every process attached to the terminal (Windows: the
 128    console's ``CTRL_C_EVENT``; POSIX: the TTY's ``SIGINT`` to the foreground
 129    process group), including a worker still bootstrapping in
 130    ``spawn_main`` - before ``PublishWorker.run()`` could install anything -
 131    which would die with an unhandled ``KeyboardInterrupt`` traceback.
 132    Workers are stopped by the pool via ``stop_event`` instead.
 133
 134    On Windows the inheritable "ignore CTRL_C" process attribute is set for the
 135    duration of the block, under ``_CTRL_C_INHERIT_LOCK``; being process-wide,
 136    the parent itself discards (does not defer) a Ctrl+C arriving meanwhile -
 137    the block must only wrap ``Process.start()``, a few milliseconds. Windows
 138    offers no API to read the attribute, so the block always ends by clearing
 139    it: a caller that was itself started with Ctrl+C ignored (e.g. through
 140    ``start /b``) handles Ctrl+C again after its first worker spawn. On POSIX
 141    ``SIGINT`` is blocked in the calling thread only: ``fork_exec`` hands that
 142    mask to the child, while the parent keeps receiving ``SIGINT`` through its
 143    other threads, so nothing is lost. ``PublishWorker._signals_install``
 144    then sets ``SIG_IGN`` and unblocks it in the child.
 145
 146    Args:
 147        logger (Logger): Logger used to report a failed Windows call.
 148
 149    Yields:
 150        None: Control returns to the caller with the shield active.
 151
 152    Returns:
 153        Iterator[None]: Generator (wrapped as a context manager) that yields
 154            once, while the shield is active.
 155
 156    Complexity:
 157        - Time: O(1), two Win32 calls or two ``pthread_sigmask`` calls.
 158        - Space: O(1), at most one saved signal mask.
 159    """
 160    # sys.platform (not IS_WINDOWS) so type checkers narrow each branch to the
 161    # platform whose API it uses (pthread_sigmask does not exist on Windows)
 162    if sys.platform == "win32":
 163        with _CTRL_C_INHERIT_LOCK:
 164            error = _set_console_ctrl_c_ignored(ignored=True)
 165            if error != _WIN32_ERROR_SUCCESS and logger.isEnabledFor(
 166                level=logging.DEBUG
 167            ):
 168                logger.debug(
 169                    msg=f"SetConsoleCtrlHandler(NULL, TRUE) failed (winerror={error}): "
 170                    "the worker may still receive Ctrl+C while bootstrapping"
 171                )
 172            try:
 173                yield
 174            finally:
 175                _set_console_ctrl_c_ignored(ignored=False)
 176    else:
 177        # pylint does not narrow on sys.platform: these POSIX-only members are
 178        # reported as missing when linting on Windows
 179        previous_mask = signal.pthread_sigmask(
 180            signal.SIG_BLOCK,  # pylint: disable=no-member
 181            {signal.SIGINT},
 182        )
 183        try:
 184            yield
 185        finally:
 186            signal.pthread_sigmask(
 187                signal.SIG_SETMASK,  # pylint: disable=no-member
 188                previous_mask,
 189            )

 246class PublishWorkerPool:
 247    """Owns the lifecycle of the publish worker process pool.
 248
 249    That is: binding the pool's single endpoint, spawning workers,
 250    health-checking/restarting, and shutdown.
 251
 252    Every ``multiprocessing`` primitive this pool creates (``Process``,
 253    ``Event``, ``Value``, ``Queue``) comes from an explicit ``spawn``
 254    context (``self._mp_context``), never the ambient default - Linux
 255    otherwise defaults to ``fork``, which is unsafe once the parent is
 256    multi-threaded (as it already is by the time this pool starts, since
 257    ``SyncComManager.start()`` starts its own background thread first).
 258
 259    The one ZMQ ``PUSH`` socket this pool owns is bound once, to one
 260    fixed port, in ``start()`` - every worker connects a ``PULL`` socket
 261    to that same endpoint, so the pool's own socket never needs to
 262    connect/disconnect per worker. Restarting a worker therefore never
 263    touches the socket.
 264
 265    Synchronization: one ``threading.Condition`` (``_cond``, wrapping an
 266    ``RLock``) guards the lifecycle state (``_state``), the socket
 267    (``_sock``/``_context``) and the per-worker bookkeeping (``_workers``,
 268    ``_ready_events``, ``_stop_events``, ``_heartbeats``, ``_restarting``,
 269    ``_restart_backoff``). It is only ever held for short reads/mutations and
 270    for the (non-blocking) ``send``, never across a blocking call such as
 271    ``proc.join()``, ``_await_ready()``'s poll loop or ``context.term()``.
 272    Slow work is instead isolated by the state machine: ``start()`` and
 273    ``close()`` claim a transitional state (``STARTING``/``CLOSING``)
 274    under the lock, do the slow work without it, then publish the final
 275    state and ``notify_all()``.
 276
 277    ``_generation`` counts the ``start()`` calls that reached ``STARTING``.
 278    A restart is bound to the generation it was claimed under and is
 279    abandoned (its freshly started process reaped, nothing written) as soon
 280    as the pool leaves ``RUNNING`` or a newer generation begins. Normally
 281    ``close()`` waits for restarts to drain, so this only matters if a
 282    restart is stuck past the drain budget: it then can neither orphan a
 283    process nor touch the bookkeeping of a pool that was started again.
 284
 285    State transitions (``PoolState``)::
 286
 287        CLOSED --start()--> STARTING --ok--> RUNNING --close()--> CLOSING --> CLOSED
 288                               |                                                ^
 289                               +---------------- error -------------------------+
 290
 291    ``start()`` and ``close()`` are symmetric: if the pool is in a
 292    transitional state they wait for it to settle (``RUNNING``/``CLOSED``)
 293    and then either return (already in the requested state) or perform the
 294    transition themselves. An error raised while starting reaches only the
 295    thread that executed that ``start()``; waiting callers see ``CLOSED``
 296    and retry on their own. ``check_health()`` only acts while ``RUNNING``,
 297    and ``close()`` waits for restarts already in flight (``_restarting``)
 298    before tearing anything down, so no worker is orphaned.
 299
 300    Attributes:
 301        params (SyncComManagerParams): The configuration shared with
 302            every worker (each worker receives its own copy, with a
 303            distinct ``instance_name``, built in ``PublishWorker.run()``).
 304        logger (Logger): Logger instance for the pool and every worker
 305            it spawns.
 306
 307    Complexity:
 308        - Time: O(w) for start/close/check_health, where w is
 309          ``publish_worker_count``; O(1) for send/is_running.
 310        - Space: O(w) for the pool's per-worker bookkeeping.
 311    """

 406    def _attempt_budget(self) -> float:
 407        """Worst-case duration of one worker spawn attempt.
 408
 409        One attempt launches the process, polls for readiness up to
 410        ``publish_worker_startup_timeout`` and, if that expires, reaps the
 411        process with ``terminate()`` then ``kill()`` (each bounded by
 412        ``publish_forceful_join_timeout``).
 413
 414        Returns:
 415            float: ``SPAWN_OVERHEAD + startup_timeout + 2 * forceful_join_timeout``,
 416            in seconds.
 417
 418        Complexity:
 419            - Time: O(1), a fixed arithmetic expression.
 420            - Space: O(1), no additional storage is required.
 421        """
 422        return (
 423            self.SPAWN_OVERHEAD
 424            + self._params.publish_worker_startup_timeout
 425            + 2 * self._params.publish_forceful_join_timeout
 426        )

 428    def _start_budget(self) -> float:
 429        """Worst-case duration of ``start()``.
 430
 431        Workers are spawned one after the other, each with up to
 432        ``MAX_WORKER_SPAWN_ATTEMPTS`` attempts.
 433
 434        Returns:
 435            float: ``w * MAX_WORKER_SPAWN_ATTEMPTS * attempt_budget +
 436            TIMEOUT_MARGIN``, in seconds.
 437
 438        Complexity:
 439            - Time: O(1), a fixed arithmetic expression.
 440            - Space: O(1), no additional storage is required.
 441        """
 442        return (
 443            self._params.publish_worker_count
 444            * self.MAX_WORKER_SPAWN_ATTEMPTS
 445            * self._attempt_budget()
 446            + self.TIMEOUT_MARGIN
 447        )

 449    def _restart_drain_budget(self) -> float:
 450        """Worst-case time for a restart already in flight to give up once
 451        ``close()`` has entered ``CLOSING``.
 452
 453        A restart re-checks the state at every attempt and at every
 454        readiness poll, so it exits within one poll interval, plus the time
 455        to reap its own not-yet-ready process.
 456
 457        Returns:
 458            float: ``READINESS_POLL_INTERVAL + SPAWN_OVERHEAD +
 459            2 * forceful_join_timeout``, in seconds.
 460
 461        Complexity:
 462            - Time: O(1), a fixed arithmetic expression.
 463            - Space: O(1), no additional storage is required.
 464        """
 465        return (
 466            self.READINESS_POLL_INTERVAL
 467            + self.SPAWN_OVERHEAD
 468            + 2 * self._params.publish_forceful_join_timeout
 469        )

 471    def _close_budget(self) -> float:
 472        """Worst-case duration of ``close()``.
 473
 474        Draining in-flight restarts, then the graceful join (one shared
 475        deadline covering the linger flush and the workers' drain), then
 476        ``terminate()``/``kill()`` escalation for every worker in turn.
 477
 478        Returns:
 479            float: ``restart_drain_budget + 2 * graceful_shutdown_timeout +
 480            w * 2 * forceful_join_timeout + TIMEOUT_MARGIN``, in seconds.
 481
 482        Complexity:
 483            - Time: O(1), a fixed arithmetic expression.
 484            - Space: O(1), no additional storage is required.
 485        """
 486        return (
 487            self._restart_drain_budget()
 488            + 2 * self._params.publish_graceful_shutdown_timeout
 489            + self._params.publish_worker_count
 490            * 2
 491            * self._params.publish_forceful_join_timeout
 492            + self.TIMEOUT_MARGIN
 493        )

 534    def _wait_settled(self) -> None:
 535        """Block until the pool is in a stable state (``RUNNING`` or ``CLOSED``).
 536
 537        Must be called with ``_cond`` held; the lock is released while
 538        waiting and re-acquired before returning, so the caller can act on
 539        the settled state atomically.

 620    def _reset_bookkeeping(self) -> None:
 621        """Forget every worker, the restart backoff and the bound endpoint, so the
 622        pool can be (re)started from a clean slate.
 623
 624        The GeoNetworking sequence number counter is kept: it lives as long as
 625        the pool.

 633        """

 635            self._workers.clear()
 636            self._ready_events.clear()
 637            self._stop_events.clear()
 638            self._heartbeats.clear()
 639            self._restarting.clear()
 640            self._restart_backoff.clear()
 641            self._endpoint = None

 643    def _dead_workers(self) -> list[int]:
 644        """List the indices of workers whose OS process has terminated.
 645
 646        Snapshots ``self._workers`` under ``_cond`` before checking
 647        ``is_alive()`` on each, so a concurrent restart's writes to that
 648        dict are never observed half-applied.
 649
 650        Returns:
 651            list[int]: The indices of workers for which ``is_alive()`` is
 652            False.
 653
 654        Complexity:
 655            - Time: O(w), where w is the number of workers, to check
 656              ``is_alive()`` on each.
 657            - Space: O(d), where d is the number of dead workers found.
 658        """
 659        with self._cond:
 660            workers = list(self._workers.items())
 661        return [worker_index for worker_index, proc in workers if not proc.is_alive()]

 663    def _blocked_workers(self) -> list[int]:
 664        """List the indices of workers that are alive but whose heartbeat has
 665        gone stale.
 666
 667        A stale heartbeat means every task on that worker's event loop -
 668        including its own heartbeat refresh - has stopped making progress,
 669        which ``is_alive()`` alone cannot detect. Snapshots
 670        ``self._workers``/``self._heartbeats`` under ``_cond`` before
 671        checking, for the same reason as ``_dead_workers``.
 672
 673        Returns:
 674            list[int]: The indices of workers that are alive but whose
 675            heartbeat is older than ``publish_heartbeat_timeout``.
 676
 677        Complexity:
 678            - Time: O(w), where w is the number of workers, to check
 679              ``is_alive()`` and the heartbeat timestamp of each.
 680            - Space: O(b), where b is the number of blocked workers found.
 681        """
 682        now = monotonic()
 683        with self._cond:
 684            workers = list(self._workers.items())
 685            heartbeats = dict(self._heartbeats)
 686        return [
 687            worker_index
 688            for worker_index, proc in workers
 689            if proc.is_alive()
 690            and (now - heartbeats[worker_index].value)
 691            > self._params.publish_heartbeat_timeout
 692        ]

 694    def _claim_worker_for_restart(
 695        self, worker_index: int, auto_restart: bool, expected_proc: BaseProcess
 696    ) -> bool:
 697        """Atomically check a worker is still the one that was found
 698        unhealthy, is not already being restarted and is past its restart
 699        backoff, and if so mark it in-flight.
 700
 701        This is the check-and-mark step that closes the races between
 702        concurrent ``check_health()`` passes, and between ``check_health()``
 703        and ``close()``: every condition and the ``_restarting`` mark
 704        happen in the same ``_cond`` critical section, so a second caller
 705        can never observe the worker as eligible after the first has
 706        claimed it, and no restart can be claimed once ``close()`` has left
 707        the ``RUNNING`` state. The identity check on ``expected_proc``
 708        rejects a stale detection: if another restart already replaced the
 709        process in that slot, the (healthy) replacement must not be
 710        restarted again. A slot whose last restart failed keeps the dead
 711        process it was given, so it is found dead again and retried, but
 712        only once its backoff delay (``_restart_backoff``) has elapsed; a
 713        diagnostic pass (``auto_restart=False``) ignores the backoff.
 714
 715        Args:
 716            worker_index (int): The worker slot to claim.
 717            auto_restart (bool): If False, this is a read-only diagnostic
 718                pass - the worker is not marked in-flight, only the
 719                eligibility checks are applied.
 720            expected_proc (BaseProcess): The process that was found
 721                unhealthy; the claim only succeeds while it still occupies
 722                the slot.
 723
 724        Returns:
 725            bool: True if the caller should proceed (log and, if
 726            ``auto_restart``, restart this worker); False if it should be
 727            skipped.
 728
 729        Complexity:
 730            - Time: O(1), a single lock-protected membership check and set
 731              mutation.
 732            - Space: O(1), no additional storage is required.
 733        """
 734        with self._cond:
 735            if self._state is not PoolState.RUNNING:
 736                return False
 737            if worker_index in self._restarting:
 738                return False
 739            if self._workers.get(worker_index) is not expected_proc:
 740                return False
 741            if auto_restart:
 742                next_attempt = self._restart_backoff.get(worker_index)
 743                if next_attempt is not None and monotonic() < next_attempt[0]:
 744                    if self._logger.isEnabledFor(level=logging.DEBUG):
 745                        self._logger.debug(
 746                            msg=f"Restart of worker {worker_index} deferred for "
 747                            f"{next_attempt[0] - monotonic():.1f} s"
 748                        )
 749                    return False
 750                self._restarting.add(worker_index)
 751            return True

 753    def _record_restart_outcome(
 754        self, worker_index: int, generation: int, succeeded: bool
 755    ) -> None:
 756        """Update the restart backoff of a slot after a restart attempt.
 757
 758        A success clears the backoff; a failure schedules the next attempt
 759        after ``RESTART_BACKOFF_INITIAL`` seconds, doubling the delay
 760        (``RESTART_BACKOFF_FACTOR``) at every further failure up to
 761        ``RESTART_BACKOFF_MAX``. An outcome from an older generation is ignored.
 762
 763        Args:
 764            worker_index (int): The restarted slot.
 765            generation (int): The generation the restart was claimed under.
 766            succeeded (bool): Whether the restart produced a ready worker.
 767
 768        Returns:
 769            None: ``_restart_backoff`` is updated as a side effect.
 770
 771        Complexity:
 772            - Time: O(1), a lock-protected dictionary update.
 773            - Space: O(1), at most one entry per slot.
 774        """
 775        with self._cond:
 776            if generation != self._generation:
 777                return
 778            if succeeded:
 779                self._restart_backoff.pop(worker_index, None)
 780                return
 781            previous = self._restart_backoff.get(worker_index)
 782            delay = (
 783                self.RESTART_BACKOFF_INITIAL
 784                if previous is None
 785                else min(
 786                    previous[1] * self.RESTART_BACKOFF_FACTOR, self.RESTART_BACKOFF_MAX
 787                )
 788            )
 789            self._restart_backoff[worker_index] = (monotonic() + delay, delay)

 791    def _release_restart(
 792        self, worker_index: int, generation: int | None = None
 793    ) -> None:
 794        """Mark a worker slot as no longer being restarted and wake any
 795        thread (typically ``close()``) waiting for restarts to drain.
 796
 797        Idempotent: releasing a slot that is not marked is a no-op. A
 798        restart from an older generation never releases the slot: by then
 799        the slot belongs to the current generation's own restart (or was
 800        already cleared when the pool was reset), and discarding its mark
 801        would let a second restart of the same slot start concurrently.
 802
 803        Args:
 804            worker_index (int): The worker slot to release.
 805            generation (int | None): The generation the releasing restart
 806                was claimed under, or None to release unconditionally.
 807                Defaults to None.
 808
 809        Returns:
 810            None: The slot is removed from ``_restarting`` as a side effect.
 811
 812        Complexity:
 813            - Time: O(1), a single lock-protected set mutation.
 814            - Space: O(1), no additional storage is required.
 815        """
 816        with self._cond:
 817            if generation is None or generation == self._generation:
 818                self._restarting.discard(worker_index)
 819            self._cond.notify_all()

 821    def _terminate_then_kill(self, proc: BaseProcess) -> None:
 822        """Forcefully reap a process: ``terminate()`` and, if it is still
 823        alive after ``publish_forceful_join_timeout``, ``kill()``.
 824
 825        Args:
 826            proc (BaseProcess): The process to reap.
 827
 828        Returns:
 829            None: The process has been signalled (and joined) as a side
 830            effect; a process still alive after ``kill()`` is abandoned with an
 831            ERROR record.
 832
 833        Complexity:
 834            - Time: O(publish_forceful_join_timeout), bounded by two joins.
 835            - Space: O(1), no additional storage is required.
 836        """
 837        proc.terminate()
 838        proc.join(timeout=self._params.publish_forceful_join_timeout)
 839        if proc.is_alive():
 840            proc.kill()
 841            proc.join(timeout=self._params.publish_forceful_join_timeout)
 842        if proc.is_alive() and self._logger.isEnabledFor(level=logging.ERROR):
 843            self._logger.error(
 844                msg=f"Worker process {proc.name} (pid={proc.pid}) still alive after "
 845                "kill(), abandoned"
 846            )

 848    def _spawn_worker_with_retry(
 849        self, worker_index: int, generation: int | None = None
 850    ) -> None:
 851        """Spawn one worker on the pool's endpoint, retrying if it fails to
 852        become ready.
 853
 854        A worker connecting to an already-bound endpoint has no port race
 855        to lose, so a failure here means the process itself didn't come up
 856        cleanly (e.g. it crashed during its own startup). Retrying a fresh
 857        process on the same endpoint, up to ``MAX_WORKER_SPAWN_ATTEMPTS``
 858        times, gives ``start()``/``_restart_worker`` the same tolerance for
 859        that as they always had for a port collision. The pool state is
 860        re-checked before every attempt, so a restart gives up early once
 861        ``close()`` has begun.
 862
 863        Args:
 864            worker_index (int): The slot to spawn.
 865            generation (int | None): The generation a restart was claimed
 866                under, or None when called by ``start()``. Once the work is
 867                superseded (see ``_abort_requested``) it is abandoned and
 868                every process it created is reaped. Defaults to None.
 869
 870        Returns:
 871            None: On success the worker is ready; otherwise StepLibError is
 872            raised.
 873
 874        Raises:
 875            StepLibError: If the work was superseded (pool closing, or a
 876                newer start began), or if no worker could be started within
 877                ``MAX_WORKER_SPAWN_ATTEMPTS`` attempts.
 878
 879        Complexity:
 880            - Time: O(MAX_WORKER_SPAWN_ATTEMPTS) in the worst case, each
 881              attempt dominated by ``_await_ready``.
 882            - Space: O(1), no additional storage beyond the spawned
 883              worker's own bookkeeping.
 884        """
 885        for _ in range(self.MAX_WORKER_SPAWN_ATTEMPTS):
 886            if self._should_abort(generation=generation):
 887                raise StepLibError("Worker pool is closing")
 888            proc = self._spawn_worker(worker_index=worker_index, generation=generation)
 889            try:
 890                self._await_ready(worker_index=worker_index)
 891            except StepLibError:
 892                if self._should_abort(generation=generation):
 893                    raise
 894                if self._logger.isEnabledFor(level=logging.WARNING):
 895                    self._logger.warning(
 896                        msg=f"Worker {worker_index} failed to start, retrying"
 897                    )
 898                continue
 899            if self._should_abort(generation=generation):
 900                self._terminate_then_kill(proc=proc)
 901                raise StepLibError("Worker pool is closing")
 902            return
 903        raise StepLibError(
 904            f"Worker {worker_index} could not be started within "
 905            f"{self.MAX_WORKER_SPAWN_ATTEMPTS} attempts"
 906        )

 908    def _spawn_worker(
 909        self, worker_index: int, generation: int | None = None
 910    ) -> BaseProcess:
 911        """Construct and start one worker process, recording its bookkeeping.
 912
 913        The bookkeeping is only recorded if the work is still current when
 914        the process has started (see ``_abort_requested``); otherwise the
 915        fresh process is reaped and nothing is written, so a superseded
 916        restart can never leave an orphan or touch a newer generation.
 917
 918        The process is started inside ``_interrupt_shielded_spawn``, so it
 919        ignores Ctrl+C from its first instruction and is only ever stopped
 920        by the pool (``stop_event``, then ``terminate()``/``kill()``).
 921
 922        Args:
 923            worker_index (int): The slot this worker occupies in the pool.
 924            generation (int | None): The generation a restart was claimed
 925                under, or None when called by ``start()``. Defaults to None.
 926
 927        Returns:
 928            BaseProcess: The new worker process, whose bookkeeping is
 929            recorded on the pool as a side effect.
 930
 931        Raises:
 932            StepLibError: If the work was superseded while the process was
 933                starting.
 934            RuntimeError: If the log queue or the endpoint has not been
 935                set up yet (i.e. ``start()`` has not begun its setup).
 936
 937        Complexity:
 938            - Time: O(1), creates a fixed number of multiprocessing
 939              primitives and starts one process.
 940            - Space: O(1), a fixed number of per-worker references is
 941              stored.
 942        """
 943        # Log queue must be set before spawning a worker
 944        if self._log_queue is None:
 945            raise RuntimeError("Log queue must be set before spawning a worker")
 946        if self._endpoint is None:
 947            raise RuntimeError("Pool endpoint must be bound before spawning a worker")
 948        ready_event = self._mp_context.Event()
 949        stop_event = self._mp_context.Event()
 950        heartbeat = self._mp_context.Value("d", monotonic())
 951        worker = PublishWorker(
 952            endpoint=self._endpoint,
 953            worker_index=worker_index,
 954            ready_event=ready_event,
 955            stop_event=stop_event,
 956            heartbeat=heartbeat,
 957            logger_name=self._logger.name,
 958            effective_level=self._logger.getEffectiveLevel(),
 959            log_queue=self._log_queue,
 960            params=self._params,
 961        )
 962        process = self._mp_context.Process(
 963            target=worker.run,
 964            daemon=True,
 965            name=f"{self._params.instance_name}-publish-worker-{worker_index}",
 966        )
 967        # The worker must start with Ctrl+C ignored: see _interrupt_shielded_spawn
 968        with _interrupt_shielded_spawn(logger=self._logger):
 969            process.start()
 970        with self._cond:
 971            superseded = self._abort_requested(generation=generation)
 972            if not superseded:
 973                self._workers[worker_index] = process
 974                self._ready_events[worker_index] = ready_event
 975                self._stop_events[worker_index] = stop_event
 976                self._heartbeats[worker_index] = heartbeat
 977        if superseded:
 978            self._terminate_then_kill(proc=process)
 979            raise StepLibError(f"Worker {worker_index} spawn abandoned: pool closing")
 980        return process

 982    def _await_ready(self, worker_index: int) -> None:
 983        """Poll for one worker to signal readiness, bounded by
 984        ``publish_worker_startup_timeout``.
 985
 986        Polls rather than blocking on one long wait, so a worker that
 987        fails for any reason - or a pool that starts closing - is detected
 988        within roughly ``READINESS_POLL_INTERVAL``, not the full timeout.
 989        The slot's objects are read once, under ``_cond``; the slot itself
 990        is owned by the calling thread for the whole call (``start()`` or
 991        the restart that claimed it).
 992
 993        Args:
 994            worker_index (int): The worker to wait for.
 995
 996        Returns:
 997            None: Returns normally once the worker signals readiness;
 998            otherwise raises.
 999
1000        Raises:
1001            StepLibError: If the worker exits before becoming ready, if the
1002                pool starts closing meanwhile, or if it does not become
1003                ready within the timeout (in the last two cases it is
1004                terminated and reaped, escalating to ``kill()`` if it does
1005                not exit promptly).
1006
1007        Complexity:
1008            - Time: O(timeout / READINESS_POLL_INTERVAL) in the worst
1009              case.
1010            - Space: O(1), no additional storage is required.
1011        """
1012        deadline = monotonic() + self._params.publish_worker_startup_timeout
1013        with self._cond:
1014            ready_event = self._ready_events.get(worker_index)
1015            proc = self._workers.get(worker_index)
1016        if ready_event is None or proc is None:
1017            raise StepLibError(f"Worker {worker_index} slot was cleared while waiting")
1018        while monotonic() < deadline:
1019            if ready_event.wait(timeout=self.READINESS_POLL_INTERVAL):
1020                return
1021            if not proc.is_alive():
1022                raise StepLibError(
1023                    f"Worker {worker_index} exited before becoming ready "
1024                    f"(exitcode={proc.exitcode})"
1025                )
1026            if self._is_closing():
1027                self._terminate_then_kill(proc)
1028                raise StepLibError(
1029                    f"Worker {worker_index} startup aborted: pool is closing"
1030                )
1031        self._terminate_then_kill(proc)
1032        raise StepLibError(
1033            f"Worker {worker_index} did not become ready within "
1034            f"{self._params.publish_worker_startup_timeout}s"
1035        )

1037    def _restart_worker(self, worker_index: int, generation: int | None = None) -> None:
1038        """Replace a dead/blocked worker with a freshly spawned one.
1039
1040        The pool's own PUSH socket's single bind never changes, so this
1041        has nothing to disconnect or reconnect on the pool's side - only
1042        the replacement process connecting to the same, already-known
1043        ``self._endpoint`` matters. The slot is marked in-flight
1044        (``_restarting``) under ``_cond`` up front, so a concurrent
1045        ``check_health()`` pass doesn't try to restart it again and
1046        ``close()`` waits for this restart before tearing down.
1047        Spawn/await-ready (potentially slow) then runs with no lock held,
1048        so ``send()`` remains usable against the other workers for the
1049        whole duration. If every attempt fails - or any other exception is
1050        raised while spawning, e.g. an ``OSError`` from ``process.start()`` -
1051        the error is logged and the slot keeps its dead process, so the next
1052        health check finds it dead again and retries once the restart backoff
1053        (``_record_restart_outcome``) has elapsed; if the pool starts closing
1054        meanwhile, the restart is abandoned quietly.
1055
1056        Args:
1057            worker_index (int): The worker slot to restart.
1058            generation (int | None): The generation the restart was claimed
1059                under (see ``_generation``); None captures the current one.
1060                A restart whose generation is already superseded returns
1061                at once, and one superseded later abandons its work,
1062                without touching the newer generation's bookkeeping.
1063                Defaults to None.
1064
1065        Returns:
1066            None: The slot is either replaced with a freshly spawned
1067            worker or left for the next health check, after its backoff.
1068
1069        Complexity:
1070            - Time: O(MAX_WORKER_SPAWN_ATTEMPTS * publish_worker_startup_timeout
1071              / READINESS_POLL_INTERVAL) in the worst case, dominated by
1072              ``_await_ready`` across every retry.
1073            - Space: O(1), no additional storage beyond the replacement
1074              worker's own bookkeeping.
1075        """
1076        with self._cond:
1077            if generation is None:
1078                generation = self._generation
1079            elif self._abort_requested(generation=generation):
1080                return
1081            self._restarting.add(worker_index)
1082        try:
1083            self._spawn_worker_with_retry(
1084                worker_index=worker_index, generation=generation
1085            )
1086            self._record_restart_outcome(
1087                worker_index=worker_index, generation=generation, succeeded=True
1088            )
1089            if self._logger.isEnabledFor(level=logging.INFO):
1090                self._logger.info(msg=f"Worker {worker_index} restarted")
1091        except StepLibError:
1092            if self._should_abort(generation=generation):
1093                if self._logger.isEnabledFor(level=logging.INFO):
1094                    self._logger.info(
1095                        msg=f"Restart of worker {worker_index} abandoned: pool closing"
1096                    )
1097            else:
1098                self._record_restart_outcome(
1099                    worker_index=worker_index, generation=generation, succeeded=False
1100                )
1101                if self._logger.isEnabledFor(level=logging.ERROR):
1102                    self._logger.exception(
1103                        msg=f"Failed to restart worker {worker_index}"
1104                    )
1105        except Exception:
1106            # Any other failure (e.g. OSError/MemoryError from process.start()) leaves
1107            # the slot for the next health check instead of propagating
1108            self._record_restart_outcome(
1109                worker_index=worker_index, generation=generation, succeeded=False
1110            )
1111            if self._logger.isEnabledFor(level=logging.ERROR):
1112                self._logger.exception(msg=f"Failed to restart worker {worker_index}")
1113        finally:
1114            self._release_restart(worker_index=worker_index, generation=generation)

1116    def _bind_and_spawn(self) -> tuple[ZmqContext[ZmqSocket[bytes]], ZmqSocket[bytes]]:
1117        """Do the slow part of ``start()``, with no lock held.
1118
1119        Sets up the log queue and listener (which hands every worker record
1120        to the parent's logger of the same name, see ``_LoggerDispatchHandler``),
1121        binds the PUSH socket (HWM ``publish_hwm``, ``LINGER=0``), spawns every
1122        worker and waits until each has completed the ZMQ handshake
1123        (``_await_handshakes``), so a ``send()`` right after ``start()`` never
1124        finds the socket without peers. Only the thread that claimed
1125        ``STARTING`` runs this, so it may write ``_log_queue``/``_listener``/
1126        ``_endpoint`` directly. On any failure (including ``BaseException``)
1127        every still-alive spawned worker is reaped (``terminate()`` then
1128        ``kill()``), the socket and context are closed, the listener is
1129        stopped, and the bookkeeping is reset before the error propagates.
1130
1131        Returns:
1132            tuple[ZmqContext, ZmqSocket]: The ZMQ context and its bound PUSH
1133            socket, in that order, not yet published on the pool.
1134
1135        Raises:
1136            StepLibError: If a worker can't be started within
1137                ``MAX_WORKER_SPAWN_ATTEMPTS`` attempts, or if the workers do not
1138                all complete the ZMQ handshake within
1139                ``publish_worker_startup_timeout``.
1140            zmq.ZMQError: If the socket cannot be created or bound (wrapped
1141                into StepLibError by ``start()``).
1142
1143        Complexity:
1144            - Time: O(w), where w is ``publish_worker_count``, to spawn and
1145              await readiness for every worker.
1146            - Space: O(w), for the per-worker bookkeeping created.
1147        """
1148        context: ZmqContext[ZmqSocket[bytes]] | None = None
1149        sock: ZmqSocket[bytes] | None = None
1150        try:
1151            self._log_queue = self._mp_context.Queue()
1152            self._listener = QueueListener(self._log_queue, _LoggerDispatchHandler())
1153            self._listener.start()
1154            context = ZmqContext()
1155            sock = context.socket(zmq.PUSH)
1156            sock.set_hwm(self._params.publish_hwm)
1157            sock.setsockopt(zmq.LINGER, 0)
1158            sock.bind(f"tcp://{PublishWorkerPool.ZMQ_BIND_IP}:*")
1159            self._endpoint = sock.getsockopt_string(option=zmq.LAST_ENDPOINT)
1160            if self._logger.isEnabledFor(level=logging.INFO):
1161                self._logger.info(msg=f"Publish endpoint {self._endpoint}")
1162            monitor = sock.get_monitor_socket(events=zmq.EVENT_HANDSHAKE_SUCCEEDED)
1163            try:
1164                for worker_index in range(self._params.publish_worker_count):
1165                    self._spawn_worker_with_retry(worker_index=worker_index)
1166                self._await_handshakes(monitor=monitor)
1167            finally:
1168                sock.disable_monitor()
1169                monitor.close(linger=0)
1170        except BaseException:
1171            for proc in list(self._workers.values()):
1172                if proc.is_alive():
1173                    self._terminate_then_kill(proc=proc)
1174            if sock is not None:
1175                sock.close()
1176            if context is not None:
1177                context.term()
1178            if self._listener is not None:
1179                self._listener.stop()
1180                self._listener = None
1181            self._reset_bookkeeping()
1182            raise
1183        return context, sock

1185    def _await_handshakes(self, monitor: ZmqSocket[bytes]) -> None:
1186        """Wait until every worker has completed the ZMQ handshake.
1187
1188        A worker's readiness only means its ``connect()`` was issued; until the
1189        handshake completes the PUSH socket has no peer and a ``NOBLOCK`` send
1190        fails with ``zmq.Again``. The monitor socket reports one
1191        ``EVENT_HANDSHAKE_SUCCEEDED`` per worker connection.
1192
1193        Args:
1194            monitor (ZmqSocket[bytes]): The PUSH socket's monitor, subscribed to
1195                ``EVENT_HANDSHAKE_SUCCEEDED``.
1196
1197        Returns:
1198            None: Returns once ``publish_worker_count`` handshakes were seen.
1199
1200        Raises:
1201            StepLibError: If not every worker completed the handshake within
1202                ``publish_worker_startup_timeout``.
1203
1204        Complexity:
1205            - Time: O(w), w = publish_worker_count, bounded by the timeout.
1206            - Space: O(1), one monitor message at a time.
1207        """
1208        deadline = monotonic() + self._params.publish_worker_startup_timeout
1209        handshakes = 0
1210        while handshakes < self._params.publish_worker_count:
1211            remaining_ms = round((deadline - monotonic()) * 1000)
1212            if remaining_ms <= 0 or not monitor.poll(timeout=remaining_ms):
1213                raise StepLibError(
1214                    f"Only {handshakes} of {self._params.publish_worker_count} "
1215                    "publish workers completed the ZMQ handshake in time"
1216                )
1217            if recv_monitor_message(monitor)["event"] == zmq.EVENT_HANDSHAKE_SUCCEEDED:
1218                handshakes += 1

1220    def _handle_unhealthy_worker(
1221        self,
1222        worker_index: int,
1223        proc: BaseProcess | None,
1224        auto_restart: bool,
1225        blocked: bool,
1226        generation: int,
1227    ) -> None:
1228        """Claim, log and (if requested) restart one worker found unhealthy.
1229
1230        The claim (``_claim_worker_for_restart``) is what makes the calling
1231        thread the only one allowed to signal/join that process and respawn
1232        that slot; it is released in a ``finally`` so a failure while
1233        handling the worker can never leave the slot marked in-flight.
1234
1235        Args:
1236            worker_index (int): The unhealthy worker's slot.
1237            proc (BaseProcess | None): The process that was found unhealthy
1238                (from the snapshot taken at the start of the pass), or None
1239                if the slot was not in the snapshot, in which case it is
1240                skipped.
1241            auto_restart (bool): If False, only log; nothing is terminated
1242                or restarted.
1243            blocked (bool): True if the worker is alive but blocked (it is
1244                force-terminated before being restarted); False if it is
1245                dead.
1246            generation (int): The pool generation the snapshot was taken
1247                in; the restart is bound to it (see ``_generation``).
1248
1249        Returns:
1250            None: The worker is logged and, if ``auto_restart`` is True,
1251            restarted as a side effect.
1252
1253        Complexity:
1254            - Time: O(1) plus the cost of ``_restart_worker`` (and of the
1255              forceful termination for a blocked worker) when restarting.
1256            - Space: O(1), no additional storage is required.
1257        """
1258        if proc is None or not self._claim_worker_for_restart(
1259            worker_index=worker_index, auto_restart=auto_restart, expected_proc=proc
1260        ):
1261            return
1262        try:
1263            if self._logger.isEnabledFor(level=logging.ERROR):
1264                reason = "blocked" if blocked else f"dead, exitcode={proc.exitcode}"
1265                self._logger.error(
1266                    msg=f"Worker {worker_index} (pid={proc.pid}) {reason}"
1267                )
1268            if auto_restart:
1269                if blocked:
1270                    self._terminate_then_kill(proc=proc)
1271                self._restart_worker(worker_index=worker_index, generation=generation)
1272        finally:
1273            if auto_restart:
1274                self._release_restart(worker_index=worker_index, generation=generation)

1276    def _shutdown_workers(
1277        self,
1278        procs: list[BaseProcess],
1279        stop_events: list[MpEvent],
1280        context: ZmqContext[ZmqSocket[bytes]] | None,
1281    ) -> None:
1282        """Stop the given workers: graceful first, forceful for stragglers.
1283
1284        Runs with no lock held. First terminates the ZMQ context, which blocks
1285        until the closed PUSH socket has handed its queued messages to the
1286        workers (its linger, see ``close()``); only then signals every stop
1287        event, so each worker drains a complete ``PULL`` buffer before exiting.
1288        Every worker is joined against one shared deadline (twice
1289        ``publish_graceful_shutdown_timeout``: the flush, then the drain), then
1290        ``terminate()``/``kill()`` escalation reaps any still alive.
1291
1292        Args:
1293            procs (list[BaseProcess]): The worker processes to stop.
1294            stop_events (list[MpEvent]): Their stop events.
1295            context (ZmqContext | None): The pool's ZMQ context, already
1296                detached from the pool with its socket closed, or None.
1297
1298        Returns:
1299            None: Every worker has exited (or been killed) as a side
1300            effect.
1301
1302        Complexity:
1303            - Time: O(w), where w is the number of workers, to signal,
1304              join, and (if needed) forcefully terminate each one.
1305            - Space: O(1), no additional storage is required.
1306        """
1307        deadline = monotonic() + 2 * self._params.publish_graceful_shutdown_timeout
1308        if context is not None:
1309            context.term()  # blocks until the linger flush is done
1310        for stop_event in stop_events:  # only now: every PULL buffer is complete
1311            stop_event.set()
1312        for proc in procs:
1313            proc.join(timeout=max(0.0, deadline - monotonic()))
1314        for proc in procs:
1315            if proc.is_alive():
1316                self._terminate_then_kill(proc=proc)

1339    def start(self) -> None:
1340        """Bind the pool's one PUSH socket, then spawn every worker to connect to it.
1341
1342        If the pool is in a transitional state (``STARTING``/``CLOSING``)
1343        this first waits for it to settle. If the pool is then already
1344        ``RUNNING`` it returns immediately (idempotent); if it is
1345        ``CLOSED`` this call claims ``STARTING`` atomically with that
1346        check and performs the start. The slow work (queue/listener setup,
1347        the bind, spawning and awaiting every worker) runs with no lock
1348        held, so ``send()``/``check_health()`` merely see ``STARTING`` and
1349        refuse, and a concurrent ``close()`` waits for the start to finish.
1350
1351        The binding uses ``tcp://{ZMQ_BIND_IP}:*`` - the wildcard
1352        port tells libzmq to ask the OS for a free port atomically, in the
1353        same call, so there is no separate probe-then-bind step and no
1354        TOCTOU window to lose a race in. The port actually chosen is read
1355        back via ``LAST_ENDPOINT``. Every worker in the pool then connects
1356        a ``PULL`` socket to this one endpoint; ZMQ fans work out to them
1357        with round-robin fair-queuing regardless of which side bound vs.
1358        connected. A worker that fails to become ready (e.g. it crashed on
1359        startup for an unrelated reason) is retried on the same endpoint
1360        via ``_spawn_worker_with_retry`` rather than aborting the whole
1361        pool.
1362
1363        The socket is created and bound as a local variable before any
1364        worker is spawned, but only published to ``self._sock``/
1365        ``self._context`` - in the same critical section that moves the
1366        state to ``RUNNING`` - once every worker has started successfully.
1367        If ``start()`` ends up raising, the state returns to ``CLOSED``.
1368
1369        Returns:
1370            None: The pool's socket and workers are ready for use when
1371            this returns without raising.
1372
1373        Raises:
1374            StepLibError: If the bind itself fails, if a worker still
1375                can't be started within ``MAX_WORKER_SPAWN_ATTEMPTS``
1376                attempts, or if the pool did not settle in time. Any
1377                worker already spawned in this call is terminated, and the
1378                socket/context closed, before the error propagates. The
1379                error is raised only in the thread that executed this
1380                start; threads that were waiting on it retry on their own.
1381
1382        Complexity:
1383            - Time: O(w), where w is ``publish_worker_count``, to spawn and
1384              await readiness for every worker.
1385            - Space: O(w), for the per-worker bookkeeping created.
1386        """
1387        with self._cond:
1388            self._wait_settled()
1389            if self._state is PoolState.RUNNING:
1390                return
1391            self._set_state(new_state=PoolState.STARTING)
1392            self._generation += 1
1393            self._reset_bookkeeping()
1394        try:
1395            context, sock = self._bind_and_spawn()
1396        except BaseException as error:
1397            with self._cond:
1398                self._set_state(new_state=PoolState.CLOSED)
1399                self._cond.notify_all()
1400            if isinstance(error, StepLibError) or not isinstance(error, Exception):
1401                raise
1402            raise StepLibError("Failed to start the publish worker pool") from error
1403        with self._cond:
1404            self._context = context
1405            self._sock = sock
1406            self._set_state(new_state=PoolState.RUNNING)
1407            self._cond.notify_all()
1408        if self._logger.isEnabledFor(level=logging.INFO):
1409            self._logger.info(
1410                msg=f"Publish worker pool ready ({len(self._workers)} workers)"
1411            )

1431    def send(self, envelope: PublishEnvelope) -> None:
1432        """Send one envelope to the pool, round-robin routed to a worker by ZMQ.
1433
1434        The envelope is stamped with the next pool-wide GeoNetworking sequence
1435        number (``sequenceNumber``). The JSON encoding happens before ``_cond`` is
1436        acquired - it is pure
1437        computation on the caller-owned ``envelope``, touching no shared
1438        state, so it does not belong in the same critical section as the
1439        socket call. The socket write itself is ``NOBLOCK`` and so short
1440        enough to run under the lock, which is what makes the state check
1441        and the write atomic with respect to ``close()``.
1442
1443        Args:
1444            envelope (PublishEnvelope): The envelope to send.
1445
1446        Returns:
1447            None: The envelope is handed to ZMQ for delivery; nothing is
1448            returned to the caller.
1449
1450        Raises:
1451            StepLibError: If the pool is not running (never started,
1452                still starting, or closing/closed), or if the socket has
1453                already been closed by a concurrent ``close()``.
1454            zmq.Again: If every worker's send pipe has reached
1455                ``publish_hwm`` (propagated to the caller as backpressure).
1456
1457        Complexity:
1458            - Time: O(n), where n is the number of fields in the envelope,
1459              for JSON encoding; O(1) for the socket send itself.
1460            - Space: O(n), where n is the number of fields in the envelope,
1461              for the encoded payload.
1462        """
1463        payload = (
1464            envelope.model_copy(update={"sequenceNumber": self._next_sequence_number()})
1465            .model_dump_json()
1466            .encode(encoding="utf-8")
1467        )
1468        with self._cond:
1469            sock = self._sock
1470            if self._state is not PoolState.RUNNING or sock is None:
1471                raise StepLibError(f"Worker pool is not running (state={self._state})")
1472            try:
1473                sock.send(data=payload, flags=zmq.NOBLOCK)
1474            except ZmqAgain:
1475                raise
1476            except ZmqError as error:
1477                raise StepLibError("Worker pool is closing") from error

1479    def check_health(self, auto_restart: bool = True) -> None:
1480        """Detect dead/blocked workers and, if requested, restart them.
1481
1482        Does nothing unless the pool is ``RUNNING``. Blocked workers are
1483        force-terminated (``terminate()``, then ``kill()`` if that alone
1484        does not reap them in time) before being restarted, since a worker
1485        stuck deep enough might not react cleanly to ``terminate()``
1486        either; dead workers need no such escalation, since their process
1487        has already exited.
1488
1489        The pass has three phases: (1) a snapshot of the worker processes
1490        under ``_cond``; (2) detection via ``_dead_workers``/
1491        ``_blocked_workers``, outside the lock; (3) for every unhealthy
1492        worker, an atomic claim (``_claim_worker_for_restart``, which also
1493        rejects a stale detection whose process has since been replaced),
1494        then the slow work - logging, terminating, restarting - with no
1495        lock held, and finally a release of the claim. The claim makes the
1496        calling thread the only one allowed to signal/join that process and
1497        respawn that slot, and ``close()`` waits for claimed restarts to
1498        finish, so ``send()`` stays available throughout and no worker is
1499        orphaned. This method never touches the PUSH socket.
1500
1501        Args:
1502            auto_restart (bool): If True (the default), restart every
1503                dead/blocked worker found. If False, only log what was
1504                found, without terminating or restarting anything - a
1505                read-only diagnostic pass.
1506
1507        Returns:
1508            None: Unhealthy workers are logged and, if auto_restart is
1509            True, restarted in place as a side effect.
1510
1511        Complexity:
1512            - Time: O(w), where w is the number of workers, for detection,
1513              plus the cost of ``_restart_worker`` for each unhealthy one
1514              found.
1515            - Space: O(w) for the snapshot of worker processes.
1516        """
1517        with self._cond:
1518            if self._state is not PoolState.RUNNING:
1519                return
1520            snapshot = dict(self._workers)
1521            generation = self._generation
1522        for worker_index in self._dead_workers():
1523            self._handle_unhealthy_worker(
1524                worker_index=worker_index,
1525                proc=snapshot.get(worker_index),
1526                auto_restart=auto_restart,
1527                blocked=False,
1528                generation=generation,
1529            )
1530        for worker_index in self._blocked_workers():
1531            self._handle_unhealthy_worker(
1532                worker_index=worker_index,
1533                proc=snapshot.get(worker_index),
1534                auto_restart=auto_restart,
1535                blocked=True,
1536                generation=generation,
1537            )

1539    def close(self) -> None:
1540        """Gracefully stop every worker, then forcefully reap any that don't
1541        cooperate.
1542
1543        If the pool is in a transitional state (``STARTING``/``CLOSING``)
1544        this first waits for it to settle. If the pool is then already
1545        ``CLOSED`` it returns immediately (idempotent, and safe to call
1546        from a ``finally`` block); if it is ``RUNNING`` this call claims
1547        ``CLOSING`` atomically with that check and performs the shutdown.
1548        From then on ``send()`` and any new restart claim are refused.
1549
1550        Sequence, under ``_cond``: wait for restarts already in flight to
1551        drain (bounded by ``_restart_drain_budget()``; they abandon their
1552        work as soon as they see ``CLOSING``), snapshot the workers and
1553        stop events, and close the PUSH socket with a linger of
1554        ``publish_graceful_shutdown_timeout``, so the messages already queued
1555        are not dropped. Then, without the lock: terminate the ZMQ context,
1556        which returns once those messages reached the workers; signal every
1557        worker's ``stop_event``, so each drains its ``PULL`` buffer to MQTT
1558        and exits; wait for graceful exit against one shared deadline (so
1559        total shutdown time stays bounded regardless of worker count);
1560        escalate to
1561        ``terminate()``/``kill()`` for any worker still alive after that;
1562        stop the log listener last, so it can still capture log lines from
1563        the steps above. Finally the bookkeeping is cleared and the state
1564        returns to ``CLOSED``, even if a step above raised.
1565
1566        Returns:
1567            None: Every worker and the pool's own socket/context are torn
1568            down as a side effect before this returns.
1569
1570        Raises:
1571            StepLibError: If the pool did not settle in time.
1572
1573        Complexity:
1574            - Time: O(w), where w is the number of workers, to signal,
1575              join, and (if needed) forcefully terminate each one.
1576            - Space: O(w), for the snapshot of worker processes and stop
1577              events.
1578        """
1579        with self._cond:
1580            self._wait_settled()
1581            if self._state is PoolState.CLOSED:
1582                return
1583            self._set_state(new_state=PoolState.CLOSING)
1584            drained = self._cond.wait_for(
1585                predicate=lambda: not self._restarting,
1586                timeout=self._restart_drain_budget(),
1587            )
1588            stop_events = list(self._stop_events.values())
1589            procs = list(self._workers.values())
1590            if self._sock is not None:
1591                # Let ZMQ hand queued messages to the workers instead of dropping them
1592                self._sock.close(
1593                    linger=round(self._params.publish_graceful_shutdown_timeout * 1000)
1594                )
1595                self._sock = None
1596            context, self._context = self._context, None
1597        try:
1598            if not drained and self._logger.isEnabledFor(level=logging.WARNING):
1599                self._logger.warning(
1600                    msg="Closing while a worker restart is still in flight"
1601                )
1602            self._shutdown_workers(
1603                procs=procs, stop_events=stop_events, context=context
1604            )
1605            if self._logger.isEnabledFor(level=logging.INFO):
1606                self._logger.info(msg="Publish worker pool closed")
1607            if self._listener is not None:
1608                self._listener.stop()
1609                self._listener = None
1610        finally:
1611            self._reset_bookkeeping()
1612            with self._cond:
1613                self._set_state(new_state=PoolState.CLOSED)
1614                self._cond.notify_all()
