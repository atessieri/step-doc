 60    """Owns the lifecycle of the publish worker process pool: binding the
 61    pool's single endpoint, spawning workers, health-checking/restarting,
 62    and shutdown.
 63
 64    Every ``multiprocessing`` primitive this pool creates (``Process``,
 65    ``Event``, ``Value``, ``Queue``) comes from an explicit ``spawn``
 66    context (``self._mp_context``), never the ambient default - Linux
 67    otherwise defaults to ``fork``, which is unsafe once the parent is
 68    multi-threaded (as it already is by the time this pool starts, since
 69    ``SyncComManager.start()`` starts its own background thread first).
 70
 71    The one ZMQ ``PUSH`` socket this pool owns is bound once, to one
 72    fixed port, in ``start()`` - every worker connects a ``PULL`` socket
 73    to that same endpoint, so the pool's own socket never needs to
 74    connect/disconnect per worker (that dance only existed in the old
 75    per-worker-port design; see ``_restart_worker``). The socket is still
 76    guarded by an internal ``_socket_lock``, acquired only by ``send()``
 77    and ``close()``'s socket-closing step, each for the duration of that
 78    call only.

173    def start(self) -> None:
174        """Bind the pool's one PUSH socket, then spawn every worker to connect to it.
175
176        Sets up the shared log queue and its ``QueueListener`` first (so
177        worker startup failures are still logged through the application's
178        configured handlers), then binds to ``tcp://{PUBLISH_ZMQ_BIND_IP}:*``
179        - the wildcard port tells libzmq to ask the OS for a free port
180        atomically, in the same call, so there is no separate probe-then-bind
181        step and no TOCTOU window to lose a race in. The port actually
182        chosen is read back via ``LAST_ENDPOINT``. Every worker in the pool
183        then connects a ``PULL`` socket to this one endpoint; ZMQ fans work
184        out to them with round-robin fair-queuing regardless of which side
185        bound vs. connected. A worker that fails to become ready (e.g. it
186        crashed on startup for an unrelated reason) is retried on the same
187        endpoint via ``_spawn_worker_with_retry`` rather than aborting the
188        whole pool.
189
190        The socket is created and bound as a local variable before any
191        worker is spawned, but only published to ``self._sock``/
192        ``self._context`` once every worker has started successfully - so
193        ``is_running`` stays False, and the socket is closed again, if
194        ``start()`` ends up raising.
195
196        Returns:
197            None: The pool's socket and workers are ready for use when
198            this returns without raising.
199
200        Raises:
201            StepLibError: If the bind itself fails, or if a worker still
202                can't be started within ``MAX_WORKER_SPAWN_ATTEMPTS``
203                attempts. Any worker already spawned in this call is
204                terminated, and the socket/context closed, before the error
205                propagates.
206
207        Complexity:
208            - Time: O(w), where w is ``publish_worker_count``, to spawn and
209              await readiness for every worker.
210            - Space: O(w), for the per-worker bookkeeping created.
211        """
212        self._log_queue = self._mp_context.Queue()
213        self._listener = QueueListener(
214            self._log_queue, *self._logger.handlers, respect_handler_level=True
215        )
216        self._listener.start()
217        context = ZmqContext()
218        sock = context.socket(zmq.PUSH)
219        sock.set_hwm(self._params.publish_hwm)
220        sock.setsockopt(zmq.LINGER, 0)
221        try:
222            sock.bind(f"tcp://{PUBLISH_ZMQ_BIND_IP}:*")
223            self._endpoint = sock.getsockopt_string(option=zmq.LAST_ENDPOINT)
224            if self._logger.isEnabledFor(level=logging.INFO):
225                self._logger.info(msg=f"Publish endpoint {self._endpoint}")
226            for worker_index in range(self._params.publish_worker_count):
227                self._spawn_worker_with_retry(worker_index=worker_index)
228        except Exception as error:
229            for proc in self._workers.values():
230                if proc.is_alive():
231                    proc.terminate()
232            sock.close()
233            context.term()
234            self._listener.stop()
235            self._listener = None
236            if isinstance(error, StepLibError):
237                raise
238            raise StepLibError("Failed to start the publish worker pool") from error
239        self._context = context
240        self._sock = sock
241        if self._logger.isEnabledFor(level=logging.INFO):
242            self._logger.info(
243                msg=f"Publish worker pool ready ({len(self._workers)} workers)"
244            )

246    def send(self, envelope: PublishEnvelope) -> None:
247        """Send one envelope to the pool, round-robin routed to a worker by ZMQ.
248
249        The JSON encoding happens before ``_socket_lock`` is acquired -
250        it is pure computation on the caller-owned ``envelope``, touching
251        no shared state, so it does not belong in the same critical
252        section as the socket call.
253
254        Args:
255            envelope (PublishEnvelope): The envelope to send.
256
257        Returns:
258            None: The envelope is handed to ZMQ for delivery; nothing is
259            returned to the caller.
260
261        Raises:
262            StepLibError: If the pool is not running, or if the socket has
263                already been closed by a concurrent ``close()``.
264            zmq.Again: If every worker's send pipe has reached
265                ``publish_hwm`` (propagated to the caller as backpressure).
266
267        Complexity:
268            - Time: O(n), where n is the number of fields in the envelope,
269              for JSON encoding; O(1) for the socket send itself.
270            - Space: O(n), where n is the number of fields in the envelope,
271              for the encoded payload.
272        """
273        if self._sock is None:
274            raise StepLibError("Worker pool is not running")

280            raise

282            raise StepLibError("Worker pool is closing") from error

328    def check_health(self, auto_restart: bool = True) -> None:
329        """Detect dead/blocked workers and, if requested, restart them.
330
331        Blocked workers are force-terminated (``terminate()``, then
332        ``kill()`` if that alone does not reap them in time) before being
333        restarted, since a worker stuck deep enough might not react
334        cleanly to ``terminate()`` either; dead workers need no such
335        escalation, since their process has already exited.
336
337        Args:
338            auto_restart (bool): If True (the default), restart every
339                dead/blocked worker found. If False, only log what was
340                found, without terminating or restarting anything - a
341                read-only diagnostic pass.
342
343        Returns:
344            None: Unhealthy workers are logged and, if auto_restart is
345            True, restarted in place as a side effect.
346
347        Complexity:
348            - Time: O(w), where w is the number of workers, for detection,
349              plus the cost of ``_restart_worker`` for each unhealthy one
350              found.
351            - Space: O(1) beyond the lists returned by
352              ``dead_workers``/``blocked_workers``.
353        """
354        for worker_index in self.dead_workers():
355            if worker_index not in self._connected:
356                continue
357            proc = self._workers[worker_index]
358            if self._logger.isEnabledFor(level=logging.ERROR):
359                self._logger.error(
360                    msg=(
361                        f"Worker {worker_index} (pid={proc.pid}) dead, "
362                        f"exitcode={proc.exitcode}"
363                    )
364                )
365            if auto_restart:
366                self._restart_worker(worker_index=worker_index)
367        for worker_index in self.blocked_workers():
368            if worker_index not in self._connected:
369                continue
370            proc = self._workers[worker_index]
371            if self._logger.isEnabledFor(level=logging.ERROR):
372                self._logger.error(
373                    msg=f"Worker {worker_index} (pid={proc.pid}) blocked"
374                )
375            if auto_restart:
376                proc.terminate()
377                proc.join(timeout=self._params.publish_forceful_join_timeout)
378                if proc.is_alive():
379                    proc.kill()
380                    proc.join(timeout=self._params.publish_forceful_join_timeout)
381                self._restart_worker(worker_index=worker_index)

383    def close(self) -> None:
384        """Gracefully stop every worker, then forcefully reap any that don't
385        cooperate.
386
387        Sequence: signal every worker's ``stop_event`` (graceful, since
388        ``mp.Event`` is independent of message traffic and so does not
389        depend on the ZMQ channel being deliverable); close the PUSH socket
390        (safe/fast even under load, since it was created with
391        ``LINGER=0``); wait for graceful exit against one shared deadline
392        (so total shutdown time stays bounded regardless of worker count);
393        escalate to ``terminate()``/``kill()`` for any worker still alive
394        after that; stop the log listener last, so it can still capture log
395        lines from the steps above.
396
397        Returns:
398            None: Every worker and the pool's own socket/context are torn
399            down as a side effect before this returns.
400
401        Complexity:
402            - Time: O(w), where w is the number of workers, to signal,
403              join, and (if needed) forcefully terminate each one.
404            - Space: O(1), no additional storage is required.
405        """
406        deadline = monotonic() + self._params.publish_graceful_shutdown_timeout
407        for stop_event in self._stop_events.values():
408            stop_event.set()
409        if self._sock is not None:
410            with self._socket_lock:
411                self._sock.close()
412            self._sock = None
413        if self._context is not None:
414            self._context.term()
415            self._context = None
416        for proc in self._workers.values():
417            proc.join(timeout=max(0.0, deadline - monotonic()))
418        for proc in self._workers.values():
419            if proc.is_alive():
420                proc.terminate()
421                proc.join(timeout=self._params.publish_forceful_join_timeout)
422                if proc.is_alive():
423                    proc.kill()
424                    proc.join(timeout=self._params.publish_forceful_join_timeout)
425        self._connected.clear()
426        if self._logger.isEnabledFor(level=logging.INFO):
427            self._logger.info(msg="Publish worker pool closed")
428        if self._listener is not None:
429            self._listener.stop()
430            self._listener = None

432    def _spawn_worker_with_retry(self, worker_index: int) -> None:
433        """Spawn one worker on the pool's endpoint, retrying if it fails to
434        become ready.
435
436        A worker connecting to an already-bound endpoint has no port race
437        to lose, so a failure here means the process itself didn't come up
438        cleanly (e.g. it crashed during its own startup). Retrying a fresh
439        process on the same endpoint, up to ``MAX_WORKER_SPAWN_ATTEMPTS``
440        times, gives ``start()``/``_restart_worker`` the same tolerance for
441        that as they always had for a port collision.
442
443        Args:
444            worker_index (int): The slot to spawn.
445
446        Returns:
447            None: On success the worker is recorded as connected;
448            otherwise StepLibError is raised.
449
450        Raises:
451            StepLibError: If no worker could be started within
452                ``MAX_WORKER_SPAWN_ATTEMPTS`` attempts.
453
454        Complexity:
455            - Time: O(MAX_WORKER_SPAWN_ATTEMPTS) in the worst case, each
456              attempt dominated by ``_await_ready``.
457            - Space: O(1), no additional storage beyond the spawned
458              worker's own bookkeeping.
459        """
460        for _ in range(self.MAX_WORKER_SPAWN_ATTEMPTS):
461            self._spawn_worker(worker_index=worker_index)
462            try:
463                self._await_ready(worker_index=worker_index)
464            except StepLibError:
465                if self._logger.isEnabledFor(level=logging.WARNING):
466                    self._logger.warning(
467                        msg=f"Worker {worker_index} failed to start, retrying"
468                    )
469                continue
470            self._connected.add(worker_index)
471            return
472        raise StepLibError(
473            f"Worker {worker_index} could not be started within "
474            f"{self.MAX_WORKER_SPAWN_ATTEMPTS} attempts"
475        )

477    def _spawn_worker(self, worker_index: int) -> None:
478        """Construct and start one worker process, recording its bookkeeping.
479
480        Args:
481            worker_index (int): The slot this worker occupies in the pool.
482
483        Returns:
484            None: The new worker process and its bookkeeping are recorded
485            on the pool as a side effect.
486
487        Complexity:
488            - Time: O(1), creates a fixed number of multiprocessing
489              primitives and starts one process.
490            - Space: O(1), a fixed number of per-worker references is
491              stored.
492        """
493        # Log queue must be set before spawning a worker
494        if self._log_queue is None:
495            raise RuntimeError("Log queue must be set before spawning a worker")
496        if self._endpoint is None:
497            raise RuntimeError("Pool endpoint must be bound before spawning a worker")
498        ready_event = self._mp_context.Event()
499        stop_event = self._mp_context.Event()
500        heartbeat = self._mp_context.Value("d", monotonic())
501        worker = PublishWorker(
502            endpoint=self._endpoint,
503            worker_index=worker_index,
504            ready_event=ready_event,
505            stop_event=stop_event,

507            parent_pid=self._parent_pid,
508            parent_create_time=self._parent_create_time,
509            logger_name=self._logger.name,
510            effective_level=self._logger.getEffectiveLevel(),
511            log_queue=self._log_queue,
512            params=self._params,

516            daemon=True,
517            name=f"{self._params.instance_name}-publish-worker-{worker_index}",
518        )
519        process.start()
520        self._workers[worker_index] = process
521        self._ready_events[worker_index] = ready_event
522        self._stop_events[worker_index] = stop_event
523        self._heartbeats[worker_index] = heartbeat

525    def _await_ready(self, worker_index: int) -> None:
526        """Poll for one worker to signal readiness, bounded by
527        ``publish_worker_startup_timeout``.
528
529        Polls rather than blocking on one long wait, so a worker that
530        fails for any reason is detected within roughly
531        ``READINESS_POLL_INTERVAL``, not the full timeout.
532
533        Args:
534            worker_index (int): The worker to wait for.
535
536        Returns:
537            None: Returns normally once the worker signals readiness;
538            otherwise raises.
539
540        Raises:
541            StepLibError: If the worker exits before becoming ready, or
542                does not become ready within the timeout (in which case it
543                is terminated).
544
545        Complexity:
546            - Time: O(timeout / READINESS_POLL_INTERVAL) in the worst
547              case.
548            - Space: O(1), no additional storage is required.
549        """
550        deadline = monotonic() + self._params.publish_worker_startup_timeout
551        ready_event = self._ready_events[worker_index]
552        proc = self._workers[worker_index]
553        while monotonic() < deadline:
554            if ready_event.wait(timeout=self.READINESS_POLL_INTERVAL):
555                return
556            if not proc.is_alive():
557                raise StepLibError(
558                    f"Worker {worker_index} exited before becoming ready "
559                    f"(exitcode={proc.exitcode})"
560                )
561        proc.terminate()
562        raise StepLibError(
563            f"Worker {worker_index} did not become ready within "
564            f"{self._params.publish_worker_startup_timeout}s"
565        )

567    def _restart_worker(self, worker_index: int) -> None:
568        """Replace a dead/blocked worker with a freshly spawned one.
569
570        The pool's own PUSH socket's single bind never changes, so unlike
571        the old per-worker-endpoint design, this has nothing to disconnect
572        or reconnect on the pool's side - only the replacement process
573        connecting to the same, already-known ``self._endpoint`` matters.
574        The slot is marked not-connected up front so a concurrent
575        ``check_health()`` pass doesn't try to restart it again while this
576        one is still in flight; spawn/await-ready (potentially slow) then
577        runs with no lock held, so ``send()`` remains usable against the
578        other workers for the whole duration. If every attempt fails, the
579        slot is simply left disconnected and picked up again on the next
580        scheduled health check.
581
582        Args:
583            worker_index (int): The worker slot to restart.
584
585        Returns:
586            None: The slot is either replaced with a freshly spawned
587            worker or left disconnected for the next health check.
588
589        Complexity:
590            - Time: O(MAX_WORKER_SPAWN_ATTEMPTS * publish_worker_startup_timeout
591              / READINESS_POLL_INTERVAL) in the worst case, dominated by
592              ``_await_ready`` across every retry.
593            - Space: O(1), no additional storage beyond the replacement
594              worker's own bookkeeping.
595        """
596        self._connected.discard(worker_index)
597        try:
598            self._spawn_worker_with_retry(worker_index=worker_index)
599        except StepLibError:
600            if self._logger.isEnabledFor(level=logging.ERROR):
601                self._logger.exception(msg=f"Failed to restart worker {worker_index}")
602            return
603        if self._logger.isEnabledFor(level=logging.INFO):
604            self._logger.info(msg=f"Worker {worker_index} restarted")
