  2
  3This module contains unit tests for the ``GpxLoader`` class, which parses a
  4GPX track file into ``CamMessageV1`` instances, raising ``LoaderError`` when
  5the file does not match the expected structure.

 30def _gpx(trk_name: str = "Path actor 42_none", trkpt: str = "") -> str:
 31    """Build a minimal well-formed GPX 1.1 document for tests.
 32
 33    Args:
 34        trk_name (str): Value for the '<trk><name>' tag.
 35        trkpt (str): Raw XML for a single '<trkpt>...</trkpt>' element to
 36            embed in the track segment; a default point is used if empty.
 37
 38    Returns:
 39        str: A complete GPX 1.1 document as a string.
 40    """
 41    if not trkpt:
 42        trkpt = (
 43            '<trkpt lat="45.0" lon="8.0"><ele>100.0</ele>'
 44            "<time>2025-08-18T15:44:20.638Z</time>"
 45            "<extensions><gxt:TrackPointExtension><gxt:speed>1.5</gxt:speed>"
 46            "<gxt:course>90</gxt:course></gxt:TrackPointExtension></extensions>"
 47            "</trkpt>"
 48        )
 49    return (
 50        '<?xml version="1.0" encoding="UTF-8"?>'
 51        '<gpx version="1.1" creator="test" '
 52        'xmlns="http://www.topografix.com/GPX/1/1" '
 53        'xmlns:gxt="http://www8.garmin.com/xmlschemas/TrackPointExtension/v2">'
 54        f"<trk><name>{trk_name}</name><trkseg>{trkpt}</trkseg></trk></gpx>"
 55    )

124    def test_call_multiple_points(self, tmp_path: Path) -> None:
125        """Test that ``__call__`` returns one ``CamMessageV1`` per track
126        point across every track, each carrying its own track's station id.
127
128        ``simulation.gpx`` has two tracks tracing the same path: station id
129        130, and station id 131 whose points are delayed by 2 minutes.
130
131        Args:
132            tmp_path (Path): Pytest's temporary directory fixture.
133
134        Returns:
135            None
136        """
137        file_path = tmp_path / "data.gpx"
138        file_path.write_text(

140        )
141        result = GpxLoader()(file_path=file_path)
142        assert len(result) > 1
143        assert {message.station_id for message in result} == {130, 131}
144        by_station = {130: 0, 131: 0}
145        for message in result:
146            by_station[message.station_id] += 1
147        assert by_station[130] == by_station[131]
