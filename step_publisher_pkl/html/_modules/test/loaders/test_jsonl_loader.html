123                    stationType=TrafficParticipantType.PASSENGER_CAR,
124                    stationTypeConfidence=None,
125                ),
126            }
127        )
128        cam_second = CamDataV1.get_default().model_copy(
129            update={
130                "creationTime": "2022-04-05T17:31:01.000Z",
131                "stationInfo": Identification(
132                    stationId=1,
133                    stationType=TrafficParticipantType.PASSENGER_CAR,
134                    stationTypeConfidence=None,
135                ),
136            }
137        )
138        file_path = tmp_path / "data.jsonl"
139        file_path.write_text(
140            data=(

159    @pytest.mark.parametrize(argnames="log_enabled", argvalues=[True, False])
160    def test_call_unsupported_message_type(
161        self, mocker: MockerFixture, tmp_path: Path, log_enabled: bool
162    ) -> None:
163        """Test the defensive ``case _:`` branch reached when a validated root is
164        neither CamDataV1, CpmDataV1, nor DenmDataV1: the line is skipped (with a
165        warning only when WARNING-level logging is enabled) and not included in the
166        returned messages.
167
168        This branch cannot be reached through the public API since
169        ``ItsMessageRoot.root`` is a discriminated union of exactly those three
170        types; it is reached here by mocking ``ItsMessageRoot.model_validate_json``
171        directly, returning a valid first line (so the file is accepted as JSONL)
172        followed by an unsupported second line.
173
174        Args:
175            mocker (MockerFixture): The pytest-mock fixture for creating mocks.
176            tmp_path (Path): Pytest's temporary directory fixture.
177            log_enabled (bool): Whether WARNING level logging is enabled.
178
179        Returns:
180            None
181        """
182        mock_logger = mocker.MagicMock(spec=Logger)
183        mock_logger.isEnabledFor.return_value = log_enabled
184        loader = JsonlLoader(logger=mock_logger)
185        file_path = tmp_path / "data.jsonl"
186        file_path.write_text(data="line1\nline2\n", encoding="utf-8")
187        valid_root = ItsMessageRoot.model_validate_json(
188            json_data=CamDataV1.get_default().model_dump_json()
189        )
190        unsupported_root = mocker.MagicMock()
191        unsupported_root.root = object()
192        mocker.patch.object(
193            target=ItsMessageRoot,
194            attribute="model_validate_json",
195            side_effect=[valid_root, unsupported_root],
196        )
197        result = loader(file_path=file_path)
198        assert len(result) == 1
199        if log_enabled:
200            mock_logger.warning.assert_called_once()
201            warning_msg = mock_logger.warning.call_args.kwargs["msg"]
202            assert "Skipping unsupported ITS message type" in warning_msg
203        else:
204            mock_logger.warning.assert_not_called()
