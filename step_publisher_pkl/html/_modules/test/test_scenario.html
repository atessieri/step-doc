  61def _expected_real_iso(original_iso: str, time_reference: TimeReference) -> str:
  62    """Compute the expected real-world ISO time for a scenario ISO time.
  63
  64    This helper independently reproduces the scenario-to-real time conversion using
  65    only ``ItsTime`` primitives, so tests that exercise ``Scenario``'s private
  66    ``_update_time_*`` helpers do not depend on ``Scenario._scenario_time_to_real``
  67    itself for their expected values.
  68
  69    Args:
  70        original_iso (str): The scenario-relative ISO-8601 time string to convert.
  71        time_reference (TimeReference): The time reference mapping used for
  72            conversion.
  73
  74    Returns:
  75        str: The expected real-world ISO-8601 time string.
  76    """
  77    original_time = ItsTime.create_from_iso_time(time_str=original_iso)
  78    delta_ms = original_time.time_difference(
  79        other_time=time_reference.start_scenario_time
  80    )
  81    return str(object=time_reference.start_real_time + delta_ms)
