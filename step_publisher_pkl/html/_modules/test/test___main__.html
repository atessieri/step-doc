140async def test_publisher_run(
141    mocker: MockerFixture, delta: int, logger_enable: bool
142) -> None:
143    """Test that ``_publisher_run`` publishes every message popped from the
144    scenario for a vehicle, sleeping to preserve timing only when the message time
145    is still in the future.
146
147    Args:
148        mocker (MockerFixture): MockerFixture for creating mock objects during
149            testing.
150        delta (int): The simulated time difference in milliseconds between the
151            message time and "now".
152        logger_enable (bool): Boolean indicating if the logger is enabled for debug
153            level.
154
155    Returns:
156        None
157    """
158    # Arrange
159    mock_client = mocker.MagicMock(spec=StepBroker)
160    mock_client.client_publish_its_message = mocker.AsyncMock()
161    mock_scenario = mocker.MagicMock(spec=Scenario)
162    mock_message = mocker.MagicMock(spec=FacilityMessage)
163    mock_message_time = mocker.MagicMock(spec=ItsTime)
164    mock_message_time.time_difference.return_value = delta
165    mock_message.its_time = mock_message_time
166    mock_message.message_type.message_type = "CAM"
167    mock_scenario.pop_message.return_value = [mock_message]
168    mock_logger = mocker.MagicMock(spec=Logger)
169    mock_logger.isEnabledFor.return_value = logger_enable
170    mock_its_time_now = mocker.MagicMock(spec=ItsTime)
171    mock_its_time_class = mocker.patch(
172        "step_publisher_pkl.__main__.ItsTime", return_value=mock_its_time_now
173    )
174    mock_its_time_class.SEC_TO_MS_FACTOR = 1000
175    mock_asyncio_sleep = mocker.patch(
176        "step_publisher_pkl.__main__.asyncio_sleep", new_callable=mocker.AsyncMock
177    )
178    # Act
179    await _publisher_run(
180        client=mock_client, scenario=mock_scenario, vehicle_id=1, logger=mock_logger
181    )
182    # Assert
183    mock_scenario.pop_message.assert_called_once_with(vehicle_id=1)
184    mock_message_time.time_difference.assert_called_once_with(
185        other_time=mock_its_time_now
186    )
187    if delta > 0:
188        mock_asyncio_sleep.assert_awaited_once_with(delay=delta / 1000)
189    else:
190        mock_asyncio_sleep.assert_not_awaited()
191    mock_client.client_publish_its_message.assert_awaited_once_with(
192        message=mock_message
193    )
194    if logger_enable:
195        expected_debug_calls = 3 if delta > 0 else 2
196        assert mock_logger.debug.call_count == expected_debug_calls
197    else:
198        mock_logger.debug.assert_not_called()

654def test_dunder_main_guard(mocker: MockerFixture) -> None:
655    """Test that running the module as a script (``python -m step_publisher_pkl``)
656    invokes ``main()`` and exits the process with its return code.
657
658    This exercises the ``if __name__ == "__main__": sys.exit(main())`` guard by
659    executing the module fresh with ``runpy.run_module`` under ``run_name="__main__"``,
660    simulating no command-line arguments so ``main()`` takes its fast USAGE path.

662    Args:

665    Returns:
666        None

672    )
