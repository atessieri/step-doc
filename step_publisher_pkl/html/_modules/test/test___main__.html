 51def _mock_app_runner(mocker: MockerFixture, *, target: str) -> tuple[Any, Any]:
 52    """Patch ``AppRunner`` so that ``AppRunner[ExitCode](...)`` returns a mock
 53    runner, keeping the real singleton from installing the process-wide signal
 54    handlers and ``sys.unraisablehook`` during the tests.
 55
 56    Args:
 57        mocker (MockerFixture): pytest-mock fixture for creating mock objects.
 58        target (str): The dotted name of the ``AppRunner`` attribute to patch.
 59
 60    Returns:
 61        tuple[Any, Any]: The mock ``AppRunner`` class and the mock runner its
 62            subscripted form returns.
 63    """
 64    mock_app_runner = mocker.MagicMock()
 65    mock_app_runner_class = mocker.patch(target)
 66    mock_app_runner_class.__getitem__.return_value.return_value = mock_app_runner
 67    return mock_app_runner_class, mock_app_runner

534def test_dunder_main_guard(mocker: MockerFixture) -> None:
535    """Test that running the module as a script (``python -m step_publisher_pkl``)
536    invokes ``main()`` and exits the process with its return code.
537
538    This exercises the ``if __name__ == "__main__": sys.exit(main())`` guard by
539    executing the module fresh with ``runpy.run_module`` under ``run_name="__main__"``,
540    simulating no command-line arguments so ``main()`` takes its fast USAGE path.
541
542    Args:
543        mocker (MockerFixture): pytest-mock fixture for creating mock objects
544
545    Returns:
546        None
547    """
548    # Arrange
549    mocker.patch("sys.argv", new=["step_publisher_pkl"])
550    # The module is executed afresh, so it imports AppRunner from its package again
551    _mock_app_runner(mocker, target="step_cloud_schemas.AppRunner")
552    cached_module: ModuleType | None = sys.modules.pop(
553        "step_publisher_pkl.__main__", None
554    )
555    # Act & Assert
556    try:
557        with pytest.raises(expected_exception=SystemExit) as exc_info:
558            runpy.run_module(
559                mod_name="step_publisher_pkl.__main__", run_name="__main__"
560            )
561    finally:
562        if cached_module is not None:
563            sys.modules["step_publisher_pkl.__main__"] = cached_module
564    assert exc_info.value.code == ExitCode.USAGE.value
