  1"""Vehicle Trajectory Scenario Management Module.
  2
  3This module provides functionality for managing and processing vehicle trajectory data
  4for crash prediction analysis. It handles loading trajectory data from pickle files
  5and generating standardized CAM messages compatible with the STEP cloud schemas.
  6
  7The module is primarily designed for use in crash prediction scenarios where vehicle
  8trajectory data needs to be published in real-time or near-real-time fashion.
  9It supports loading multiple vehicle trajectories from pickle files and converting
 10them into CamDataV1 objects for further analysis.
 11
 12Key Features:
 13    - Loading vehicle trajectory data from pickle files
 14    - Conversion to standardized STEP cloud schema CAM messages
 15    - Memory-efficient data handling using deque
 16    - Support for multiple vehicles in a single scenario
 17    - Scenario-to-real-time timestamp alignment
 18
 19Example:
 20    >>> from logging import Logger
 21    >>> logger = Logger("demo")
 22    >>> scenario = Scenario(logger=logger)
 23    >>> scenario.load_data(input_pkl_files=["/path/to/scenario.pkl"])
 24    >>> for vehicle_id in scenario.get_vehicle_ids():
 25    ...     for position_result in scenario.pop_position_arrays(vehicle_id):
 26    ...         # Process position result for crash prediction analysis
 27    ...         pass
 28
 29Note:
 30    - Vehicle trajectory data is expected to be in pickle format
 31    - Data structure should contain 'trajectories'['vehicles'] dictionary
 32    - GNSS coordinates, speed, and heading data are required for each trajectory point
 33    - All measurements are converted to appropriate units (cm, cm/s, degrees)
 34"""

102class Scenario:
103    """A scenario management class for vehicle trajectory simulation and real-time data
104    processing.
105
106    This class handles loading vehicle trajectory data from pickle files, managing time references
107    for scenario-to-real-time conversion, and providing vehicle position data for simulation or
108    analysis purposes. It serves as the main orchestrator for vehicle data processing in
109    step-by-step scenario execution.
110    The Scenario class maintains a collection of vehicle trajectories loaded from pickle files,
111    each containing timestamped position data. It establishes a time reference that maps scenario
112    time to real time, enabling synchronized playback of recorded vehicle movements.
113
114    Key Features:
115        - Loads vehicle trajectory data from multiple pickle files
116        - Converts scenario timestamps to real-time for synchronized playback
117        - Manages vehicle position data as queues for efficient sequential access
118        - Converts raw trajectory data to structured CAM (Cooperative Awareness Message) format
119        - Supports iterative consumption of vehicle positions through generators
120
121    Attributes:
122        logger (Logger): Logger instance for recording events and debugging information
123
124    Usage:
125        The typical workflow involves:
126
127            #. Initialize with a logger
128            #. Call load_data() to populate vehicle trajectories from pickle files
129            #. Use get_vehicle_ids() to enumerate available vehicles
130            #. Call pop_position_arrays() to consume vehicle positions sequentially
131
132        Vehicle trajectory data is expected to be in pickle format with a specific structure
133        containing a 'trajectories' dictionary with vehicle data. Once position data is
134        consumed via pop_position_arrays(), it is removed from internal storage.
135    """
136
137    __slots__ = ("_logger", "_stations", "_time")

139    def __init__(self, logger: Logger) -> None:

161    def _dict_to_cam_data(self, vehicle_id: int, data: dict[str, Any]) -> CamDataV1:
162        """Converts a dictionary containing vehicle and sensor data into a CamDataV1
163        object.
164
165        Args:
166            vehicle_id (int): The unique identifier for the vehicle.
167            data (dict[str, Any]): A dictionary containing vehicle data, including timestamp, geo_location, heading, and velocity.
168
169        Returns:
170            CamDataV1: An instance of CamDataV1 populated with the provided data.
171        """
172        dt: datetime.datetime = data["timestamp"]
173        self._time.dt = dt
174        return CamDataV1(
175            creationTime=str(object=self._time),
176            stationInfo=Identification(
177                stationId=vehicle_id, stationType=TrafficParticipantType.PASSENGER_CAR
178            ),
179            position=ReferencePosition(
180                latitude=float(data["geo_location"]["latitude"]),
181                longitude=float(data["geo_location"]["longitude"]),
182                positionConfidence=None,
183                altitude=Altitude(
184                    value=round(
185                        data["geo_location"]["altitude"] * Altitude.SCALE_FACTOR
186                    ),
187                    confidence=None,
188                ),
189            ),
190            heading=Heading(
191                value=round(data["heading"] * Heading.SCALE_FACTOR), confidence=None
192            )
193            if "heading" in data
194            else None,
195            speed=Speed(
196                value=round(data["velocity"]["speed"] * Speed.SCALE_FACTOR),
197                confidence=None,
198            )
199            if "velocity" in data
200            else None,
201            driveDirection=None,
202            vehicleLength=None,
203            vehicleWidth=None,
204            longitudinalAcceleration=None,
205            curvature=None,
206            curvatureCalculationMode=None,
207            yawRate=None,
208            vehicleRole=None,
209        )

211    def _scenario_time_to_real(
212        self, scenario_time: ItsTime, time_reference: TimeReference
213    ) -> ItsTime:
214        """Convert scenario time to real time using the time reference.
215
216        Args:
217            scenario_time (ItsTime): The scenario time to convert.
218            time_reference (TimeReference): The time reference mapping for conversion.
219
220        Returns:
221            ItsTime: The corresponding real time.
222        """
223        delta = scenario_time.time_difference(
224            other_time=time_reference.start_scenario_time
225        )
226        return time_reference.start_real_time + delta

228    def _update_time_cam_data(
229        self, data: CamDataV1, time_reference: TimeReference
230    ) -> CamDataV1:
231        """Update the creation time of CAM (Cooperative Awareness Message) data from
232        scenario time to real time.
233
234        This method converts the scenario-relative creation time of a CAM data object to real-world time
235        using the provided time reference, and returns a new CAM data object with the updated timestamp.
236
237        Args:
238            data (CamDataV1): The CAM data object containing the original scenario-relative creation time.
239            time_reference (TimeReference): The time reference object used to convert scenario time to real time.
240
241        Returns:
242            CamDataV1: A new CAM data object with the creation time updated to real-world time as a string.
243
244        Note:
245            The method uses model_copy to create a new instance rather than modifying the original data object.
246            The creation time is converted to a string representation in the returned object.
247        """
248        self._time.iso_format = data.creationTime
249        return data.model_copy(
250            update={
251                "creationTime": str(
252                    object=self._scenario_time_to_real(
253                        scenario_time=self._time, time_reference=time_reference
254                    )
255                )
256            }
257        )

259    def load_data(self, input_pkl_files: list[str]) -> None:  # pylint: disable=too-many-locals
260        """Load vehicle trajectory data from pickle files.
261
262        This method loads vehicle trajectory data from the specified pickle files.
263        Each file is expected to contain a
264        ``trajectories`` dictionary with a ``vehicles`` key that maps vehicle IDs to
265        lists of trajectory points.
266
267        For each file, a time reference is established mapping scenario timestamps to
268        real-world time. This allows synchronized playback of recorded trajectories
269        by converting all position timestamps from scenario time to real time.
270
271        Args:
272            input_pkl_files (list[str]): List of paths to the pickle files to read and publish.
273
274        Returns:
275            None
276
277        Note:
278            - Vehicle IDs are converted to integers for storage
279            - Progress information is logged for each file and vehicle loaded
280            - Existing vehicle data will be overwritten if the same vehicle ID
281              is found in multiple files
282            - Timestamps are converted from scenario time to real time using the
283              minimum start time across all vehicles in each file as the reference
284        """
285        files_path = (Path(file) for file in input_pkl_files)
286        # Load each pickle file and extract vehicle trajectory data
287        for file_path in files_path:
288            if self._logger.isEnabledFor(level=logging.INFO):
289                self._logger.info(msg=f"Loading data from {file_path}")
290            with file_path.open(mode="rb") as file:
291                data: dict[str, Any] = pickle_load(file=file)  # noqa: S301
292                vehicles_data: dict[str, list[dict[str, Any]]] = data["trajectories"][
293                    "vehicles"
294                ]
295                # Traverse all points of all vehicles creating a dictionary of deque
296                vehicles_file: list[int] = []
297                for vehicle_id, vehicle_points in vehicles_data.items():
298                    vehicle_id_int = int(vehicle_id)
299                    vehicle_deque = deque[CamDataV1]()
300                    for point in vehicle_points:
301                        vehicle_deque.append(
302                            self._dict_to_cam_data(
303                                vehicle_id=vehicle_id_int, data=point
304                            )
305                        )
306                    self._stations[vehicle_id_int] = vehicle_deque
307                    if vehicle_deque and self._logger.isEnabledFor(level=logging.INFO):
308                        self._logger.info(
309                            msg=(
310                                f"Vehicle {vehicle_id_int}: range {vehicle_deque[0].creationTime} - "
311                                f"{vehicle_deque[-1].creationTime} with {len(vehicle_deque)} points loaded"
312                            )
313                        )
314                    vehicles_file.append(vehicle_id_int)
315                if vehicles_file:
316                    # Determine the minimum start time across all vehicles in the file
317                    vehicle_min_start_time = min(
318                        vehicles_file,
319                        key=lambda vehicle: self._get_epoch_ms(
320                            time_str=self._stations[vehicle][0].creationTime
321                        ),
322                    )
323                    time_reference = TimeReference(
324                        start_scenario_time=ItsTime.create_from_iso_time(
325                            time_str=self._stations[vehicle_min_start_time][
326                                0
327                            ].creationTime
328                        ),
329                        start_real_time=ItsTime(),
330                    )
331                    for vehicle_id_int in vehicles_file:
332                        self._stations[vehicle_id_int] = deque(
333                            self._update_time_cam_data(
334                                data=item, time_reference=time_reference
335                            )
336                            for item in self._stations[vehicle_id_int]
337                        )
338                    if self._logger.isEnabledFor(level=logging.INFO):
339                        self._logger.info(
340                            msg=f"File {file_path} complete. Time reference: {time_reference}."
341                        )

354    def _pop_vehicle_position(self, vehicle_id: int) -> StationPositionResult:
355        """Retrieve and remove the next vehicle position from the station queue.
356
357        This method pops the next position for a given vehicle from its station queue,
358        ensuring the position's timestamp is not in the future. If the timestamp is in
359        the future, it returns None for the position and schedules the next attempt.
360
361        Args:
362            vehicle_id (int): The unique identifier of the vehicle whose position
363                is to be retrieved.
364
365        Returns:
366            StationPositionResult: A result object containing:
367
368                - position: The current vehicle position data, or None if the timestamp
369                  is in the future or no positions are available.
370                - next_position_time: The timestamp of the next position in the queue,
371                  or None if the queue is empty.
372
373        Raises:
374            KeyError: If the vehicle_id does not exist in the _stations dictionary.
375            IndexError: If the station_positions deque is empty.
376
377        Note:
378            - Issues a warning if the vehicle's timestamp is in the future.
379            - Logs debug information when a new message is created.
380            - Modifies the internal state by removing the processed position from the queue.
381        """
382        station_positions = self._stations[vehicle_id]
383        current_vehicle_time = ItsTime.create_from_iso_time(
384            time_str=station_positions[0].creationTime
385        )
386        now = ItsTime()
387        if current_vehicle_time > now:
388            if self._logger.isEnabledFor(level=logging.WARNING):
389                self._logger.warning(
390                    msg=(
391                        f"Station id {vehicle_id}: current vehicle time {current_vehicle_time} "
392                        f"is in the future compared to now {now}"
393                    )
394                )
395            return StationPositionResult(
396                position=None, next_position_time=current_vehicle_time
397            )
398        current_vehicle_point = station_positions.popleft()
399        next_vehicle_point = station_positions[0] if station_positions else None
400        next_vehicle_time = (
401            ItsTime.create_from_iso_time(time_str=next_vehicle_point.creationTime)
402            if next_vehicle_point
403            else None
404        )
405        if self._logger.isEnabledFor(level=logging.DEBUG):
406            self._logger.debug(
407                msg=f"Station id {vehicle_id}: created new message time {current_vehicle_time}"
408            )
409        return StationPositionResult(
410            position=current_vehicle_point, next_position_time=next_vehicle_time
411        )

421            vehicle_id (int): The unique identifier of the vehicle whose positions to pop.
