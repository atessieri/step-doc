  1"""Vehicle Trajectory/ITS Message Scenario Management Module.
  2
  3This module provides functionality for managing and processing vehicle trajectory/ITS
  4message data for publishing to the STEP broker. It handles loading data from JSONL,
  5pickle, or GPX scenario files (via a pluggable set of FileLoader implementations) and
  6serving standardized CAM, CPM, or DENM messages compatible with the STEP cloud schemas.
  7
  8It supports loading multiple vehicles' trajectories/messages from one or more scenario
  9files and serving them per vehicle in real-world time order.
 10
 11Key Features:
 12    - Loading vehicle trajectory/ITS message data via pluggable FileLoader implementations
 13      (JSONL, pickle actor, pickle camera, GPX)
 14    - Conversion to standardized STEP cloud schema CAM, CPM, or DENM messages
 15    - Memory-efficient data handling using a sorted queue
 16    - Support for multiple vehicles in a single scenario
 17    - Scenario-to-real-time timestamp alignment
 18
 19Example:
 20    >>> from logging import Logger
 21    >>> from step_publisher_pkl.loaders import (
 22    ...     JsonlLoader,
 23    ...     PickleActorLoader,
 24    ...     PickleCameraLoader,
 25    ... )
 26    >>> logger = Logger("demo")
 27    >>> scenario = Scenario(
 28    ...     stations_blacklist=set(),
 29    ...     wait_time_before_starting_ms=1000,
 30    ...     loaders=[
 31    ...         JsonlLoader(logger=logger),
 32    ...         PickleActorLoader(),
 33    ...         PickleCameraLoader(),
 34    ...     ],
 35    ...     logger=logger,
 36    ... )
 37    >>> scenario.load_data(input_files=["/path/to/scenario.pkl"])
 38    >>> for vehicle_id in scenario.get_vehicle_ids():
 39    ...     for message in scenario.pop_message(vehicle_id):
 40    ...         # Process the ITS message for crash prediction analysis
 41    ...         pass
 42
 43Note:
 44    - Vehicle trajectory/message data may be provided in JSONL, pickle, or GPX format;
 45      see step_publisher_pkl.loaders for the format each FileLoader implementation expects
 46    - GNSS coordinates, speed, and heading data are required for each trajectory point
 47    - All measurements are converted to appropriate units (cm, cm/s, degrees)
 48
 49Complexity:
 50    - Time: O(s), dominated by Scenario.load_data()/_update_time_stations(),
 51      s = total number of trajectory points/messages loaded across all input files.
 52    - Space: O(s), dominated by Scenario.load_data()/_update_time_stations(),
 53      s = total number of trajectory points/messages loaded across all input files.
 54
 55Classes:
 56    TimeReference: Maps a scenario-time reference point to a real-time reference
 57        point for playback timing conversion. Complexity: time O(1), space O(1).
 58    Scenario: Loads vehicle trajectories/ITS messages from JSONL or pickle files,
 59        rewrites their timestamps from scenario time to real time, and serves them
 60        per vehicle in time order. Complexity (worst case across its methods, in
 61        load_data()/_update_time_stations()): time O(s), space O(s), s = total
 62        number of trajectory points/messages loaded across all input files.
 63"""

110class Scenario:
111    """A scenario management class for vehicle trajectory simulation and real-time data
112    processing.
113
114    This class handles loading vehicle trajectory/ITS message data via pluggable FileLoader
115    implementations, managing time references for scenario-to-real-time conversion, and
116    providing vehicle message data for simulation or analysis purposes. It serves as the
117    main orchestrator for vehicle data processing in step-by-step scenario execution.
118    The Scenario class maintains a collection of vehicle messages loaded from the configured
119    input files, each containing timestamped position/message data. It establishes a time
120    reference that maps scenario time to real time, enabling synchronized playback of
121    recorded vehicle movements.
122
123    Key Features:
124        - Loads vehicle trajectory/ITS message data from multiple input files via pluggable
125          FileLoader implementations
126        - Converts scenario timestamps to real-time for synchronized playback
127        - Manages vehicle message data as time-ordered sorted queues for efficient sequential access
128        - Converts raw trajectory/message data to structured CAM, CPM, or DENM format
129        - Supports iterative consumption of vehicle messages through generators
130
131    Attributes:
132        stations_blacklist (set[int]): Set of station IDs to be ignored in the scenario.
133        wait_time_before_starting_ms (int): Time to wait before starting the scenario in
134            milliseconds.
135        loaders (Sequence[FileLoader]): File loaders tried, in order, against
136            each input file passed to load_data().
137        logger (Logger): Logger instance for recording events and debugging information.
138
139    Usage:
140        The typical workflow involves:
141
142            #. Initialize with stations_blacklist, wait_time_before_starting_ms, loaders
143               (the FileLoader implementations to try, in order), and a logger
144            #. Call load_data() to populate vehicle trajectories/messages from the
145               configured input files
146            #. Use get_vehicle_ids() to enumerate available vehicles
147            #. Call pop_message() to consume each vehicle's messages in time order
148
149        Once a message is consumed via pop_message(), it is removed from internal storage.
150
151    Complexity:
152        - Time: O(s), dominated by load_data()/_update_time_stations(), s = total
153          number of trajectory points/messages loaded across all input files.
154        - Space: O(s), dominated by load_data()/_update_time_stations(), s = total
155          number of trajectory points/messages loaded across all input files.
156    """
157
158    __slots__ = (
159        "_loaders",
160        "_loaders_by_suffix",
161        "_logger",
162        "_stations",
163        "_stations_blacklist",
164        "_wait_time_before_starting_ms",
165    )
166
167    def __init__(
168        self,
169        stations_blacklist: set[int],
170        wait_time_before_starting_ms: int,
171        loaders: Sequence[FileLoader],
172        logger: Logger,
173    ) -> None:
174        """Initialize the scenario with a station blacklist, start delay,
175        file loaders, and logger.
176
177        Args:
178            stations_blacklist (set[int]): Set of station IDs to be ignored in the scenario.
179            wait_time_before_starting_ms (int): Time to wait before starting the scenario in milliseconds.
180            loaders (Sequence[FileLoader]): File loaders to try, in order, against
181                each input file passed to load_data().
182            logger (Logger): Logger instance for recording events and debugging information.
183
184        Complexity:
185            - Time: O(k), where k is the number of loaders, to build the
186              per-suffix loader ordering.
187            - Space: O(k), no vehicle data is stored until load_data() is
188              called.
189        """
190        self._stations_blacklist = stations_blacklist
191        self._wait_time_before_starting_ms = wait_time_before_starting_ms
192        self._logger = logger
193        self._stations: dict[int, SortedList[FacilityMessage]] = {}
194        self._loaders = list(loaders)
195        self._loaders_by_suffix = self._build_loaders_by_suffix(loaders=self._loaders)
196
197    @staticmethod
198    def _build_loaders_by_suffix(
199        loaders: list[FileLoader],
200    ) -> dict[str, list[FileLoader]]:
201        """Group loaders by the file extension each one owns, for fast
202        dispatch in load_data().
203
204        For each extension present among loaders, the loader(s) owning that
205        extension come first (in their given relative order), followed by
206        every other loader (also in their given relative order) as a
207        fallback in case the extension-based guess is wrong.
208
209        Args:
210            loaders (list[FileLoader]): The loaders to group, in the order
211                they should be tried when no extension matches.
212
213        Returns:
214            dict[str, list[FileLoader]]: Mapping from file extension to the
215                ordered list of loaders to try for a file with that
216                extension.
217
218        Complexity:
219            - Time: O(k^2), where k is the number of loaders (a small,
220              fixed-size collection in practice).
221            - Space: O(k^2), one reordered copy of loaders per distinct
222              extension.
223        """
224        extensions = dict.fromkeys(loader.get_extension for loader in loaders)
225        return {
226            extension: (
227                [loader for loader in loaders if loader.get_extension == extension]
228                + [loader for loader in loaders if loader.get_extension != extension]
229            )
230            for extension in extensions
231        }

620    def _update_time_stations(self, time_reference: TimeReference) -> None:
621        """Rebuild the internal station collections with all message times
622        converted from scenario time to real time.
623
624        Iterates over every message of every station, converts its time fields
625        according to its concrete message type (CAM, CPM, or DENM), and replaces
626        the internal station collections with the converted messages. Messages
627        of an unsupported type are skipped with a warning.
628
629        Args:
630            time_reference (TimeReference): The time reference mapping used for
631                conversion.
632
633        Returns:
634            None: The internal station collections are replaced in place; no
635                value is returned.
636
637        Complexity:
638            - Time: O(s), where s is the total number of messages across all
639              stations, each converted and re-inserted in amortized logarithmic
640              time relative to its station's collection size.
641            - Space: O(s), builds a new set of station collections holding the
642              same total number of converted messages.
643        """
644        new_stations: dict[int, SortedList[FacilityMessage]] = {}
645        for station_id, message_queue in self._stations.items():
646            new_stations[station_id] = self._create_queue()
647            for message in message_queue:
648                new_message: FacilityMessage
649                match message:
650                    case CamMessageV1():
651                        new_message = self._update_time_cam_message(
652                            message=message, time_reference=time_reference
653                        )
654                    case CpmMessageV1():
655                        new_message = self._update_time_cpm_message(
656                            message=message, time_reference=time_reference
657                        )
658                    case DenmMessageV1():
659                        new_message = self._update_time_denm_message(
660                            message=message, time_reference=time_reference
661                        )
662                    case _:
663                        if self._logger.isEnabledFor(level=logging.WARNING):
664                            self._logger.warning(
665                                msg=f"Skipping unsupported ITS message type: {message.__class__.__name__}"
666                            )
667                        continue
668                new_stations[station_id].add(value=new_message)
669            if self._logger.isEnabledFor(level=logging.INFO):
670                self._logger.info(
671                    msg=(
672                        f"Station ID {station_id}: new time "
673                        f"from {new_stations[station_id][0].its_time} to {new_stations[station_id][-1].its_time}"
674                    )
675                )
676        self._stations = new_stations

678    def load_data(self, input_files: list[str]) -> None:  # pylint: disable=too-many-locals
679        """Load vehicle trajectory data from the given input files and align
680        message times to real time.
681
682        For each input file, detects whether it is a JSONL or pickle file and
683        loads it accordingly, skipping files that match neither format. If any
684        messages were loaded, computes a time reference from the earliest
685        scenario message time and the current real time (offset by
686        wait_time_before_starting_ms), and rewrites all station messages with
687        real-world timestamps.
688
689        Args:
690            input_files (list[str]): List of paths to the JSONL or pickle files
691                to load.
692
693        Returns:
694            None: The internal station collections are populated in place; no
695                value is returned.
696
697        Complexity:
698            - Time: O(s), where s is the total number of messages loaded across
699              all input files, dominated by loading the files and rewriting
700              message times.
701            - Space: O(s), stores one FacilityMessage per loaded, non-blacklisted
702              trajectory point or JSONL line.
703        """
704        files_path = (Path(file) for file in input_files)
705        # Load each file, trying its extension's loaders first and falling
706        # back to every loader (in the given order) for an unknown extension
707        for file_path in files_path:
708            messages: list[FacilityMessage] | None = None
709            for loader in self._loaders_by_suffix.get(file_path.suffix, self._loaders):
710                try:
711                    messages = loader(file_path=file_path)
712                except LoaderError as exc:
713                    if self._logger.isEnabledFor(level=logging.INFO):
714                        self._logger.info(msg=str(exc))
715                    continue
716                break
717            if messages is None:
718                continue
719            self._store_messages(file_path=file_path, messages=messages)
720        if len(self._stations) == 0:
721            if self._logger.isEnabledFor(level=logging.WARNING):
722                self._logger.warning(
723                    msg="No valid vehicle trajectory data loaded from provided files"
724                )
725            return
726        if self._logger.isEnabledFor(level=logging.INFO):
727            for station_id, message_queue in self._stations.items():
728                self._logger.info(
729                    msg=(
730                        f"Station ID {station_id} has {len(message_queue)} messages loaded "
731                        f"from {message_queue[0].its_time} to {message_queue[-1].its_time} "
732                        f"delta {message_queue[-1].its_time.time_difference(other_time=message_queue[0].its_time)} ms"
733                    )
734                )
735        start_scenario_time = min(
736            station[0].its_time for station in self._stations.values()
737        )
738        start_real_time = ItsTime()
739        start_real_time += self._wait_time_before_starting_ms
740        reference_time = TimeReference(
741            start_scenario_time=start_scenario_time, start_real_time=start_real_time
742        )
743        self._update_time_stations(time_reference=reference_time)
