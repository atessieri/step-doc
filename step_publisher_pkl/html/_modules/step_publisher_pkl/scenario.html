   1"""Vehicle Trajectory Scenario Management Module.
   2
   3This module provides functionality for managing and processing vehicle trajectory data
   4for crash prediction analysis. It handles loading trajectory data from pickle files
   5and generating standardized CAM messages compatible with the STEP cloud schemas.
   6
   7The module is primarily designed for use in crash prediction scenarios where vehicle
   8trajectory data needs to be published in real-time or near-real-time fashion.
   9It supports loading multiple vehicle trajectories from pickle files and converting
  10them into CamDataV1 objects for further analysis.
  11
  12Key Features:
  13    - Loading vehicle trajectory data from pickle files
  14    - Conversion to standardized STEP cloud schema CAM messages
  15    - Memory-efficient data handling using queue
  16    - Support for multiple vehicles in a single scenario
  17    - Scenario-to-real-time timestamp alignment
  18
  19Example:
  20    >>> from logging import Logger
  21    >>> logger = Logger("demo")
  22    >>> scenario = Scenario(
  23    ...     stations_blacklist=set(), wait_time_before_starting_ms=1000, logger=logger
  24    ... )
  25    >>> scenario.load_data(input_files=["/path/to/scenario.pkl"])
  26    >>> for vehicle_id in scenario.get_vehicle_ids():
  27    ...     for message in scenario.pop_message(vehicle_id):
  28    ...         # Process the ITS message for crash prediction analysis
  29    ...         pass
  30
  31Note:
  32    - Vehicle trajectory data is expected to be in pickle format
  33    - Data structure should contain 'trajectories'['vehicles'] dictionary
  34    - GNSS coordinates, speed, and heading data are required for each trajectory point
  35    - All measurements are converted to appropriate units (cm, cm/s, degrees)
  36
  37Complexity:
  38    - Time: O(s), dominated by Scenario.load_data()/_update_time_stations(),
  39      s = total number of trajectory points/messages loaded across all input files.
  40    - Space: O(s), dominated by Scenario.load_data()/_update_time_stations(),
  41      s = total number of trajectory points/messages loaded across all input files.
  42
  43Classes:
  44    TimeReference: Maps a scenario-time reference point to a real-time reference
  45        point for playback timing conversion. Complexity: time O(1), space O(1).
  46    Scenario: Loads vehicle trajectories/ITS messages from JSONL or pickle files,
  47        rewrites their timestamps from scenario time to real time, and serves them
  48        per vehicle in time order. Complexity (worst case across its methods, in
  49        load_data()/_update_time_stations()): time O(s), space O(s), s = total
  50        number of trajectory points/messages loaded across all input files.
  51"""

 108class Scenario:
 109    """A scenario management class for vehicle trajectory simulation and real-time data
 110    processing.
 111
 112    This class handles loading vehicle trajectory data from pickle files, managing time references
 113    for scenario-to-real-time conversion, and providing vehicle position data for simulation or
 114    analysis purposes. It serves as the main orchestrator for vehicle data processing in
 115    step-by-step scenario execution.
 116    The Scenario class maintains a collection of vehicle trajectories loaded from pickle files,
 117    each containing timestamped position data. It establishes a time reference that maps scenario
 118    time to real time, enabling synchronized playback of recorded vehicle movements.
 119
 120    Key Features:
 121        - Loads vehicle trajectory data from multiple pickle files
 122        - Converts scenario timestamps to real-time for synchronized playback
 123        - Manages vehicle position data as queues for efficient sequential access
 124        - Converts raw trajectory data to structured CAM (Cooperative Awareness Message) format
 125        - Supports iterative consumption of vehicle positions through generators
 126
 127    Attributes:
 128        stations_blacklist (set[int]): Set of station IDs to be ignored in the scenario.
 129        wait_time_before_starting_ms (int): Time to wait before starting the scenario in
 130            milliseconds.
 131        logger (Logger): Logger instance for recording events and debugging information.
 132
 133    Usage:
 134        The typical workflow involves:
 135
 136            #. Initialize with a logger
 137            #. Call load_data() to populate vehicle trajectories from pickle files
 138            #. Use get_vehicle_ids() to enumerate available vehicles
 139            #. Call pop_position_arrays() to consume vehicle positions sequentially
 140
 141        Vehicle trajectory data is expected to be in pickle format with a specific structure
 142        containing a 'trajectories' dictionary with vehicle data. Once position data is
 143        consumed via pop_position_arrays(), it is removed from internal storage.
 144
 145    Complexity:
 146        - Time: O(s), dominated by load_data()/_update_time_stations(), s = total
 147          number of trajectory points/messages loaded across all input files.
 148        - Space: O(s), dominated by load_data()/_update_time_stations(), s = total
 149          number of trajectory points/messages loaded across all input files.
 150    """
 151
 152    __slots__ = (
 153        "_logger",
 154        "_stations",
 155        "_stations_blacklist",
 156        "_wait_time_before_starting_ms",
 157    )

 184    def _is_file_jsonl(self, file_path: Path) -> bool:
 185        """Check whether a file is a valid JSONL file of ITS messages.
 186
 187        Attempts to read the first line of the file and validate it against
 188        ItsMessageRoot. The file is not modified or fully consumed by this check.
 189
 190        Args:
 191            file_path (Path): Path to the file to check.
 192
 193        Returns:
 194            bool: True if the file's first line is a valid ITS message in JSONL
 195                format, False otherwise.
 196
 197        Complexity:
 198            - Time: O(1), reads and validates a single line of the file.
 199            - Space: O(1), holds only the single line and its validated model in
 200              memory.
 201        """
 202        try:
 203            with file_path.open(mode="r", encoding="utf-8") as file:
 204                ItsMessageRoot.model_validate_json(json_data=file.readline())
 205                return True
 206        except (OSError, UnicodeDecodeError, ValidationError) as exc:
 207            if self._logger.isEnabledFor(level=logging.INFO):
 208                self._logger.info(
 209                    msg=f"File {file_path} is not a valid JSONL file with ITS messages: {exc}"
 210                )
 211            return False

 213    def _validate_pickle_point(self, point: dict[str, Any]) -> None:
 214        """Validate that a single trajectory point contains the required keys.
 215
 216        Args:
 217            point (dict[str, Any]): A single trajectory point expected to contain
 218                a 'timestamp' key and a 'geo_location' key with 'latitude',
 219                'longitude', and 'altitude' sub-keys. The 'heading' and
 220                'velocity' keys are optional, matching _dict_to_cam_data().
 221
 222        Returns:
 223            None
 224
 225        Raises:
 226            ValueError: If 'timestamp', 'geo_location', or any of the required
 227                'geo_location' sub-keys are missing from the point.
 228
 229        Complexity:
 230            - Time: O(1), checks a constant number of dictionary keys.
 231            - Space: O(1), no additional data structures are allocated.
 232        """
 233        if not isinstance(point["timestamp"], datetime.datetime):
 234            raise TypeError("Trajectory point 'timestamp' is not a datetime instance")
 235        geo_location = point["geo_location"]
 236        for key in ("latitude", "longitude", "altitude"):
 237            if not isinstance(geo_location[key], (float, int)):
 238                raise TypeError(
 239                    f"Trajectory point 'geo_location' '{key}' is not a float or int"
 240                )
 241        if ("heading" in point) and not isinstance(point["heading"], (float, int)):
 242            raise TypeError("heading is not float or int")
 243        if ("velocity" in point) and not isinstance(
 244            point["velocity"]["speed"], (float, int)
 245        ):
 246            raise TypeError("speed is not float or int")

 248    def _validate_pickle_actor_data(self, data: Any) -> None:
 249        """Validate the overall structure of unpickled trajectory data.
 250
 251        Checks that the unpickled object is a dictionary containing a
 252        'trajectories' dictionary with a 'vehicles' key, and validates the first
 253        trajectory point of the first vehicle found, if any.
 254
 255        Args:
 256            data (Any): The object obtained from unpickling a file, expected to be
 257                a dictionary with a 'trajectories'['vehicles'] structure.
 258
 259        Returns:
 260            None
 261
 262        Raises:
 263            TypeError: If data is not a dictionary.
 264            ValueError: If the 'trajectories' or 'vehicles' keys are missing, or if
 265                the first validated trajectory point is malformed.
 266
 267        Complexity:
 268            - Time: O(1), inspects a constant number of keys and validates only the
 269              first point of the first vehicle, independent of the total data
 270              size.
 271            - Space: O(1), no additional data structures are allocated beyond a
 272              reference to the existing vehicles dictionary.
 273        """
 274        if not isinstance(data, dict):
 275            raise TypeError("Pickle actor file does not contain a dictionary")
 276        vehicles_data: dict[str, list[dict[str, Any]]] = data["trajectories"][
 277            "vehicles"
 278        ]
 279        if len(vehicles_data) == 0:
 280            raise ValueError("Pickle actor file contains no vehicle trajectories")
 281        for vehicle_id, vehicle_points in vehicles_data.items():
 282            if not isinstance(vehicle_id, int):
 283                raise TypeError("Vehicle ID is not an int")
 284            if len(vehicle_points) == 0:
 285                raise ValueError(f"Vehicle ID {vehicle_id} has no trajectory points")
 286            for point in vehicle_points:
 287                self._validate_pickle_point(point=point)

 289    def _is_file_pickle_actor(self, file_path: Path) -> bool:
 290        """Check whether a file is a valid pickle file containing vehicle
 291        trajectories.
 292
 293        Attempts to unpickle the file and validate its structure via
 294        _validate_pickle_data().
 295
 296        Args:
 297            file_path (Path): Path to the file to check.
 298
 299        Returns:
 300            bool: True if the file can be unpickled and contains a valid vehicle
 301                trajectories structure, False otherwise.
 302
 303        Complexity:
 304            - Time: O(m), where m is the size of the pickled data, dominated by
 305              unpickling the entire file content.
 306            - Space: O(m), the unpickled data is held in memory during
 307              validation.
 308        """
 309        try:
 310            with file_path.open(mode="rb") as file:
 311                data = pickle_load(file)  # noqa: S301
 312                self._validate_pickle_actor_data(data=data)
 313                return True
 314        except (
 315            OSError,
 316            TypeError,
 317            ValueError,
 318            UnpicklingError,
 319            KeyError,
 320            AttributeError,
 321        ) as exc:
 322            if self._logger.isEnabledFor(level=logging.INFO):
 323                self._logger.info(
 324                    msg=f"File {file_path} is not a valid pickle actor file with vehicle trajectories: {exc}"
 325                )
 326            return False

 328    def _validate_pickle_camera_data(self, data: Any) -> None:
 329        """Validate the overall structure of unpickled camera (CPM) data.
 330
 331        Checks that the unpickled object is a non-empty list and validates
 332        every item against CpmDataV1.
 333
 334        Args:
 335            data (Any): The object obtained from unpickling a file, expected
 336                to be a list of dictionaries each convertible to CpmDataV1.
 337
 338        Returns:
 339            None
 340
 341        Raises:
 342            TypeError: If data is not a list.
 343            ValueError: If data is an empty list.
 344            pydantic.ValidationError: If any item in data does not validate
 345                against CpmDataV1.
 346
 347        Complexity:
 348            - Time: O(m), where m is the number of items in data, each
 349              validated against CpmDataV1.
 350            - Space: O(1), no additional data structures are allocated beyond
 351              the transient CpmDataV1 instances created for validation.
 352        """
 353        if not isinstance(data, list):
 354            raise TypeError("Pickle camera file does not contain a list")
 355        if len(data) == 0:
 356            raise ValueError("Pickle camera file contains no camera data")
 357        for item in data:

 462    def _load_from_jsonl_file(self, file_path: Path) -> None:
 463        """Load ITS messages from a JSONL file into the internal station
 464        collections.
 465
 466        Reads the file line by line, validates and converts each line into a
 467        CamMessageV1, CpmMessageV1, or DenmMessageV1 via ItsMessageRoot, and adds
 468        the resulting message to the corresponding station's sorted collection.
 469        Lines with an unsupported message type or that fail validation are
 470        skipped with a warning. Messages from blacklisted station IDs are
 471        skipped.
 472
 473        Args:
 474            file_path (Path): Path to the JSONL file to load.
 475
 476        Returns:
 477            None
 478
 479        Complexity:
 480            - Time: O(l), where l is the number of lines in the file, each parsed,
 481              validated, and inserted in amortized logarithmic time relative to
 482              the station's collection size.
 483            - Space: O(l), one FacilityMessage is stored per valid,
 484              non-blacklisted line.
 485        """
 486        if self._logger.isEnabledFor(level=logging.INFO):
 487            self._logger.info(msg=f"Loading data from JSONL file {file_path}")
 488        with file_path.open(mode="r", encoding="utf-8") as file:
 489            for line in file:
 490                try:
 491                    message_root = ItsMessageRoot.model_validate_json(json_data=line)
 492                    message: FacilityMessage
 493                    match message_root.root:
 494                        case CamDataV1():
 495                            message = CamMessageV1(message_param=message_root.root)
 496                        case CpmDataV1():
 497                            message = CpmMessageV1(message_param=message_root.root)
 498                        case DenmDataV1():
 499                            message = DenmMessageV1(message_param=message_root.root)
 500                        case _:
 501                            raise ValidationError(  # noqa: TRY301
 502                                self._logger.warning(
 503                                    msg=(
 504                                        f"Skipping unsupported ITS message type in {file_path}: "
 505                                        f"{message_root.root.__class__.__name__}"
 506                                    )
 507                                )
 508                            )
 509                    station_id = message.station_id
 510                    if station_id in self._stations_blacklist:
 511                        if self._logger.isEnabledFor(level=logging.DEBUG):
 512                            self._logger.debug(
 513                                msg=(
 514                                    f"Skipping ITS message from blacklisted station ID {station_id} "
 515                                    f"in {file_path} - JSONL file"
 516                                )
 517                            )
 518                        continue
 519                    if self._stations.get(station_id) is None:
 520                        self._stations[station_id] = self._create_queue()
 521                    self._stations[station_id].add(value=message)
 522                except ValidationError as exc:
 523                    self._logger.warning(
 524                        msg=f"Skipping invalid ITS message in {file_path}: {exc}"
 525                    )

 527    def _load_from_pickle_actor_file(self, file_path: Path) -> None:
 528        """Load vehicle trajectory data from a pickle file into the internal
 529        station collections.
 530
 531        Unpickle the file, iterates over each vehicle's trajectory points,
 532        converts each point into a CamMessageV1 via _dict_to_cam_data(), and adds
 533        it to the corresponding station's sorted collection. Vehicles with a
 534        blacklisted station ID are skipped entirely.
 535
 536        Args:
 537            file_path (Path): Path to the pickle file to load.
 538
 539        Returns:
 540            None
 541
 542        Complexity:
 543            - Time: O(p), where p is the total number of trajectory points across
 544              all vehicles in the file, each converted and inserted in amortized
 545              logarithmic time relative to the station's collection size.
 546            - Space: O(p), one CamMessageV1 is stored per non-blacklisted
 547              trajectory point.
 548        """
 549        if self._logger.isEnabledFor(level=logging.INFO):
 550            self._logger.info(msg=f"Loading data from pickle actor file {file_path}")
 551        with file_path.open(mode="rb") as file:
 552            data: dict[str, Any] = pickle_load(file=file)  # noqa: S301
 553            vehicles_data: dict[str, list[dict[str, Any]]] = data["trajectories"][
 554                "vehicles"
 555            ]
 556            # Traverse all points of all vehicles creating a dictionary of queue
 557            for vehicle_id, vehicle_points in vehicles_data.items():
 558                vehicle_id_int = int(vehicle_id)
 559                if vehicle_id_int in self._stations_blacklist:
 560                    if self._logger.isEnabledFor(level=logging.DEBUG):
 561                        self._logger.debug(
 562                            msg=(
 563                                f"Skipping vehicle ID {vehicle_id_int} from blacklisted stations "
 564                                f"in {file_path} - pickle actor file"
 565                            )
 566                        )
 567                    continue
 568                station_deque = self._stations.get(vehicle_id_int)
 569                if station_deque is None:
 570                    station_deque = self._create_queue()
 571                    self._stations[vehicle_id_int] = station_deque
 572                for point in vehicle_points:
 573                    station_deque.add(
 574                        value=CamMessageV1(
 575                            message_param=self._dict_to_cam_data(
 576                                vehicle_id=vehicle_id_int, data=point
 577                            )
 578                        )
 579                    )

 623    def _scenario_time_to_real(
 624        self, scenario_time: ItsTime, time_reference: TimeReference
 625    ) -> ItsTime:
 626        """Convert scenario time to real time using the time reference.
 627
 628        Args:
 629            scenario_time (ItsTime): The scenario time to convert.
 630            time_reference (TimeReference): The time reference mapping for conversion.
 631
 632        Returns:
 633            ItsTime: The corresponding real time.
 634
 635        Complexity:
 636            - Time: O(1), computes a single time difference and offset.
 637            - Space: O(1), allocates a single ItsTime instance.
 638        """
 639        delta = scenario_time.time_difference(
 640            other_time=time_reference.start_scenario_time
 641        )
 642        return time_reference.start_real_time + delta

 644    def _update_time_cam_message(
 645        self, message: CamMessageV1, time_reference: TimeReference
 646    ) -> CamMessageV1:
 647        """Update the creation time of CAM (Cooperative Awareness Message) data from
 648        scenario time to real time.
 649
 650        This method converts the scenario-relative creation time of a CAM data object to real-world time
 651        using the provided time reference, and returns a new CAM data object with the updated timestamp.
 652
 653        Args:
 654            message (CamMessageV1): The CAM message object containing the original scenario-relative creation time.
 655            time_reference (TimeReference): The time reference object used to convert scenario time to real time.
 656
 657        Returns:
 658            CamMessageV1: A new CAM message object with the creation time updated to real-world time as a string.
 659
 660        Note:
 661            The method uses model_copy to create a new instance rather than modifying the original data object.
 662            The creation time is converted to a string representation in the returned object.
 663
 664        Complexity:
 665            - Time: O(1), copies a single fixed-size object with one updated field.
 666            - Space: O(1), allocates one new CamMessageV1 instance via model_copy.
 667        """
 668        message_data = message.message
 669        return CamMessageV1(
 670            message_param=message.message.model_copy(
 671                update={
 672                    "creationTime": str(
 673                        object=self._scenario_time_to_real(
 674                            scenario_time=ItsTime.create_from_iso_time(
 675                                time_str=message_data.creationTime
 676                            ),
 677                            time_reference=time_reference,
 678                        )
 679                    )
 680                }
 681            )
 682        )

 960    def _update_time_stations(self, time_reference: TimeReference) -> None:
 961        """Rebuild the internal station collections with all message times
 962        converted from scenario time to real time.
 963
 964        Iterates over every message of every station, converts its time fields
 965        according to its concrete message type (CAM, CPM, or DENM), and replaces
 966        the internal station collections with the converted messages. Messages
 967        of an unsupported type are skipped with a warning.
 968
 969        Args:
 970            time_reference (TimeReference): The time reference mapping used for
 971                conversion.
 972
 973        Returns:
 974            None
 975
 976        Complexity:
 977            - Time: O(s), where s is the total number of messages across all
 978              stations, each converted and re-inserted in amortized logarithmic
 979              time relative to its station's collection size.
 980            - Space: O(s), builds a new set of station collections holding the
 981              same total number of converted messages.
 982        """
 983        new_stations: dict[int, SortedList[FacilityMessage]] = {}
 984        for station_id, message_queue in self._stations.items():
 985            new_stations[station_id] = self._create_queue()
 986            for message in message_queue:
 987                new_message: FacilityMessage
 988                match message:
 989                    case CamMessageV1():
 990                        new_message = self._update_time_cam_message(
 991                            message=message, time_reference=time_reference
 992                        )
 993                    case CpmMessageV1():
 994                        new_message = self._update_time_cpm_message(
 995                            message=message, time_reference=time_reference
 996                        )
 997                    case DenmMessageV1():
 998                        new_message = self._update_time_denm_message(
 999                            message=message, time_reference=time_reference
1000                        )
1001                    case _:
1002                        if self._logger.isEnabledFor(level=logging.WARNING):
1003                            self._logger.warning(
1004                                msg=f"Skipping unsupported ITS message type: {message.__class__.__name__}"
1005                            )
1006                        continue
1007                new_stations[station_id].add(value=new_message)
1008            if self._logger.isEnabledFor(level=logging.INFO):
1009                self._logger.info(

1017    def load_data(self, input_files: list[str]) -> None:  # pylint: disable=too-many-locals
1018        """Load vehicle trajectory data from the given input files and align
1019        message times to real time.
1020
1021        For each input file, detects whether it is a JSONL or pickle file and
1022        loads it accordingly, skipping files that match neither format. If any
1023        messages were loaded, computes a time reference from the earliest
1024        scenario message time and the current real time (offset by
1025        wait_time_before_starting_ms), and rewrites all station messages with
1026        real-world timestamps.
1027
1028        Args:
1029            input_files (list[str]): List of paths to the JSONL or pickle files
1030                to load.
1031
1032        Returns:
1033            None
1034
1035        Complexity:
1036            - Time: O(s), where s is the total number of messages loaded across
1037              all input files, dominated by loading the files and rewriting
1038              message times.
1039            - Space: O(s), stores one FacilityMessage per loaded, non-blacklisted
1040              trajectory point or JSONL line.
1041        """
1042        files_path = (Path(file) for file in input_files)
1043        # Load each pickle file and extract vehicle trajectory data
1044        for file_path in files_path:
1045            if self._is_file_jsonl(file_path=file_path):
1046                self._load_from_jsonl_file(file_path=file_path)
1047            elif self._is_file_pickle_actor(file_path=file_path):
1048                self._load_from_pickle_actor_file(file_path=file_path)
1049            elif self._is_file_pickle_camera(file_path=file_path):
1050                self._load_from_pickle_camera_file(file_path=file_path)
1051            else:
1052                self._logger.warning(
1053                    msg=f"File {file_path} is neither a valid JSONL nor a valid pickle file"
1054                )
1055        if len(self._stations) == 0:
1056            if self._logger.isEnabledFor(level=logging.WARNING):
1057                self._logger.warning(
1058                    msg="No valid vehicle trajectory data loaded from provided files"
1059                )
1060            return
1061        if self._logger.isEnabledFor(level=logging.INFO):
1062            for station_id, message_queue in self._stations.items():
1063                self._logger.info(
1064                    msg=(
1065                        f"Station ID {station_id} has {len(message_queue)} messages loaded "
1066                        f"from {message_queue[0].its_time} to {message_queue[-1].its_time} "
1067                        f"delta {message_queue[-1].its_time.time_difference(other_time=message_queue[0].its_time)} ms"
1068                    )
1069                )
1070        start_scenario_time = min(
1071            station[0].its_time for station in self._stations.values()
1072        )
1073        start_real_time = ItsTime()
1074        start_real_time += self._wait_time_before_starting_ms
1075        reference_time = TimeReference(
1076            start_scenario_time=start_scenario_time, start_real_time=start_real_time
1077        )
1078        self._update_time_stations(time_reference=reference_time)
