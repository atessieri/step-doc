  1"""STEP Gateway Module for MQTT and Redis Integration.
  2
  3This module provides the necessary components for a secure, configurable MQTT-based gateway
  4within the STEP ecosystem.
  5
  6The module implements a gateway that supports:
  7
  8    - Secure TLS/SSL MQTT communications with configurable parameters
  9    - Bidirectional message conversion between MQTT and Redis streams
 10    - Asynchronous operation with context management for clean resource handling
 11    - Quality of Service (QoS) controls for reliable message delivery
 12    - Support for various ITS message types, with focus on CAM (Cooperative Awareness Messages)
 13    - Clean error handling and comprehensive logging
 14
 15The STEP Broker class leverages async/await patterns throughout to ensure non-blocking
 16operations, making it suitable for high-throughput messaging environments in
 17ITS applications.
 18"""

 43class StepBroker(AsyncComManager):
 44    """STEP gateway implementing AsyncComManager.
 45
 46    This class provides a high-level client for connecting to a STEP service
 47    using MQTT protocol. It handles secure connections, topic subscriptions
 48    and message management according to the provided configuration.
 49
 50    The class manages:
 51
 52        - TLS/SSL connections with certificate validation
 53        - MQTT connection parameters (host, port, auth, timeouts, etc.)
 54        - Topic subscriptions with configurable QoS levels
 55        - Message queuing and delivery
 56
 57    It implements the AsyncComManager interface and supports asynchronous
 58    context manager protocol for safe resource management.
 59
 60    Attributes:
 61        instance_name (str): Unique identifier for the STEP client instance.
 62        config (StepBrokerConfig): Configuration object containing STEP broker settings.
 63        logger (Logger): Logger instance for logging messages and errors.
 64    """
 65
 66    MAC_ID: ClassVar[int] = 123456789
 67
 68    __slots__ = ("_config", "_geo_network_address", "_logger")

 70    def __init__(
 71        self, instance_name: str, config: StepBrokerConfig, logger: Logger
 72    ) -> None:
 73        """Initialize the StepBroker instance.
 74
 75        Args:
 76            instance_name (str): The name identifier for this broker instance.
 77            config (StepBrokerConfig): Configuration object containing STEP broker settings
 78                including MQTT client parameters, protocol settings, and SSL configuration.
 79            logger (Logger): Logger instance for recording broker operations and events.
 80
 81        Returns:
 82            None
 83
 84        Note:
 85            This initializer sets up the STEP broker with MQTT communication capabilities,
 86            configures SSL/TLS parameters if enabled, and initializes the async communication
 87            manager with all necessary parameters for STEP protocol communication.
 88        """
 89        self._config = config
 90        self._geo_network_address = GeoNetworkAddress(
 91            is_manual_configured=True,
 92            traffic_participant_type=TrafficParticipantType.PASSENGER_CAR,
 93            mac_id=self.MAC_ID,
 94        )
 95        self._logger = logger
 96        tls_context: Optional[ssl.SSLContext] = None
 97        if self._config.mqttClient.sslEnable:
 98            tls_context = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
 99            tls_context.minimum_version = ssl.TLSVersion.TLSv1_2
100            if self._config.mqttClient.sslCaCertsFile:
101                tls_context.load_verify_locations(
102                    cafile=self._config.mqttClient.sslCaCertsFile
103                )
104            if not self._config.mqttClient.sslCertRequested:
105                tls_context.check_hostname = False
106                tls_context.verify_mode = ssl.CERT_NONE
107                self._logger.warning(
108                    "TLS certificate verification is DISABLED. MITM attacks possible."
109                )
110            if self._config.mqttClient.sslClientCertFile:
111                tls_context.load_cert_chain(
112                    certfile=self._config.mqttClient.sslClientCertFile,
113                    keyfile=self._config.mqttClient.sslClientKeyFile,
114                    password=self._config.mqttClient.sslClientKeyFilePassword,
115                )
116        async_com_manager_params = AsyncComManagerParams(
117            instance_name=instance_name,
118            host=self._config.mqttClient.host,
119            port=self._config.mqttClient.port,
120            username=self._config.mqttClient.username or "",
121            password=self._config.mqttClient.password or "",
122            timeout=self._config.mqttProtocol.timeout,
123            station_id=None,
124            keepalive=self._config.mqttProtocol.keepalive,
125            bind_address="",
126            bind_port=0,
127            clean_start=self._config.mqttProtocol.cleanStart,
128            queue_type=None,
129            max_queued_incoming_messages=self._config.mqttProtocol.maxQueuedIncomingMessages,
130            max_queued_outgoing_messages=self._config.mqttProtocol.maxQueuedOutgoingMessages,
131            max_inflight_messages=self._config.mqttProtocol.maxInflightMessages,
132            max_concurrent_outgoing_calls=self._config.mqttProtocol.maxConcurrentOutgoingCalls,
133            tls_context=tls_context,
134            tls_params=None,
135            proxy=None,
136            socket_options=None,
137            session_expiry=self._config.mqttProtocol.sessionExpiry,
138            receive_maximum=10,
139            maximum_packet_size=4096,
140            topic_alias_maximum=0,
141            publish_topics_params=(
142                [
143                    PublishTopicParams(
144                        message_type=MessageId(topic.messageType),
145                        header_topic=topic.headerTopic,
146                        geohash_precision=topic.geohashPrecision,
147                        qos=MqttQoS(topic.qos),
148                    )
149                    for topic in self._config.mqttProtocol.publishTopics
150                ]
151                if self._config.mqttProtocol.publishTopics is not None
152                else None
153            ),
154            cache_dir=self._config.cacheDir,
155            logger=self._logger,
156        )
157        super().__init__(params=async_com_manager_params)

173    async def subscribe_execution(self) -> None:
174        """Subscribe to a list of topics with specified MQTT options.
175
176        This method subscribes to a list of topics specified in the configuration
177        with the following MQTT options:
178
179            - QoS level from config
180            - No local messages (noLocal=True)
181            - Do not retain as published (retainAsPublished=False)
182            - Do not send retained messages (retainHandling=RETAIN_DO_NOT_SEND)
183
184        Raises:
185            StepPublisherPklError: If subscription fails or QoS level doesn't match the configured
186                one.
187
188        Returns:
189            None
190        """
191        if self._config.mqttProtocol.subscriptionTopics is not None:
192            sub_results = await self.subscribe(
193                topic=[
194                    (
195                        topic_params.topic,
196                        SubscribeOptions(
197                            qos=self._config.mqttProtocol.maxSubscriptionQos,
198                            noLocal=False,
199                            retainAsPublished=False,
200                            retainHandling=SubscribeOptions.RETAIN_DO_NOT_SEND,
201                        ),
202                    )
203                    for topic_params in self._config.mqttProtocol.subscriptionTopics
204                ]
205            )
206            if len(sub_results) != len(self._config.mqttProtocol.subscriptionTopics):
207                raise StepPublisherPklError(
208                    f"Subscription results {len(sub_results)} do not match the number "
209                    f"of topics {len(self._config.mqttProtocol.subscriptionTopics)}."
210                )
211            for topic, result in zip(
212                self._config.mqttProtocol.subscriptionTopics, sub_results, strict=True
213            ):
214                if result != self._config.mqttProtocol.maxSubscriptionQos:
215                    raise StepPublisherPklError(
216                        f"Failed to subscribe to {topic}: {result}"
217                    )
218                self._logger.info("Subscribed to %s: %s", topic, result)
