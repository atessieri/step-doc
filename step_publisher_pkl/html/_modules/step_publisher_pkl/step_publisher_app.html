Source code for step_publisher_pkl.step_publisher_app

  1"""Workload of the STEP Publisher Pickle program.
  2
  3Defines the ``AbstractApp`` subclass handed to ``AppRunner`` by
  4``step_publisher_pkl.__main__``: it reads the MQTT credential from the environment,
  5connects to the STEP MQTT broker, loads the scenario from the configured input files
  6and publishes the ITS messages of every vehicle with one concurrent publisher task
  7per vehicle, preserving the timing of the scenario. The workload ends on its own once
  8every publisher task has published all its messages.
  9
 10Complexity:
 11    - Time: O(s log s), dominated by StepPublisherApp.work loading the scenario
 12      (Scenario.load_data) and by _publisher_run aggregated over every vehicle
 13      (s = total number of messages loaded and published across all vehicles;
 14      each insert into/pop from a sorted collection is O(log s)), excluding
 15      unbounded wall-clock sleep/network time.
 16    - Space: O(s), dominated by StepPublisherApp.work holding the loaded
 17      scenario (every message in memory until published) plus one concurrent
 18      task per vehicle (O(v), v <= s).
 19
 20Classes:
 21    - ExitCode: Enumeration of the program exit codes. Complexity: time O(1),
 22      space O(1).
 23    - StepPublisherApp: Workload run by ``AppRunner``: the STEP broker connection,
 24      the scenario and one publisher task per vehicle. Complexity: time
 25      O(s log s), space O(s).
 26"""

 74class StepPublisherApp(AbstractApp[ExitCode]):
 75    """Workload of the program, run by ``AppRunner``: the STEP broker connection, the
 76    scenario and one publisher task per vehicle.
 77
 78    Holds the state shared by ``work`` and ``on_shutdown``: the settings and the
 79    MQTT credential read from the environment.
 80
 81    Attributes:
 82        instance_name (str): The name of the STEP client instance.
 83        config (StepPublisherPklConfig): Configuration object containing all
 84            necessary settings.
 85        logger (Logger): Logger instance for recording program events.
 86
 87    Complexity:
 88        - Time: O(s log s), dominated by ``work`` (scenario loading plus
 89          ``_publisher_run`` over every vehicle), where s is the total number of
 90          messages loaded and published across all vehicles, excluding the
 91          wall-clock time spent sleeping between messages.
 92        - Space: O(s), dominated by ``work`` holding the loaded scenario; the
 93          v publisher tasks add O(v), v <= s.
 94    """
 95
 96    __slots__ = ("_config", "_instance_name", "_logger", "_mqtt_credential")
 97
 98    def __init__(
 99        self, instance_name: str, config: StepPublisherPklConfig, logger: Logger
100    ) -> None:
101        """Store the settings and read the MQTT credential from the environment.
102
103        Args:
104            instance_name (str): The name of the STEP client instance.
105            config (StepPublisherPklConfig): Configuration object containing all
106                necessary settings.
107            logger (Logger): Logger instance for recording program events.
108
109        Raises:
110            ValueError: If the MQTT credential cannot be read from the
111                environment (see ``MqttCredential.from_env``).
112
113        Complexity:
114            - Time: O(1), a few attribute assignments and a fixed number of
115              environment variable lookups.
116            - Space: O(1), a fixed set of references.
117        """
118        self._instance_name = instance_name
119        self._config = config
120        self._logger = logger
121        # Read here and not in work(): a missing variable ends the program
122        # before the event loop is started
123        self._mqtt_credential = MqttCredential.from_env(logger=logger)

125    async def _publisher_run(
126        self, client: StepBroker, scenario: Scenario, vehicle_id: int
127    ) -> None:
128        """Asynchronously publishes ITS messages for a specific vehicle from scenario
129        data.
130
131        This coroutine iterates through the vehicle's messages retrieved from the scenario
132        using pop_message(), sleeping when necessary to preserve the original timing between
133        messages before publishing each one through the STEP broker.
134
135        Args:
136            client (StepBroker): STEP broker client used to publish ITS messages via
137                client_publish_its_message().
138            scenario (Scenario): Scenario object providing vehicle message data through
139                pop_message().
140            vehicle_id (int): Unique identifier for the vehicle whose messages to publish.
141
142        Returns:
143            None: Runs until the vehicle's messages are exhausted; no value is
144                returned.
145
146        Processing Flow:
147            #. Retrieve the vehicle's next message via scenario.pop_message(vehicle_id)
148            #. Log the message type and time if DEBUG level enabled
149            #. Compute the time delta between the message's time and the current ItsTime
150            #. Sleep for the computed delta (converted from ms to seconds) if it is positive
151            #. Publish the message via client.client_publish_its_message()
152            #. Log the published message type and time if DEBUG level enabled
153
154        Note:
155            - Uses ItsTime for current time and time difference calculations
156            - Converts sleep duration from milliseconds to seconds using ItsTime.SEC_TO_MS_FACTOR
157            - Skips sleeping when the message time is not in the future
158
159        Complexity:
160            - Time: O(n log n), where n is the number of messages available for the
161              vehicle (each pop is O(log n)), excluding the wall-clock time spent
162              sleeping between messages.
163            - Space: O(1), processes one message at a time without accumulating
164              additional data structures.
165        """
166        for message in scenario.pop_message(vehicle_id=vehicle_id):
167            now = ItsTime()
168            message_time = message.its_time
169            delta = message_time.time_difference(other_time=now)
170            if self._logger.isEnabledFor(level=logging.DEBUG):
171                self._logger.debug(
172                    msg=(
173                        f"Station id: {vehicle_id}, message type {message.message_type.message_type}, "
174                        f"time: {message_time}, current time: {now}, delta: {delta} ms"
175                    )
176                )
177            if delta > 0:
178                if self._logger.isEnabledFor(level=logging.DEBUG):
179                    self._logger.debug(
180                        msg=f"Station id: {vehicle_id} sleeping for {delta} ms"
181                    )
182                await asyncio_sleep(delay=delta / ItsTime.SEC_TO_MS_FACTOR)
183            await client.client_publish_its_message(message=message)
184            if self._logger.isEnabledFor(level=logging.DEBUG):
185                self._logger.debug(
186                    msg=(
187                        f"Station id: {vehicle_id}, published message type {message.message_type.message_type}, "
188                        f"time: {message_time}"
189                    )
190                )

192    @override
193    async def work(self) -> ExitCode:
194        """Connect to the STEP broker, load the scenario and publish the messages of
195        every vehicle.
196
197        Opens a ``StepBroker``, loads the scenario from the configured input files
198        and runs one ``_publisher_run`` task per vehicle in a ``TaskGroup`` nested
199        in the broker context, so the broker only disconnects once the group has
200        unwound. Returns once every publisher task has finished on its own; on a
201        shutdown signal ``AppRunner`` cancels it wherever it is waiting.
202
203        Returns:
204            ExitCode: ExitCode.OK, once every publisher task has finished.
205
206        Raises:
207            ExceptionGroup: Any exception raised by a publisher task, wrapped by
208                the ``TaskGroup``.
209
210        Complexity:
211            - Time: O(s log s), where s is the total number of messages across
212              all vehicles: Scenario.load_data is O(s log s), creating one
213              publisher task per vehicle is O(v), and the tasks together pop
214              and publish every message in O(s log s); excludes the wall-clock
215              time spent sleeping and on the network.
216            - Space: O(s), the loaded scenario holds every message in memory
217              until it is published, plus one concurrent task per vehicle
218              (O(v), v <= s).
219        """
220        async with (
221            StepBroker(
222                instance_name=self._instance_name,
223                config=self._config.stepBroker,
224                mqtt_credential=self._mqtt_credential,
225                logger=self._logger,
226            ) as client,
227            TaskGroup() as group,
228        ):
229            scenario = Scenario(
230                stations_blacklist=set(self._config.stationsBlacklist or []),
231                wait_time_before_starting_ms=self._config.waitTimeBeforeStarting,
232                loaders=[
233                    JsonlLoader(),
234                    PickleActorLoader(),
235                    PickleCameraLoader(),
236                    GpxLoader(),
237                ],
238                logger=self._logger,
239            )
240            scenario.load_data(input_files=self._config.inputFiles)
241            # Create one publisher task per vehicle
242            for vehicle_id in scenario.get_vehicle_ids():
243                group.create_task(
244                    coro=self._publisher_run(
245                        client=client, scenario=scenario, vehicle_id=vehicle_id
246                    )
247                )
248        return ExitCode.OK

250    @override
251    def on_shutdown(self, stop_signal: Signals) -> ExitCode:
252        """Log the shutdown signal that stopped ``work``.
253
254        Called by ``AppRunner`` after every task has unwound, so the publisher
255        tasks are cancelled and the STEP broker is already disconnected. The
256        messages not yet published are dropped.
257
258        Args:
259            stop_signal (Signals): The signal that stopped the run.
260
261        Returns:
262            ExitCode: ExitCode.ABORTED.
263
264        Complexity:
265            - Time: O(1), a single conditional check and one log call.
266            - Space: O(1), no additional data structures are allocated.
267        """
268        if self._logger.isEnabledFor(level=logging.INFO):
269            self._logger.info(msg=f"Program interrupted by {stop_signal.name}")
270        return ExitCode.ABORTED
