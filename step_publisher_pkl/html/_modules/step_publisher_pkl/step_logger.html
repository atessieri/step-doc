  1"""Step Publisher Pickle Logging Module.
  2
  3This module provides a comprehensive logging solution for the step_publisher_pkl package with
  4support for various logging handlers, UTC timezone formatting, and advanced file rotation
  5capabilities with compression.
  6
  7Key Features:
  8    - UTC-based timestamp formatting for consistent logging across timezones
  9    - Multiple handler types: stream, file, null, and rotating file handlers
 10    - Automatic log file compression using BZIP2 for space efficiency
 11    - Size-based and time-based log rotation with configurable backup counts
 12    - Flexible configuration through LogConfig objects
 13    - Instance-specific logging for multi-instance applications
 14
 15Handler Types Supported:
 16    - StreamHandler: Output to stdout/stderr
 17    - FileHandler: Write to static log files with optional append mode
 18    - NullHandler: Discard all log messages (for testing/production silence)
 19    - RotatingFileHandler: Size-based rotation with automatic compression
 20    - TimedRotatingFileHandler: Time-based rotation with automatic compression
 21
 22Configuration:
 23    The module uses LogConfig objects from step_publisher_pkl_config to define:
 24
 25        - Log levels (DEBUG, INFO, WARNING, ERROR, CRITICAL)
 26        - Handler types and their specific parameters
 27        - Message formatting and date formatting
 28        - File paths, rotation intervals, and backup counts
 29
 30Note:
 31    This module is part of the step_publisher_pkl package and requires the step_publisher_pkl_config
 32    module for configuration classes and enums.
 33"""

 67class UtcFormatter(Formatter):
 68    """This class definition creates a custom logging formatter called UtcFormatter that
 69    inherits from Python's built-in logging.Formatter class.
 70
 71    Here's what the class method does:
 72    converter: This class attribute sets the time conversion function to ``gmtime``,
 73    which converts a time in seconds since the epoch to a struct_time in UTC
 74    (Coordinated Universal Time) zone.
 75    This means that any timestamps generated by this formatter will be in UTC.
 76    Note that this class doesn't define any instance methods, only a class attribute.
 77    """

 94def _rotator(source: str, dest: str) -> None:
 95    """Compresses a file using BZIP2 compression and saves it to the specified
 96    destination, then removes the original file.
 97
 98    Args:
 99        source (str): The path to the source file to be compressed.
100        dest (str): The path where the compressed file will be saved.
101
102    Returns:
103        None
104    """
105    with ZipFile(
106        file=dest,
107        mode="w",
108        compression=zipfile.ZIP_BZIP2,
109        allowZip64=True,
110        compresslevel=9,
111    ) as zip_file:
112        zip_file.write(filename=source, arcname=Path(source).name)
113        Path(source).unlink()

128def get_logger(  # noqa: C901
129    config: Optional[LogConfig] = None, instance_name: Optional[str] = None
130) -> Logger:
131    """Create and configure a logger instance with the specified configuration.
132
133    This function creates a logger with customizable handlers and formatters based on the
134    provided configuration. If no configuration is provided, it defaults to INFO level
135    logging with a stream handler that outputs to stdout.
136
137    Args:
138        config (Optional[LogConfig], optional): Configuration object that specifies the
139            logging level, handler type, format, and other logging settings. If None,
140            defaults to INFO level with stream handler. Defaults to None.
141        instance_name (Optional[str], optional): Name identifier for the logger instance,
142            required when using file-based handlers to create unique log filenames.
143            Defaults to None.
144
145    Returns:
146        Logger: Configured logger instance ready for use.
147
148    Raises:
149        ValueError: If instance_name is None when using file-based handlers
150            (LogFileHandler, LogSizeFileRotateHandler, or LogTimeFileRotateHandler).
151
152    Note:
153        - The function clears any existing handlers before adding the new one
154        - File handlers will create the directory path if it doesn't exist
155        - Rotating file handlers use custom rotator and namer functions
156        - All handlers use UTF-8 encoding for file operations
157        - Time-based rotation uses UTC timezone
158    """
159    logger = getLogger()
160    logger.handlers.clear()
161    handler: Handler
162    if config is None:
163        logger.setLevel(level=logging.INFO)
164        handler = StreamHandler(stream=sys.stdout)
165        formatter = UtcFormatter(
166            fmt="%(asctime)s.%(msecs)03d [%(levelname)s][pid=%(process)d]: %(message)s",
167            datefmt="%Y-%m-%d %H:%M:%S",
168        )
169        handler.setFormatter(fmt=formatter)
170        handler.setLevel(level=logging.INFO)
171        logger.addHandler(hdlr=handler)
172    else:
173        logger.setLevel(level=LogMode.to_logging_level(mode=config.globalLevel))
174        for handler_config in config.handlersConfig:
175            match handler_config.handler:
176                case LogStreamHandler():
177                    handler = StreamHandler(stream=sys.stdout)
178                case LogNullHandler():
179                    handler = NullHandler()
180                case LogFileHandler():
181                    if instance_name is None:
182                        raise ValueError(
183                            "Instance name must be provided for file logging."
184                        )
185                    Path(handler_config.handler.path).mkdir(parents=True, exist_ok=True)
186                    full_path = f"{handler_config.handler.path}/{instance_name}{handler_config.handler.filenameTrailer}"
187                    handler = FileHandler(
188                        filename=full_path,
189                        mode="a" if handler_config.handler.append else "w",
190                        encoding="utf-8",
191                        delay=False,
192                        errors=None,
193                    )
194                case LogSizeFileRotateHandler():
195                    if instance_name is None:
196                        raise ValueError(
197                            "Instance name must be provided for file logging."
198                        )
199                    Path(handler_config.handler.path).mkdir(parents=True, exist_ok=True)
200                    full_path = f"{handler_config.handler.path}/{instance_name}{handler_config.handler.filenameTrailer}"
201                    handler = RotatingFileHandler(
202                        filename=full_path,
203                        maxBytes=handler_config.handler.maxBytes,
204                        backupCount=handler_config.handler.backupCount,
205                        encoding="utf-8",
206                        delay=False,
207                        errors=None,
208                    )
209                    handler.rotator = _rotator
210                    handler.namer = _namer
211                case LogTimeFileRotateHandler():
212                    if instance_name is None:
213                        raise ValueError(
214                            "Instance name must be provided for file logging."
215                        )
216                    Path(handler_config.handler.path).mkdir(parents=True, exist_ok=True)
217                    full_path = f"{handler_config.handler.path}/{instance_name}{handler_config.handler.filenameTrailer}"
218                    handler = TimedRotatingFileHandler(
219                        filename=full_path,
220                        when=LogIntervalType.to_when(
221                            interval_type=handler_config.handler.intervalType
222                        ),
223                        interval=handler_config.handler.interval,
224                        backupCount=handler_config.handler.backupCount,
225                        encoding="utf-8",
226                        delay=False,
227                        utc=True,
228                    )
229                    handler.rotator = _rotator
230                    handler.namer = _namer
231                case _:
232                    raise ValueError(
233                        f"Unknown handler type: {type(handler_config.handler)}"
234                    )
235            formatter = UtcFormatter(
236                fmt=handler_config.format, datefmt=handler_config.dateFormat
237            )
238            handler.setFormatter(fmt=formatter)
239            handler.setLevel(level=LogMode.to_logging_level(mode=handler_config.level))
240            logger.addHandler(hdlr=handler)
241    return logger

244def logger_force_write(logger: Logger, msg: str) -> None:
245    """Forces the logger to write an INFO level message, temporarily elevating all
246    handler levels if necessary.
247
248    Args:
249        logger (Logger): The logger instance to use for writing the message.
250        msg (str): The message to log at INFO level.
251
252    Notes:
253        This function temporarily sets all handlers and the logger itself to INFO level to ensure the message is logged,
254        then restores their original levels.
255    """
256    original_handlers_level: list[int] = []
257    for handler in logger.handlers:
258        original_handlers_level.append(handler.level)
259        handler.setLevel(level=logging.INFO)
260    original_level = logger.level
261    logger.setLevel(level=logging.INFO)
262    logger.info(msg=msg)
263    logger.setLevel(level=original_level)
264    for original_handler_level, handler in zip(
265        original_handlers_level, logger.handlers, strict=True
266    ):
267        handler.setLevel(level=original_handler_level)
