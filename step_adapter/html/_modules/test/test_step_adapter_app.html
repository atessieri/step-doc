403    @pytest.mark.parametrize(argnames="log_enabled", argvalues=[True, False])
404    @pytest.mark.parametrize(argnames="converted_number", argvalues=[0, 1, 3])
405    async def test__stream_receive_run(
406        self, mocker: MockerFixture, converted_number: int, log_enabled: bool
407    ) -> None:
408        """Test that every received message is converted and its arrays published.
409
410        The receive generator yields one message, then the second receive loop
411        raises to end the endless loop.
412
413        Args:
414            mocker (MockerFixture): The pytest-mock fixture.
415            converted_number (int): Number of position arrays the conversion returns.
416            log_enabled (bool): Whether the logger reports every level as enabled.
417
418        Returns:
419            None
420        """
421        app = _app(mocker=mocker, enabled=log_enabled)
422        streams = app._config.redis.streams
423        adapter_stream = _async_context(mocker=mocker, target="StepAdapterRedisStream")
424        crash_stream = _async_context(
425            mocker=mocker, target="StepCrashPredictionRedisStream"
426        )
427        adapter_client = adapter_stream.return_value.__aenter__.return_value
428        adapter_client.create_input_stream = mocker.AsyncMock()
429        crash_client = crash_stream.return_value.__aenter__.return_value
430        crash_client.publish_position_arrays = mocker.AsyncMock()
431        message = _message(kind="cam")
432        receive_calls = 0
433
434        async def receive_adapter_message() -> AsyncIterator[object]:
435            """Yield the message once, then end the loop on the next call.
436
437            Yields:
438                object: The received message.
439            """
440            nonlocal receive_calls
441            receive_calls += 1
442            if receive_calls > 1:
443                raise _StopLoopError
444            yield message
445
446        adapter_client.receive_adapter_message = receive_adapter_message
447        arrays = [PositionArrayV1.get_default() for _ in range(converted_number)]
448        convert = mocker.patch.object(
449            target=StepAdapterApp,
450            attribute="_message_to_position_arrays",
451            return_value=arrays,
452        )
453        actor_dataset = mocker.MagicMock(spec=ActorDataset)
454        perceived_dataset = mocker.MagicMock(spec=ObjectPerceivedDataset)
455        with pytest.raises(expected_exception=_StopLoopError):
456            await app._stream_receive_run(
457                config=streams,
458                pool=mocker.sentinel.pool,
459                actor_dataset=actor_dataset,
460                perceived_dataset=perceived_dataset,
461            )
462        adapter_stream.assert_called_once_with(
463            pool=mocker.sentinel.pool,
464            config=app._input_stream_params(config=streams),
465            logger=app._logger,
466        )
467        crash_stream.assert_called_once_with(
468            pool=mocker.sentinel.pool,
469            config=app._crash_prediction_stream_params(config=streams),
470            logger=app._logger,
471        )
472        adapter_client.create_input_stream.assert_awaited_once_with()
473        assert receive_calls == 2
474        convert.assert_called_once_with(
475            message=message,
476            actor_dataset=actor_dataset,
477            perceived_dataset=perceived_dataset,
478        )
479        if arrays:
480            crash_client.publish_position_arrays.assert_awaited_once_with(
481                position_arrays=arrays
482            )
483        else:
484            crash_client.publish_position_arrays.assert_not_awaited()
485        assert _logger_of(app).info.called is log_enabled
