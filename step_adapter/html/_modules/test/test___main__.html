 88async def _mock_adapter_messages(
 89    message_list: list[Any], exc: BaseException
 90) -> AsyncGenerator[Any]:
 91    """Asynchronously yields mock messages from a provided list.
 92
 93    This function is designed for testing purposes to simulate an asynchronous
 94    iterator of messages. It takes a list of MagicMock objects and yields them
 95    one by one in an async generator pattern.
 96
 97    Args:
 98        message_list (list[Any]): A list of objects representing
 99            messages to be yielded asynchronously.
100        exc (BaseException): The exception instance to be raised after all messages are yielded.
101
102    Yields:
103        Any: Individual mock message objects from the input list.
104
105    Raises:
106        Exception: If an Exception object is encountered in the list.
107
108    Returns:
109        AsyncGenerator[Any, None]: An async generator that yields objects
110        from the input list and returns None when exhausted.
111
112    Time complexity:
113        O(n), where n is the number of messages in ``message_list``.
114
115    Space complexity:
116        O(1) additional space beyond the input ``message_list``.
117    """
118    for message in message_list:
119        yield message
120    raise exc
