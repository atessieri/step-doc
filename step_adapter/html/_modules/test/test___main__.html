 41def _patch_main(
 42    mocker: MockerFixture, log_enabled: bool = True, is_windows: bool = True
 43) -> SimpleNamespace:
 44    """Patch the logger factory, AppRunner and StepAdapterApp used by ``main``.
 45
 46    Args:
 47        mocker (MockerFixture): The pytest-mock fixture.
 48        log_enabled (bool): The value returned by the logger ``isEnabledFor``.
 49        is_windows (bool): The value of the ``IS_WINDOWS`` flag seen by ``main``.
 50
 51    Returns:
 52        SimpleNamespace: The mocks: ``step_logger`` (StepLogger class), ``logger``
 53            (the logger it returns), ``app_runner_cls`` (the parametrized AppRunner
 54            class), ``app_runner`` (its instance) and ``app_cls`` (StepAdapterApp).
 55    """
 56    step_logger = mocker.patch(f"{MODULE}.StepLogger")
 57    logger = step_logger.get_logger.return_value
 58    logger.isEnabledFor.return_value = log_enabled
 59    app_runner_cls = mocker.patch(f"{MODULE}.AppRunner").__getitem__.return_value
 60    app_runner_cls.return_value.run.return_value = ExitCode.OK
 61    mocker.patch(f"{MODULE}.IS_WINDOWS", new=is_windows)
 62    return SimpleNamespace(
 63        step_logger=step_logger,
 64        logger=logger,
 65        app_runner_cls=app_runner_cls,
 66        app_runner=app_runner_cls.return_value,
 67        app_cls=mocker.patch(f"{MODULE}.StepAdapterApp"),
 68    )
