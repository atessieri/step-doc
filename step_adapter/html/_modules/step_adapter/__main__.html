  1"""STEP Adapter Main Module.
  2
  3This module serves as the main entry point for the STEP (Safer Transport for Europe Platform)
  4Adapter service, which processes communication messages and generates position arrays
  5for traffic monitoring and analysis.
  6
  7The STEP Adapter is designed to:
  8
  9    - Process CAM (Cooperative Awareness Messages) and CPM (Collective Perception Messages)
 10      from Redis streams
 11    - Generate position array messages based on the processed data
 12    - Maintain an actor dataset with temporal position tracking
 13    - Provide continuous data cleaning and memory management
 14    - Support multiple concurrent stream consumers for high-throughput scenarios
 15    - Handle graceful shutdown and error recovery mechanisms
 16
 17Key Features:
 18    - Asynchronous Redis stream processing with connection pooling
 19    - Configurable actor dataset with temporal data management
 20    - Multiple receiver client instances for load distribution
 21    - Automatic data cleanup with configurable intervals
 22    - Signal handling for graceful application shutdown
 23    - Comprehensive configuration management with JSON schema support
 24    - Cross-platform support (Windows and Unix-like systems)
 25
 26Command Line Interface:
 27    The module can be executed directly and supports the following operations:
 28
 29        - Generate configuration JSON schema files
 30        - Create configuration template files
 31        - Load configuration from file and run the adapter service
 32        - Display version information
 33
 34Usage:
 35    python -m step_adapter [options]
 36
 37    Options:
 38        -v, --version           Show program version
 39        -s, --conf_schema      Generate configuration JSON schema
 40        -t, --conf_template    Generate configuration template
 41        -c, --config          Load configuration from file
 42        -i, --instance        Specify instance name (required with --config)
 43
 44Architecture:
 45    The service uses an event-driven architecture with asyncio for concurrent processing:
 46
 47    #. Main event loop with exception handling
 48    #. Redis connection pool for stream operations
 49    #. Multiple stream receiver tasks for parallel message processing
 50    #. Actor dataset for temporal position data management
 51    #. Background cleaner task for memory management
 52    #. Automatic reconnection on failures
 53"""

191def _exception_handler(context: dict[Any, Any], logger: Logger) -> None:
192    """Handle exceptions in asyncio event loop.
193
194    This function serves as an exception handler for the asyncio event loop,
195    logging any uncaught exceptions that occur during loop execution.
196
197    Args:
198        context (dict[Any, Any]): A dictionary containing contextual information about the exception.
199            Must include an 'exception' key with the actual exception object.
200        logger (Logger): Logger instance to record the error message.
201
202    Returns:
203        None
204
205    Example:
206        >>> loop = asyncio.get_running_loop()
207        ... loop.set_exception_handler(lambda ctx: _exception_handler(ctx, logger))
208
209    Time complexity:
210        O(1): A single ``isEnabledFor`` check and, at most, one log call.
211
212    Space complexity:
213        O(1): No additional data structures are allocated.
214    """
215    if logger.isEnabledFor(level=logging.ERROR):
216        logger.error(msg="Exception in loop", exc_info=context.get("exception"))

219async def _stream_receive_run(
220    instance_name: str,
221    config: RedisStreams,
222    logger: Logger,
223    pool: StepRedisPool,
224    actor_dataset: ActorDataset,
225) -> None:
226    """Asynchronously processes Redis stream messages for a STEP Adapter instance.
227
228    This function establishes a connection to a Redis stream and continuously listens for
229    incoming messages. It processes CAM (Cooperative Awareness Message) data and publishes
230    position arrays when applicable.
231
232    Args:
233        instance_name (str): The unique identifier for this stream consumer instance.
234        config (RedisStreams): Configuration object containing Redis stream settings.
235        logger (Logger): Logger instance for recording operation events and debugging.
236        pool (StepRedisPool): Redis connection pool for managing connections.
237        actor_dataset (ActorDataset): Dataset handler for processing and storing actor data.
238
239    Returns:
240        None
241
242    Note:
243        This function runs in an infinite loop and should be used with proper exception
244        handling and cancellation mechanisms in the calling context.
245
246    Time complexity:
247        O(1) amortized per received message: each iteration performs a constant
248        number of dict/stream operations (message dispatch, optional publish) with
249        no data-structure growth. The function itself has unbounded total runtime
250        because the outer ``while True`` loop never terminates on its own.
251
252    Space complexity:
253        O(1) beyond the two async context managers (``adapter_redis_stream`` and
254        ``crash_prediction_redis_stream``), which hold a constant amount of
255        connection state independent of the number of messages processed.
256    """
257    async with (
258        StepAdapterRedisStream(
259            pool=pool,
260            config=StepRedisStreamParams(
261                stream_id=config.input.streamId,
262                group_id=config.input.groupId,
263                wait_timeout_ms=config.input.waitTimeout,
264                consumer_name=instance_name,
265                ttl_seconds=(config.input.waitTimeout * 2) // ItsTime.SEC_TO_MS_FACTOR,
266            ),
267            logger=logger,
268        ) as adapter_redis_stream,
269        StepCrashPredictionRedisStream(
270            pool=pool,
271            config=StepRedisStreamParams(
272                stream_id=config.outputs.crashPrediction.streamId,
273                max_length=config.outputs.crashPrediction.maxLength,
274                consumer_name=instance_name,
275            ),
276            logger=logger,
277        ) as crash_prediction_redis_stream,
278    ):
279        if logger.isEnabledFor(level=logging.INFO):
280            logger.info(
281                msg=f"Listening for messages from Redis stream instance name: {instance_name}"
282            )
283        await adapter_redis_stream.create_input_stream()
284        while True:
285            async for message in adapter_redis_stream.receive_adapter_message():
286                if logger.isEnabledFor(level=logging.DEBUG):
287                    logger.debug(msg=f"Redis message: {message}")
288                match message.root:
289                    case AdapterCloudEventsCamDataV1():
290                        position_array = actor_dataset.add_cam_message(
291                            cam=message.root.data
292                        )
293                        if position_array:
294                            await crash_prediction_redis_stream.publish_position_array(
295                                position_array=position_array
296                            )
297                    case _:
298                        if logger.isEnabledFor(level=logging.WARNING):
299                            logger.warning(
300                                msg=f"Unknown message type: {type(message.root)}"
301                            )

304async def _cleaner_run(
305    actor_dataset: ActorDataset, cleaner_interval_seconds: int, logger: Logger
306) -> None:
307    """Continuously runs a dataset cleaner task that removes old data points at regular
308    intervals.
309
310    This coroutine function implements a background cleaning service that periodically
311    removes outdated entries from an actor dataset to prevent memory buildup and
312    maintain optimal performance.
313
314    Args:
315        actor_dataset (ActorDataset): The dataset instance to be cleaned of old data points.
316        cleaner_interval_seconds (int): The time interval in seconds between consecutive
317            cleaning operations.
318        logger (Logger): Logger instance for recording cleaner activity and debug information.
319
320    Returns:
321        None
322
323    Note:
324        This function is designed to run as a background task and will continue executing
325        until the program terminates or an exception occurs. The cleaning operation is
326        performed by calling the clear_old_points() method on the provided dataset.
327
328    Time complexity:
329        Unbounded total runtime since the ``while True`` loop never terminates on its
330        own. Per iteration: O(1) for the ``asyncio_sleep`` call plus the cost of
331        ``ActorDataset.clear_old_points()``, which has its own documented complexity
332        (see ``ActorDataset.clear_old_points`` in actor_dataset.py).
333
334    Space complexity:
335        O(1) for the loop itself; no additional data structures are allocated beyond

395async def _main_async(
396    instance_name: str, config: StepAdapterConfig, logger: Logger
397) -> None:
398    """Main async function that orchestrates the STEP Adapter service.
399
400    This function sets up the main event loop with exception handling, establishes
401    Redis connection pool, and manages concurrent tasks for stream processing and
402    data cleanup. It runs indefinitely, automatically reconnecting on failures.
403
404    The function creates:
405
406        - Redis connection pool for stream operations
407        - ActorDataset for managing point path data
408        - Multiple stream receiver tasks based on receiveClientNumber config
409        - A cleaner task to maintain data hygiene at regular intervals
410
411    Args:
412        instance_name (str): Unique identifier for this service instance
413        config (StepAdapterConfig): Configuration object containing Redis settings,
414            client numbers, path parameters, and timing intervals
415        logger (Logger): Logger instance for recording service events and errors
416
417    Returns:
418        None
419
420    Time complexity:
421        O(c) where c = ``config.receiveClientNumber``: the setup path spawns one
422        ``_stream_receive_run`` task per receive client plus one ``_cleaner_run``
423        task, each an O(1) task-creation call. The outer reconnect ``while True``
424        loop runs indefinitely, so the function as a whole has unbounded total
425        runtime.
426
427    Space complexity:
428        O(c) where c = ``config.receiveClientNumber``: one concurrent task and its
429        associated state is held per receive client, plus O(1) for the cleaner task,
430        the Redis connection pool, and the ActorDataset instance (whose own space
431        complexity is documented separately in actor_dataset.py).
432    """
433    get_running_loop().set_exception_handler(
434        handler=lambda _, context: _exception_handler(context=context, logger=logger)
435    )
436    if logger.isEnabledFor(level=logging.INFO):
437        logger.info(msg="Program started")
438    while True:
439        try:
440            async with StepRedisPool(
441                params=_redis_params_convert(
442                    config=config.redis.connection,
443                    max_connections=config.receiveClientNumber + 1,
444                ),
445                logger=logger,
446            ) as redis_pool:
447                actor_dataset = ActorDataset(
448                    config=ActorDatasetConfig(
449                        input_max_point_number=config.maxActorHistoryPointNumber,
450                        input_max_period_ms=config.inputMaxPeriod,
451                        input_max_local_time_deviation_ms=config.inputLocalTimeMaxDeviation
452                        * ItsTime.SEC_TO_MS_FACTOR,
453                        input_oldest_allowed_time_ms=config.inputOldestAllowedTime
454                        * ItsTime.SEC_TO_MS_FACTOR,
455                        output_nominal_sample_frequency=config.outputNominalSampleFrequency,
456                        output_period_max_deviation_ms=config.outputPeriodMaxDeviation,
457                    ),
458                    logger=logger,
459                )
460                async with TaskGroup() as group:
461                    # Create stream receive thread
462                    for instance_number in range(config.receiveClientNumber):
463                        group.create_task(
464                            coro=_stream_receive_run(
465                                instance_name=instance_name
466                                + "-"
467                                + str(object=instance_number),
468                                config=config.redis.streams,
469                                logger=logger,
470                                pool=redis_pool,
471                                actor_dataset=actor_dataset,
472                            )
473                        )
474                    # Create cleaner task
475                    group.create_task(
476                        coro=_cleaner_run(
477                            actor_dataset=actor_dataset,
478                            cleaner_interval_seconds=config.cleanerInterval,
479                            logger=logger,
480                        )
481                    )
482        except Exception:
483            if logger.isEnabledFor(level=logging.ERROR):
484                logger.exception(msg="Exception raised")
485            await asyncio_sleep(delay=config.reconnectInterval)
486            if logger.isEnabledFor(level=logging.INFO):
487                logger.info(msg=f"Connection retry after {config.reconnectInterval}")

566def main() -> int:  # noqa: C901, PLR0912
567    """Main entry point for the application.
568
569    This function initializes the application, processes command-line arguments,
570    loads configuration, sets up logging, and executes the main asynchronous event loop.
571
572    It handles the following operations:
573
574        - Printing help when no arguments are provided
575        - Generating JSON schema file when requested
576        - Creating configuration template file when requested
577        - Loading configuration from specified file
578        - Setting up Windows event loop policy if running on Windows
579        - Running the main asynchronous function
580
581    Exception handling:
582
583        - Catches interruptions (Ctrl+C) gracefully
584        - Logs unhandled exceptions
585
586    Returns:
587        int: Exit code indicating program result status:
588
589            - ExitCode.OK (0): Success
590            - ExitCode.USAGE (1): Improper usage
591            - ExitCode.ABORTED (3): Program was interrupted
592            - ExitCode.ERROR (2): An unhandled exception occurred
593
594    Time complexity:
595        O(1) for signal registration, argument parsing (``_args_parse``), and
596        configuration loading/validation of a fixed-schema config file, plus the
597        delegated costs of ``_get_config_schema_file`` (O(s), schema size) and
598        ``_get_config_template_file`` (O(1)) when those options are requested. When
599        both ``--config`` and ``--instance`` are supplied, the function additionally
600        runs ``_main_async`` via the asyncio runner, which blocks for the lifetime
601        of the process (unbounded runtime) since its reconnect loop never returns
602        on its own.
603
604    Space complexity:
605        O(1) for the synchronous setup portion (parser, parsed args, logger), plus
606        the delegated space costs of ``_get_config_schema_file`` (O(s), schema size)
607        and ``_get_config_template_file`` (O(1)) when requested, plus the space used
608        by ``_main_async`` while the service is running (documented separately).
609    """
610    if IS_WINDOWS:
611        # Windows: Handle CTRL_BREAK_EVENT
612        sigbreak = getattr(signal, "SIGBREAK", None)
613        if sigbreak is not None:
614            signal.signal(signalnum=sigbreak, handler=_signal_handler)
615    else:
616        # Linux/Unix: Handle SIGINT (in case default behavior is overridden)
617        signal.signal(signal.SIGINT, _signal_handler)
618        # Linux/Unix: Handle SIGTERM (sent by container runtimes on stop)
619        signal.signal(signal.SIGTERM, _signal_handler)
620    result = ExitCode.OK
621    logger = StepLogger.get_logger(name=None, config=None, instance_name=None)
622    try:
623        parser = _args_parse()
624        if len(sys.argv) == 1:
625            parser.print_help()
626            return ExitCode.USAGE
627        args = parser.parse_args()
628        if args.conf_schema_file is not None:
629            _get_config_schema_file(path=Path(args.conf_schema_file))
630        if args.conf_template_file is not None:
631            _get_config_template_file(path=Path(args.conf_template_file))
632        if (args.config_file is not None) and (args.instance_name is None):
633            raise StepCliError(  # noqa: TRY301
634                "The -i/--instance option must be specified when using -c/--config."
635            )
636        if (args.config_file is None) and (args.instance_name is not None):
637            raise StepCliError(  # noqa: TRY301
638                "The -c/--config option must be specified when using -i/--instance."
639            )
640        if (args.config_file is not None) and (args.instance_name is not None):
641            config = StepAdapterConfig.model_validate_json(
642                json_data=Path(args.config_file).read_text(encoding="utf-8")
643            )
644            logger = StepLogger.get_logger(
645                name=None, config=config.log, instance_name=args.instance_name
646            )
647            StepLogger.logger_force_write(
648                logger=logger,
649                msg=f"STEP Adapter: name {args.instance_name} version {step_adapter.__version__}",
650            )
651            StepLogger.logger_force_write(logger=logger, msg="Library:")
652            StepLogger.logger_force_write(
653                logger=logger,
654                msg=f"    - step_cloud_schemas: {step_cloud_schemas.__version__}",
655            )
656            StepLogger.logger_force_write(
657                logger=logger, msg=f"    - step_redis: {step_redis.__version__}"
658            )
659            loop_factory = (
660                asyncio.SelectorEventLoop
661                if (sys.platform == "win32") or (os.name.lower() == "nt")
662                else None
663            )
664            with Runner(loop_factory=loop_factory) as runner:
665                runner.run(
666                    coro=_main_async(
667                        instance_name=args.instance_name, config=config, logger=logger
668                    )
669                )
670    except KeyboardInterrupt:
671        if logger.isEnabledFor(level=logging.INFO):

674    except Exception:
675        if logger.isEnabledFor(level=logging.ERROR):
676            logger.exception(msg="Exception unmanaged")
677        result = ExitCode.ERROR
