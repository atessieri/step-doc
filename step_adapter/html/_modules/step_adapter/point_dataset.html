   1"""Position dataset management with cubic spline interpolation.
   2
   3Processes raw, irregularly-timed position measurements (e.g. GNSS fixes from CAM
   4messages) into smooth, regularly-sampled trajectories, using cubic spline
   5interpolation between real measurements.
   6
   7Two components:
   8    - ``PointDatasetConfig``: immutable, validated processing parameters (queue
   9      size bounds, period/deviation limits, output frequency).
  10    - ``PointDataset``: per-actor dataset manager. Validates each incoming
  11      position for temporal/spatial coherence, then either appends it directly
  12      (consecutive positions close to the nominal output period) or generates
  13      interpolated positions to fill the gap (larger temporal gaps). Large gaps
  14      beyond ``input_max_period_ms`` clear the history instead of interpolating
  15      across a likely discontinuity.
  16
  17Complexity is documented per-method; broadly, ``add_point`` is O(1) for the common
  18case (no interpolation needed) and O(n + k) when interpolating, where n is the
  19input queue size and k is the number of generated output points - both bounded by
  20configuration.
  21
  22Thread Safety:
  23    This module is **NOT thread-safe**. Concurrent access requires external
  24    synchronization (e.g. one ``PointDataset`` per thread per actor, or a lock
  25    around a shared instance).
  26
  27Note:
  28    Large temporal gaps exceeding ``input_max_period_ms`` clear history rather than
  29    interpolate across a likely discontinuity, which may produce temporary output
  30    gaps; downstream consumers needing continuous output should buffer accordingly.
  31"""

 235class PointDataset:
 236    """Per-actor position dataset with cubic spline interpolation.
 237
 238    Maintains a bounded history of recent position measurements for a single actor
 239    and, on each ``add_point()`` call, either appends the new position directly
 240    (consecutive positions close to the nominal output period) or fills the gap
 241    with cubic-spline-interpolated positions (larger gaps). Large temporal gaps or
 242    coherence failures trigger a history reset instead of interpolating across a
 243    likely discontinuity.
 244
 245    Attributes:
 246        actor_par (ActorId): The actor identifier for this dataset instance.
 247        config (PointDatasetConfig): Configuration object containing dataset parameters
 248            including input_max_point_number for queue and array sizing.
 249        logger (Logger): Logger instance for recording events and debugging information.
 250
 251    Complexity is documented per-method; see ``add_point`` and
 252    ``_create_positions_list`` for the two execution paths.
 253
 254    Thread Safety:
 255        This class is **NOT thread-safe**. Concurrent access from multiple threads
 256        requires external synchronization (e.g. one instance per thread per actor).
 257
 258    Example:
 259        >>> config = PointDatasetConfig(
 260        ...     input_max_point_number=20,
 261        ...     input_max_period_ms=2000,
 262        ...     input_max_local_time_deviation_ms=5000,
 263        ...     output_nominal_period_ms=200,  # 5 Hz
 264        ...     output_max_period_deviation_ms=20,
 265        ... )

 483    def _find_optimal_period(
 484        self, actor: ActorId, time_last_ms: int, time_new_ms: int
 485    ) -> tuple[int, int, bool]:
 486        """Find the optimal output period and interval count for interpolation.
 487
 488        This method computes the best possible output period (in ms) and number of intervals
 489        between two timestamps to achieve evenly-spaced interpolated points. The algorithm
 490        attempts to match the configured nominal output period while respecting the allowed
 491        deviation constraints.
 492
 493        Algorithm Overview:
 494            #. Define the valid period range: [nominal - deviation, nominal + deviation]
 495            #. Calculate the time difference between the two input timestamps
 496            #. Determine the valid range of interval counts that satisfy period constraints
 497            #. If a valid solution exists:
 498
 499                - Choose the interval count closest to the ideal (diff_time / nominal_period)
 500                - Calculate the period using integer division: diff_time // interval_count
 501
 502            #. If no exact solution exists (constraint conflict):
 503
 504                - Select the boundary interval count closest to ideal
 505                - Calculate the period using integer division: diff_time // interval_count
 506                - Mark as non-ideal solution
 507
 508        The algorithm guarantees that:
 509
 510            - All generated timestamps fall within [time_last_ms, time_new_ms]
 511            - The spacing between consecutive points respects the configured deviation
 512            - The total number of intervals optimally balances the nominal period target
 513            - Period calculation uses integer division to ensure whole millisecond values
 514
 515        Time Complexity: O(1) - All operations are constant time arithmetic
 516
 517        Args:
 518            actor (ActorId): The actor identifier for logging/debugging purposes.
 519            time_last_ms (int): Timestamp of the last recorded point in milliseconds.
 520            time_new_ms (int): Timestamp of the new incoming point in milliseconds.
 521
 522        Returns:
 523            tuple[int, int, bool]:
 524
 525            - period_optimal (int): The chosen output period in milliseconds (integer division result).
 526            - interval_number_optimal (int): The number of intervals between points.
 527            - period_ideal (bool): True if an exact solution within constraints was found,
 528              False if an approximation at the boundary was necessary.
 529
 530        Example:
 531            With nominal_period=1000ms, max_deviation=50ms:
 532
 533                - Valid period range: [950ms, 1050ms]
 534                - For diff_time=3000ms: Returns (1000, 3, True) - exact solution
 535                - For diff_time=2800ms: Returns (933, 3, False) - approximation needed
 536        """
 537        output_period_min = (
 538            self._config.output_nominal_period_ms
 539            - self._config.output_max_period_deviation_ms
 540        )
 541        output_period_max = (
 542            self._config.output_nominal_period_ms
 543            + self._config.output_max_period_deviation_ms
 544        )
 545        diff_time = time_new_ms - time_last_ms
 546        interval_number_ideal = diff_time / self._config.output_nominal_period_ms
 547        # Valid range for n: ceil(d/pt_max) <= n <= floor(d/pt_min)
 548        interval_number_min = ceil(diff_time / output_period_max)
 549        interval_number_max = floor(diff_time / output_period_min)
 550        # Solution exists with exact diff_time
 551        if interval_number_min <= interval_number_max:
 552            # Clamp to valid range
 553            interval_number_optimal = max(
 554                interval_number_min,
 555                min(interval_number_max, round(number=interval_number_ideal)),
 556            )
 557            period_optimal = diff_time // interval_number_optimal
 558            period_ideal = True
 559        # No exact solution - choose closest interval number to ideal
 560        elif abs(interval_number_max - interval_number_ideal) <= abs(
 561            interval_number_min - interval_number_ideal
 562        ):
 563            # Use interval number max
 564            interval_number_optimal = interval_number_max
 565            period_optimal = diff_time // interval_number_optimal
 566            period_ideal = False
 567        else:
 568            # Use interval number min
 569            interval_number_optimal = interval_number_min
 570            period_optimal = diff_time // interval_number_optimal
 571            period_ideal = False
 572        if self._logger.isEnabledFor(level=logging.DEBUG):
 573            self._logger.debug(
 574                msg=(
 575                    f"{actor}: period {'ideal' if period_ideal else 'optimal'}: "
 576                    f"period={period_optimal} ms, intervals={interval_number_optimal} "
 577                    f"for time diff time={diff_time} ms => range {time_last_ms} - {time_new_ms} ms, "
 578                    f"error={abs(period_optimal - self._config.output_nominal_period_ms)} ms"
 579                )
 580            )
 581        return (period_optimal, interval_number_optimal, period_ideal)

 583    def _create_interpolated_timestamp_array(
 584        self, actor: ActorId, new_position: PositionInfoRecord
 585    ) -> np.typing.NDArray[np.uint64]:
 586        """Creates an array of interpolated timestamps between the last queued position
 587        and a new position.
 588
 589        This method generates timestamps at regular intervals based on an optimal period calculated
 590        from the time difference between positions. If an ideal period cannot be found, it falls back
 591        to a non-optimal period calculation with constraints based on configured maximum deviations.
 592
 593        Time Complexity:
 594            O(k) where k is the number of intervals between the last and new position timestamps.
 595            The dominant operation is np.arange() which generates k timestamps.
 596
 597        Space Complexity:
 598            O(k) where k is the number of generated timestamps. The array size depends on the
 599            time difference and the optimal period calculated.
 600
 601        Args:
 602            actor (ActorId): The identifier of the actor for which timestamps are being generated.
 603            new_position (PositionInfoRecord): The new position record containing the target timestamp.
 604
 605        Returns:
 606            np.typing.NDArray[np.uint64]: Array of timestamps in milliseconds (epoch time) between
 607            the last position and the new position, excluding the new position timestamp itself.
 608
 609        Notes:
 610            - When ideal period is found, timestamps are generated from last_position + period
 611              to last_position + (period * interval_count) not included
 612            - When ideal period is not found (no exact multiple of output_nominal_period_ms
 613              fits within the deviation tolerance), the algorithm:
 614
 615                - Calculates the period using integer division: diff_time // interval_count
 616                - Works backwards from new_position_time to determine the first interpolated point
 617                - Ensures the first point is >= min_allowed (last_position + nominal - deviation)
 618                - May generate fewer interpolated points than the ideal case to maintain
 619                  period constraints
 620                - In extreme cases, may generate no interpolated points, outputting only
 621                  the new position
 622
 623            - Warning logs are generated when non-ideal periods are used, showing the actual
 624              time range and number of generated points
 625        """
 626        last_position_time = self._input_points_queue[-1].time
 627        new_position_time = new_position.time
 628        (period, interval_number, ideal_period_found) = self._find_optimal_period(
 629            actor=actor,
 630            time_last_ms=last_position_time.epoch_ms,
 631            time_new_ms=new_position_time.epoch_ms,
 632        )
 633        if ideal_period_found:
 634            # Create timestamp array with ideal period without the last point
 635            # substituted by new position time
 636            timestamp_ms_array = np.arange(
 637                int(last_position_time.epoch_ms + period),
 638                int(last_position_time.epoch_ms + (period * interval_number)),
 639                int(period),
 640                dtype=np.uint64,
 641            )
 642        else:
 643            # Create timestamp array with optimal period
 644            new_position_time_ms = new_position_time.epoch_ms
 645            # Minimum allowed first point
 646            min_allowed = (
 647                last_position_time.epoch_ms
 648                + self._config.output_nominal_period_ms
 649                - self._config.output_max_period_deviation_ms
 650            )
 651            # First point is new_point minus max_intervals * p
 652            first_point_ms = new_position_time_ms - (
 653                ((new_position_time_ms - min_allowed) // period) * period
 654            )
 655            # Generate timestamps from first_point to last_point with period p
 656            timestamp_ms_array = np.arange(
 657                int(first_point_ms),
 658                int(new_position_time_ms),
 659                int(period),
 660                dtype=np.uint64,
 661            )
 662            if self._logger.isEnabledFor(level=logging.WARNING):
 663                if timestamp_ms_array.size > 0:
 664                    self._logger.warning(
 665                        msg=(
 666                            f"{actor}: no ideal period => {period}, range {last_position_time} - {new_position_time} "
 667                            f"=> {ItsTime.create_from_epoch_ms(epoch_ms=timestamp_ms_array[0])} - {new_position_time} "
 668                            f"with {timestamp_ms_array.size} points without new point"
 669                        )
 670                    )
 671                else:
 672                    self._logger.warning(
 673                        msg=(
 674                            f"{actor}: no ideal period => {period}, range {last_position_time} - {new_position_time} "
 675                            f"=> only new point"
 676                        )
 677                    )
 678        return timestamp_ms_array

 680    def _create_interpolated_arrays(
 681        self, actor: ActorId, new_position: PositionInfoRecord
 682    ) -> tuple[
 683        np.typing.NDArray[np.uint64],
 684        np.typing.NDArray[np.float64],
 685        np.typing.NDArray[np.float64],
 686        np.typing.NDArray[np.int64],
 687    ]:
 688        """Interpolate latitude, longitude, and (if available) altitude via cubic spline.
 689
 690        Populates the pre-allocated input arrays from the queue plus ``new_position``,
 691        then fits a single vector-valued ``CubicSpline`` over (latitude, longitude,
 692        [altitude]) - sharing one tridiagonal solve across columns instead of
 693        building 2-3 independent splines - and evaluates it at the timestamps from
 694        ``_create_interpolated_timestamp_array``.
 695
 696        Time complexity:
 697            O(n + k*log(n)): O(n) array population and spline construction, O(k*log(n))
 698            evaluation for k output points, where n is the input queue size + 1.
 699
 700        Space complexity:
 701            O(n + k): input arrays are reused pre-allocated buffers; O(k) for the
 702            returned output arrays.
 703
 704        Args:
 705            actor (ActorId): The actor identifier.
 706            new_position (PositionInfoRecord): The new position to include in interpolation.
 707
 708        Returns:
 709            tuple[NDArray[uint64], NDArray[float64], NDArray[float64], NDArray[int64]]:
 710            Interpolated (timestamps_ms, latitudes, longitudes, altitudes), each of
 711            length k, monotonically increasing in time. Altitudes are filled with
 712            ``Altitude.VALUE_INVALID`` when any input point (or ``new_position``)
 713            lacks altitude.
 714
 715        Raises:
 716            ValueError: If the input queue is empty.
 717        """
 718        queue_len = len(self._input_points_queue)
 719        if queue_len == 0:
 720            raise ValueError(
 721                f"{self._actor}: insufficient points to create interpolators, the queue is empty"
 722            )
 723        np_points_len = min(self._np_input_points_timestamp_ms.size, queue_len + 1)
 724        # Single vectorized pass through queue to populate arrays directly
 725        altitude_not_none = True
 726        for i, position_info_record in enumerate(self._input_points_queue):
 727            self._np_input_points_timestamp_ms[i] = position_info_record.time.epoch_ms
 728            self._np_input_points_latitude[i] = (
 729                position_info_record.position.position.latitude
 730            )
 731            self._np_input_points_longitude[i] = (
 732                position_info_record.position.position.longitude
 733            )
 734            if altitude_not_none:
 735                if position_info_record.position.position.altitude:
 736                    self._np_input_points_altitude[i] = (
 737                        position_info_record.position.position.altitude.value
 738                    )
 739                else:
 740                    altitude_not_none = False
 741        self._np_input_points_timestamp_ms[np_points_len - 1] = (
 742            new_position.time.epoch_ms
 743        )
 744        self._np_input_points_latitude[np_points_len - 1] = (
 745            new_position.position.position.latitude
 746        )
 747        self._np_input_points_longitude[np_points_len - 1] = (
 748            new_position.position.position.longitude
 749        )
 750        if altitude_not_none:
 751            if new_position.position.position.altitude:
 752                self._np_input_points_altitude[np_points_len - 1] = (
 753                    new_position.position.position.altitude.value
 754                )
 755            else:
 756                altitude_not_none = False
 757        # Lat/lon (and altitude, when available) share the same x (timestamps), so a
 758        # single vector-valued CubicSpline solves one tridiagonal system for all
 759        # columns at once instead of building 2-3 independent spline objects.
 760        column_count = 3 if altitude_not_none else 2
 761        combined_y = np.empty(shape=(np_points_len, column_count), dtype=np.float64)
 762        combined_y[:, 0] = self._np_input_points_latitude[:np_points_len]
 763        combined_y[:, 1] = self._np_input_points_longitude[:np_points_len]
 764        if altitude_not_none:
 765            combined_y[:, 2] = self._np_input_points_altitude[:np_points_len]
 766        interpolator = CubicSpline(
 767            x=self._np_input_points_timestamp_ms[:np_points_len], y=combined_y
 768        )
 769        interpolation_time_ms = self._create_interpolated_timestamp_array(
 770            actor=actor, new_position=new_position
 771        )
 772        interpolated = interpolator(x=interpolation_time_ms, extrapolate=False)
 773        return (
 774            interpolation_time_ms,
 775            interpolated[:, 0],
 776            interpolated[:, 1],
 777            (
 778                interpolated[:, 2].astype(np.int64)
 779                if altitude_not_none
 780                else np.full(
 781                    shape=interpolation_time_ms.shape,
 782                    fill_value=Altitude.VALUE_INVALID,
 783                    dtype=np.int64,
 784                )
 785            ),
 786        )

 789    def _speed_heading_from_motion(
 790        distance_m: float, heading_deg: float, time_diff_ms: int
 791    ) -> tuple[Optional[Speed], Optional[Heading]]:
 792        """Compute speed and heading from a distance/heading pair and a time delta.
 793
 794        Pure helper shared by the fast path and the interpolation path: both already
 795        have a (distance, heading) pair on hand (from a coherence check or from the
 796        batched interpolation-chain geodesic call) and only need the motion math
 797        applied, without any further ``Geocode`` call.
 798
 799        Args:
 800            distance_m (float): Geodesic distance to the previous point, in meters.
 801            heading_deg (float): Forward azimuth from the previous point, in degrees.
 802            time_diff_ms (int): Time elapsed since the previous point, in milliseconds.
 803
 804        Time complexity:
 805            O(1).
 806
 807        Space complexity:
 808            O(1).
 809
 810        Returns:
 811            tuple[Optional[Speed], Optional[Heading]]: ``(None, None)`` when
 812            ``time_diff_ms`` is not strictly positive, otherwise the computed speed
 813            (0.01 m/s units, clamped to ``Speed.VALUE_MAX`` by its validator) and
 814            heading (0.1 degree units).
 815        """
 816        if time_diff_ms <= 0:
 817            return None, None
 818        speed_obj = Speed(
 819            value=round(
 820                (distance_m * Speed.SCALE_FACTOR * ItsTime.SEC_TO_MS_FACTOR)
 821                / time_diff_ms
 822            ),
 823            confidence=None,
 824        )
 825        heading_obj = Heading(
 826            value=round(heading_deg * Heading.SCALE_FACTOR), confidence=None
 827        )
 828        return speed_obj, heading_obj

 944    @staticmethod
 945    def _compute_chain_motion(
 946        new_position: PositionInfoRecord,
 947        last_queued: PositionInfoRecord,
 948        times_ms: np.typing.NDArray[np.uint64],
 949        latitudes: np.typing.NDArray[np.float64],
 950        longitudes: np.typing.NDArray[np.float64],
 951    ) -> ChainMotion:
 952        """Compute geodesic distance/heading and derived speed/heading for one chain.
 953
 954        Issues a single vectorized geodesic call over the whole chain (last queued
 955        point -> interpolated point 0 -> ... -> interpolated point k-1 -> new real
 956        position, i.e. k+1 pairs) instead of one scalar ``Geocode`` call per pair,
 957        then derives per-pair time deltas and scaled speed/heading values.
 958
 959        Args:
 960            new_position (PositionInfoRecord): The newly arrived real position,
 961                the chain's last point.
 962            last_queued (PositionInfoRecord): The last position already in the
 963                queue, the chain's origin.
 964            times_ms (np.typing.NDArray[np.uint64]): k interpolated timestamps (ms epoch).
 965            latitudes (np.typing.NDArray[np.float64]): k interpolated latitudes.
 966            longitudes (np.typing.NDArray[np.float64]): k interpolated longitudes.
 967
 968        Time complexity:
 969            O(k): one vectorized geodesic call over k+1 pairs plus O(k) numpy math
 970            (down from O(k) scalar geodesic calls).
 971
 972        Space complexity:
 973            O(k).
 974
 975        Returns:
 976            ChainMotion: The k+1-row distance/heading/time-diff/speed/heading
 977            arrays for the chain; row k is the motion into the trailing real
 978            position.
 979        """
 980        k = int(times_ms.size)
 981        chain = np.empty(shape=k + 1, dtype=Geocode.SPHERICAL_POINT_DTYPE)
 982        chain["latitude"][:k] = latitudes
 983        chain["longitude"][:k] = longitudes
 984        chain["latitude"][k] = new_position.position.position.latitude
 985        chain["longitude"][k] = new_position.position.position.longitude
 986        previous_point = np.void(
 987            (
 988                last_queued.position.position.latitude,
 989                last_queued.position.position.longitude,
 990            ),
 991            dtype=Geocode.SPHERICAL_POINT_DTYPE,
 992        )
 993        distance_heading = Geocode.get_distances_and_headings_vector_numpy(
 994            points=chain,
 995            previous_point=previous_point,
 996            dist_unit=DistanceUnit.METERS,
 997            angle_unit=AngleUnit.DEGREES,
 998        )
 999        chain_times_ms = np.empty(shape=k + 2, dtype=np.int64)
1000        chain_times_ms[0] = last_queued.time.epoch_ms
1001        chain_times_ms[1 : k + 1] = times_ms.astype(np.int64)
1002        chain_times_ms[k + 1] = new_position.time.epoch_ms
1003        time_diff_ms = np.diff(chain_times_ms)
1004        has_motion = time_diff_ms > 0
1005        speed_values = np.zeros(shape=k + 1, dtype=np.int64)
1006        heading_values = np.zeros(shape=k + 1, dtype=np.int64)
1007        speed_values[has_motion] = np.minimum(
1008            np.round(
1009                (distance_heading["distance"][has_motion] * Speed.SCALE_FACTOR)
1010                * ItsTime.SEC_TO_MS_FACTOR
1011                / time_diff_ms[has_motion]
1012            ),
1013            Speed.VALUE_MAX,
1014        ).astype(np.int64)
1015        heading_values[has_motion] = (
1016            np.round(distance_heading["heading"][has_motion] * Heading.SCALE_FACTOR)
1017            % (Heading.VALUE_MAX - Heading.VALUE_MIN + 1)
1018        ).astype(np.int64)
1019        return ChainMotion(
1020            distance_heading=distance_heading,
1021            time_diff_ms=time_diff_ms,
1022            has_motion=has_motion,
1023            speed_values=speed_values,
1024            heading_values=heading_values,
1025        )

1027    def _build_interpolated_positions(
1028        self,
1029        new_position: PositionInfoRecord,
1030        last_queued: PositionInfoRecord,
1031        times_ms: np.typing.NDArray[np.uint64],
1032        latitudes: np.typing.NDArray[np.float64],
1033        longitudes: np.typing.NDArray[np.float64],
1034        altitudes: np.typing.NDArray[np.int64],
1035    ) -> list[PositionInfo]:
1036        """Build the interpolated trajectory plus the trailing real position.
1037
1038        Delegates the geodesic/motion math to ``_compute_chain_motion`` (a single
1039        vectorized call over the whole chain, rather than one scalar ``Geocode``
1040        call per point), then constructs one ``PositionInfo`` per interpolated
1041        point plus the trailing real position. Interpolated ``PositionInfo``/
1042        ``ReferencePosition``/``Speed``/``Heading``/``Altitude`` objects are built
1043        via ``model_construct`` (skipping pydantic validation) since their values
1044        are computed internally; the range clamping/wrapping each validator would
1045        have applied is replicated manually via ``_clamp_latitude``,
1046        ``_wrap_longitude`` and ``_clamp_altitude_value``.
1047
1048        Args:
1049            new_position (PositionInfoRecord): The newly arrived real position, appended
1050                as the final element of the returned list.
1051            last_queued (PositionInfoRecord): The last position already in the queue,
1052                used as the origin of the geodesic chain.
1053            times_ms (np.typing.NDArray[np.uint64]): k interpolated timestamps (ms epoch).
1054            latitudes (np.typing.NDArray[np.float64]): k interpolated latitudes.
1055            longitudes (np.typing.NDArray[np.float64]): k interpolated longitudes.
1056            altitudes (np.typing.NDArray[np.int64]): k interpolated altitudes, or
1057                ``Altitude.VALUE_INVALID`` where altitude is unavailable.
1058
1059        Time complexity:
1060            O(k): see ``_compute_chain_motion``, plus O(k) object construction
1061            (down from O(k) validated object construction).
1062
1063        Space complexity:
1064            O(k).
1065
1066        Returns:
1067            list[PositionInfo]: k interpolated positions (``POSITION_*_ESTIMATED``)
1068            followed by the real ``new_position`` with recomputed speed/heading.
1069        """
1070        k = int(times_ms.size)
1071        chain_motion = self._compute_chain_motion(
1072            new_position=new_position,
1073            last_queued=last_queued,
1074            times_ms=times_ms,
1075            latitudes=latitudes,
1076            longitudes=longitudes,
1077        )
1078        position_type = (
1079            PositionType.POSITION_GNSS_ESTIMATED
1080            if new_position.position.positionType == PositionType.POSITION_GNSS
1081            else PositionType.POSITION_PERCEIVED_ESTIMATED
1082        )
1083        time_tmp = ItsTime(its_time_ms=0)
1084        positions: list[PositionInfo] = []
1085        for i in range(k):
1086            reference_position = ReferencePosition.model_construct(
1087                latitude=self._clamp_latitude(float(latitudes[i])),
1088                longitude=self._wrap_longitude(float(longitudes[i])),
1089                positionConfidence=None,
1090                altitude=(
1091                    Altitude.model_construct(
1092                        value=self._clamp_altitude_value(int(altitudes[i])),
1093                        confidence=None,
1094                    )
1095                    if altitudes[i] != Altitude.VALUE_INVALID
1096                    else None
1097                ),
1098            )
1099            time_tmp.epoch_ms = int(times_ms[i])
1100            positions.append(
1101                PositionInfo.model_construct(
1102                    stationInfo=new_position.position.stationInfo,
1103                    objectId=new_position.position.objectId,
1104                    positionType=position_type,
1105                    time=str(object=time_tmp),
1106                    position=reference_position,
1107                    speed=(
1108                        Speed.model_construct(
1109                            value=int(chain_motion.speed_values[i]), confidence=None
1110                        )
1111                        if chain_motion.has_motion[i]
1112                        else None
1113                    ),
1114                    heading=(
1115                        Heading.model_construct(
1116                            value=int(chain_motion.heading_values[i]), confidence=None
1117                        )
1118                        if chain_motion.has_motion[i]
1119                        else None
1120                    ),
1121                    driveDirection=None,
1122                )
1123            )
1124        new_real_position = self._apply_motion(
1125            position=new_position.position,
1126            distance_m=float(chain_motion.distance_heading["distance"][k]),
1127            heading_deg=float(chain_motion.distance_heading["heading"][k]),
1128            time_diff_ms=int(chain_motion.time_diff_ms[k]),
1129        )
1130        positions.append(new_real_position)
1131        if self._logger.isEnabledFor(level=logging.DEBUG):
1132            self._logger.debug(
1133                msg=(
1134                    f"{self._actor}: built {k} interpolated positions plus the real "
1135                    f"position from a single batched geodesic call over {k + 1} pairs"
1136                )
1137            )
1138        return positions

1140    def _create_positions_list(
1141        self,
1142        actor: ActorId,
1143        new_position: PositionInfoRecord,
1144        last_position_distance_heading: Optional[DistanceHeading],
1145    ) -> list[PositionInfo]:
1146        """Create the list of positions from the last queued position to the new one.
1147
1148        Two paths, chosen by the temporal gap to the last queued position:
1149
1150        1. **Fast path** (gap within the nominal output period +/- deviation): the new
1151           position is returned as-is with speed/heading recomputed from
1152           ``last_position_distance_heading`` - the distance/heading pair the caller's
1153           coherence check already computed for this exact pair of points, so no
1154           further ``Geocode`` call is made here.
1155        2. **Interpolation path** (gap exceeds the threshold): ``_create_interpolated_arrays``
1156           produces k interpolated coordinates via cubic spline, and
1157           ``_build_interpolated_positions`` turns them into k+1 positions (k
1158           interpolated + the real new position) using a single vectorized geodesic
1159           call over the whole chain, instead of one scalar call per point.
1160
1161        In both paths, speed/heading reported by the CAM sender on ``new_position``
1162        are not preserved - they are always recomputed from the geodesic displacement
1163        relative to the previous point.
1164
1165        Time complexity:
1166            Fast path: O(1). Interpolation path: O(n + k) for array population and
1167            spline construction/evaluation, plus one vectorized geodesic call and
1168            O(k) object construction (down from O(k) scalar geodesic calls).
1169
1170        Space complexity:
1171            O(1) fast path; O(n + k) interpolation path.
1172
1173        Args:
1174            actor (ActorId): The actor identifier, used for logging.
1175            new_position (PositionInfoRecord): The newly arrived position record.
1176            last_position_distance_heading (Optional[DistanceHeading]): Distance/heading
1177                from the last queued position to ``new_position``, as computed by
1178                ``_check_position_coherence``. Required (non-None) since this method is
1179                only reached when the queue is non-empty and coherence succeeded.
1180
1181        Returns:
1182            list[PositionInfo]: Fast path: a single-element list with the real
1183            position. Interpolation path: k interpolated positions
1184            (``POSITION_*_ESTIMATED``) followed by the real position.
1185
1186        Raises:
1187            StepAdapterError: If the input queue is empty, or if
1188                ``last_position_distance_heading`` is missing on the fast path.
1189        """
1190        if len(self._input_points_queue) == 0:
1191            raise StepAdapterError(
1192                f"{actor}: positions list error due to input queue being empty"
1193            )
1194        last_queued = self._input_points_queue[-1]
1195        diff_time_ms = new_position.time.time_difference(other_time=last_queued.time)
1196        # No interpolation needed - just return new position
1197        if (
1198            diff_time_ms
1199            <= self._config.output_nominal_period_ms
1200            + self._config.output_max_period_deviation_ms
1201        ):
1202            if last_position_distance_heading is None:
1203                raise StepAdapterError(
1204                    f"{actor}: missing precomputed distance/heading for the fast path"
1205                )
1206            new_real_position = self._apply_motion(
1207                position=new_position.position,
1208                distance_m=last_position_distance_heading.distance,
1209                heading_deg=last_position_distance_heading.heading,
1210                time_diff_ms=diff_time_ms,
1211            )
1212            self._input_points_queue.append(new_position)
1213            return [new_real_position]
1214        (times_ms, latitudes, longitudes, altitudes) = self._create_interpolated_arrays(
1215            actor=actor, new_position=new_position
1216        )
1217        positions = self._build_interpolated_positions(
1218            new_position=new_position,
1219            last_queued=last_queued,
1220            times_ms=times_ms,
1221            latitudes=latitudes,
1222            longitudes=longitudes,
1223            altitudes=altitudes,
1224        )
1225        self._input_points_queue.append(new_position)
1226        return positions

1228    def _check_position_coherence(
1229        self,
1230        new_position_time: ItsTime,
1231        new_position: PositionInfo,
1232        last_position_time: Optional[ItsTime],
1233        last_position: Optional[PositionInfo],
1234    ) -> tuple[InsertPointResult, Optional[DistanceHeading]]:
1235        """Check the temporal and spatial coherence of a new position relative to the
1236        last recorded position.
1237
1238        This method validates the new position by performing the following checks:
1239            - Minimum period: Ensures the time difference between the new and
1240              last position is not below the configured minimum.
1241            - Maximum period: Ensures the time difference does not exceed the
1242              configured maximum.
1243            - Speed coherence: Calculates the speed between the two positions and
1244              ensures it does not exceed the defined threshold.
1245            - Local time deviation: Checks if the new position's timestamp is
1246              coherent with the system's local time (always checked, regardless
1247              of whether a previous position exists).
1248
1249        If any check fails, an appropriate InsertPointResult is returned and a warning is
1250        logged. The distance/heading computed for the speed-coherence check (when a last
1251        position exists) is returned alongside the result so callers on the fast path
1252        (see ``_create_positions_list``) can reuse it instead of issuing another
1253        ``Geocode`` call for the same pair of points.
1254
1255        Args:
1256            new_position_time (ItsTime): Time of the new position.
1257            new_position (PositionInfo): Information about the new position.
1258            last_position_time (Optional[ItsTime]): Time of the last recorded position.
1259            last_position (Optional[PositionInfo]): Information about the last recorded position.
1260
1261        Returns:
1262            tuple[InsertPointResult, Optional[DistanceHeading]]: The coherence result,
1263            and the (distance, heading) from ``last_position`` to ``new_position`` when
1264            a last position was available (``None`` otherwise, e.g. for the first
1265            position).
1266
1267        Time Complexity:
1268            O(1): All checks — time difference, period bounds, local-time deviation,
1269            haversine distance, and speed comparison — execute in constant time.
1270
1271        Space Complexity:
1272            O(1): A fixed number of local variables are used; no data structures
1273            grow with input size.
1274        """
1275        distance_heading: Optional[DistanceHeading] = None
1276        if last_position_time and last_position:
1277            diff_time_ms = new_position_time.time_difference(
1278                other_time=last_position_time
1279            )
1280            # Minimum period
1281            minimum_period = (
1282                self._config.output_nominal_period_ms
1283                - self._config.output_max_period_deviation_ms
1284            )
1285            if diff_time_ms < minimum_period:
1286                if self._logger.isEnabledFor(level=logging.WARNING):
1287                    self._logger.warning(
1288                        msg=f"{self._actor}: new position time {new_position_time} "
1289                        f"is too close (min {minimum_period} ms) to last position time "
1290                        f"{last_position_time}, diff time {diff_time_ms} ms"
1291                    )
1292                return InsertPointResult.PERIOD_TOO_SHORT_ERROR, None
1293            # Maximum period
1294            if diff_time_ms > self._config.input_max_period_ms:
1295                if self._logger.isEnabledFor(level=logging.WARNING):
1296                    self._logger.warning(
1297                        msg=f"{self._actor}: new position time {new_position_time} "
1298                        f"is too far (max {self._config.input_max_period_ms} ms) from the last recorded time "
1299                        f"{last_position_time}, diff time {diff_time_ms} ms"
1300                    )
1301                return InsertPointResult.PERIOD_TOO_LONG_ERROR, None
1302            # Speed and heading (distance/heading solved together, single Geod call).
1303            # A local, throwaway Geocode is used instead of a shared mutable instance
1304            # attribute - there is no per-call state worth reusing here.
1305            geocode = Geocode(
1306                point_a_par=LatLonPoint(
1307                    last_position.position.latitude, last_position.position.longitude
1308                ),
1309                point_b_par=LatLonPoint(
1310                    new_position.position.latitude, new_position.position.longitude
1311                ),
1312            )
1313            distance_m = geocode.get_distance(unit=DistanceUnit.METERS)
1314            heading_deg = geocode.get_heading(unit=AngleUnit.DEGREES)
1315            distance_heading = DistanceHeading(distance_m, heading_deg)
1316            # Speed calculation in m/s
1317            speed = (distance_m * ItsTime.SEC_TO_MS_FACTOR) / diff_time_ms
1318            if speed > self.SPEED_COHERENCE_THRESHOLD:
1319                if self._logger.isEnabledFor(level=logging.WARNING):
1320                    self._logger.warning(
1321                        msg=f"{self._actor}: position speed {speed} m/s exceeds threshold "
1322                        f"{self.SPEED_COHERENCE_THRESHOLD} m/s"
1323                    )
1324                return InsertPointResult.POSITION_ERROR, None
1325        # Local time deviation
1326        now = ItsTime()
1327        if not (
1328            now.epoch_ms - self._config.input_max_local_time_deviation_ms
1329            <= new_position_time.epoch_ms
1330            <= now.epoch_ms + self._config.input_max_local_time_deviation_ms
1331        ):
1332            if self._logger.isEnabledFor(level=logging.WARNING):
1333                self._logger.warning(
1334                    msg=f"{self._actor}: new position time {new_position_time} "
1335                    f"is not coherent with local system time {now} "
1336                    f"due to deviation threshold of {self._config.input_max_local_time_deviation_ms} ms"
1337                )
1338            return InsertPointResult.LOCAL_TIME_NOT_COHERENT_ERROR, None
1339        return InsertPointResult.SUCCESS, distance_heading

1428

1430            Optional[list[PositionInfo]]:

1443        Note:
1444            Not thread-safe; external synchronization is required for concurrent
1445            access. Debug logs are emitted for first positions/resets, warnings for
1446            validation failures - consider log volume in high-throughput scenarios.
1447        """
1448        input_actor = ActorId(
1449            station_id=new_position.stationInfo.stationId,
1450            object_id=new_position.objectId,
1451        )
1452        if input_actor != self._actor:
1453            if self._logger.isEnabledFor(level=logging.WARNING):
1454                self._logger.warning(
1455                    msg=f"Actor ID mismatch: expected {self._actor}, got {input_actor}"
1456                )
1457            return None
1458        new_position_record = PositionInfoRecord(position=new_position)
1459        new_position_time = new_position_record.time
1460        position_record = (
1461            self._input_points_queue[-1] if len(self._input_points_queue) > 0 else None
1462        )
1463        (coherence_result, last_position_distance_heading) = (
1464            self._check_position_coherence(
1465                new_position_time=new_position_time,
1466                new_position=new_position,
1467                last_position_time=position_record.time if position_record else None,
1468                last_position=position_record.position if position_record else None,
1469            )
1470        )
1471        match coherence_result:
1472            case InsertPointResult.SUCCESS:
1473                self._apply_vehicle_metadata(
1474                    vehicle_length=vehicle_length, vehicle_width=vehicle_width
1475                )
1476                if len(self._input_points_queue) == 0:
1477                    return self._seed_first_position(
1478                        new_position_record=new_position_record,
1479                        new_position=new_position,
1480                        new_position_time=new_position_time,
1481                    )
1482            case (
1483                InsertPointResult.PERIOD_TOO_SHORT_ERROR
1484                | InsertPointResult.LOCAL_TIME_NOT_COHERENT_ERROR
1485            ):
1486                return None
1487            case (
1488                InsertPointResult.PERIOD_TOO_LONG_ERROR
1489                | InsertPointResult.POSITION_ERROR
1490            ):
1491                self._clear_history()
1492                self._apply_vehicle_metadata(
1493                    vehicle_length=vehicle_length, vehicle_width=vehicle_width
1494                )
1495                return self._seed_first_position(
1496                    new_position_record=new_position_record,
1497                    new_position=new_position,
1498                    new_position_time=new_position_time,
1499                )
1500            case _:
1501                raise StepAdapterError(
1502                    f"Unhandled InsertPointResult: {coherence_result!r}"
1503                )
1504        return self._create_positions_list(
1505            actor=self._actor,
1506            new_position=new_position_record,
1507            last_position_distance_heading=last_position_distance_heading,
1508        )
