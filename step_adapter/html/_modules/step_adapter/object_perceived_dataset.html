  1"""Conversion of CPM perceived objects into position arrays at the output period.
  2
  3Defines ``ObjectPerceivedDataset``, used by ``StepAdapterApp`` to turn each
  4perceived object of a CPM message (``CpmDataV1``) into a ``PositionArrayV1``.
  5The measured position of the object and its predicted path points are mapped to
  6``PositionInfo`` points sharing one ``Identification`` (sender station id, object
  7type and type confidence); points not strictly following the previous one are
  8dropped, the path is cut at the configured maximum CPM span, and the others
  9are resampled on the output period grid by ``GapFiller.resample`` (interpolated
 10when the CPM period is longer, decimated when shorter); the interpolated points
 11are typed ``POSITION_PERCEIVED_PREDICTED`` like the predicted path. An object whose kept
 12points span less than the minimum output period is skipped with a warning. The sample
 13frequency is the nominal output frequency, and the object
 14dimensions become the vehicle length (X) and width (Y). Unlike ``ActorDataset``,
 15no history is kept between messages.
 16
 17Let:
 18    - o = number of perceived objects in one CPM message, at most
 19      ``CpmDataV1.PERCEIVED_OBJECTS_MAX_LENGTH`` (255).
 20    - p = number of predicted path points of one perceived object, at most
 21      ``PerceivedObject.POSITION_PREDICTED_MAX_LENGTH`` (255).
 22    - g = number of resampled points of one perceived object, at most
 23      ``PositionArrayV1.POINTS_MAX_LENGTH`` (101).
 24
 25Complexity:
 26    - Time: O(o*(p + g*log(p))), dominated by
 27      ObjectPerceivedDataset.add_cpm_message.
 28    - Space: O(o*g + p), dominated by ObjectPerceivedDataset.add_cpm_message: o
 29      returned position arrays of at most g points each plus the transient
 30      resampling buffers of one object at a time.
 31
 32Classes:
 33    - ObjectPerceivedDatasetConfig: Immutable configuration of
 34      ObjectPerceivedDataset. Complexity: time O(1), space O(1).
 35    - ObjectPerceivedDataset: Stateless converter of CPM perceived objects to
 36      position arrays. Complexity: time O(o*(p + g*log(p))), space O(o*g + p).
 37"""

 99class ObjectPerceivedDataset:
100    """Stateless converter of CPM perceived objects to position arrays.
101
102    Each ``add_cpm_message`` call converts the perceived objects of one CPM message
103    independently, resampling their points at the output period; nothing is
104    retained between calls apart from the configuration, the gap filler and the
105    logger.
106
107    Attributes:
108        config (ObjectPerceivedDatasetConfig): Configuration converted once into the
109            GapFillerConfig of the gap filler resampling the points of every object.
110        logger (Logger): Logger used for the warnings on skipped perceived objects
111            and points, and for the debug traces of the created position arrays.
112
113    Complexity:
114        - Time: O(o*(p + g*log(p))), driven by add_cpm_message, o = perceived objects
115          in the message, p = predicted path points and g = resampled points per
116          object.
117        - Space: O(o*g + p), driven by add_cpm_message: o returned position arrays of
118          at most g points each plus the transient resampling buffers of one object
119          at a time.
120    """
121
122    __slots__ = ("_config", "_gap_filler", "_logger")
123
124    def __init__(self, config: ObjectPerceivedDatasetConfig, logger: Logger) -> None:
125        """Store the configuration and build the gap filler.
126
127        Logs the configuration at INFO level and builds the ``GapFiller`` with the
128        output nominal period computed as 1000 // ``output_nominal_sample_frequency``
129        ms.
130
131        Args:
132            config (ObjectPerceivedDatasetConfig): Configuration converted once into the
133                GapFillerConfig of the gap filler resampling the points of every object.
134            logger (Logger): Logger used for the warnings on skipped perceived objects
135                and points, and for the debug traces of the created position arrays.
136
137        Raises:
138            ValueError: If a parameter of the derived GapFillerConfig is out of range
139                (see ``GapFillerConfig.__post_init__``).
140
141        Complexity:
142            - Time: O(1), a fixed number of fields is copied and checked.
143            - Space: O(1), one configuration and one gap filler are allocated.
144        """
145        self._config = config
146        self._logger = logger
147        if self._logger.isEnabledFor(level=logging.INFO):
148            self._logger.info(msg=f"ObjectPerceivedDataset config: {self._config}")
149        self._gap_filler = GapFiller(
150            config_par=GapFillerConfig(
151                input_min_altitude_trailing_point_percent=config.input_min_altitude_trailing_point_percent,
152                output_nominal_period_ms=ItsTime.SEC_TO_MS_FACTOR
153                // config.output_nominal_sample_frequency,
154                output_max_period_deviation_ms=config.output_period_max_deviation_ms,
155            ),
156            logger=logger,
157        )
158
159    def _time_ordered_records(
160        self, actor: ActorId, points: list[PositionInfo]
161    ) -> list[PositionInfoRecord]:
162        """Wrap the points in records, dropping those not following the previous one.
163
164        The first point (the measured position) is always kept; a point whose time
165        does not strictly follow the last kept one is dropped with a warning, since
166        the spline of ``GapFiller.resample`` needs strictly increasing times.
167
168        Args:
169            actor (ActorId): The perceived object identifier, used for logging.
170            points (list[PositionInfo]): Measured position followed by the predicted
171                path points.
172
173        Returns:
174            list[PositionInfoRecord]: The kept points with their parsed time, in
175                strictly increasing time order.
176
177        Complexity:
178            - Time: O(p), one time parse and comparison per point.
179            - Space: O(p), one record per kept point.
180        """
181        records: list[PositionInfoRecord] = []
182        for point in points:
183            record = PositionInfoRecord(position=point)
184            if records and record.time.epoch_ms <= records[-1].time.epoch_ms:
185                if self._logger.isEnabledFor(level=logging.WARNING):
186                    self._logger.warning(
187                        msg=(
188                            f"{actor}: point time {record.time} does not follow "
189                            f"{records[-1].time}, dropped"
190                        )
191                    )
192                continue
193            records.append(record)
194        return records

196    def _records_within_span(
197        self, actor: ActorId, records: list[PositionInfoRecord]
198    ) -> list[PositionInfoRecord]:
199        """Cut the records at the maximum CPM span, with a warning.
200
201        A record exactly at the maximum is kept. The first record (the measured
202        position) is always kept.
203
204        Args:
205            actor (ActorId): The perceived object identifier, used for logging.
206            records (list[PositionInfoRecord]): Points with their parsed time, in
207                strictly increasing time order, at least one.
208
209        Returns:
210            list[PositionInfoRecord]: The records within the maximum CPM span from
211                the first one.
212
213        Complexity:
214            - Time: O(p), one bisection plus the slice copy.
215            - Space: O(p), the sliced list.
216        """
217        max_time_ms = records[0].time.epoch_ms + self._config.input_max_cpm_span_ms
218        kept_number = bisect_right(
219            records, max_time_ms, key=lambda record: record.time.epoch_ms
220        )
221        if kept_number < len(records) and self._logger.isEnabledFor(
222            level=logging.WARNING
223        ):
224            self._logger.warning(
225                msg=(
226                    f"{actor}: path cut at the maximum CPM span of "
227                    f"{self._config.input_max_cpm_span_ms} ms, "
228                    f"{len(records) - kept_number} points dropped"
229                )
230            )
231        return records[:kept_number]

233    def _to_position_array(
234        self, station_id: int, perceived_object: PerceivedObject
235    ) -> PositionArrayV1 | None:
236        """Convert one perceived object into a position array at the output period.
237
238        Builds the ``PositionInfo`` points of the object: the measured position
239        (``POSITION_PERCEIVED``, with the object speed and heading) followed by one
240        ``POSITION_PERCEIVED_PREDICTED`` point per predicted path entry (position,
241        confidence and altitude copied, no speed or heading). The points not
242        strictly following the previous one are dropped
243        (``_time_ordered_records``) and the path is cut at
244        ``input_max_cpm_span_ms`` (``_records_within_span``). The object is
245        skipped when the kept points span less than the minimum output period
246        (``GapFillerConfig.output_period_min_ms``), which includes an object with no
247        predicted point left; the kept points are resampled by
248        ``GapFiller.resample`` to at most ``PositionArrayV1.POINTS_MAX_LENGTH``
249        points: the measured position is kept unchanged, the following ones get
250        speed/heading recomputed and the grid points between real ones, estimated by
251        the resampling (``POSITION_PERCEIVED_ESTIMATED``), are typed
252        ``POSITION_PERCEIVED_PREDICTED``, as every point after the measured position
253        is a prediction. The ``PositionArrayV1`` is then
254        constructed with ``sourceType`` ``CAMERA_SRC``, the nominal output frequency
255        of the gap filler as sample frequency, a ``VehicleLength`` built from the
256        ``objectDimensionX`` value (no confidence indication) as vehicle length, the
257        ``objectDimensionY`` value as vehicle width (each None when the dimension is
258        missing) and the resampled points as ``points``. Following the ETSI CPM
259        convention, X is the dimension along the object's longitudinal axis (length)
260        and Y the transverse one (width).
261
262        Args:
263            station_id (int): Station id of the CPM sender, used as the stationId of
264                the points' Identification and in the log messages.
265            perceived_object (PerceivedObject): Perceived object to convert.
266
267        Returns:
268            PositionArrayV1 | None: The position array of the object, or None (with a
269                warning logged) when the object has no objectId, no predicted
270                positions, or kept points spanning less than the minimum output
271                period.
272
273        Raises:
274            ValueError: If the resampling fails (see ``GapFiller.resample``).
275
276        Complexity:
277            - Time: O(p + g*log(p)), p PositionInfo built and parsed, plus
278              ``GapFiller.resample``.
279            - Space: O(p + g), the input points, the records, the spline and the g
280              resampled points held by the returned array.
281        """
282        if perceived_object.objectId is None:
283            if self._logger.isEnabledFor(level=logging.WARNING):
284                self._logger.warning(
285                    msg=f"Perceived object with stationId {station_id} has no objectId"
286                )
287            return None
288        if not perceived_object.positionPredicted:
289            if self._logger.isEnabledFor(level=logging.WARNING):
290                self._logger.warning(
291                    msg=(
292                        f"Perceived object with stationId {station_id} and objectId "
293                        f"{perceived_object.objectId} has no position predicted"
294                    )
295                )
296            return None
297        station_info = Identification(
298            stationId=station_id,
299            stationType=perceived_object.objectType,
300            stationTypeConfidence=perceived_object.objectTypeConfidence,
301        )
302        points: list[PositionInfo] = [
303            PositionInfo(
304                stationInfo=station_info,
305                objectId=perceived_object.objectId,
306                positionType=PositionType.POSITION_PERCEIVED,
307                time=perceived_object.measurementTime,
308                position=perceived_object.position,
309                speed=perceived_object.speed,
310                heading=perceived_object.heading,
311                driveDirection=None,
312            )
313        ]
314        points.extend(
315            PositionInfo(
316                stationInfo=station_info,
317                objectId=perceived_object.objectId,
318                positionType=PositionType.POSITION_PERCEIVED_PREDICTED,
319                time=predicted.time,
320                position=ReferencePosition(
321                    latitude=predicted.position.latitude,
322                    longitude=predicted.position.longitude,
323                    positionConfidence=predicted.position.positionConfidence,
324                    altitude=(
325                        Altitude(
326                            value=predicted.position.altitude.value,
327                            confidence=predicted.position.altitude.confidence,
328                        )
329                        if predicted.position.altitude is not None
330                        else None
331                    ),
332                ),
333                speed=None,
334                heading=None,
335                driveDirection=None,
336            )
337            for predicted in perceived_object.positionPredicted
338        )
339        actor = ActorId(station_id=station_id, object_id=perceived_object.objectId)
340        records = self._records_within_span(
341            actor=actor, records=self._time_ordered_records(actor=actor, points=points)
342        )
343        span_ms = records[-1].time.epoch_ms - records[0].time.epoch_ms
344        span_min_ms = self._gap_filler.config.output_period_min_ms
345        if span_ms < span_min_ms:
346            if self._logger.isEnabledFor(level=logging.WARNING):
347                self._logger.warning(
348                    msg=(
349                        f"{actor}: span {span_ms} ms below the minimum "
350                        f"{span_min_ms} ms, object skipped"
351                    )
352                )
353            return None
354        # Every estimated point follows the measured position: it is a prediction
355        resampled_points = [
356            point.model_copy(
357                update={"positionType": PositionType.POSITION_PERCEIVED_PREDICTED}
358            )
359            if point.positionType == PositionType.POSITION_PERCEIVED_ESTIMATED
360            else point
361            for point in self._gap_filler.resample(
362                records=records, max_point_number=PositionArrayV1.POINTS_MAX_LENGTH
363            )
364        ]
365        position_array = PositionArrayV1(
366            sourceType=SourceType.CAMERA_SRC,
367            sampleFrequency=self._gap_filler.output_nominal_sample_frequency,
368            vehicleLength=VehicleLength(
369                vehicleLengthValue=perceived_object.objectDimensionX.value,
370                vehicleLengthConfidenceIndication=None,
371            )
372            if perceived_object.objectDimensionX is not None
373            else None,
374            vehicleWidth=perceived_object.objectDimensionY.value
375            if perceived_object.objectDimensionY is not None
376            else None,
377            points=resampled_points,
378        )
379        if self._logger.isEnabledFor(level=logging.DEBUG):
380            self._logger.debug(
381                msg=(
382                    f"Created PositionArrayV1 for station {station_id}, object {perceived_object.objectId} "
383                    f"type {station_info.stationType}, position confidence "
384                    f"{perceived_object.position.positionConfidence}, "
385                    f"with {len(position_array.points)} points from {len(points)}, "
386                    f"sample frequency: {position_array.sampleFrequency}"
387                )
388            )
389        return position_array

391    def add_cpm_message(self, message: CpmDataV1) -> list[PositionArrayV1]:
392        """Convert every perceived object of a CPM message into a position array.
393
394        Calls ``_to_position_array`` once per perceived object, in message order, with
395        the message stationId, timing each call at DEBUG level through a
396        ``LoggingTimer`` (no clock is read when DEBUG is disabled), and logs the
397        number of entries at DEBUG level.
398
399        Args:
400            message (CpmDataV1): CPM message whose perceived objects are converted; its
401                stationId is used for every object.
402
403        Returns:
404            list[PositionArrayV1]: One entry per converted perceived object, in message
405                order. Objects skipped by ``_to_position_array`` (no objectId, no
406                predicted positions or kept points spanning less than the minimum output
407                period) are filtered out.
408
409        Raises:
410            ValueError: If the resampling of an object fails (see
411                ``GapFiller.resample``).
412
413        Complexity:
414            - Time: O(o*(p + g*log(p))), o = perceived objects in the message, p =
415              predicted path points and g = resampled points per object.
416            - Space: O(o*g + p), o returned position arrays of at most g points each
417              plus the transient resampling buffers of one object at a time, released
418              before the next object.
419        """
420        station_id = message.stationId
421        logging_timer = LoggingTimer(
422            logger=self._logger, prefix="step_adapter", log_level=LogMode.DEBUG
423        )
424        position_array_list: list[PositionArrayV1] = []
425        for perceived_object in message.perceivedObjects:
426            with logging_timer(self._to_position_array):
427                position_array = self._to_position_array(
428                    station_id=station_id, perceived_object=perceived_object
429                )
430            if position_array is not None:
431                position_array_list.append(position_array)
432        if self._logger.isEnabledFor(level=logging.DEBUG):
433            self._logger.debug(
434                msg=(
435                    f"Added {len(position_array_list)} PositionArrayV1 objects received from station {station_id}"
436                )
437            )
438        return position_array_list
