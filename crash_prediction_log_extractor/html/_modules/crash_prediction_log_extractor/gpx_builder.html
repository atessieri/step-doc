  1"""GPX file generation from crash-prediction collection-actor data.
  2
  3This module converts the real-position points recorded for each actor in a
  4``CollectionActors`` payload into a single multi-track GPX file, one track
  5per actor, in the format written by step_message_save's ``GpxWriter`` and
  6read by step_publisher_pkl. Consecutive sessions of an actor are merged into
  7a single chronologically-ordered track and actors that produce no track
  8points are skipped.
  9
 10Complexity:
 11    - Time: O(a log a + N + S log p), dominated by GpxBuilder.build_file;
 12      a = actors, N = total real positions, S = total sessions, p = real
 13      positions per session.
 14    - Space: O(N), one GPX waypoint per accepted real position.
 15
 16Classes:
 17    - GpxBuilder: Writes one GPX track per actor from a CollectionActors
 18      payload. Complexity: time O(a log a + N + S log p), space O(N).
 19
 20Typical usage example::
 21
 22    from crash_prediction_log_extractor.gpx_builder import GpxBuilder
 23    from pathlib import Path
 24
 25    builder = GpxBuilder(logger=logger)
 26    builder.build_file(file_path=Path("output/log.gpx"), data=collection_actors)

 54class GpxBuilder:
 55    """Builds a multi-track GPX file from collection-actor data.
 56
 57    One track is written per actor, named
 58    ``Path actor <stationId>_<objectId>`` (the literal ``none`` when the
 59    actor has no object id), so that step_publisher_pkl can read the station
 60    id back. Each track has a single segment of chronologically-ordered
 61    points; tracks are sorted by station id, then object id (``none`` first).
 62    Heading and speed are written to the Garmin TrackPointExtension v2
 63    ``course``/``speed`` elements.
 64
 65    Class Attributes:
 66        EXTENSION (ClassVar[str]): File extension of the generated GPX files
 67            (``".gpx"``).
 68
 69    Attributes:
 70        logger (Logger): Logger instance used to report GPX file creation.
 71
 72    Complexity:
 73        - Time: O(a log a + N + S log p), dominated by build_file; a =
 74          actors, N = total real positions, S = total sessions, p = real
 75          positions per session.
 76        - Space: O(N), dominated by build_file, one waypoint per accepted
 77          point.
 78    """
 79
 80    __slots__ = ("_logger",)
 81
 82    EXTENSION: ClassVar[str] = ".gpx"

 84    _GARMIN_TPX_NAMESPACE: ClassVar[str] = (
 85        "http://www.garmin.com/xmlschemas/TrackPointExtension/v2"
 86    )
 87    _GARMIN_TPX_PREFIX: ClassVar[str] = "gpxtpx"
 88    _TRACK_NAME_FORMAT: ClassVar[str] = "Path actor {station_id}_{object_id}"
 89    _NO_OBJECT_ID: ClassVar[str] = "none"
 90    _TRACK_POINT_EXTENSION_TAG: ClassVar[str] = "TrackPointExtension"

 92    def __init__(self, logger: Logger) -> None:
 93        """Initializes the builder and registers the Garmin namespace prefix.
 94
 95        ElementTree keeps a process-wide prefix registry: registering it here
 96        makes the extension elements serialize as ``gpxtpx:`` instead of the
 97        default ``ns0:``. Registering the same prefix/uri pair again is a
 98        no-op.
 99
100        Args:
101            logger (Logger): Logger instance used to report GPX file creation.
102
103        Complexity:
104            - Time: O(r), r = number of prefixes already registered in the
105              ElementTree registry, scanned for conflicts.
106            - Space: O(1), one registry entry at most.
107        """
108        self._logger = logger
109        register_namespace(
110            prefix=self._GARMIN_TPX_PREFIX, uri=self._GARMIN_TPX_NAMESPACE
111        )

113    def build_file(self, file_path: Path, data: CollectionActors) -> None:
114        """Builds and writes the multi-track GPX file of all collection actors.
115
116        Actors without track points produce no track; a data set without
117        track points produces a valid GPX file with no tracks.
118
119        Args:
120            file_path (Path): Path of the GPX file to write; its folder must
121                already exist.
122            data (CollectionActors): Collection-actors payload providing the
123                per-actor sessions to convert into GPX tracks.
124
125        Returns:
126            None: The GPX file is written to ``file_path`` and its creation is
127                logged.
128
129        Complexity:
130            - Time: O(a log a + N + S log p), a = actors (sorted by id), N =
131              total real positions, S = total sessions, p = real positions
132              per session (see _build_track); writing the XML is O(N).
133            - Space: O(N), one Waypoint per accepted point.
134        """
135        tracks = [
136            track
137            for collection_actor in sorted(
138                data.collections,
139                key=lambda actor: (
140                    actor.actorId.stationId,
141                    actor.actorId.objectId is not None,
142                    actor.actorId.objectId or 0,
143                ),
144            )
145            if (track := self._build_track(collection_actor=collection_actor))
146            is not None
147        ]
148        GPX(trk=tracks).write_gpx(file_path)
149        StepLogger.logger_force_write(
150            logger=self._logger,
151            msg=f"GPX file with {len(tracks)} tracks saved to {file_path}",
152        )

154    def _build_track(self, collection_actor: CollectionActor) -> Track | None:
155        """Builds the GPX track of a single collection actor.
156
157        Merges the real positions of all the actor's sessions into a single
158        chronologically-ordered list of track points: in each session, a
159        binary search skips the leading positions whose timestamp is not
160        strictly after the last accepted timestamp, and all remaining
161        positions of that session are accepted. Assumes each session's real
162        positions are sorted by time.
163
164        Args:
165            collection_actor (CollectionActor): Actor whose sessions'
166                real positions are converted into a GPX track.
167
168        Returns:
169            Track | None: A track named
170                ``Path actor <stationId>_<objectId|none>`` with a single
171                segment, or ``None`` if no track points result.
172
173        Complexity:
174            - Time: O(n + s log p), n = total real positions of the actor,
175              s = its sessions, p = real positions per session (one
176              ``bisect_right`` per session, then a linear pass).
177            - Space: O(n), the accumulated track points list.
178        """
179        last_time: dt.datetime = dt.datetime.min.replace(tzinfo=dt.UTC)
180        track_points: list[Waypoint] = []
181        for session in collection_actor.sessions:
182            real_positions = session.realPositions
183            new_start_index = bisect_right(
184                a=real_positions,
185                x=last_time,
186                key=lambda position: dt.datetime.fromisoformat(position.time),
187            )
188            for position in real_positions[new_start_index:]:
189                last_time = dt.datetime.fromisoformat(position.time)
190                track_points.append(
191                    self._position_to_point(position=position, time=last_time)
192                )
193        if not track_points:
194            return None
195        actor_id = collection_actor.actorId
196        return Track(
197            name=self._TRACK_NAME_FORMAT.format(
198                station_id=actor_id.stationId,
199                object_id=self._NO_OBJECT_ID
200                if actor_id.objectId is None
201                else actor_id.objectId,
202            ),
203            trkseg=[TrackSegment(trkpt=track_points)],
204        )

206    def _position_to_point(self, position: Position, time: dt.datetime) -> Waypoint:
207        """Converts a real position into a GPX track point.
208
209        Args:
210            position (Position): The real position to convert.
211            time (dt.datetime): The position timestamp, already parsed.
212
213        Returns:
214            Waypoint: A track point with latitude, longitude and time; heading
215                and speed, when present, are stored in the Garmin
216                TrackPointExtension v2 ``course``/``speed`` elements. Values
217                are rounded to the step_cloud_schemas precisions
218                (COORDINATE_PRECISION, HEADING_VALUE_PRECISION,
219                SPEED_VALUE_PRECISION). Units already match GPX: degrees,
220                degrees, m/s. The extensions are omitted when both heading
221                and speed are ``None``.
222
223        Complexity:
224            - Time: O(1), a fixed number of roundings and element builds.
225            - Space: O(1), one waypoint with at most two extension values.
226        """
227        extensions = Extensions()
228        if position.heading is not None:
229            extensions.set_text(
230                tag="course",
231                value=self._rounded_text(
232                    value=position.heading, ndigits=HEADING_VALUE_PRECISION
233                ),
234                namespace=self._GARMIN_TPX_NAMESPACE,
235                parent_tag=self._TRACK_POINT_EXTENSION_TAG,
236            )
237        if position.speed is not None:
238            extensions.set_text(
239                tag="speed",
240                value=self._rounded_text(
241                    value=position.speed, ndigits=SPEED_VALUE_PRECISION
242                ),
243                namespace=self._GARMIN_TPX_NAMESPACE,
244                parent_tag=self._TRACK_POINT_EXTENSION_TAG,
245            )
246        return Waypoint(
247            lat=Latitude(
248                value=self._rounded_text(
249                    value=position.latitude, ndigits=COORDINATE_PRECISION
250                )
251            ),
252            lon=Longitude(
253                value=self._rounded_text(
254                    value=position.longitude, ndigits=COORDINATE_PRECISION
255                )
256            ),
257            time=time,
258            extensions=extensions or None,
259        )

261    @staticmethod
262    def _rounded_text(value: float, ndigits: int) -> str:
263        """Rounds value to ndigits decimal places and renders it as text.
264
265        Args:
266            value (float): The value to round.
267            ndigits (int): The number of decimal places to keep, one of the
268                step_cloud_schemas ``*_PRECISION`` constants.
269
270        Returns:
271            str: The shortest text representation of the rounded value.
272
273        Complexity:
274            - Time: O(1), one rounding and one float-to-text conversion.
275            - Space: O(1), one short string.
276        """
277        return str(object=round(number=value, ndigits=ndigits))
