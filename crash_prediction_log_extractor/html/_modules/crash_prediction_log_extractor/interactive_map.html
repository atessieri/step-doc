  1"""Interactive map generation for crash prediction log data.
  2
  3This module provides three main components:
  4
  5- ``MapBounds``: frozen dataclass representing a geographic bounding box
  6  (south, west, north, east edges).
  7- ``MapData``: frozen dataclass pairing a filtered road-network
  8  ``GeoDataFrame`` with its ``MapBounds``, one per loaded OSM file.
  9- ``InteractiveMap``: builds interactive Folium HTML maps from
 10  ``CollectionSessions`` and ``CollectionActors`` data structures.
 11  HTML panel templates and JavaScript data layers are loaded from the
 12  bundled ``templates`` package directory; the serialized collection is
 13  injected at render time as a sequence of ``<script>`` blocks, each one
 14  holding a chunk of the collection list, so the browser-side panel can
 15  display session and actor details without additional server requests.
 16  Chunking keeps every script source well below the maximum string length
 17  of the browser JavaScript engines (about 512 Mi characters in V8), which
 18  a single JSON literal of a large log would exceed, preventing the whole
 19  map script from running.
 20
 21The OSM files are parsed once, at ``InteractiveMap`` construction; each map
 22then draws the road network of the first loaded OSM file whose bounds
 23overlap the collection, one selectable layer per road type.
 24
 25Complexity:
 26    - Time: O(M + n + E), dominated by InteractiveMap (OSM parsing at
 27      construction and the ``*_create_map`` methods); M = total size of the
 28      OSM files, n = size of the serialized collection, E = road edges of the

257    def _filter_edges(self, edges: GeoDataFrame, road_types: set[str]) -> GeoDataFrame:
258        """Return the edges whose ``highway`` tag matches any given road type.
259
260        A ``highway`` value may be a single string or, for OSM multi-value
261        tags, a list; the edge matches when any of its values is in
262        ``road_types``. An empty ``road_types`` keeps no edge.
263
264        Args:
265            edges (GeoDataFrame): Road network edges to filter.
266            road_types (set[str]): ``highway`` tag values to keep.
267
268        Returns:
269            GeoDataFrame: The edges with at least one ``highway`` value in
270                ``road_types``.
271
272        Complexity:
273            - Time: O(E * h), E = len(edges), h = highway values per edge.
274            - Space: O(E), the boolean mask and the filtered frame.
275        """
276
277        def _matches(highway: str | list[str]) -> bool:
278            """Tell whether an edge ``highway`` value matches a selected road type.
279
280            Args:
281                highway (str | list[str]): ``highway`` tag of the edge, a single
282                    value or a list of values.
283
284            Returns:
285                bool: True if any value is in ``road_types``.
286            """
287            return any(
288                value in road_types
289                for value in (highway if isinstance(highway, list) else [highway])
290            )
291
292        return edges[edges["highway"].apply(func=_matches)]

294    def _get_network_bounds(
295        self, osm_paths: list[Path], map_road_type_filter: set[str]
296    ) -> dict[Path, MapData]:
297        """Load OSM files and build a road network index keyed by file path.
298
299        Parses each OSM file into a graph, converts it to a GeoDataFrame of
300        edges, retains only edges whose ``highway`` tag matches a value in
301        ``map_road_type_filter``, and computes the resulting bounding box.
302
303        Args:
304            osm_paths (list[Path]): Paths to the OSM XML files to load.
305            map_road_type_filter (set[str]): ``highway`` tag values to keep;
306                edges that do not match any value in this set are dropped.
307
308        Returns:
309            dict[Path, MapData]: Mapping from each OSM file path to its
310                filtered edges and computed geographic bounding box (NaN
311                bounds when no edge survives the filter).
312
313        Complexity:
314            - Time: O(M), M = total size of the OSM files, dominated by the
315              ``osmnx`` XML parsing and graph conversion.
316            - Space: O(M), the parsed graphs and filtered edge frames.
317        """
318        bounds: dict[Path, MapData] = {}
319        for path in osm_paths:
320            graph = graph_from_xml(path)
321            edges = self._filter_edges(
322                edges=graph_to_gdfs(graph, nodes=False), road_types=map_road_type_filter
323            )
324            lon_min, lat_min, lon_max, lat_max = edges.total_bounds
325            bounds[path] = MapData(
326                edges=edges,
327                bounds=MapBounds(
328                    lat_min=float(lat_min),
329                    lon_min=float(lon_min),
330                    lat_max=float(lat_max),
331                    lon_max=float(lon_max),
332                ),
333            )
334            if self._logger.isEnabledFor(level=logging.DEBUG):
335                self._logger.debug(
336                    msg=f"Loaded map file {path} with bounds {bounds[path].bounds}"
337                )
338        return bounds

340    def _get_local_map(self, map_bounds: MapBounds) -> GeoDataFrame | None:
341        """Return the edges of the first loaded map overlapping the bounds.
342
343        Iterates over all loaded OSM files in load order and performs an
344        axis-aligned bounding box intersection test (touching edges count as
345        overlapping). Returns the edges of the first file whose bounds
346        intersect ``map_bounds``.
347
348        Args:
349            map_bounds (MapBounds): Geographic bounding box of the area to
350                match.
351
352        Returns:
353            GeoDataFrame | None: Filtered edge GeoDataFrame for the first
354                overlapping OSM file, or ``None`` if no loaded map overlaps
355                the requested bounds.
356
357        Complexity:
358            - Time: O(m), m = number of loaded OSM files.
359            - Space: O(1), the frame is returned by reference.
360        """
361        for path, data in self._map_data_dict.items():
362            if self._logger.isEnabledFor(level=logging.DEBUG):
363                self._logger.debug(
364                    msg=f"Checking map file {path} with bounds {data.bounds} against requested bounds {map_bounds}"
365                )
366            if (
367                (data.bounds.lat_min <= map_bounds.lat_max)
368                and (map_bounds.lat_min <= data.bounds.lat_max)
369            ) and (
370                (data.bounds.lon_min <= map_bounds.lon_max)
371                and (map_bounds.lon_min <= data.bounds.lon_max)
372            ):
373                if self._logger.isEnabledFor(level=logging.INFO):
374                    self._logger.info(
375                        msg=f"Found local map for bounds {map_bounds} in file {path}"
376                    )
377                return data.edges
378        if self._logger.isEnabledFor(level=logging.INFO):
379            self._logger.info(msg=f"No local map found for bounds {map_bounds}")
380        return None

625        r"""Make a JSON string safe to embed inside an HTML ``<script>`` element.

643    def _iter_payload_chunks(self, items: Sequence[BaseModel]) -> Iterator[str]:
644        """Serialize a list of models as consecutive JSON array chunks.
645
646        Each item is serialized with ``model_dump_json`` (the same output it
647        has inside the serialized collection), escaped with
648        ``_escape_script_json``, and consecutive items are grouped while the
649        chunk stays within ``PAYLOAD_CHUNK_MAX_CHARS`` characters. An item
650        longer than the budget is yielded alone in its own chunk and a
651        warning is logged.
652
653        Args:
654            items (Sequence[BaseModel]): Items of the collection list, in
655                order.
656
657        Yields:
658            str: JSON array text holding the next group of items; the
659                concatenation of all the arrays, in order, is ``items``.
660
661        Returns:
662            None: The generator ends after the last chunk; nothing is yielded
663                for an empty ``items``.
664
665        Complexity:
666            - Time: O(n), n = total serialized length of ``items``.
667            - Space: O(C + l), C = ``PAYLOAD_CHUNK_MAX_CHARS`` (one pending
668              chunk), l = length of the longest item.
669        """
670        chunk: list[str] = []
671        chunk_len = 2  # "[" and "]"
672        for idx, item in enumerate(items):
673            item_json = self._escape_script_json(json_str=item.model_dump_json())
674            if chunk and chunk_len + 1 + len(item_json) > self.PAYLOAD_CHUNK_MAX_CHARS:
675                yield "[" + ",".join(chunk) + "]"
676                chunk = []
677                chunk_len = 2
678            if len(item_json) + 2 > self.PAYLOAD_CHUNK_MAX_CHARS and (
679                self._logger.isEnabledFor(level=logging.WARNING)
680            ):
681                self._logger.warning(
682                    msg=(
683                        f"Payload item {idx} is {len(item_json)} characters long, over the "
684                        f"chunk budget of {self.PAYLOAD_CHUNK_MAX_CHARS} characters"
685                    )
686                )
687            chunk_len += len(item_json) + (1 if chunk else 0)  # "," separator
688            chunk.append(item_json)
689        if chunk:
690            yield "[" + ",".join(chunk) + "]"

692    def _add_payload_scripts(
693        self, figure_root: Figure, collection: BaseModel, list_field: str
694    ) -> None:
695        """Add the serialized collection to the map body as chunked ``<script>`` blocks.
696
697        A first script stores the collection without ``list_field`` in
698        ``window.__PAYLOAD_HEAD`` and creates the empty array
699        ``window.__PAYLOAD_PARTS``; each following script pushes one chunk
700        of ``list_field`` items (see ``_iter_payload_chunks``) into it. The
701        scripts go in ``figure_root.html``, which the ``Figure`` renders in
702        ``<body>`` before its main ``<script>``, so the map JavaScript can
703        rebuild the whole payload when it runs.
704
705        Args:
706            figure_root (Figure): Root element of the map.
707            collection (BaseModel): Collection to serialize.
708            list_field (str): Name of the list field of ``collection`` to
709                split into chunks.
710
711        Returns:
712            None: The scripts are added to ``figure_root.html`` in place.
713
714        Complexity:
715            - Time: O(n), n = size of the serialized collection.
716            - Space: O(n), every chunk is kept as an element of the figure.
717        """
718        head_json = self._escape_script_json(
719            json_str=collection.model_dump_json(exclude={list_field})
720        )
721        figure_root.html.add_child(
722            Element(
723                template=(
724                    f"<script>window.__PAYLOAD_HEAD = {head_json}; "
725                    "window.__PAYLOAD_PARTS = [];</script>"
726                )
727            )
728        )
729        chunk_sizes: list[int] = []
730        for chunk in self._iter_payload_chunks(items=getattr(collection, list_field)):
731            figure_root.html.add_child(
732                Element(
733                    template=f"<script>window.__PAYLOAD_PARTS.push({chunk});</script>"
734                )
735            )
736            chunk_sizes.append(len(chunk))
737        if self._logger.isEnabledFor(level=logging.DEBUG):
738            self._logger.debug(
739                msg=f"Added {len(chunk_sizes)} payload chunks for '{list_field}' with sizes {chunk_sizes}"
740            )

742    def _get_collection_sessions_bounds(
743        self, collection: CollectionSessions
744    ) -> MapBounds:
745        """Calculate the geographic bounding box for a collection of sessions.
746
747        Derives bounds from the first real position of the first session in
748        which each distinct ``ActorIdModel`` (compared on all fields)
749        appears; later positions are ignored.
750
751        Args:
752            collection (CollectionSessions): Collection of session data.
753
754        Returns:
755            MapBounds: Geographic bounding box enclosing those first real
756                positions. Returns
757                ``MapBounds(lat_min=0.0, lon_min=0.0, lat_max=0.0, lon_max=0.0)``
758                when the collection has no sessions.
759
760        Complexity:
761            - Time: O(S * A), S = sessions, A = actor sessions per session.
762            - Space: O(U), U = distinct actor IDs.
763        """
764        if not collection.sessions:
765            return MapBounds(lat_min=0.0, lon_min=0.0, lat_max=0.0, lon_max=0.0)
766        positions_dict = {}
767        for session in collection.sessions:
768            for actor_session in session.actorsSession:
769                if actor_session.actorId not in positions_dict:
770                    positions_dict[actor_session.actorId] = (
771                        actor_session.data.realPositions[0]
772                    )
773        min_lat = min(position.latitude for position in positions_dict.values())
774        max_lat = max(position.latitude for position in positions_dict.values())
775        min_lon = min(position.longitude for position in positions_dict.values())
776        max_lon = max(position.longitude for position in positions_dict.values())
777        return MapBounds(
778            lat_min=min_lat, lon_min=min_lon, lat_max=max_lat, lon_max=max_lon
779        )
