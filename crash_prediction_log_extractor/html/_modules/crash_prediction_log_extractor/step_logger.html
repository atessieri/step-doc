  1"""Logging utilities for the crash_prediction_log_extractor package.
  2
  3This module provides a lightweight logging system with UTC timestamp formatting and
  4support for both console and file-based logging. It offers a simple configuration
  5interface through the get_logger function that creates a pre-configured logger
  6with consistent UTC timezone handling across all handlers.
  7
  8Key Features:
  9    - UTC timestamp formatting for consistent time handling across timezones
 10    - Stream handler for console output (stdout)
 11    - File handler for optional file-based logging with enhanced formatting
 12    - Configurable logging levels via the LogMode enumeration
 13    - Force-write capability for critical messages that must be logged regardless
 14      of the current logging level
 15    - File operations use UTF-8 encoding
 16"""

 48    @staticmethod
 49    def to_logging_level(mode: "LogMode") -> int:
 50        """Converts a LogMode enum value to the corresponding logging level integer.
 51
 52        Args:
 53            mode (LogMode): The logging mode enum value to convert.
 54
 55        Returns:
 56            int: The corresponding logging level constant from the logging module.
 57
 58        Raises:
 59            ValueError: If the provided mode is not a valid LogMode value.
 60        """
 61        level: int
 62        match mode:
 63            case LogMode.CRITICAL:
 64                level = logging.CRITICAL
 65            case LogMode.ERROR:
 66                level = logging.ERROR
 67            case LogMode.WARNING:
 68                level = logging.WARNING
 69            case LogMode.INFO:
 70                level = logging.INFO
 71            case LogMode.DEBUG:
 72                level = logging.DEBUG
 73            case _:
 74                raise ValueError(f"Invalid log mode: {mode}")
 75        return level

101def get_logger(file_log: Optional[Path], log_level: Optional[LogMode]) -> Logger:
102    """Create and configure a logger with console and optional file handlers.
103
104    Sets up a logger with UTC formatting that writes to stdout and optionally to
105    a file. All handlers use the UtcFormatter for consistent timezone handling.
106    If a file path is provided, a file handler is added with enhanced formatting
107    that includes thread and function information.
108
109    Args:
110        file_log (Optional[Path]): Path to log file. If None, only console logging
111            is configured. If provided, creates a file handler that writes to this
112            location in write mode (overwrites existing content).
113        log_level (Optional[LogMode]): Logging level to use. If None, defaults to
114            ERROR level. Applies to both console and file handlers.
115
116    Returns:
117        Logger: Configured logger instance with handlers and UTC formatting applied.
118    """
119    log_level_int = LogMode.to_logging_level(mode=log_level or LogMode.ERROR)
120    logger = getLogger()
121    logger.handlers.clear()
122    logger.setLevel(level=log_level_int)
123    handler = StreamHandler(stream=sys.stdout)
124    formatter = UtcFormatter(
125        fmt="%(asctime)s.%(msecs)03d [%(levelname)s][pid=%(process)d]: %(message)s",
126        datefmt="%Y-%m-%d %H:%M:%S",
127    )
128    handler.setFormatter(fmt=formatter)
129    handler.setLevel(level=logging.ERROR)
130    logger.addHandler(hdlr=handler)
131    if file_log:
132        file_log.parent.mkdir(parents=True, exist_ok=True)
133        file_handler = FileHandler(
134            filename=file_log, mode="w", encoding="utf-8", delay=False, errors=None
135        )
136        formatter = UtcFormatter(
137            fmt="%(asctime)s.%(msecs)03d [%(levelname)s][pid=%(process)d][thread=%(threadName)s][module=%(module)s][function=%(funcName)s]: %(message)s",
138            datefmt="%Y-%m-%d %H:%M:%S",
139        )
140        file_handler.setFormatter(fmt=formatter)
141        file_handler.setLevel(level=log_level_int)
142        logger.addHandler(hdlr=file_handler)
143    return logger

146def logger_force_write(logger: Logger, msg: str) -> None:
147    """Force the logger to write a message at INFO level, overriding handler levels.
148
149    Temporarily elevates the logger and all its handlers to INFO level to ensure
150    the message is logged, regardless of their original levels. Original levels
151    are restored after the message is written. Useful for critical messages that
152    must be logged even when the logger is set to WARNING or higher levels.
153
154    Args:
155        logger (Logger): The logger instance to use for writing the message.
156        msg (str): The message to log at INFO level.
157
158    Returns:
159        None
160
161    Note:
162        This function modifies handler levels during execution and restores them
163        afterward, so it is thread-safe for the duration of the operation.
164    """
165    original_handlers_level: list[int] = []
166    for handler in logger.handlers:
167        original_handlers_level.append(handler.level)
168        handler.setLevel(level=logging.INFO)
169    original_level = logger.level
170    logger.setLevel(level=logging.INFO)
171    logger.info(msg=msg)
172    logger.setLevel(level=original_level)
173    for original_handler_level, handler in zip(
174        original_handlers_level, logger.handlers, strict=True
175    ):
176        handler.setLevel(level=original_handler_level)
