  1"""Log extractor for crash prediction scenario data.
  2
  3This module parses structured log files produced by the ``step_crash_prediction``
  4service and assembles the extracted data into ``CollectionsPoints`` payloads.
  5
  6Log file format
  7---------------
  8
  9Each line in the log file follows the standard Python logging layout::
 10
 11    YYYY-MM-DD HH:MM:SS.mmm [LEVEL][pid=...][thread=...][module=...][function=...]: <marker>
 12
 13The extractor ignores regular log output and only reacts to **marker** lines
 14embedded in the log. The markers delimit scenarios, section types (normalized /
 15anomalous, real / forecasted), crash events, and individual data points.
 16
 17Marker reference
 18~~~~~~~~~~~~~~~~
 19
 20* Scenario boundary::
 21
 22    --Scenario start--
 23    --Scenario end--
 24
 25* Section data boundaries (three blocks: ``normalized real``, ``normalized
 26  forecasted``, and ``anomaly real``):::
 27
 28    --Section normalized real start--
 29    --Section normalized real end--
 30    --Section normalized forecasted start--
 31    --Section normalized forecasted end--
 32    --Section anomaly real start--
 33    --Section anomaly real end--
 34
 35* Actor identifier (inside a section data block)::
 36
 37    --Id: sid <station_id>, oid <object_id|None>--
 38
 39    Example: --Id: sid 37, oid None--
 40             --Id: sid 36, oid 5--
 41
 42* Position data point (following an actor identifier)::
 43
 44    --t: <ISO8601>, p: (<lat>, <lon>), s: <speed|None>, h: <heading|None>--
 45
 46    Example (real):      --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), s: 0.00, h: 21.10--
 47    Example (forecast):  --t: 2026-02-24T05:26:06.379Z, p: (45.609514, 8.863836), s: None, h: None--
 48
 49* Crash section boundaries::
 50
 51    --Section crash start--
 52    --Section crash end--
 53
 54* Individual crash record (inside a crash section)::
 55
 56    --Crash start--
 57    --Time: <ISO8601>, pos: (<lat>, <lon>), risk: <0.000-1.000>--
 58    --Id: sid <station_id>, oid <object_id|None>--
 59    ... (one --Id:-- line per actor involved)
 60    --Crash end--
 61
 62    Log example::
 63
 64        --Crash start--
 65        --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 0.850--
 66        --Id: sid 36, oid None--
 67        --Id: sid 37, oid None--
 68        --Crash end--
 69
 70Full minimal scenario example
 71~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
 72::
 73
 74    ... [INFO]...: --Scenario start--
 75    ... [INFO]...: --Section normalized real start--
 76    ... [INFO]...: --Section normalized real end--
 77    ... [INFO]...: --Section normalized forecasted start--
 78    ... [INFO]...: --Section normalized forecasted end--
 79    ... [INFO]...: --Section anomaly real start--
 80    ... [INFO]...: --Id: sid 37, oid None--
 81    ... [INFO]...: --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), s: 0.00, h: 21.10--
 82    ... [INFO]...: --Section anomaly real end--
 83    ... [INFO]...: --Section crash start--
 84    ... [INFO]...: --Crash start--
 85    ... [INFO]...: --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 0.850--
 86    ... [INFO]...: --Id: sid 36, oid None--
 87    ... [INFO]...: --Id: sid 37, oid None--
 88    ... [INFO]...: --Crash end--
 89    ... [INFO]...: --Section crash end--
 90    ... [INFO]...: --Scenario end--
 91"""

238class LogExtractor:
239    """Parses a ``step_crash_prediction`` log file and builds a
240    :class:`CollectionsPoints` payload.
241
242    The extractor reads the log file line by line and drives a hierarchy of
243    finite-state machines to recognize structured marker lines embedded in the
244    standard Python logging output.  A full scenario block looks like::
245
246        --Scenario start--
247          --Section normalized real start--
248          --Id: sid 37, oid None--
249          --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), s: 0.00, h: 21.10--
250          --Section normalized real end--
251          --Section normalized forecasted start--
252          --t: 2026-02-24T05:26:00.379Z, p: (45.609514, 8.863836), s: None, h: None--
253          --Section normalized forecasted end--
254          --Section anomaly real start--
255          --Section anomaly real end--
256          --Section crash start--
257          --Crash start--
258          --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 0.850--
259          --Id: sid 36, oid None--
260          --Id: sid 37, oid None--
261          --Crash end--
262          --Section crash end--
263        --Scenario end--
264
265    Multiple scenarios may appear in one log file; each is processed
266    independently and the results are merged into a single
267    :class:`CollectionsPoints` object available through the
268    :attr:`collections` property after :meth:`extract` has been called.
269
270    Attributes:
271        SCENARIO_PATTERN (Pattern[str]): Compiled regex that matches
272            ``--Scenario (start|end)--``.
273        SECTION_DATA_PATTERN (Pattern[str]): Compiled regex that matches a
274            section data boundary, e.g.
275            ``--Section anomaly real start--``.
276        SECTION_CRASH_PATTERN (Pattern[str]): Compiled regex that matches
277            ``--Section crash (start|end)--``.
278        CRASH_PATTERN (Pattern[str]): Compiled regex that matches
279            ``--Crash (start|end)--``.
280        CRASH_RISK_PATTERN (Pattern[str]): Compiled regex that matches
281            ``--Time: <ISO8601>, pos: (<lat>, <lon>), risk: <0.000-1.000>--``
282            and captures the time, position, and risk float string.
283        ACTOR_ID_PATTERN (Pattern[str]): Compiled regex that matches
284            ``--Id: sid <station_id>, oid <object_id|None>--`` and captures
285            both groups.
286        POINT_PATTERN (Pattern[str]): Compiled regex that matches a position
287            data line and captures time, latitude, longitude, speed and heading.
288    """
289
290    # Scenario markers
291    SCENARIO_PATTERN: ClassVar[Pattern[str]] = re_compile(
292        pattern=r"--Scenario (start|end)--"
293    )
294    # Section data markers
295    SECTION_DATA_PATTERN: ClassVar[Pattern[str]] = re_compile(
296        pattern=r"--Section (anomaly|normalized) (real|forecasted) (start|end)--"
297    )
298    # Section crash markers
299    SECTION_CRASH_PATTERN: ClassVar[Pattern[str]] = re_compile(
300        pattern=r"--Section crash (start|end)--"
301    )
302    # Crash markers
303    CRASH_PATTERN: ClassVar[Pattern[str]] = re_compile(pattern=r"--Crash (start|end)--")
304    # Risk marker
305    CRASH_RISK_PATTERN: ClassVar[Pattern[str]] = re_compile(
306        pattern=r"--Time: ([^,]+), pos: \((-?\d+(?:\.\d+)?), (-?\d+(?:\.\d+)?)\), risk: (-?\d+(?:\.\d+)?)--"
307    )
308    # Actor ID pattern
309    ACTOR_ID_PATTERN: ClassVar[Pattern[str]] = re_compile(
310        pattern=r"--Id: sid (\d+), oid (None|\d+)--"
311    )
312    # Data point patterns
313    POINT_PATTERN: ClassVar[Pattern[str]] = re_compile(
314        pattern=r"--t: ([^,]+), p: \(([^,]+), ([^)]+)\), s: (None|[^,]+), h: (None|[^)]+)--"
315    )

328    def __init__(self, input_file: Path, logger: Logger) -> None:
329        """Initialise the log extractor.
330
331        Args:
332            input_file (Path): Path to the ``step_crash_prediction`` log file
333                to parse.  The file must be UTF-8 encoded plain text whose
334                lines follow the format::
335
336                    YYYY-MM-DD HH:MM:SS.mmm [LEVEL][...][function=...]: <marker>
337
338            logger (Logger): Standard library logger used to emit diagnostic
339                messages during parsing (debug, info, and error level).
340        """
341        self._input_file = input_file
342        self._collections: Optional[CollectionsPoints] = None
343        self._session_normalized_real_data: dict[ActorId, list[Position]] = {}
344        self._session_normalized_forecasted_data: dict[ActorId, list[Position]] = {}
345        self._session_anomaly_real_data: dict[ActorId, list[Position]] = {}
346        self._session_crash_data: list[CrashInfo] = []
347        self._tables_data_markers: list[ParserTableDataMarkerItem] = [
348            ParserTableDataMarkerItem(
349                data_type=DataType.NORMALIZED,
350                section_data_type=SectionDataType.REAL,
351                data_collection=self._session_normalized_real_data,
352            ),
353            ParserTableDataMarkerItem(
354                data_type=DataType.NORMALIZED,
355                section_data_type=SectionDataType.FORECASTED,
356                data_collection=self._session_normalized_forecasted_data,
357            ),
358            ParserTableDataMarkerItem(
359                data_type=DataType.ANOMALY,
360                section_data_type=SectionDataType.REAL,
361                data_collection=self._session_anomaly_real_data,
362            ),
363        ]
364        self._logger = logger
365

371            Optional[CollectionsPoints]: The assembled :class:`CollectionsPoints`
372            object containing all actor sessions extracted from the log file, or
373            ``None`` if :meth:`extract` has not been called yet.
374        """
375        return self._collections

377    def clear_table(self) -> None:
378        """Reset all in-memory session data buffers for the current scenario.
379
380        Clears the three position dictionaries (normalized real, normalized
381        forecasted, and anomaly real) and the crash info list so that a new
382        scenario can be parsed from a clean state.  Called automatically by

387        """

389        self._session_normalized_forecasted_data.clear()
390        self._session_anomaly_real_data.clear()

406        Returns:
407            ActorId: The constructed actor identifier with ``station_id`` set to
408            the parsed integer and ``object_id`` set to ``None`` when the raw
409            value is the string ``"None"`` or to the parsed integer otherwise.

411        station_id, object_id_str = match.groups()
412        self._logger.info(
413            msg=(
414                f"Found actor ID - station_id: {station_id}, object_id: {object_id_str}"
415            )
416        )
417        return ActorId(
418            station_id=int(station_id),
419            object_id=None if object_id_str == "None" else int(object_id_str),

422    def _insert_position_data(
423        self,
424        match: Match[str],
425        last_actor_id: Optional[ActorId],
426        data: dict[ActorId, list[Position]],
427    ) -> None:
428        """Parse a position data point and append it to the actor's list.
429
430        Converts the raw string groups captured by :attr:`POINT_PATTERN` from a
431        log line of the form::
432
433            --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), s: 0.00, h: 21.10--
434            --t: 2026-02-24T05:26:06.379Z, p: (45.609514, 8.863836), s: None, h: None--

436        into a :class:`Position` instance and appends it to ``data[last_actor_id]``.
437        If ``last_actor_id`` is ``None`` (no preceding ``--Id:--`` line was seen)

439

441            match (Match[str]): Regex match object whose groups are
442                ``(time_str, lat_str, lon_str, speed_str, heading_str)``.
443            last_actor_id (Optional[ActorId]): The most recently parsed actor
444                identifier in the current section.  Points are associated with
445                this actor.
446            data (dict[ActorId, list[Position]]): Mutable mapping that is
447                updated in place; the new position is appended to the list at
448                key ``last_actor_id``.
449
450        Returns:
451            None
452        """
453        time_str, lat_str, lon_str, speed_str, heading_str = match.groups()
454        self._logger.info(
455            msg=(
456                f"Extracted data - time: {time_str}, latitude: {lat_str}, "

458            )
459        )

461            position = Position(

463                latitude=float(lat_str),
464                longitude=float(lon_str),
465                speed=None if speed_str == "None" else float(speed_str),
466                heading=None if heading_str == "None" else float(heading_str),
467            )
468            self._logger.debug(msg=f"Created position: {position}")
469            data[last_actor_id].append(position)

471    def _extract_section_data(  # noqa: C901, PLR0912
472        self, file: TextIO, data_type: DataType, section_data_type: SectionDataType
473    ) -> Optional[dict[ActorId, list[Position]]]:
474        """Extract all position data points from one section data block.
475
476        Reads lines from ``file`` and drives a
477        :class:`ParserSectionDataState` state machine until the matching
478        ``end`` marker is found.  The section is delimited by::
479
480            --Section <data_type> <section_data_type> start--
481            --Id: sid <station_id>, oid <object_id|None>--
482            --t: <ISO8601>, p: (<lat>, <lon>), s: <speed|None>, h: <heading|None>--
483            ... (more --Id:-- and --t:-- lines)
484            --Section <data_type> <section_data_type> end--
485
486        Multiple actors can appear in a single section; each ``--Id:--`` line
487        switches the current actor, and subsequent ``--t:--`` lines belong to
488        that actor until the next ``--Id:--`` or the ``end`` marker.
489
490        Example section (anomaly real)::
491
492            --Section anomaly real start--
493            --Id: sid 37, oid None--
494            --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), s: 0.00, h: 21.10--
495            --t: 2026-02-24T05:25:51.439Z, p: (45.610087, 8.863476), s: 0.97, h: 21.10--
496            --Section anomaly real end--
497
498        Args:
499            file (TextIO): Open log file positioned just after the scenario
500                ``start`` marker (or after the previous section ``end``).
501            data_type (DataType): The expected section data type token
502                (``normalized`` or ``anomaly``).
503            section_data_type (SectionDataType): The expected section kind
504                token (``real`` or ``forecasted``).
505
506        Returns:
507            Optional[dict[ActorId, list[Position]]]: Mapping from each actor ID
508            found in the section to its ordered list of parsed
509            :class:`Position` objects, or ``None`` if the ``end`` marker was
510            never reached (truncated file).
511
512        Raises:
513            ValueError: If a section data marker is found but its embedded
514                ``data_type`` or ``section_data_type`` tokens do not match the
515                expected values, or if the state machine reaches an invalid
516                state.
517        """
518        data: dict[ActorId, list[Position]] = {}
519        last_actor_id: Optional[ActorId] = None
520        state = ParserSectionDataState.WAIT_START_SECTION_DATA
521        while line := file.readline():
522            line_stripped = line.strip()
523            new_state = state
524            section_state_read = None
525            if match := self.SECTION_DATA_PATTERN.search(string=line_stripped):
526                self._logger.debug(msg=f"Section line: {line_stripped}")
527                data_type_read, section_data_type_read, section_state_read = (
528                    match.groups()
529                )
530                if data_type_read != data_type.value:
531                    raise ValueError(
532                        f"Section data type read {data_type_read} does not match expected data type {data_type}"
533                    )
534                if section_data_type_read != section_data_type.value:
535                    raise ValueError(
536                        f"Section data type read {section_data_type_read} does not match expected section data type {section_data_type}"
537                    )
538            match state:
539                case ParserSectionDataState.WAIT_START_SECTION_DATA:
540                    if section_state_read == "start":
541                        new_state = ParserSectionDataState.WAIT_END_SECTION_DATA
542                case ParserSectionDataState.WAIT_END_SECTION_DATA:
543                    if section_state_read == "end":
544                        new_state = ParserSectionDataState.DONE
545                    elif match := self.ACTOR_ID_PATTERN.search(string=line_stripped):
546                        self._logger.debug(msg=f"Data processing line: {line_stripped}")
547                        last_actor_id = self._get_actor_id(match=match)
548                        if last_actor_id not in data:
549                            data[last_actor_id] = []
550                    elif match := self.POINT_PATTERN.search(string=line_stripped):
551                        self._logger.debug(msg=f"Data processing line: {line_stripped}")
552                        self._insert_position_data(
553                            match=match, last_actor_id=last_actor_id, data=data
554                        )
555                case _:
556                    raise ValueError(f"Invalid parser section state {state}")
557            if state != new_state:
558                self._logger.info(
559                    msg=(
560                        f"Parser section state changed from {state} to {new_state} for "
561                        f"data type {data_type_read} and section data type {section_data_type_read}"
562                    )
563                )
564                state = new_state
565            if state == ParserSectionDataState.DONE:
566                break
567        return data if state == ParserSectionDataState.DONE else None

569    def _extract_crash_data(  # noqa: C901, PLR0912  # pylint: disable=too-many-locals
570        self, file: TextIO
571    ) -> Optional[list[CrashInfo]]:
572        """Extract one crash record or detect the end of the crash section.
573
574        Reads lines from ``file`` and drives a :class:`ParserCrashState` state
575        machine.  Returns an empty list when a ``--Section crash end--`` marker
576        is found (signalling that no more crash records follow), a single-item
577        list when a full crash record is parsed, or ``None`` on a truncated
578        file.
579
580        A crash record has the following structure::
581
582            --Crash start--
583            --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 0.850--
584            --Id: sid 36, oid None--
585            --Id: sid 37, oid None--
586            --Crash end--
587
588        The risk value is a float in ``[0.0, 1.0]`` with three decimal places.
589        At least two ``--Id:--`` lines are expected (the actors involved).
590
591        Args:
592            file (TextIO): Open log file positioned inside a crash section
593                (after ``--Section crash start--``).
594
595        Returns:
596            Optional[list[CrashInfo]]: One of:
597
598            * A single-element list containing the parsed :class:`CrashInfo`
599              when a complete crash record (``--Crash start--`` …
600              ``--Crash end--``) is found.
601            * An empty list when ``--Section crash end--`` is encountered,
602              indicating no more crash records in the current section.
603            * ``None`` if EOF is reached before the record is complete
604              (truncated file).
605
606        Raises:
607            ValueError: If the state machine reaches an unrecognized state.
608        """
609        state = ParserCrashState.WAIT_START_CRASH
610        estimated_timestamp = ""
611        estimated_latitude = 0.0
612        estimated_longitude = 0.0
613        risk = 0.0
614        actors_involved: list[ActorId] = []
615        while line := file.readline():
616            line_stripped = line.strip()
617            new_state = state
618            if match := self.SECTION_CRASH_PATTERN.search(string=line_stripped):
619                self._logger.debug(msg=f"Section crash line: {line_stripped}")
620                if match.group(1) == "end":
621                    return []
622            if match := self.CRASH_PATTERN.search(string=line_stripped):
623                self._logger.debug(msg=f"Crash data processing line: {line_stripped}")
624                crash_state_read = match.group(1)
625            else:
626                crash_state_read = None
627            match state:
628                case ParserCrashState.WAIT_START_CRASH:
629                    if crash_state_read == "start":
630                        new_state = ParserCrashState.WAIT_RISK
631                case ParserCrashState.WAIT_RISK:
632                    if crash_state_read == "end":
633                        new_state = ParserCrashState.DONE
634                    elif match := self.CRASH_RISK_PATTERN.search(string=line_stripped):
635                        self._logger.debug(
636                            msg=f"Crash data processing line: {line_stripped}"
637                        )
638                        estimated_timestamp, lat_str, lon_str, risk_str = match.groups()
639                        estimated_latitude = float(lat_str)
640                        estimated_longitude = float(lon_str)
641                        risk = float(risk_str)
642                        self._logger.info(
643                            msg=(
644                                f"Extracted crash data - time: {estimated_timestamp}, "
645                                f"latitude: {estimated_latitude:.7f}, longitude: {estimated_longitude:.7f}, "
646                                f"risk: {risk:.3f}"
647                            )
648                        )
649                        new_state = ParserCrashState.WAIT_END_CRASH
650                case ParserCrashState.WAIT_END_CRASH:
651                    if crash_state_read == "end":
652                        new_state = ParserCrashState.DONE
653                    elif match := self.ACTOR_ID_PATTERN.search(string=line_stripped):
654                        self._logger.debug(
655                            msg=f"Crash data processing line: {line_stripped}"
656                        )
657                        station_id, object_id_str = match.groups()
658                        self._logger.info(
659                            msg=(
660                                f"Found actor ID - station_id: {station_id}, "
661                                f"object_id: {object_id_str}"
662                            )
663                        )
664                        actors_involved.append(
665                            ActorId(
666                                station_id=int(station_id),
667                                object_id=None
668                                if object_id_str == "None"
669                                else int(object_id_str),
670                            )
671                        )
672                case _:
673                    raise ValueError(f"Invalid parser crash state {state}")
674            if state != new_state:
675                self._logger.info(
676                    msg=f"Parser section state changed from {state} to {new_state}"
677                )
678                state = new_state
679            if state == ParserCrashState.DONE:
680                break
681        return (
682            [
683                CrashInfo(
684                    estimatedTimestamp=estimated_timestamp,
685                    estimatedLatitude=estimated_latitude,
686                    estimatedLongitude=estimated_longitude,
687                    risk=risk,
688                    actorsInvolved=actors_involved,
689                )
690            ]
691            if (state == ParserCrashState.DONE) and (len(actors_involved) > 1)
692            else None
693        )

695    def _extract_section_crash_data(self, file: TextIO) -> Optional[list[CrashInfo]]:
696        """Extract all crash records from the crash section of a scenario.
697
698        Reads lines from ``file`` until the opening ``--Section crash start--``
699        marker is found, then delegates to :meth:`_extract_crash_data` in a
700        loop until it signals the end of the section (empty list) or an error
701        (``None``).
702
703        A crash section may contain zero or more individual crash records::
704
705            --Section crash start--
706            --Crash start--
707            --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 0.850--
708            --Id: sid 36, oid None--
709            --Id: sid 37, oid None--
710            --Crash end--
711            --Section crash end--
712
713        When no crashes occurred the section is empty::
714
715            --Section crash start--
716            --Section crash end--
717
718        Args:
719            file (TextIO): Open log file positioned after the last section data
720                ``end`` marker of the current scenario.
721
722        Returns:
723            Optional[list[CrashInfo]]: The complete list of :class:`CrashInfo`
724            objects for the scenario (may be empty), or ``None`` if a parsing
725            error occurred (truncated file or malformed record).
726        """
727        crash_info_list: list[CrashInfo] = []
728        while line := file.readline():
729            line_stripped = line.strip()
730            if match := self.SECTION_CRASH_PATTERN.search(string=line_stripped):
731                self._logger.debug(msg=f"Section crash line: {line_stripped}")
732                if match.group(1) == "start":
733                    while True:
734                        crashes_info = self._extract_crash_data(file=file)
735                        if crashes_info is None:
736                            return None
737                        if len(crashes_info) == 0:
738                            break
739                        crash_info_list.extend(crashes_info)
740                    break
741        return crash_info_list

743    def _merge_session_data_collections(
744        self,
745        data_type: DataType,
746        real_data: dict[ActorId, list[Position]],
747        forecasted_data: dict[ActorId, list[Position]],
748        crash_data: list[CrashInfo],
749    ) -> dict[ActorId, SessionData]:
750        """Build :class:`SessionData` objects for one data-type category.
751
752        Iterates over every actor that has at least one real position and
753        combines its real positions, forecasted positions, and the crash events
754        it is involved in into a :class:`SessionData` instance.
755
756        Args:
757            data_type (DataType): Category of the data being merged
758                (``normalized`` or ``anomaly``).
759            real_data (dict[ActorId, list[Position]]): Mapping from actor ID to
760                its list of real position points for the current scenario.
761            forecasted_data (dict[ActorId, list[Position]]): Mapping from actor
762                ID to its forecasted position points.  Actors absent from this
763                mapping receive an empty forecasted list.
764            crash_data (list[CrashInfo]): All crash records for the current
765                scenario; only those where the actor appears in
766                ``actorsInvolved`` are included in the session.
767
768        Returns:
769            dict[ActorId, SessionData]: Mapping from each actor ID (with at
770            least one real position) to its assembled :class:`SessionData`.
771        """
772        result: dict[ActorId, SessionData] = {}
773        for actor_id, real_positions in real_data.items():
774            if len(real_positions) > 0:
775                result[actor_id] = SessionData(
776                    lastRealTime=real_positions[-1].time,
777                    dataType=data_type,
778                    realPositions=real_positions,
779                    forecastedPositions=forecasted_data.get(actor_id, []),
780                    crashInfo=[
781                        crash_info
782                        for crash_info in crash_data
783                        if actor_id in crash_info.actorsInvolved
784                    ],
785                )
786        return result

788    def _merge_session_data(self) -> dict[ActorId, SessionData]:
789        """Merge all three in-memory buffers into a single per-actor session map.
790
791        Calls :meth:`_merge_session_data_collections` for both ``normalized``
792        and ``anomaly`` categories and combines the results.  Anomaly entries
793        overwrite normalized entries when the same actor appears in both (the
794        ``anomaly`` data is considered more specific).
795
796        Returns:
797            dict[ActorId, SessionData]: Combined mapping from actor ID to
798            :class:`SessionData` for the current scenario.
799        """
800        result: dict[ActorId, SessionData] = {}
801        # Merge normalized data
802        result.update(
803            self._merge_session_data_collections(
804                data_type=DataType.NORMALIZED,
805                real_data=self._session_normalized_real_data,
806                forecasted_data=self._session_normalized_forecasted_data,
807                crash_data=self._session_crash_data,
808            )
809        )
810        # Merge anomaly data
811        result.update(
812            self._merge_session_data_collections(
813                data_type=DataType.ANOMALY,
814                real_data=self._session_anomaly_real_data,
815                forecasted_data={},
816                crash_data=[],
817            )
818        )
819        return result

821    def _extract_section(self, file: TextIO) -> Optional[dict[ActorId, SessionData]]:  # noqa: C901, PLR0912
822        """Parse one complete scenario block from the log file.
823
824        Drives the top-level :class:`ParserScenarioState` state machine.  When
825        the ``--Scenario start--`` marker is found it sequentially delegates to
826        :meth:`_extract_section_data` (three times, covering normalized real,
827        normalized forecasted, and anomaly real) and
828        :meth:`_extract_section_crash_data`.
829
830        The expected structure is::
831
832            --Scenario start--
833              --Section normalized real start-- ... --Section normalized real end--
834              --Section normalized forecasted start-- ... --Section normalized forecasted end--
835              --Section anomaly real start-- ... --Section anomaly real end--
836              --Section crash start-- ... --Section crash end--
837            --Scenario end--
838
839        Args:
840            file (TextIO): Open log file; reading continues from the current
841                position.
842
843        Returns:
844            Optional[dict[ActorId, SessionData]]: Per-actor session data
845            assembled from the parsed scenario, or ``None`` if EOF is reached
846            before the scenario is complete or a sub-parser signals failure.
847
848        Raises:
849            ValueError: If the state machine reaches an invalid state.
850        """
851        self.clear_table()
852        state = ParserScenarioState.WAIT_FOR_START_SCENARIO
853        while line := file.readline():
854            line_stripped = line.strip()
855            new_state = state
856            if match := self.SCENARIO_PATTERN.search(string=line_stripped):
857                self._logger.debug(msg=f"Scenario line: {line_stripped}")
858                scenario_state_read = match.group(1)
859            else:
860                scenario_state_read = None
861            match state:
862                case ParserScenarioState.WAIT_FOR_START_SCENARIO:
863                    if scenario_state_read == "start":
864                        # Data scenario
865                        for marker in self._tables_data_markers:
866                            data_result = self._extract_section_data(
867                                file=file,
868                                data_type=marker.data_type,
869                                section_data_type=marker.section_data_type,
870                            )
871                            if data_result is None:
872                                return None
873                            marker.data_collection.update(data_result)
874                        # Crash scenario
875                        crash_data_result = self._extract_section_crash_data(file=file)
876                        if crash_data_result is None:
877                            return None
878                        self._session_crash_data.extend(crash_data_result)
879                        new_state = ParserScenarioState.WAIT_FOR_END_SCENARIO
880                case ParserScenarioState.WAIT_FOR_END_SCENARIO:
881                    if scenario_state_read == "end":
882                        new_state = ParserScenarioState.DONE
883                case _:
884                    raise ValueError(f"Invalid parser state {state}")
885            if state != new_state:
886                if self._logger.isEnabledFor(level=logging.INFO):
887                    self._logger.info(
888                        msg=f"Parser state changed from {state} to {new_state}"
889                    )
890                state = new_state
891            if state == ParserScenarioState.DONE:
892                break
893        if state != ParserScenarioState.DONE:
894            return None
895        return self._merge_session_data()

897    def extract(self) -> None:
898        """Parse the entire log file and populate :attr:`collections`.
899
900        Opens :attr:`_input_file` in read mode and iteratively calls
901        :meth:`_extract_section` until the end of the file is reached or a
902        :class:`ValueError` is raised by a sub-parser (which is caught and
903        logged, allowing partial results to be returned).
904
905        After this method returns, :attr:`collections` contains a
906        :class:`CollectionsPoints` payload with one :class:`CollectionSections`
907        entry per actor, each holding all :class:`SessionData` instances
908        extracted across every parsed scenario.
909
910        The result is always a valid :class:`CollectionsPoints` object (possibly
911        with an empty ``collections`` list) even when parsing errors occur.
912
913        Returns:
914            None
915
916        Example usage::
917
918            from pathlib import Path
919            import logging
920
921            extractor = LogExtractor(
922                input_file=Path("step_crash_prediction.txt"),
923                logger=logging.getLogger(__name__),
924            )
925            extractor.extract()
926            payload = extractor.collections
927            if payload:
928                payload.print_to_file(Path("output.json"))
929        """
930        with self._input_file.open(mode="r", encoding="utf-8") as file:
931            collections: dict[ActorId, list[SessionData]] = {}
932            try:
933                while True:
934                    session_scenario = self._extract_section(file=file)
935                    if session_scenario is None:
936                        break
937                    for actor_id, session_data in session_scenario.items():
938                        if actor_id not in collections:
939                            collections[actor_id] = []
940                        collections[actor_id].append(session_data)
941            except ValueError as e:
942                if self._logger.isEnabledFor(level=logging.ERROR):
943                    self._logger.exception(msg=f"Error while parsing log file: {e}")
944            collection_list = [
945                CollectionSections(actorId=actor_id, sessions=sessions)
946                for actor_id, sessions in collections.items()
947            ]
948            self._collections = CollectionsPoints(
949                formatVersion=CollectionsPointsDataModelVersion.V1,
950                collections=collection_list,
951            )
