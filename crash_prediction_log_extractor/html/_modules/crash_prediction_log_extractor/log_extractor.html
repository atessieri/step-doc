   1"""Log extractor for crash prediction scenario data.
   2
   3This module parses structured log files produced by the ``step_crash_prediction``
   4service and assembles the extracted data into ``CollectionSessions`` payloads.
   5
   6The primary public interface is ``LogExtractor``.  Instantiate it with the
   7path to a log file and a ``logging.Logger``, call ``LogExtractor.extract``,
   8then read the result from ``LogExtractor.collections``.
   9
  10Log file format
  11---------------
  12
  13Each line in the log file follows the standard Python logging layout::
  14
  15    YYYY-MM-DD HH:MM:SS.mmm [LEVEL][pid=...][thread=...][module=...][function=...]: <marker>
  16
  17The extractor ignores regular log output and only reacts to **marker** lines
  18embedded in the log. The markers delimit scenarios, section types (normalized /
  19anomalous, real / forecasted), crash events, and individual data points.
  20
  21Marker reference
  22~~~~~~~~~~~~~~~~
  23
  24* Scenario boundary::
  25
  26    --Scenario start--
  27    --Scenario end--
  28
  29* Scenario time (required after ``--Scenario start--``, before the
  30  sections)::
  31
  32    --Scenario t: <ISO8601>--
  33
  34* Section data boundaries (three blocks: ``normalized real``, ``normalized
  35  forecasted``, and ``anomaly real``, always in this order)::
  36
  37    --Section normalized real start--
  38    --Section normalized real end--
  39    --Section normalized forecasted start--
  40    --Section normalized forecasted end--
  41    --Section anomaly real start--
  42    --Section anomaly real end--
  43
  44* Actor identifier (inside a section data block)::
  45
  46    --sid: <station_id>, oid: <object_id|None>, type: <TrafficParticipantType>, confidence: <0-100|None>--
  47
  48    Example: --sid: 37, oid: None, type: passengerCar, confidence: 90--
  49             --sid: 36, oid: 5, type: cyclist, confidence: None--
  50
  51* Position data point (following an actor identifier)::
  52
  53    --t: <ISO8601>, p: (<lat>, <lon>), t: <PositionType>, s: <speed|None>, h: <heading|None>--
  54
  55    Example (real):      --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), t: GNSS, s: 0.00, h: 21.10--
  56    Example (forecast):  --t: 2026-02-24T05:26:06.379Z, p: (45.609514, 8.863836), t: GNSS_predicted, s: None, h: None--
  57
  58* Checkpoint marker (timing / progress probe)::
  59
  60    --Checkpoint: <name>, timer: <ms.3f>--
  61
  62    where ``<ms>`` is a monotonic timer value in milliseconds (any number
  63    of decimals is accepted).
  64
  65    Accepted checkpoint names are the ``CheckpointType`` values:
  66
  67    * ``scenario start``
  68    * ``scenario norm start``
  69    * ``scenario norm to dataframe``
  70    * ``forecasting norm``
  71    * ``forecasted snapping``
  72    * ``log forecasted norm``
  73    * ``scenario norm end``
  74    * ``scenario anomaly start``
  75    * ``scenario anomaly end``
  76    * ``send forecasted data``
  77    * ``process collision detection start``
  78    * ``process collision detection end``
  79    * ``send crash detection results``
  80    * ``scenario end``
  81
  82    A checkpoint with any other name is skipped with a warning. Checkpoint
  83    lines are collected anywhere in a scenario except inside individual
  84    crash records.
  85
  86* Crash section boundaries::
  87
  88    --Section crash start--
  89    --Section crash end--
  90
  91* Individual crash record (inside a crash section)::
  92
  93    --Crash start--
  94    --Time: <ISO8601>, pos: (<lat>, <lon>), risk: <1-7>--
  95    --sid: <station_id>, oid: <object_id|None>, type: <TrafficParticipantType>, confidence: <0-100|None>--
  96    ... (one --sid:-- line per actor involved)
  97    --Crash end--
  98
  99    Log example::
 100
 101        --Crash start--
 102        --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 5--
 103        --sid: 36, oid: None, type: passengerCar, confidence: 90--
 104        --sid: 37, oid: None, type: pedestrian, confidence: 80--
 105        --Crash end--
 106

 373class LogExtractor:
 374    """Parses a ``step_crash_prediction`` log file into ``CollectionSessions``.
 375
 376    The extractor reads the log file line by line and drives a hierarchy of
 377    finite-state machines to recognize structured marker lines embedded in the
 378    standard Python logging output.  A full scenario block looks like::
 379
 380        --Scenario start--
 381          --Scenario t: 2026-02-24T05:25:51.339Z--
 382          --Checkpoint: scenario start, timer: 1000.000--
 383          --Checkpoint: scenario norm start, timer: 1000.000--
 384          --Section normalized real start--
 385          --sid: 37, oid: None, type: passengerCar, confidence: 90--
 386          --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), t: GNSS, s: 0.00, h: 21.10--
 387          --Section normalized real end--
 388          --Checkpoint: log forecasted norm, timer: 1001.000--
 389          --Section normalized forecasted start--
 390          --sid: 37, oid: None, type: passengerCar, confidence: 90--
 391          --t: 2026-02-24T05:26:00.379Z, p: (45.609514, 8.863836), t: GNSS_predicted, s: None, h: None--
 392          --Section normalized forecasted end--
 393          --Checkpoint: scenario norm end, timer: 1001.000--
 394          --Checkpoint: scenario anomaly start, timer: 1001.000--
 395          --Section anomaly real start--
 396          --Section anomaly real end--
 397          --Checkpoint: scenario anomaly end, timer: 1002.000--
 398          --Checkpoint: send forecasted data, timer: 1002.000--
 399          --Checkpoint: send crash detection results, timer: 1002.000--
 400          --Section crash start--
 401          --Crash start--
 402          --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 5--
 403          --sid: 36, oid: None, type: passengerCar, confidence: 90--
 404          --sid: 37, oid: None, type: pedestrian, confidence: 80--
 405          --Crash end--
 406          --Section crash end--
 407          --Checkpoint: scenario end, timer: 1002.000--
 408        --Scenario end--
 409
 410    Multiple scenarios may appear in one log file; each is processed
 411    independently and every complete scenario with at least one actor becomes
 412    one ``Session`` of a single ``CollectionSessions`` object, available
 413    through the ``collections`` property after ``extract`` has been called.
 414    Not thread-safe: the parser keeps per-scenario state (checkpoints and a
 415    shared ``ItsTime`` helper) on the instance.
 416
 417    Class Attributes:
 418        MINIMUM_ACTORS_IN_CRASH (ClassVar[int]): Minimum number of distinct
 419            actors a crash record needs to be kept (2).
 420        SCENARIO_PATTERN (ClassVar[Pattern[str]]): ``--Scenario start|end--``.
 421        SCENARIO_TIME_PATTERN (ClassVar[Pattern[str]]): ``--Scenario t: ...--``.
 422        SECTION_DATA_PATTERN (ClassVar[Pattern[str]]): Section data
 423            ``start``/``end`` markers.
 424        SECTION_CRASH_PATTERN (ClassVar[Pattern[str]]): Crash section
 425            ``start``/``end`` markers.
 426        CRASH_PATTERN (ClassVar[Pattern[str]]): ``--Crash start|end--``.
 427        CRASH_RISK_PATTERN (ClassVar[Pattern[str]]): Crash time, position,
 428            and integer risk line.
 429        ACTOR_ID_PATTERN (ClassVar[Pattern[str]]): Actor identifier line.
 430        POINT_PATTERN (ClassVar[Pattern[str]]): Position data point line.
 431        CHECKPOINT_PATTERN (ClassVar[Pattern[str]]): Checkpoint line.
 432        DATA_MARKERS (ClassVar[list[ParserTableDataMarkerItem]]): The three
 433            section data blocks of a scenario, in parsing order.
 434
 435    Attributes:
 436        input_file (Path): Path to the ``step_crash_prediction`` log file
 437            to parse.  The file must be UTF-8 encoded plain text whose
 438            lines follow the format::
 439
 440                YYYY-MM-DD HH:MM:SS.mmm [LEVEL][...][function=...]: <marker>
 441
 442        logger (Logger): Standard library logger used to emit diagnostic
 443            messages during parsing (debug, info, warning, and error level).
 444
 445    Complexity:
 446        - Time: O(B + K * a**2), dominated by extract; B = size of the log
 447          file in characters, K = crash records, a = actors per crash record.
 448        - Space: O(P), dominated by extract; P = positions, crashes, and
 449          checkpoints kept for the parsed sessions.
 450    """

 489        r"t: (?P<fix_type>[^,]+), s: (?P<speed>None|[^,]+), h: (?P<heading>None|[^)]+)--"

 491    CHECKPOINT_PATTERN: ClassVar[Pattern[str]] = re_compile(
 492        pattern=r"--Checkpoint: (?P<checkpoint_type>.+?), timer: (?P<timer>\d+(?:\.\d+)?)--"
 493    )
 494    DATA_MARKERS: ClassVar[list[ParserTableDataMarkerItem]] = [
 495        ParserTableDataMarkerItem(
 496            data_type=DataType.NORMALIZED, section_data_type=SectionDataType.REAL
 497        ),
 498        ParserTableDataMarkerItem(
 499            data_type=DataType.NORMALIZED, section_data_type=SectionDataType.FORECASTED
 500        ),
 501        ParserTableDataMarkerItem(
 502            data_type=DataType.ANOMALY, section_data_type=SectionDataType.REAL
 503        ),
 504    ]

 508    def __init__(self, input_file: Path, logger: Logger) -> None:
 509        """Initialise the log extractor.
 510
 511        Args:
 512            input_file (Path): Path to the ``step_crash_prediction`` log file
 513                to parse.  The file must be UTF-8 encoded plain text whose
 514                lines follow the format::
 515
 516                    YYYY-MM-DD HH:MM:SS.mmm [LEVEL][...][function=...]: <marker>
 517
 518            logger (Logger): Standard library logger used to emit diagnostic
 519                messages during parsing (debug, info, warning, and error level).
 520
 521        Complexity:
 522            - Time: O(1), stores the arguments; the file is not opened.
 523            - Space: O(1), empty result and checkpoint state.
 524        """
 525        self._input_file = input_file

 547    def _get_checkpoint(self, line: str) -> CheckpointDataModel | None:
 548        """Extract checkpoint metadata from a raw log line.
 549
 550        Applies ``CHECKPOINT_PATTERN`` to lines that contain a checkpoint
 551        marker and captures:
 552
 553        - the checkpoint name token from
 554          ``--Checkpoint: <name>, timer: <ms.3f>--``
 555        - the monotonic timer value in milliseconds (an unsigned number with
 556          any number of decimals)
 557
 558        A matching line whose name is not a ``CheckpointType`` value, or whose
 559        timer fails ``CheckpointDataModel`` validation, is logged as a warning
 560        and yields ``None``.
 561
 562        Example input line::
 563
 564            2026-06-02T12:45:18.350Z [INFO][...]: --Checkpoint: scenario end, timer: 610174404.123--
 565
 566        Args:
 567            line (str): Full log line to inspect.
 568
 569        Returns:
 570            CheckpointDataModel | None: A populated checkpoint model when
 571                the line matches :attr:`CHECKPOINT_PATTERN` with a valid name
 572                and timer, otherwise ``None``.
 573
 574        Complexity:
 575            - Time: O(c), c = len(line), one regex search.
 576            - Space: O(1), one small model.
 577        """
 578        match = self.CHECKPOINT_PATTERN.search(string=line)
 579        if match is None:
 580            return None
 581        checkpoint_type, timer_str = (
 582            match.group("checkpoint_type"),
 583            match.group("timer"),
 584        )
 585        try:
 586            cp_type = CheckpointType(checkpoint_type)
 587            return CheckpointDataModel(checkpoint_type=cp_type, timer=float(timer_str))
 588        except ValueError as exc:
 589            if self._logger.isEnabledFor(logging.WARNING):
 590                self._logger.warning(msg=f"Checkpoint type {checkpoint_type}: {exc}")
 591        return None

 648    def _calculate_speed_and_heading(
 649        self,
 650        last_time: str | None,
 651        last_position: LatLonPoint | None,
 652        new_time: str,
 653        new_position: LatLonPoint,
 654    ) -> SpeedHeading | None:
 655        """Compute speed and forward bearing between two consecutive positions.
 656
 657        Distance and forward azimuth are solved together by a single geodesic
 658        call (:meth:`Geocode.get_distances_and_headings_vector`); the speed in
 659        m/s is the distance over the elapsed time in milliseconds (from
 660        :attr:`ItsTime.epoch_ms`) scaled by :attr:`ItsTime.SEC_TO_MS_FACTOR`,
 661        the same formula used by ``GapFiller.apply_motion``.  When the time does
 662        not advance (elapsed time not strictly positive) no speed can be
 663        derived and ``None`` is returned.
 664
 665        Args:
 666            last_time (str | None): ISO 8601 UTC timestamp of the previous
 667                position (e.g. ``"2026-02-24T05:25:51.339Z"``).  ``None`` when
 668                no preceding position is available.
 669            last_position (LatLonPoint | None): WGS-84
 670                ``(latitude, longitude)`` of the previous position in decimal
 671                degrees.  ``None`` when no preceding position is available.
 672            new_time (str): ISO 8601 UTC timestamp of the current position.
 673            new_position (LatLonPoint): WGS-84 ``(latitude, longitude)``
 674                of the current position in decimal degrees.
 675
 676        Returns:
 677            SpeedHeading | None: An object containing ``speed`` in m/s and
 678                ``heading`` as the forward azimuth in degrees ``[0, 360)``, or
 679                ``None`` when either ``last_time`` or ``last_position`` is
 680                ``None``, when ``new_time`` is not strictly after
 681                ``last_time``, or when the geodesic solver returns no result.
 682
 683        Complexity:
 684            - Time: O(1), two timestamp conversions and one geodesic solve.
 685            - Space: O(1), one small tuple.
 686        """
 687        if (last_position is None) or (last_time is None):
 688            return None
 689        time_diff_ms = self._epoch_ms(iso_time=new_time) - self._epoch_ms(
 690            iso_time=last_time
 691        )
 692        if time_diff_ms <= 0:
 693            return None
 694        distance_heading = Geocode.get_distances_and_headings_vector(
 695            points=[new_position],
 696            previous_point=last_position,
 697            dist_unit=DistanceUnit.METERS,
 698            angle_unit=AngleUnit.DEGREES,
 699        )[0]
 700        if distance_heading is None:
 701            return None
 702        return SpeedHeading(
 703            speed=distance_heading.distance * ItsTime.SEC_TO_MS_FACTOR / time_diff_ms,
 704            heading=distance_heading.heading,
 705        )

 707    def _epoch_ms(self, iso_time: str) -> int:
 708        """Convert an ISO 8601 UTC timestamp to milliseconds since the Unix epoch.
 709
 710        Uses (and overwrites) the shared ``ItsTime`` helper; results are not
 711        cached.
 712
 713        Args:
 714            iso_time (str): ISO 8601 UTC timestamp
 715                (e.g. ``"2026-02-24T05:25:51.339Z"``).
 716
 717        Returns:
 718            int: Milliseconds since the Unix epoch.
 719
 720        Complexity:
 721            - Time: O(1), one fixed-length timestamp parse.
 722            - Space: O(1), no allocation.
 723        """
 724        self._time.iso_format = iso_time
 725        return self._time.epoch_ms

 727    def _insert_position_data(  # pylint: disable=too-many-locals
 728        self,
 729        match: Match[str],
 730        last_actor_id: ActorIdModel | None,
 731        data: dict[ActorIdModel, list[Position]],
 732    ) -> None:
 733        """Parse a position data point and append it to the actor's list.
 734
 735        Converts the raw string groups captured by ``POINT_PATTERN`` from a
 736        log line of the form::
 737
 738            --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), t: GNSS, s: 0.00, h: 21.10--
 739            --t: 2026-02-24T05:26:06.379Z, p: (45.609514, 8.863836), t: GNSS_predicted, s: None, h: None--
 740
 741        into a ``Position`` instance and appends it to ``data[last_actor_id]``.
 742        The raw fields are logged at INFO level first. If ``last_actor_id`` is
 743        ``None`` (no preceding ``--sid:--`` line was seen) the point is then
 744        dropped without a warning.  A point whose time is not strictly after
 745        the time of the actor's previous point (e.g. a duplicated timestamp) is
 746        discarded with a warning, so speed and heading can always be derived
 747        from a positive elapsed time. When the logged speed or heading is
 748        ``None``, only the missing value is filled in from
 749        ``_calculate_speed_and_heading`` against the actor's previous point
 750        (it stays ``None`` for the actor's first point).
 751
 752        Args:
 753            match (Match[str]): Regex match object with named groups
 754                ``timestamp``, ``latitude``, ``longitude``, ``fix_type``,
 755                ``speed``, and ``heading`` (see ``POINT_PATTERN``).
 756            last_actor_id (ActorIdModel | None): The most recently parsed actor
 757                identifier in the current section.  Points are associated with
 758                this actor.
 759            data (dict[ActorIdModel, list[Position]]): Mutable mapping that is
 760                updated in place; the new position is appended to the list at
 761                key ``last_actor_id``, which must already exist.
 762
 763        Returns:
 764            None: ``data`` is updated in place (or left unchanged when the
 765                point is dropped).
 766
 767        Raises:
 768            ValueError: If a numeric field is not a valid float, the fix type
 769                is not a ``PositionType`` value, or (as
 770                ``pydantic.ValidationError``) the position fails validation.
 771
 772        Complexity:
 773            - Time: O(1), a fixed number of conversions, at most one geodesic
 774              solve, and one model build.
 775            - Space: O(1), one appended position.
 776        """
 777        time_str = match.group("timestamp")
 778        lat_str = match.group("latitude")
 779        lon_str = match.group("longitude")
 780        position_type_str = match.group("fix_type")
 781        speed_str = match.group("speed")
 782        heading_str = match.group("heading")
 783        self._logger.info(
 784            msg=(
 785                f"Extracted data - time: {time_str}, latitude: {lat_str}, "
 786                f"longitude: {lon_str}, position_type: {position_type_str}, "
 787                f"speed: {speed_str}, heading: {heading_str}"
 788            )
 789        )
 790        if last_actor_id:
 791            last_position = data[last_actor_id][-1] if data[last_actor_id] else None
 792            if last_position and self._epoch_ms(iso_time=time_str) <= self._epoch_ms(
 793                iso_time=last_position.time
 794            ):
 795                self._logger.warning(
 796                    msg=(
 797                        f"Discarded position of actor {last_actor_id}: time {time_str} "
 798                        f"is not after the previous position time {last_position.time}"
 799                    )
 800                )
 801                return
 802            latitude = float(lat_str)
 803            longitude = float(lon_str)
 804            if speed_str == "None" or heading_str == "None":
 805                speed_heading_result = self._calculate_speed_and_heading(
 806                    last_time=last_position.time if last_position else None,
 807                    last_position=LatLonPoint(
 808                        latitude=last_position.latitude,
 809                        longitude=last_position.longitude,
 810                    )
 811                    if last_position
 812                    else None,
 813                    new_time=time_str,
 814                    new_position=LatLonPoint(latitude=latitude, longitude=longitude),
 815                )
 816                speed = (
 817                    float(speed_str)
 818                    if speed_str != "None"
 819                    else (speed_heading_result.speed if speed_heading_result else None)
 820                )
 821                heading = (
 822                    float(heading_str)
 823                    if heading_str != "None"
 824                    else (
 825                        speed_heading_result.heading if speed_heading_result else None
 826                    )
 827                )
 828            else:
 829                speed = float(speed_str)
 830                heading = float(heading_str)
 831            position = Position(
 832                time=time_str,
 833                latitude=latitude,
 834                longitude=longitude,
 835                positionType=PositionType(position_type_str),
 836                speed=speed,
 837                heading=heading,
 838            )
 839            self._logger.debug(msg=f"Created position: {position}")
 840            data[last_actor_id].append(position)

 842    def _extract_section_data(  # noqa: C901, PLR0912, PLR0915 pylint: disable=too-many-branches, too-many-locals, too-many-statements
 843        self, file: TextIO, data_type: DataType, section_data_type: SectionDataType
 844    ) -> dict[ActorIdModel, list[Position]] | None:
 845        """Extract all position data points from one section data block.
 846
 847        Reads lines from ``file`` and drives a
 848        ``ParserSectionDataState`` state machine until the matching
 849        ``end`` marker is found.  The section is delimited by::
 850
 851            --Section <data_type> <section_data_type> start--
 852            --sid: <station_id>, oid: <object_id|None>, type: <TrafficParticipantType>, confidence: <0-100|None>--
 853            --t: <ISO8601>, p: (<lat>, <lon>), t: <PositionType>, s: <speed|None>, h: <heading|None>--
 854            ... (more --sid:-- and --t:-- lines)
 855            --Section <data_type> <section_data_type> end--
 856
 857        Multiple actors can appear in a single section; each ``--sid:--`` line
 858        switches the current actor, and subsequent ``--t:--`` lines belong to
 859        that actor until the next ``--sid:--`` or the ``end`` marker.  Actors
 860        are identified by ``actor_key`` (``stationId``, ``objectId``): if the
 861        same actor appears again with a different type or confidence, its
 862        points are appended to the first-seen ``ActorIdModel``. Checkpoint
 863        lines met anywhere are appended to the scenario checkpoints, and
 864        lines before the ``start`` marker that are not markers are ignored.
 865        If a ``--Scenario start|end--`` marker is met, the file is rewound to
 866        that line so the scenario parser can handle it.
 867
 868        Example section (anomaly real)::
 869
 870            --Section anomaly real start--
 871            --sid: 37, oid: None, type: passengerCar, confidence: 90--
 872            --t: 2026-02-24T05:25:51.339Z, p: (45.610087, 8.863476), t: GNSS, s: 0.00, h: 21.10--
 873            --t: 2026-02-24T05:25:51.439Z, p: (45.610087, 8.863476), t: GNSS, s: 0.97, h: 21.10--
 874            --Section anomaly real end--
 875
 876        Args:
 877            file (TextIO): Open log file positioned just after the scenario
 878                ``start`` marker (or after the previous section ``end``).
 879            data_type (DataType): The expected section data type token
 880                (``normalized`` or ``anomaly``).
 881            section_data_type (SectionDataType): The expected section kind
 882                token (``real`` or ``forecasted``).
 883
 884        Returns:
 885            dict[ActorIdModel, list[Position]] | None: Mapping from each actor
 886                ID found in the section to its ordered list of parsed
 887                ``Position`` objects (possibly empty), or ``None`` if the
 888                ``end`` marker was never reached (truncated file or scenario
 889                marker met first).
 890
 891        Raises:
 892            ValueError: If a section data marker is found but its embedded
 893                ``data_type`` or ``section_data_type`` tokens do not match the
 894                expected values, if the state machine reaches an invalid
 895                state, or if an actor ID or point line fails to parse (see
 896                ``_get_actor_id`` and ``_insert_position_data``).
 897
 898        Complexity:
 899            - Time: O(L * c), L = lines read, c = line length; each line is
 900              scanned by a fixed set of regular expressions.
 901            - Space: O(p), p = positions parsed in the section.
 902        """
 903        data: dict[ActorIdModel, list[Position]] = {}
 904        known_actors: dict[ActorKey, ActorIdModel] = {}
 905        last_actor_id: ActorIdModel | None = None
 906        state = ParserSectionDataState.WAIT_START_SECTION_DATA
 907        while True:
 908            position = file.tell()
 909            if not (line := file.readline()):
 910                break
 911            line_stripped = line.strip()
 912            new_state = state
 913            data_type_read: str | None = None
 914            section_data_type_read: str | None = None
 915            section_state_read = None
 916            if match := self.SCENARIO_PATTERN.search(string=line_stripped):
 917                if self._logger.isEnabledFor(logging.WARNING):
 918                    self._logger.warning(
 919                        msg=f"Encountered scenario marker while parsing section data: {line_stripped}. "
 920                    )
 921                file.seek(position)
 922                return None
 923            if checkpoint := self._get_checkpoint(line=line_stripped):
 924                if self._logger.isEnabledFor(logging.DEBUG):
 925                    self._logger.debug(msg=f"Checkpoint line: {line_stripped}")
 926                self._checkpoints.append(checkpoint)
 927                continue
 928            if match := self.SECTION_DATA_PATTERN.search(string=line_stripped):
 929                if self._logger.isEnabledFor(logging.DEBUG):
 930                    self._logger.debug(msg=f"Section line: {line_stripped}")
 931                data_type_read, section_data_type_read, section_state_read = (
 932                    match.group("section_type"),
 933                    match.group("section_point_type"),
 934                    match.group("section_state"),
 935                )
 936                if data_type_read != data_type.value:
 937                    raise ValueError(
 938                        f"Section data type read {data_type_read} does not match expected data type {data_type}"
 939                    )
 940                if section_data_type_read != section_data_type.value:
 941                    raise ValueError(
 942                        f"Section data type read {section_data_type_read} does not match expected section data type {section_data_type}"
 943                    )
 944            match state:
 945                case ParserSectionDataState.WAIT_START_SECTION_DATA:
 946                    if section_state_read == "start":
 947                        new_state = ParserSectionDataState.WAIT_END_SECTION_DATA
 948                case ParserSectionDataState.WAIT_END_SECTION_DATA:
 949                    if section_state_read == "end":
 950                        new_state = ParserSectionDataState.DONE
 951                    elif match := self.ACTOR_ID_PATTERN.search(string=line_stripped):
 952                        if self._logger.isEnabledFor(logging.DEBUG):
 953                            self._logger.debug(
 954                                msg=f"Data processing line: {line_stripped}"
 955                            )
 956                        actor_id = self._get_actor_id(match=match)
 957                        last_actor_id = known_actors.setdefault(
 958                            actor_id.actor_key, actor_id
 959                        )
 960                        data.setdefault(last_actor_id, [])
 961                    elif match := self.POINT_PATTERN.search(string=line_stripped):
 962                        if self._logger.isEnabledFor(logging.DEBUG):
 963                            self._logger.debug(
 964                                msg=f"Data processing line: {line_stripped}"
 965                            )
 966                        self._insert_position_data(
 967                            match=match, last_actor_id=last_actor_id, data=data
 968                        )
 969                case _:
 970                    raise ValueError(f"Invalid parser section state {state}")
 971            if state != new_state:
 972                if self._logger.isEnabledFor(logging.INFO):
 973                    self._logger.info(
 974                        msg=(
 975                            f"Parser section state changed from {state} to {new_state} for "
 976                            f"data type {data_type_read} and section data type {section_data_type_read}"
 977                        )
 978                    )
 979                state = new_state
 980            if state == ParserSectionDataState.DONE:
 981                break
 982        return data if state == ParserSectionDataState.DONE else None

 984    def _extract_crash_data(  # noqa: C901, PLR0912, PLR0915 pylint: disable=too-many-locals, too-many-branches, too-many-statements
 985        self, file: TextIO
 986    ) -> tuple[bool, CrashInfo | None]:
 987        """Extract one crash record or detect the end of the crash section.
 988
 989        Reads lines from ``file`` and drives a ``ParserCrashState`` state
 990        machine until a ``--Crash end--`` marker completes a record, a
 991        ``--Section crash end--`` marker ends the section, a scenario marker
 992        is met (the file is then rewound to that line), or EOF. Checkpoint
 993        lines inside a crash record are ignored. Duplicate actors (same
 994        ``actor_key``) in one record are kept once.
 995
 996        A crash record has the following structure::
 997
 998            --Crash start--
 999            --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 5--
1000            --sid: 36, oid: None, type: passengerCar, confidence: 90--
1001            --sid: 37, oid: None, type: pedestrian, confidence: 80--
1002            --Crash end--
1003
1004        The risk value is an information quality integer in ``[1, 7]``.
1005        At least two ``--sid:--`` lines are expected (the actors involved).
1006
1007        Args:
1008            file (TextIO): Open log file positioned inside a crash section
1009                (after ``--Section crash start--``).
1010
1011        Returns:
1012            tuple[bool, CrashInfo | None]: A two-element tuple where:
1013
1014                * First element (bool): ``True`` if a fatal error occurred (an
1015                  unexpected scenario marker was encountered), ``False``
1016                  otherwise.
1017                * Second element (CrashInfo | None): The parsed
1018                  ``CrashInfo`` when a complete crash record is found, or
1019                  ``None`` when the crash section ends normally
1020                  (``--Section crash end--``) or when EOF is reached before
1021                  the record is complete.
1022
1023        Raises:
1024            ValueError: If a complete record has fewer than
1025                ``MINIMUM_ACTORS_IN_CRASH`` distinct actors (including a
1026                record whose risk line is missing or unparsable, whose actor
1027                lines are then ignored) or a risk outside ``[1, 7]``, which
1028                aborts the whole extraction; also if the state machine
1029                reaches an unrecognized state, or an actor ID line fails to
1030                parse (see ``_get_actor_id``).
1031
1032        Complexity:
1033            - Time: O(L * c + a**2), L = lines read, c = line length, a =
1034              actors in the record (linear duplicate check per actor).
1035            - Space: O(a), the actors involved.
1036        """
1037        state = ParserCrashState.WAIT_START_CRASH
1038        estimated_timestamp = ""
1039        estimated_latitude = 0.0
1040        estimated_longitude = 0.0
1041        risk = 0
1042        actors_involved: list[ActorIdModel] = []
1043        while True:
1044            position = file.tell()
1045            if not (line := file.readline()):
1046                break
1047            line_stripped = line.strip()
1048            new_state = state
1049            if match := self.SCENARIO_PATTERN.search(string=line_stripped):
1050                if self._logger.isEnabledFor(logging.WARNING):
1051                    self._logger.warning(
1052                        msg=f"Encountered scenario marker while parsing crash data: {line_stripped}. "
1053                    )
1054                file.seek(position)
1055                return (True, None)
1056            if match := self.SECTION_CRASH_PATTERN.search(string=line_stripped):
1057                if self._logger.isEnabledFor(logging.DEBUG):
1058                    self._logger.debug(msg=f"Section crash line: {line_stripped}")
1059                if match.group("section_crash_state") == "end":
1060                    return (False, None)
1061            if match := self.CRASH_PATTERN.search(string=line_stripped):
1062                if self._logger.isEnabledFor(logging.DEBUG):
1063                    self._logger.debug(
1064                        msg=f"Crash data processing line: {line_stripped}"
1065                    )
1066                crash_state_read = match.group("crash_state")
1067            else:
1068                crash_state_read = None
1069            match state:
1070                case ParserCrashState.WAIT_START_CRASH:
1071                    if crash_state_read == "start":
1072                        new_state = ParserCrashState.WAIT_RISK
1073                case ParserCrashState.WAIT_RISK:
1074                    if crash_state_read == "end":
1075                        new_state = ParserCrashState.DONE
1076                    elif match := self.CRASH_RISK_PATTERN.search(string=line_stripped):
1077                        if self._logger.isEnabledFor(logging.DEBUG):
1078                            self._logger.debug(
1079                                msg=f"Crash data processing line: {line_stripped}"
1080                            )

1082                            match.group("estimated_timestamp"),
1083                            match.group("latitude"),
1084                            match.group("longitude"),
1085                            match.group("risk"),
1086                        )
1087                        estimated_latitude = float(lat_str)
1088                        estimated_longitude = float(lon_str)
1089                        risk = int(risk_str)
1090                        if self._logger.isEnabledFor(logging.INFO):
1091                            self._logger.info(
1092                                msg=(
1093                                    f"Extracted crash data - time: {estimated_timestamp}, "
1094                                    f"latitude: {estimated_latitude:.7f}, longitude: {estimated_longitude:.7f}, "
1095                                    f"risk: {risk}"
1096                                )
1097                            )
1098                        new_state = ParserCrashState.WAIT_END_CRASH
1099                case ParserCrashState.WAIT_END_CRASH:
1100                    if crash_state_read == "end":
1101                        new_state = ParserCrashState.DONE
1102                    elif match := self.ACTOR_ID_PATTERN.search(string=line_stripped):

1104                            self._logger.debug(
1105                                msg=f"Crash data processing line: {line_stripped}"
1106                            )

1108                        if all(
1109                            actor.actor_key != actor_id.actor_key
1110                            for actor in actors_involved
1111                        ):
1112                            actors_involved.append(actor_id)

1114                    raise ValueError(f"Invalid parser crash state {state}")

1116                if self._logger.isEnabledFor(logging.INFO):
1117                    self._logger.info(
1118                        msg=f"Parser section state changed from {state} to {new_state}"
1119                    )
1120                state = new_state
1121            if state == ParserCrashState.DONE:
1122                break
1123        if state != ParserCrashState.DONE:
1124            self._logger.warning(
1125                msg="End of file reached before crash record was complete"
1126            )
1127            return (False, None)
1128        if len(actors_involved) < self.MINIMUM_ACTORS_IN_CRASH:
1129            raise ValueError(
1130                f"Parsed crash record with insufficient actors involved ({len(actors_involved)}), "
1131                f"at least {self.MINIMUM_ACTORS_IN_CRASH} expected"
1132            )

1134            raise ValueError(
1135                f"Parsed crash record with risk {risk} outside the range "
1136                f"[{INFORMATION_QUALITY_MIN}, {INFORMATION_QUALITY_MAX}]"
1137            )
1138        return (
1139            False,
1140            CrashInfo(
1141                estimatedTimestamp=estimated_timestamp,
1142                estimatedLatitude=estimated_latitude,
1143                estimatedLongitude=estimated_longitude,
1144                risk=risk,
1145                actorsInvolved=actors_involved,

1149    def _extract_section_crash_data(self, file: TextIO) -> list[CrashInfo] | None:  # noqa: C901
1150        """Extract all crash records from the crash section of a scenario.
1151
1152        Reads lines from ``file`` until the opening ``--Section crash start--``
1153        marker is found (collecting checkpoint lines met on the way), then
1154        delegates to :meth:`_extract_crash_data` in a loop: each returned
1155        ``CrashInfo`` is appended, a ``None`` result ends the section
1156        (``--Section crash end--`` or EOF), an error flag aborts the section,
1157        and an invalid record raises ``ValueError``.
1158
1159        A crash section may contain zero or more individual crash records::
1160
1161            --Section crash start--
1162            --Crash start--
1163            --Time: 2026-02-24T05:25:51.339Z, pos: (45.610087, 8.863476), risk: 5--
1164            --sid: 36, oid: None, type: passengerCar, confidence: 90--
1165            --sid: 37, oid: None, type: pedestrian, confidence: 80--
1166            --Crash end--
1167            --Section crash end--
1168
1169        When no crashes occurred the section is empty::
1170
1171            --Section crash start--
1172            --Section crash end--
1173
1174        Args:
1175            file (TextIO): Open log file positioned after the last section data
1176                ``end`` marker of the current scenario.
1177
1178        Returns:
1179            list[CrashInfo] | None: The ``CrashInfo`` objects read for the
1180                scenario (may be empty, also when EOF is reached first), or
1181                ``None`` if a scenario marker was met before the section ended
1182                (the file is then rewound to that line).
1183
1184        Raises:
1185            ValueError: Propagated from :meth:`_extract_crash_data`.
1186
1187        Complexity:
1188            - Time: O(L * c + K * a**2), L = lines read, c = line length,
1189              K = crash records, a = actors per record.
1190            - Space: O(K * a), the crash records.
1191        """
1192        crash_info_list: list[CrashInfo] = []
1193        while True:
1194            position = file.tell()
1195            if not (line := file.readline()):
1196                break
1197            line_stripped = line.strip()
1198            if match := self.SCENARIO_PATTERN.search(string=line_stripped):
1199                if self._logger.isEnabledFor(logging.WARNING):
1200                    self._logger.warning(
1201                        msg=f"Encountered scenario marker while parsing crash section: {line_stripped}. "
1202                    )
1203                file.seek(position)
1204                return None
1205            if checkpoint := self._get_checkpoint(line=line_stripped):
1206                if self._logger.isEnabledFor(logging.DEBUG):
1207                    self._logger.debug(msg=f"Checkpoint line: {line_stripped}")
1208                self._checkpoints.append(checkpoint)
1209                continue
1210            if match := self.SECTION_CRASH_PATTERN.search(string=line_stripped):
1211                if self._logger.isEnabledFor(logging.DEBUG):
1212                    self._logger.debug(msg=f"Section crash line: {line_stripped}")
1213                if match.group("section_crash_state") == "start":
1214                    while True:
1215                        is_error, crashes_info = self._extract_crash_data(file=file)
1216                        if is_error:
1217                            return None
1218                        if crashes_info is None:
1219                            break
1220                        crash_info_list.append(crashes_info)
1221                    break
1222        return crash_info_list

1224    def _create_session(
1225        self,
1226        session_index: int,
1227        session_time: str,
1228        session_data: dict[
1229            ActorIdModel, dict[ParserTableDataMarkerItem, list[Position]]
1230        ],
1231        crash_list: list[CrashInfo],
1232        checkpoints: list[CheckpointDataModel],
1233    ) -> Session | None:
1234        """Build the per-actor ``ActorSession`` list and assemble the scenario ``Session``.
1235
1236        For each actor present in ``session_data``, picks one data source:
1237
1238        * If the actor has ``normalized``/``real`` positions, that is the
1239          chosen ``realPositions``, paired with whatever
1240          ``normalized``/``forecasted`` positions are available (possibly
1241          none) as ``forecastedPositions``, and ``dataType`` is set to
1242          ``DataType.NORMALIZED``.
1243        * Otherwise, the actor's ``anomaly``/``real`` positions are used as
1244          ``realPositions``, ``forecastedPositions`` is left empty, and
1245          ``dataType`` is set to ``DataType.ANOMALY``.
1246
1247        Normalized and anomaly data for the same actor are never combined;
1248        anomaly data is only used as a fallback when no normalized real
1249        points were recorded for that actor.
1250
1251        Args:
1252            session_index (int): Zero-based index of the current scenario, used
1253                to populate the ``sessionIndex`` field of each
1254                ``SessionData``.
1255            session_time (str): ISO 8601 timestamp extracted from the
1256                ``--Scenario t:`` marker; stored in the resulting
1257                ``Session``.
1258            session_data (dict[ActorIdModel, dict[ParserTableDataMarkerItem, list[Position]]]):
1259                Nested mapping built by :meth:`_extract_section`; the outer key
1260                is the actor ID and the inner key identifies the section type
1261                (normalized/anomaly * real/forecasted).
1262            crash_list (list[CrashInfo]): All crash records extracted for the
1263                current scenario.
1264            checkpoints (list[CheckpointDataModel]): All checkpoint metadata
1265                extracted for the current scenario.
1266
1267        Returns:
1268            Session | None: The assembled ``Session`` containing one
1269                ``ActorSession`` per actor in ``session_data`` plus
1270                ``crash_list`` and ``checkpoints``, or ``None`` if
1271                ``session_data`` was empty (no actor produced a session).
1272
1273        Raises:
1274            pydantic.ValidationError: If an actor has no real position in the
1275                chosen source (e.g. an actor listed only in the forecasted
1276                section, or whose points were all discarded), since
1277                ``SessionData`` requires at least one, or if more than 100
1278                checkpoints were collected.
1279
1280        Complexity:
1281            - Time: O(A + p + k), A = actors, p = positions, k = checkpoints
1282              (lists are re-validated by the models).
1283            - Space: O(A), the actor sessions; position lists are shared.
1284        """
1285        actor_session: list[ActorSession] = []
1286        for actor_id, marker_data in session_data.items():
1287            real_positions = marker_data.get(
1288                ParserTableDataMarkerItem(
1289                    data_type=DataType.NORMALIZED,
1290                    section_data_type=SectionDataType.REAL,
1291                ),
1292                [],
1293            )
1294            if len(real_positions) > 0:
1295                data_type = DataType.NORMALIZED
1296                forecasted_positions: list[Position] = marker_data.get(
1297                    ParserTableDataMarkerItem(
1298                        data_type=DataType.NORMALIZED,
1299                        section_data_type=SectionDataType.FORECASTED,
1300                    ),
1301                    [],
1302                )
1303            else:
1304                data_type = DataType.ANOMALY
1305                forecasted_positions = []
1306                real_positions = marker_data.get(
1307                    ParserTableDataMarkerItem(
1308                        data_type=DataType.ANOMALY,
1309                        section_data_type=SectionDataType.REAL,
1310                    ),
1311                    [],
1312                )
1313            actor_session_item = ActorSession(
1314                actorId=actor_id,
1315                data=SessionData(
1316                    sessionIndex=session_index,
1317                    sessionTime=session_time,
1318                    dataType=data_type,
1319                    realPositions=real_positions,
1320                    forecastedPositions=forecasted_positions,
1321                ),
1322            )
1323            actor_session.append(actor_session_item)
1324        if len(actor_session) == 0:
1325            return None
1326        return Session(
1327            sessionTime=session_time,
1328            actorsSession=actor_session,
1329            crashesInfo=crash_list,
1330            checkpoints=checkpoints,
1331        )

1333    def _extract_section(  # noqa: C901, PLR0912, PLR0915 pylint: disable=too-many-locals, too-many-branches, too-many-statements
1334        self, file: TextIO, session_index: int
1335    ) -> tuple[bool, Session | None]:
1336        """Parse one complete scenario block from the log file.
1337
1338        Drives the top-level ``ParserScenarioState`` state machine.  When
1339        the ``--Scenario start--`` marker is found the scenario checkpoints
1340        are reset; once the ``--Scenario t:`` line is read it sequentially
1341        delegates to :meth:`_extract_section_data` (three times, covering
1342        normalized real, normalized forecasted, and anomaly real, merging
1343        actors by ``actor_key`` across the three blocks) and
1344        :meth:`_extract_section_crash_data`, then waits for
1345        ``--Scenario end--``. Lines before ``--Scenario start--`` are skipped,
1346        and so is a ``--Scenario start--`` met while waiting for the end.
1347
1348        The expected structure is::
1349
1350            --Scenario start--
1351              --Scenario t: 2026-02-24T05:25:51.339Z--
1352              --Checkpoint: scenario start, timer: <ms.3f>--
1353              ... (checkpoint markers interleaved with section markers)
1354              --Section normalized real start-- ... --Section normalized real end--
1355              --Checkpoint: log forecasted norm, timer: <ms.3f>--
1356              --Section normalized forecasted start-- ... --Section normalized forecasted end--
1357              --Checkpoint: scenario norm end, timer: <ms.3f>--
1358              --Checkpoint: scenario anomaly start, timer: <ms.3f>--
1359              --Section anomaly real start-- ... --Section anomaly real end--
1360              --Checkpoint: scenario anomaly end, timer: <ms.3f>--
1361              --Checkpoint: send forecasted data, timer: <ms.3f>--
1362              --Checkpoint: send crash detection results, timer: <ms.3f>--
1363              --Section crash start-- ... --Section crash end--
1364              --Checkpoint: scenario end, timer: <ms.3f>--
1365            --Scenario end--
1366
1367        Args:
1368            file (TextIO): Open log file; reading continues from the current
1369                position.
1370            session_index (int): Index of the session being built (the number
1371                of sessions produced so far), used to set the
1372                ``sessionIndex`` field of each ``SessionData`` in the session.
1373
1374        Returns:
1375            tuple[bool, Session | None]: A two-element tuple where:
1376
1377                * First element (bool): ``True`` if EOF was reached before a
1378                  complete scenario was found, ``False`` otherwise.
1379                * Second element (Session | None): The assembled
1380                  ``Session`` for the parsed scenario, or ``None`` if the
1381                  scenario was incomplete, contained no actors, or a
1382                  sub-parser signalled failure (the caller then simply
1383                  continues with the next scenario).
1384
1385        Raises:
1386            ValueError: If ``--Scenario end--`` is found before the scenario
1387                time, if the state machine reaches an invalid state, or
1388                propagated from the sub-parsers and :meth:`_create_session`.
1389
1390        Complexity:
1391            - Time: O(L * c + K * a**2 + p), L = lines read, c = line
1392              length, K = crash records, a = actors per record, p =
1393              positions.
1394            - Space: O(p + K * a + k), the scenario positions, crashes, and
1395              k checkpoints.
1396        """
1397        crash_list: list[CrashInfo] = []
1398        session_time: str = ""
1399        session_data: dict[
1400            ActorIdModel, dict[ParserTableDataMarkerItem, list[Position]]
1401        ] = {}
1402        known_actors: dict[ActorKey, ActorIdModel] = {}
1403        state = ParserScenarioState.WAIT_FOR_START_SCENARIO
1404        while line := file.readline():
1405            line_stripped = line.strip()
1406            new_state = state
1407            if match := self.SCENARIO_PATTERN.search(string=line_stripped):
1408                if self._logger.isEnabledFor(logging.DEBUG):
1409                    self._logger.debug(msg=f"Scenario line: {line_stripped}")
1410                scenario_state_read = match.group("scenario_state")
1411            else:
1412                scenario_state_read = None
1413            match state:
1414                case ParserScenarioState.WAIT_FOR_START_SCENARIO:
1415                    if scenario_state_read == "start":
1416                        self._checkpoints = []
1417                        new_state = ParserScenarioState.WAIT_FOR_SESSION_TIME
1418                case ParserScenarioState.WAIT_FOR_SESSION_TIME:
1419                    if scenario_state_read == "end":
1420                        raise ValueError("Scenario end found before scenario time")
1421                    if checkpoint := self._get_checkpoint(line=line_stripped):
1422                        if self._logger.isEnabledFor(level=logging.DEBUG):
1423                            self._logger.debug(msg=f"Checkpoint line: {line_stripped}")
1424                        self._checkpoints.append(checkpoint)
1425                        continue
1426                    if match := self.SCENARIO_TIME_PATTERN.search(string=line_stripped):
1427                        if self._logger.isEnabledFor(logging.DEBUG):
1428                            self._logger.debug(
1429                                msg=f"Data processing line: {line_stripped}"
1430                            )
1431                        session_time = str(object=match.group("scenario_time"))
1432                        for marker in self.DATA_MARKERS:
1433                            data_result = self._extract_section_data(
1434                                file=file,
1435                                data_type=marker.data_type,
1436                                section_data_type=marker.section_data_type,
1437                            )
1438                            if data_result is None:
1439                                return (False, None)
1440                            for actor_id, actor_data in data_result.items():
1441                                canonical_id = known_actors.setdefault(
1442                                    actor_id.actor_key, actor_id
1443                                )
1444                                session_data.setdefault(canonical_id, {})[marker] = (
1445                                    actor_data
1446                                )
1447                        # Crash scenario
1448                        crash_list_tmp = self._extract_section_crash_data(file=file)
1449                        if crash_list_tmp is None:
1450                            return (False, None)
1451                        crash_list = crash_list_tmp
1452                        new_state = ParserScenarioState.WAIT_FOR_END_SCENARIO
1453                case ParserScenarioState.WAIT_FOR_END_SCENARIO:
1454                    if scenario_state_read == "end":
1455                        new_state = ParserScenarioState.DONE
1456                    elif checkpoint := self._get_checkpoint(line=line_stripped):
1457                        if self._logger.isEnabledFor(level=logging.DEBUG):
1458                            self._logger.debug(msg=f"Checkpoint line: {line_stripped}")
1459                        self._checkpoints.append(checkpoint)
1460                case _:
1461                    raise ValueError(f"Invalid parser state {state}")
1462            if state != new_state:
1463                if self._logger.isEnabledFor(level=logging.INFO):
1464                    self._logger.info(
1465                        msg=f"Parser state changed from {state} to {new_state}"
1466                    )
1467                state = new_state
1468            if state == ParserScenarioState.DONE:
1469                break
1470        if state != ParserScenarioState.DONE:
1471            return (True, None)
1472        return (
1473            False,
1474            self._create_session(
1475                session_index=session_index,
1476                session_time=session_time,
1477                session_data=session_data,
1478                crash_list=crash_list,
1479                checkpoints=self._checkpoints,
1480            ),
1481        )
