 220        """Validate attribute initialization performed by ``__init__``.

 222        Checks that every slot-backed attribute is set to its expected initial
 223        value immediately after construction: ``_input_file`` stores the

 227        ``ParserTableDataMarkerItem`` entries covering NORMALIZED/REAL,

 229        reference to the corresponding session dict.
 230
 231        Args:

 235        Returns:

 242        assert extractor._input_file == Path("dummy_path.txt")
 243        assert extractor._collections is None
 244        assert extractor._session_normalized_real_data == {}
 245        assert extractor._session_normalized_forecasted_data == {}
 246        assert extractor._session_anomaly_real_data == {}
 247        assert extractor._session_crash_data == []

 386    ) -> None:
 387        """``_insert_position_data`` appends a ``Position`` to the actor's list.

 397                ``None`` if no insertion should happen.
 398
 399        Returns:

 401        """

 404        data: dict[ActorId, list[Position]] = {}
 405        if last_actor_id is not None:
 406            data[last_actor_id] = []
 407        match = LogExtractor.POINT_PATTERN.search(string=line)
 408        assert match is not None
 409        extractor._insert_position_data(
 410            match=match, last_actor_id=last_actor_id, data=data
 411        )
