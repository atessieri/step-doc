252def test_get_logger(
253    mocker: MockerFixture, file_log: Optional[Path], log_level: Optional[LogMode]
254) -> None:
255    """Assert ``get_logger`` wires up stream and optional file handlers correctly.
256
257    Verifies that ``get_logger``:
258
259    - Calls ``getLogger()`` once with no arguments and clears existing handlers.
260    - Sets the logger level to ``log_level`` (defaults to ``ERROR`` when ``None``).
261    - Creates a ``StreamHandler`` on ``sys.stdout`` fixed at ``ERROR`` level.
262    - When ``file_log`` is given, creates the parent directory (``parents=True,
263      exist_ok=True``) and attaches a ``FileHandler`` at the resolved log level.
264    - Returns the configured logger instance.
265
266    Args:
267        mocker: pytest-mock fixture for patching dependencies.
268        file_log: Path to the log file, or ``None`` for stream-only logging.
269        log_level: Desired log level, or ``None`` to use the default (``ERROR``).
270
271    Returns:
272        None
273    """
274    # Arrange
275    mock_logger = mocker.MagicMock(spec=Logger)
276    mock_logger.handlers = mocker.MagicMock(spec=list[Handler])
277    mock_logger.handlers.clear = mocker.MagicMock()
278    mock_logger.setLevel = mocker.MagicMock()
279    mock_logger.addHandler = mocker.MagicMock()
280    mock_stream_handler = mocker.MagicMock(spec=Handler)
281    mock_stream_handler.setFormatter = mocker.MagicMock()
282    mock_stream_handler.setLevel = mocker.MagicMock()
283    mock_file_handler = mocker.MagicMock(spec=Handler)
284    mock_file_handler.setFormatter = mocker.MagicMock()
285    mock_file_handler.setLevel = mocker.MagicMock()
286    mock_get_logger = mocker.patch(
287        "crash_prediction_log_extractor.step_logger.getLogger", return_value=mock_logger
288    )
289    mock_stream_handler_class = mocker.patch(
290        "crash_prediction_log_extractor.step_logger.StreamHandler",
291        return_value=mock_stream_handler,
292    )
293    mock_file_handler_class = mocker.patch(
294        "crash_prediction_log_extractor.step_logger.FileHandler",
295        return_value=mock_file_handler,
296    )
297    mocker.patch(
298        "crash_prediction_log_extractor.step_logger.UtcFormatter",
299        return_value=mocker.MagicMock(spec=UtcFormatter),
300    )
301    mock_mkdir = mocker.patch.object(target=Path, attribute="mkdir")
302    # Act
303    result = get_logger(file_log=file_log, log_level=log_level)
304    # Assert
305    mock_get_logger.assert_called_once_with()
306    mock_logger.handlers.clear.assert_called_once()
307    expected_log_level = LogMode.to_logging_level(mode=log_level or LogMode.ERROR)
308    mock_logger.setLevel.assert_called_once_with(level=expected_log_level)
309    mock_stream_handler_class.assert_called_once_with(stream=sys.stdout)
310    mock_stream_handler.setLevel.assert_called_once_with(level=logging.ERROR)
311    mock_logger.addHandler.assert_any_call(hdlr=mock_stream_handler)
312    if file_log:
313        mock_mkdir.assert_called_once()
314        assert mock_mkdir.call_args.kwargs == {"parents": True, "exist_ok": True}
315        mock_file_handler_class.assert_called_once_with(
316            filename=file_log, mode="w", encoding="utf-8", delay=False, errors=None
317        )
318        mock_file_handler.setLevel.assert_called_once_with(level=expected_log_level)
319        mock_logger.addHandler.assert_any_call(hdlr=mock_file_handler)
320    assert result == mock_logger

323def test_logger_force_write(mocker: MockerFixture) -> None:
324    """Assert ``logger_force_write`` temporarily elevates levels to INFO, logs, then
325    restores.
326
327    Sets up a mock logger at ``WARNING`` level with two handlers also at ``WARNING``.
328    Verifies that ``logger_force_write`` raises both the logger and every handler to
329    ``INFO`` before calling ``logger.info(msg=msg)``, and then restores each back to
330    their original ``WARNING`` level.
331
332    Args:
333        mocker: pytest-mock fixture for creating mock objects.
334
335    Returns:
336        None
337    """
338    # Arrange
339    mock_logger = mocker.MagicMock(spec=Logger)
340    mock_logger.info = mocker.MagicMock()
341    mock_logger.level = logging.WARNING
342    mock_logger.setLevel = mocker.MagicMock()
343    mock_handler = mocker.MagicMock(spec=Handler)
344    mock_handler.level = logging.WARNING
345    mock_handler.setLevel = mocker.MagicMock()
346    mock_logger.handlers = [mock_handler, mock_handler]
347    msg = "Test message"
348    # Act
349    logger_force_write(logger=mock_logger, msg=msg)
350    # Assert
351    mock_logger.setLevel.assert_any_call(level=logging.INFO)
352    mock_logger.setLevel.assert_any_call(level=logging.WARNING)
353    mock_handler.setLevel.assert_any_call(level=logging.INFO)
354    mock_handler.setLevel.assert_any_call(level=logging.WARNING)
355    mock_logger.info.assert_called_once_with(msg=msg)
